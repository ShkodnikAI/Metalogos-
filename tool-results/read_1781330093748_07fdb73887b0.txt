    77→/// Parse source, build Axum router, start server on configured port.
    78→/// This is the entry point for `mlog serve <file>`.
    79→pub async fn run_server(source: &str) -> Result<(), Box<dyn std::error::Error + Send + Sync>> {
    80→    let declarations = crate::parser::parse(source)
    81→        .map_err(|e| format!("parse error: {}", e))?;
    82→
    83→    let mut interp = Interpreter::new();
    84→    // Run declarations to populate templates, patterns, etc. (skip flows)
    85→    for decl in declarations.clone() {
    86→        match decl {
    87→            Declaration::MlogServer(ref srv) => {
    88→                interp = build_interpreter_with_server(srv, interp);
    89→            }
    90→            Declaration::Flow(_) => { /* skip flows in server mode */ }
    91→            _ => {
    92→                let mut tmp_interp = Interpreter::new();
    93→                tmp_interp.set_base_dir(std::path::PathBuf::from("."));
    94→                let _ = tmp_interp.run(vec![decl]);
    95→                interp = merge_interpreter(tmp_interp, interp);
    96→            }
    97→        }
    98→    }
    99→
   100→    // Find MlogServer declaration
   101→    let server_config = declarations.iter().find_map(|d| match d {
   102→        Declaration::MlogServer(s) => Some(s.clone()),
   103→        _ => None,
   104→    });
   105→
   106→    let config = match server_config {
   107→        Some(c) => c,
   108→        None => return Err("no mlogserver block found in source".into()),
   109→    };
   110→
   111→    let port = config.port;
   112→    let state = build_state(config, interp).await;
   113→    let app = build_router(state);
   114→
   115→    println!("mlog serve: listening on 0.0.0.0:{}", port);
   116→    let listener = tokio::net::TcpListener::bind(format!("0.0.0.0:{}", port)).await?;
   117→    axum::serve(listener, app).await?;
   118→    Ok(())
   119→}

   162→async fn build_state(config: MlogServerDecl, interp: Interpreter) -> ServerState {
   163→    // Generate HMAC key
   164→    let hmac_key = generate_hmac_key();
   165→
   166→    // Collect templates from interpreter
   167→    let templates_map = interp.get_templates().clone();
   168→
   169→    // Phase 7.4: SQLite session store (in-memory for this server instance)
   170→    let conn = rusqlite::Connection::open_in_memory()
   171→        .expect("Failed to open SQLite in-memory database");
   172→    init_session_db(&conn).expect("Failed to create sessions table");
   173→
   174→    ServerState {
   175→        sessions: Arc::new(RwLock::new(HashMap::new())),
   176→        csrf_tokens: Arc::new(RwLock::new(HashMap::new())),
   177→        hmac_key: Arc::new(hmac_key),
   178→        audit_log: Arc::new(RwLock::new(Vec::new())),
   179→        templates: Arc::new(RwLock::new(templates_map)),
   180→        db_store: Arc::new(RwLock::new(Vec::new())),
   181→        memory_persist: interp.get_memory_persist_path(),
   182→        interpreter: Arc::new(RwLock::new(interp)),
   183→        routes: config.routes.clone(),
   184→        middleware: config.middleware.clone(),
   185→        db: Arc::new(tokio::sync::Mutex::new(conn)),
   186→        rate_limits: Arc::new(RwLock::new(HashMap::new())),
   187→    }
   188→}

   190→fn build_router(state: ServerState) -> Router {
   191→    let mut app = Router::new();
   192→
   193→    // Add security headers layer (always applied)
   194→    app = app.layer(SetResponseHeaderLayer::if_not_present(
   195→        header::X_FRAME_OPTIONS, HeaderValue::from_static("DENY"),
   196→    ));
   197→    app = app.layer(SetResponseHeaderLayer::if_not_present(
   198→        header::X_CONTENT_TYPE_OPTIONS, HeaderValue::from_static("nosniff"),
   199→    ));
   200→    app = app.layer(SetResponseHeaderLayer::if_not_present(
   201→        header::STRICT_TRANSPORT_SECURITY, HeaderValue::from_static("max-age=31536000; includeSubDomains"),
   202→    ));
   203→    app = app.layer(SetResponseHeaderLayer::if_not_present(
   204→        header::CONTENT_SECURITY_POLICY,
   205→        HeaderValue::from_static("default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline'"),
   206→    ));
   207→
   208→    // Register routes
   209→    for route in &state.routes {
   210→        let path = route.path.clone();
   211→        let handler = route_handler;
   212→
   213→        match route.method.as_str() {
   214→            "GET" => app = app.route(&path, get(handler)),
   215→            "POST" => app = app.route(&path, post(handler)),
   216→            "PUT" => app = app.route(&path, put(handler)),
   217→            "DELETE" => app = app.route(&path, delete(handler)),
   218→            _ => app = app.route(&path, any(handler)),
   219→        }
   220→    }
   221→
   222→    app.with_state(state)
   223→}

   225→// ── Route Handler ──────────────────────────────────────────────────
   226→
   227→async fn route_handler(
   228→    State(state): State<ServerState>,
   229→    uri: Uri,
   230→    method: Method,
   231→    headers: HeaderMap,
   232→    body: bytes::Bytes,
   233→) -> Response {
   234→    // 0. Extract client IP for rate limiting
   235→    let client_ip = extract_client_ip(&headers);
   236→
   237→    // 1. Rate limiting (Phase 7.4)
   238→    if state.middleware.contains(&"rate_limit".to_string()) {
   239→        if let Err(resp) = check_rate_limit(&state, &client_ip, 100).await {
   240→            return resp;
   241→        }
   242→    }
   243→
   244→    // 2. CSRF check for mutating methods (Phase 7.4: real double-submit)
   245→    if matches!(method, Method::POST | Method::PUT | Method::DELETE) {
   246→        if state.middleware.contains(&"csrf".to_string()) {
   247→            if let Err(resp) = check_csrf(&state, &headers).await {
   248→                return resp;
   249→            }
   250→        }
   251→    }
   252→
   253→    // 3. Session expiry check (Phase 7.4: SQLite-backed)
   254→    if state.middleware.contains(&"session".to_string()) {
   255→        if let Some(session_id) = extract_session_cookie(&headers) {
   256→            // Verify HMAC signature first
   257→            let verified = verify_cookie(&session_id, &state.hmac_key);
   258→            if let Some(raw_id) = verified {
   259→                if let Err(resp) = validate_session_in_db(&state, &raw_id).await {
   260→                    return resp;
   261→                }
   262→            }
   263→        }
   264→    }
   265→
   266→    // 4. Find matching route by path AND method
   267→    let matched_route = state.routes.iter().find(|r| {
   268→        r.path == uri.path() && r.method == method.as_str()
   269→    });
   270→
   271→    if let Some(route) = matched_route {
   272→        // Role check
   273→        if !route.requires.is_empty() {
   274→            if state.middleware.contains(&"session".to_string()) {
   275→                if let Err(resp) = check_roles(&state, &headers, &route.requires).await {
   276→                    return resp;
   277→                }
   278→            }
   279→        }
   280→
   281→        // Execute route handler body
   282→        let result = execute_route_body(&state, &route.body, &headers, &body).await;
   283→        let mut response = match result {
   284→            Ok(response) => response,
   285→            Err(e) => (
   286→                StatusCode::INTERNAL_SERVER_ERROR,
   287→                format!("Handler error: {}", e),
   288→            ).into_response(),
   289→        };
   290→
   291→        // 5. On GET with CSRF middleware, generate and set CSRF token cookie (Phase 7.4)
   292→        if method == Method::GET && state.middleware.contains(&"csrf".to_string()) {
   293→            let token = generate_csrf_token();
   294→            {
   295→                let mut tokens = state.csrf_tokens.write().await;
   296→                tokens.insert(token.clone(), token.clone());
   297→            }
   298→            let cookie_value = format!(
   299→                "_mlog_csrf={}; HttpOnly; SameSite=Strict; Path=/",
   300→                token
   301→            );
   302→            if let Ok(val) = HeaderValue::from_str(&cookie_value) {
   303→                response.headers_mut().append(header::SET_COOKIE, val);
   304→            }
   305→        }
   306→
   307→        response
   308→    } else {
   309→        (StatusCode::NOT_FOUND, "404 Not Found").into_response()
   310→    }
   311→}

   313→// ── CSRF Middleware (Phase 7.4: real double-submit) ────────────────
   314→
   315→/// Generate a cryptographically random CSRF token (32 hex chars).
   316→pub fn generate_csrf_token() -> String {
   317→    use rand::Rng;
   318→    let mut buf = [0u8; 16];
   319→    rand::thread_rng().fill(&mut buf[..]);
   320→    hex::encode(buf)
   321→}
   322→
   323→async fn check_csrf(state: &ServerState, headers: &HeaderMap) -> Result<(), Response> {
   324→    // Read CSRF token from cookie
   325→    let cookie_token = headers.get("cookie")
   326→        .and_then(|c| c.to_str().ok())
   327→        .and_then(|s| extract_cookie(s, "_mlog_csrf"));
   328→
   329→    // Read CSRF token from header (X-CSRF-Token) or form field (_csrf)
   330→    let header_token = headers.get("x-csrf-token")
   331→        .and_then(|t| t.to_str().ok())
   332→        .map(|s| s.to_string())
   333→        .or_else(|| {
   334→            // Also check content-type for form data with _csrf field
   335→            headers.get("x-csrf-field")
   336→                .and_then(|t| t.to_str().ok())
   337→                .map(|s| s.to_string())
   338→        });
   339→
   340→    match (cookie_token, header_token) {
   341→        (Some(cookie), Some(header)) if cookie == header => Ok(()),
   342→        _ => {
   343→            // Log to audit
   344→            {
   345→                let mut log = state.audit_log.write().await;
   346→                log.push("[CSRF] Rejected: missing or mismatched CSRF token".to_string());
   347→            }
   348→            Err((StatusCode::FORBIDDEN, "403 Forbidden: CSRF token validation failed").into_response())
   349→        }
   350→    }
   351→}
   352→
   353→fn extract_cookie(cookie_header: &str, name: &str) -> Option<String> {
   354→    for pair in cookie_header.split(';') {
   355→        let pair = pair.trim();
   356→        if let Some(eq_pos) = pair.find('=') {
   357→            let key = &pair[..eq_pos];
   358→            let val = &pair[eq_pos + 1..];
   359→            if key.trim() == name {
   360→                return Some(val.trim().to_string());
   361→            }
   362→        }
   363→    }
   364→    None
   365→}
   366→
   367→/// Extract the session cookie (unsigned) from the Cookie header.
   368→fn extract_session_cookie(headers: &HeaderMap) -> Option<String> {
   369→    headers.get("cookie")
   370→        .and_then(|c| c.to_str().ok())
   371→        .and_then(|s| extract_cookie(s, "_mlog_session"))
   372→}
   373→
   374→/// Extract client IP from headers (x-forwarded-for or x-real-ip).
   375→fn extract_client_ip(headers: &HeaderMap) -> String {
   376→    headers.get("x-forwarded-for")
   377→        .and_then(|v| v.to_str().ok())
   378→        .map(|s| s.split(',').next().unwrap_or("unknown").trim().to_string())
   379→        .or_else(|| {
   380→            headers.get("x-real-ip")
   381→                .and_then(|v| v.to_str().ok())
   382→                .map(|s| s.to_string())
   383→        })
   384→        .unwrap_or_else(|| "unknown".to_string())
   385→}

   387→// ── Rate Limiting (Phase 7.4) ─────────────────────────────────────
   388→
   389→/// Check rate limit using sliding window. Returns Err(429) if exceeded.
   390→pub async fn check_rate_limit(
   391→    state: &ServerState,
   392→    ip: &str,
   393→    max_per_minute: usize,
   394→) -> Result<(), Response> {
   395→    let mut limits = state.rate_limits.write().await;
   396→    let now = std::time::Instant::now();
   397→    let window_start = now - std::time::Duration::from_secs(60);
   398→
   399→    let entries = limits.entry(ip.to_string()).or_default();
   400→    // Remove entries outside the 60-second window
   401→    entries.retain(|&t| t > window_start);
   402→
   403→    if entries.len() >= max_per_minute {
   404→        {
   405→            let mut log = state.audit_log.write().await;
   406→            log.push(format!("[RATE_LIMIT] Rejected: {} exceeded {} req/min", ip, max_per_minute));
   407→        }
   408→        return Err((
   409→            StatusCode::TOO_MANY_REQUESTS,
   410→            "429 Too Many Requests: rate limit exceeded",
   411→        ).into_response());
   412→    }
   413→
   414→    entries.push(now);
   415→    Ok(())
   416→}

   418→// ── Session & Role Middleware ────────────────────────────────────────
   419→
   420→async fn check_roles(state: &ServerState, headers: &HeaderMap, required_roles: &[String]) -> Result<(), Response> {
   421→    let session_cookie = extract_session_cookie(headers);
   422→
   423→    let raw_id = match session_cookie {
   424→        Some(id) => {
   425→            // Verify HMAC signature
   426→            match verify_cookie(&id, &state.hmac_key) {
   427→                Some(raw) => raw,
   428→                None => {
   429→                    let mut log = state.audit_log.write().await;
   430→                    log.push("[AUTH] Rejected: tampered session cookie".to_string());
   431→                    return Err((StatusCode::UNAUTHORIZED, "401 Unauthorized: invalid session signature").into_response());
   432→                }
   433→            }
   434→        }
   435→        None => {
   436→            let mut log = state.audit_log.write().await;
   437→            log.push("[AUTH] Rejected: no session cookie".to_string());
   438→            return Err((StatusCode::UNAUTHORIZED, "401 Unauthorized: no session").into_response());
   439→        }
   440→    };
   441→
   442→    // Check in-memory cache first, then SQLite
   443→    let sessions = state.sessions.read().await;
   444→    if let Some(entry) = sessions.get(&raw_id) {
   445→        if entry.expires < std::time::Instant::now() {
   446→            return Err((StatusCode::UNAUTHORIZED, "401 Unauthorized: session expired").into_response());
   447→        }
   448→        let has_role = required_roles.iter().any(|role| entry.roles.contains(role));
   449→        if has_role {
   450→            Ok(())
   451→        } else {
   452→            drop(sessions);
   453→            let mut log = state.audit_log.write().await;
   454→            log.push(format!("[AUTH] Rejected: insufficient roles (need {:?}, have {:?})",
   455→                required_roles, Vec::<String>::new()));
   456→            Err((StatusCode::FORBIDDEN, "403 Forbidden: insufficient permissions").into_response())
   457→        }
   458→    } else {
   459→        // Fall through to SQLite check
   460→        drop(sessions);
   461→        validate_session_in_db(&state, &raw_id).await?;
   462→        // If valid but not in memory cache, load from DB
   463→        // For simplicity, reject here — session needs re-login
   464→        Err((StatusCode::UNAUTHORIZED, "401 Unauthorized: session not found in cache").into_response())
   465→    }
   466→}

   470→/// Initialize the sessions table in SQLite.
   471→pub fn init_session_db(conn: &rusqlite::Connection) -> Result<(), rusqlite::Error> {
   472→    conn.execute_batch(
   473→        "CREATE TABLE IF NOT EXISTS sessions (
   474→            id TEXT PRIMARY KEY,

   476→            data TEXT NOT NULL DEFAULT '{}',
   477→            created_at INTEGER NOT NULL,
   478→            expires_at INTEGER NOT NULL
   479→        );
   480→        CREATE INDEX IF NOT EXISTS idx_sessions_expires ON sessions(expires_at);
   481→
   482→        -- Phase 7.5: Audit log table for interpreter audit entries
   483→        CREATE TABLE IF NOT EXISTS audit_log (
   484→            id INTEGER PRIMARY KEY AUTOINCREMENT,
   485→            timestamp INTEGER NOT NULL,
   486→            action TEXT NOT NULL,
   487→            pattern TEXT,
   488→            result TEXT,
   489→            sandbox TEXT
   490→        );"
   491→    )?;
   492→    Ok(())
   493→}

   495→/// Create a new session in SQLite. Returns the session ID (UUID).
   496→pub async fn create_session_db(
   497→    conn: &Arc<tokio::sync::Mutex<rusqlite::Connection>>,
   498→    user_id: &str,
   499→) -> Result<String, String> {
   500→    let id = uuid::Uuid::new_v4().to_string();
   501→    let now = std::time::SystemTime::now()
   502→        .duration_since(std::time::UNIX_EPOCH)
   503→        .unwrap_or_default()
   504→        .as_secs() as i64;
   505→    let expires_at = now + 24 * 3600; // 24 hours
   506→
   507→    let conn = conn.lock().await;
   508→    conn.execute(
   509→        "INSERT INTO sessions (id, user_id, data, created_at, expires_at) VALUES (?1, ?2, '{}', ?3, ?4)",
   510→        rusqlite::params![id, user_id, now, expires_at],
   511→    ).map_err(|e| format!("Failed to create session: {}", e))?;
   512→
   513→    Ok(id)
   514→}

   516→/// Validate a session against SQLite: check existence and expiry.
   517→/// Returns Ok(()) if valid, Err(Response) if expired or not found.
   518→pub async fn validate_session_in_db(state: &ServerState, session_id: &str) -> Result<(), Response> {
   519→    let now = std::time::SystemTime::now()
   520→        .duration_since(std::time::UNIX_EPOCH)
   521→        .unwrap_or_default()
   522→        .as_secs() as i64;
   523→
   524→    let conn = state.db.lock().await;
   525→    let result: Result<String, _> = conn.query_row(
   526→        "SELECT id FROM sessions WHERE id = ?1 AND expires_at > ?2",
   527→        rusqlite::params![session_id, now],
   528→        |row| row.get(0),
   529→    );
   530→    drop(conn); // release lock before async ops
   531→
   532→    match result {
   533→        Ok(_) => Ok(()),
   534→        Err(rusqlite::Error::QueryReturnedNoRows) => {
   535→            let mut log = state.audit_log.write().await;
   536→            log.push("[AUTH] Rejected: session expired or not found in DB".to_string());
   537→            Err((StatusCode::UNAUTHORIZED, "401 Unauthorized: session expired").into_response())
   538→        }
   539→        Err(e) => {
   540→            Err((StatusCode::INTERNAL_SERVER_ERROR, format!("DB error: {}", e)).into_response())
   541→        }
   542→    }
   543→}

   586→/// Recursively convert serde_json::Value → metalogos Value.
   587→/// Supports nested objects (→ Value::Struct), arrays, strings, numbers, bools, null.
   588→pub fn json_value_to_value(val: &serde_json::Value) -> Value {
   589→    match val {
   590→        serde_json::Value::String(s) => Value::String(s.clone()),
   591→        serde_json::Value::Number(n) => Value::Float(n.as_f64().unwrap_or(0.0)),
   592→        serde_json::Value::Bool(b) => Value::Bool(*b),
   593→        serde_json::Value::Null => Value::Unit,
   594→        serde_json::Value::Array(arr) => {
   595→            Value::List(arr.iter().map(json_value_to_value).collect())
   596→        }
   597→        serde_json::Value::Object(map) => {
   598→            let fields: HashMap<String, Value> = map
   599→                .iter()
   600→                .map(|(k, v)| (k.clone(), json_value_to_value(v)))
   601→                .collect();
   602→            Value::Struct {
   603→                type_name: "JsonObject".to_string(),
   604→                fields,
   605→            }
   606→        }
   607→    }
   608→}

   610→// ── Route Body Execution ────────────────────────────────────────────
   611→
   612→async fn execute_route_body(
   613→    state: &ServerState,
   614→    body_stmts: &[Statement],
   615→    _headers: &HeaderMap,
   616→    raw_body: &bytes::Bytes,
   617→) -> Result<Response, String> {
   618→    // Set up interpreter with request context (Наряд №8: route pattern invocation fix)
   619→    let mut interp = Interpreter::new();
   620→    // Copy ALL program definitions (patterns, learnables, templates, struct types,
   621→    // rules, sandboxes, namespaces, variables, db_config, db_url) from shared interpreter.
   622→    {
   623→        let shared = state.interpreter.read().await;
   624→        shared.clone_definitions_into(&mut interp);
   625→    }
   626→    interp.set_base_dir(std::path::PathBuf::from("."));
   627→
   628→    // Initialize memory persistence (per-request SQLite connection to shared DB)
   629→    if let Some(ref persist_path) = state.memory_persist {
   630→        interp.configure_memory(&MemoryDecl { persist: Some(persist_path.clone()) });
   631→    }
   632→
   633→    // Initialize DB connection for per-request interpreter (query() / db_execute())
   634→    // Opens a NEW connection to the same database, so concurrent requests are safe.
   635→    interp.reconnect_db();
   636→
   637→    // Parse JSON body recursively and inject as json_body() server builtin (Наряд №3)
   638→    if let Ok(body_str) = std::str::from_utf8(raw_body) {
   639→        if !body_str.is_empty() {
   640→            if let Ok(json) = serde_json::from_str::<serde_json::Value>(body_str) {
   641→                let value = json_value_to_value(&json);
   642→                interp.set_server_json_body(value);
   643→            }
   644→        }
   645→    }
   646→
   647→    // Execute body statements
   648→    let mut env = HashMap::new();
   649→    for stmt in body_stmts {
   650→        match stmt {
   651→            Statement::LetBinding { name, value } => {
   652→                let val = interp.eval_expr_with_env(value, &env)?;
   653→                env.insert(name.clone(), val);
   654→            }
   655→            Statement::Assign { name, value } => {
   656→                let val = interp.eval_expr_with_env(value, &env)?;
   657→                if env.contains_key(name) {
   658→                    env.insert(name.clone(), val);
   659→                }
   660→            }
   661→            Statement::Return(expr) => {
   662→                let val = interp.eval_expr_with_env(expr, &env)?;
   663→                // Phase 7.5: Flush interpreter audit entries to SQLite
   664→                flush_audit_to_db(state, &mut interp).await;
   665→                return Ok(value_to_response(val));
   666→            }
   667→            Statement::IfThen(cond, body) => {
   668→                let cond_val = interp.eval_expr_with_env(cond, &env)?;
   669→                if cond_val.as_bool().unwrap_or(false) {
   670→                    let result = interp.eval_statements(body, &mut env)?;
   671→                    if !matches!(result, Value::Unit) {
   672→                        flush_audit_to_db(state, &mut interp).await;
   673→                        return Ok(value_to_response(result));
   674→                    }
   675→                }
   676→            }
   677→            // Block-level if/else (Наряд №2 + final integration)
   678→            Statement::IfElseBlock { condition, then_body, else_ifs, else_body } => {
   679→                let cond_val = interp.eval_expr_with_env(condition, &env)?;
   680→                let branch = if cond_val.as_bool().unwrap_or(false) {
   681→                    Some(then_body.as_slice())
   682→                } else {
   683→                    // Check else-if chain
   684→                    let mut matched = None;
   685→                    for (ei_cond, ei_body) in else_ifs {
   686→                        let ei_val = interp.eval_expr_with_env(ei_cond, &env)?;
   687→                        if ei_val.as_bool().unwrap_or(false) {
   688→                            matched = Some(ei_body.as_slice());
   689→                            break;
   690→                        }
   691→                    }
   692→                    matched.or_else(|| else_body.as_deref())
   693→                };
   694→                if let Some(stmts) = branch {
   695→                    for s in stmts {
   696→                        match s {
   697→                            Statement::Return(expr) => {
   698→                                let val = interp.eval_expr_with_env(expr, &env)?;
   699→                                flush_audit_to_db(state, &mut interp).await;
   700→                                return Ok(value_to_response(val));
   701→                            }
   702→                            Statement::LetBinding { name, value } => {
   703→                                let val = interp.eval_expr_with_env(value, &env)?;
   704→                                env.insert(name.clone(), val);
   705→                            }
   706→                            Statement::ExprStmt(expr) => {
   707→                                let val = interp.eval_expr_with_env(expr, &env)?;
   708→                                if let Value::HttpResponse { .. } = val {
   709→                                    flush_audit_to_db(state, &mut interp).await;
   710→                                    return Ok(value_to_response(val));
   711→                                }
   712→                            }
   713→                            _ => { interp.eval_statements(&[s.clone()], &mut env)?; }
   714→                        }
   715→                    }
   716→                }
   717→            }
   718→            // Bare expression statement — evaluate for side effects
   719→            Statement::ExprStmt(expr) => {
   720→                let val = interp.eval_expr_with_env(expr, &env)?;
   721→                // If expression is respond("ok") or similar HttpResponse, use as route response
   722→                if let Value::HttpResponse { .. } = val {
   723→                    flush_audit_to_db(state, &mut interp).await;
   724→                    return Ok(value_to_response(val));
   725→                }
   726→            }
   727→            _ => {
   728→                let result = interp.eval_statements(&[stmt.clone()], &mut env)?;
   729→                // If the statement produced an HttpResponse (e.g., respond("ok")),
   730→                // use it as the route response (final integration)
   731→                if let Value::HttpResponse { .. } = result {
   732→                    flush_audit_to_db(state, &mut interp).await;
   733→                    return Ok(value_to_response(result));
   734→                }
   735→            }
   736→        }
   737→    }
   738→
   739→    // Phase 7.5: Flush interpreter audit entries to SQLite before response
   740→    flush_audit_to_db(state, &mut interp).await;
   741→
   742→    Ok((StatusCode::OK, "OK").into_response())
   743→}
   744→
   745→/// Phase 7.5: Write interpreter audit entries to the SQLite audit_log table.
   746→async fn flush_audit_to_db(state: &ServerState, interp: &mut Interpreter) {
   747→    let entries = interp.take_audit_log();
   748→    if entries.is_empty() {
   749→        return;
   750→    }
   751→
   752→    let now = std::time::SystemTime::now()
   753→        .duration_since(std::time::UNIX_EPOCH)
   754→        .unwrap_or_default()
   755→        .as_secs() as i64;
   756→
   757→    // Get active sandbox name for context
   758→    let sandbox_name = interp.get_active_sandbox().map(|sb| sb.name.clone());
   759→
   760→    let conn = state.db.lock().await;
   761→    for entry in &entries {
   762→        // Parse entry to extract action and pattern name
   763→        let (action, pattern, result) = parse_audit_entry(entry);
   764→        let _ = conn.execute(
   765→            "INSERT INTO audit_log (timestamp, action, pattern, result, sandbox) VALUES (?1, ?2, ?3, ?4, ?5)",
   766→            rusqlite::params![now, action, pattern, result, sandbox_name.as_deref().unwrap_or("")],
   767→        );
   768→    }
   769→    // Also append to in-memory audit log for backward compatibility
   770→    {
   771→        let mut log = state.audit_log.write().await;
   772→        for entry in &entries {
   773→            log.push(entry.clone());
   774→        }
   775→    }
   776→}
   777→
   778→/// Parse an audit entry string into (action, pattern, result) components.
   779→/// Format: "[AUDIT] adapt PatternName: input -> output"
   780→///         "[AUDIT] mutate PatternName: N examples, accuracy=X"
   781→///         "[AUDIT] unsafe_html: rendered template 'name'"
   782→fn parse_audit_entry(entry: &str) -> (String, Option<String>, Option<String>) {
   783→    if let Some(rest) = entry.strip_prefix("[AUDIT] ") {
   784→        let parts: Vec<&str> = rest.splitn(2, ' ').collect();
   785→        let action = parts[0].to_string();
   786→        let detail = if parts.len() > 1 { Some(parts[1].to_string()) } else { None };
   787→
   788→        match action.as_str() {
   789→            "adapt" | "mutate" => {
   790→                // Extract pattern name (first word of detail)
   791→                let pattern = detail.as_ref().and_then(|d| d.split(':').next()).map(|s| s.trim().to_string());
   792→                let result = detail.as_ref().and_then(|d| {
   793→                    d.splitn(2, ':').nth(1).map(|s| s.trim().to_string())
   794→                });
   795→                (action, pattern, result)
   796→            }
   797→            "unsafe_html" => {
   798→                let pattern = detail.as_ref().and_then(|d| d.split('\'').nth(1)).map(|s| s.to_string());
   799→                (action, pattern, None)
   800→            }
   801→            _ => (action, None, None),
   802→        }
   803→    } else {
   804→        ("unknown".to_string(), None, None)
   805→    }
   806→}

   808→fn value_to_response(val: Value) -> Response {
   809→    match val {
   810→        Value::HttpResponse { status, body } => {
   811→            let code = StatusCode::from_u16(status).unwrap_or(StatusCode::OK);
   812→            (code, body).into_response()
   813→        }
   814→        Value::Html(html) => {
   815→            AxumHtml(html).into_response()
   816→        }
   817→        Value::String(s) => {

   819→        }
   820→        Value::Unit => {
   821→            StatusCode::OK.into_response()
   822→        }
   823→        other => {
   824→            (StatusCode::OK, format!("{}", other)).into_response()
   825→        }
   826→    }
   827→}

   838→pub fn sign_cookie(value: &str, key: &[u8]) -> String {
   839→    use hmac::{Hmac, Mac};
   840→    use sha2::Sha256;
   841→    type HmacSha256 = Hmac<Sha256>;
   842→
   843→    let mut mac = HmacSha256::new_from_slice(key).expect("HMAC key error");
   844→    mac.update(value.as_bytes());
   845→    let result = mac.finalize();
   846→    let signature = hex::encode(result.into_bytes());
   847→    format!("{}.{}", value, signature)
   848→}
   849→
   850→pub fn verify_cookie(cookie: &str, key: &[u8]) -> Option<String> {
   851→    use hmac::{Hmac, Mac};
   852→    use sha2::Sha256;
   853→    type HmacSha256 = Hmac<Sha256>;
   854→
   855→    let parts: Vec<&str> = cookie.rsplitn(2, '.').collect();
   856→    if parts.len() != 2 {
   857→        return None;
   858→    }
   859→    let signature = parts[0];
   860→    let value = parts[1];
   861→
   862→    let mut mac = HmacSha256::new_from_slice(key).expect("HMAC key error");
   863→    mac.update(value.as_bytes());
   864→    let expected = hex::encode(mac.finalize().into_bytes());
   865→
   866→    if signature == expected {
   867→        Some(value.to_string())
   868→    } else {
   869→        None
   870→    }
   871→}

   875→/// Escape HTML special characters to prevent XSS.
   876→pub fn escape_html(s: &str) -> String {
   877→    s.replace('&', "&amp;")
   878→     .replace('<', "&lt;")
   879→     .replace('>', "&gt;")
   880→     .replace('"', "&quot;")
   881→     .replace('\'', "&#x27;")
   882→}

   884→/// Simple template rendering: replace {{ var }} with escaped values.
   885→pub fn render_template(body: &str, vars: &HashMap<String, String>) -> String {
   886→    let mut result = body.to_string();
   887→    for (key, val) in vars {
   888→        let escaped = escape_html(val);
   889→        result = result.replace(&format!("{{{{{}}}}}", key), &escaped);
   890→    }
   891→    result
   892→}

   901→fn merge_interpreter(from: Interpreter, mut into: Interpreter) -> Interpreter {
   902→    // Merge variables (borrow, don't move)
   903→    for (k, v) in &from.variables {
   904→        into.variables.entry(k.clone()).or_insert(v.clone());
   905→    }
   906→    // Merge templates
   907→    for (k, v) in from.get_templates() {
   908→        into.templates.entry(k.clone()).or_insert(v.clone());
   909→    }
   910→    // Propagate memory persist path
   911→    if let Some(path) = from.get_memory_persist_path() {
   912→        into.set_memory_persist_path(Some(path));
   913→    }
   914→    // Merge patterns, struct types, learnable patterns, rules, sandboxes, module namespaces
   915→    from.clone_definitions_into(&mut into);
   916→    into
   917→}

    27→/// Opaque secret string with automatic memory zeroing on drop (Phase 7.3).
    28→/// Implements serde by serializing as "[SECRET]" marker — actual value is NEVER persisted.
    29→#[derive(Clone, Debug)]
    30→pub struct SecretString(pub Zeroizing<String>);
    31→
    32→impl serde::Serialize for SecretString {
    33→    fn serialize<S: serde::Serializer>(&self, s: S) -> Result<S::Ok, S::Error> {
    34→        // Never serialize the actual secret — emit a safe marker
    35→        s.serialize_str("[SECRET]")
    36→    }
    37→}
    38→
    39→impl<'de> serde::Deserialize<'de> for SecretString {
    40→    fn deserialize<D: serde::Deserializer<'de>>(d: D) -> Result<Self, D::Error> {
    41→        let inner = String::deserialize(d)?;
    42→        Ok(SecretString(Zeroizing::new(inner)))
    43→    }
    44→}

    46→impl std::ops::Deref for SecretString {
    47→    type Target = String;
    48→    fn deref(&self) -> &Self::Target {
    49→        &self.0
    50→    }
    51→}

    56→    }

    58→        self.0.as_str()
    59→    }

    62→/// Runtime value.
    63→#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
    64→pub enum Value {
    65→    String(String),
    66→    Float(f64),
    67→    Bool(bool),
    68→    Struct {
    69→        type_name: String,
    70→        fields: HashMap<String, Value>,
    71→    },
    72→    /// List value: ordered collection of items.
    73→    List(Vec<Value>),
    74→    /// Fluid value: superposition of typed variants with confidence scores.
    75→    /// Collapses lazily at point of use (see `maybe_collapse`).
    76→    Fluid(Vec<FluidValueVariant>),
    77→    Unit,
    78→    /// Opaque HTML content (Phase 6.2) — cannot be concatenated, printed, or converted to String
    79→    Html(String),
    80→    /// Opaque SQL query (Phase 6.3) — only created via query() builtin
    81→    Query(String),
    82→    /// Opaque secret value (Phase 6.4) — cannot be printed or converted to String.
    83→    /// Phase 7.3: Internally uses SecretString (Zeroizing<String>) — memory is zeroed on drop.
    84→    Secret(SecretString),
    85→    /// Opaque encrypted data (Phase 6.4)
    86→    Encrypted(Vec<u8>),
    87→    /// Opaque password hash (Phase 6.4)
    88→    Hash(String),
    89→    /// Opaque session data (Phase 6.5)
    90→    Session(std::collections::HashMap<String, String>),
    91→    /// HTTP response value (Phase 6.1)
    92→    HttpResponse { status: u16, body: String },
    93→}

    95→impl std::fmt::Display for Value {
    96→    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
    97→        match self {
    98→            Value::String(s) => write!(f, "{}", s),
    99→            Value::Float(n) => write!(f, "{}", n),
   100→            Value::Bool(b) => write!(f, "{}", b),
   101→            Value::Struct { type_name, fields } => {
   102→                write!(f, "{} {{", type_name)?;
   103→                let pairs: Vec<_> = fields.iter().collect();
   104→                for (i, (k, v)) in pairs.iter().enumerate() {
   105→                    if i > 0 { write!(f, ", ")?; }
   106→                    write!(f, "{}: {}", k, v)?;
   107→                }
   108→                write!(f, "}}")
   109→            }
   110→            Value::List(items) => {
   111→                write!(f, "[")?;
   112→                for (i, item) in items.iter().enumerate() {
   113→                    if i > 0 { write!(f, ", ")?; }
   114→                    write!(f, "{}", item)?;
   115→                }
   116→                write!(f, "]")
   117→            }
   118→            Value::Fluid(variants) => {
   119→                // Display as the highest-confidence variant
   120→                let best = variants.iter().max_by(|a, b| {
   121→                    a.confidence.partial_cmp(&b.confidence).unwrap_or(std::cmp::Ordering::Equal)
   122→                });
   123→                match best {
   124→                    Some(v) => write!(f, "{}", v.value),
   125→                    None => write!(f, "()"),
   126→                }
   127→            }
   128→            Value::Unit => write!(f, "()"),
   129→            Value::Html(_) => write!(f, "[Html]"),
   130→            Value::Query(_) => write!(f, "[Query]"),
   131→            Value::Secret(_) => write!(f, "[Secret]"),
   132→            Value::Encrypted(_) => write!(f, "[Encrypted]"),
   133→            Value::Hash(_) => write!(f, "[Hash]"),
   134→            Value::Session(_) => write!(f, "[Session]"),
   135→            Value::HttpResponse { status, .. } => write!(f, "[HttpResponse {}]", status),
   136→        }
   137→    }
   138→}

   140→impl Value {
   141→    pub fn type_name(&self) -> &'static str {
   142→        match self {
   143→            Value::String(_) => "String",
   144→            Value::Float(_) => "Float",
   145→            Value::Bool(_) => "Bool",
   146→            Value::List(_) => "List",
   147→            Value::Struct { .. } => "Struct",
   148→            Value::Fluid(_) => "Fluid",
   149→            Value::Unit => "Unit",
   150→            Value::Html(_) => "Html",
   151→            Value::Query(_) => "Query",
   152→            Value::Secret(_) => "Secret",
   153→            Value::Encrypted(_) => "Encrypted",
   154→            Value::Hash(_) => "Hash",
   155→            Value::Session(_) => "Session",
   156→            Value::HttpResponse { .. } => "HttpResponse",
   157→        }
   158→    }
   159→
   160→    /// Get a field value from a struct. Returns Err if not a struct or field missing.
   161→    pub fn get_field(&self, field: &str) -> Result<&Value, String> {
   162→        match self {
   163→            Value::Struct { fields, .. } => fields.get(field)
   164→                .ok_or_else(|| format!("field '{}' not found on struct", field)),
   165→            _ => Err(format!("cannot access field '{}' on non-struct value ({})", field, self.type_name())),
   166→        }
   167→    }
   168→
   169→    /// Set a field value on a mutable struct.
   170→    pub fn set_field(&mut self, field: &str, value: Value) -> Result<(), String> {
   171→        match self {
   172→            Value::Struct { fields, .. } => {
   173→                if fields.contains_key(field) {
   174→                    fields.insert(field.to_string(), value);
   175→                    Ok(())
   176→                } else {
   177→                    Err(format!("field '{}' not found on struct", field))
   178→                }
   179→            }
   180→            _ => Err(format!("cannot set field '{}' on non-struct value", field)),
   181→        }
   182→    }
   183→
   184→    /// Convert to f64 for numeric comparisons.
   185→    pub fn as_float(&self) -> Result<f64, String> {
   186→        match self {
   187→            Value::Float(f) => Ok(*f),
   188→            Value::Bool(b) => Ok(if *b { 1.0 } else { 0.0 }),
   189→            Value::String(s) => s.parse::<f64>()
   190→                .map_err(|_| format!("cannot convert '{}' to Float", s)),
   191→            _ => Err(format!("cannot convert {} to Float", self.type_name())),
   192→        }
   193→    }
   194→
   195→    /// Convert to bool for condition checking.
   196→    pub fn as_bool(&self) -> Result<bool, String> {
   197→        match self {
   198→            Value::Bool(b) => Ok(*b),
   199→            Value::Float(f) => Ok(*f != 0.0),
   200→            Value::String(s) => Ok(!s.is_empty()),
   201→            Value::Unit => Ok(false),
   202→            _ => Err(format!("cannot convert {} to Bool", self.type_name())),
   203→        }
   204→    }
   205→
   206→    /// Check if this value is a Fluid superposition.
   207→    pub fn is_fluid(&self) -> bool {
   208→        matches!(self, Value::Fluid(_))
   209→    }
   210→}

   426→impl EvalResult {
   427→    /// Format the eval result as a human-readable report.
   428→    pub fn format_report(&self) -> String {
   429→        let mut lines = Vec::new();
   430→        lines.push(format!("Eval: {}", self.pattern_name));
   431→        lines.push(format!("  Dataset: {} examples", self.total));
   432→        lines.push(format!("  Accuracy: {:.1}% ({}/{})", self.accuracy * 100.0, self.correct, self.total));
   433→        lines.push(format!("  Threshold: {}", self.threshold));
   434→        lines.push(format!("  Result: {}", if self.passed { "PASS" } else { "FAIL (below threshold)" }));
   435→
   436→        // Confusion matrix
   437→        if !self.confusion.is_empty() {
   438→            // Collect all unique labels
   439→            let mut all_labels: std::collections::BTreeSet<String> = std::collections::BTreeSet::new();
   440→            for (expected, predictions) in &self.confusion {
   441→                all_labels.insert(expected.clone());
   442→                for pred in predictions.keys() {
   443→                    all_labels.insert(pred.clone());
   444→                }
   445→            }
   446→
   447→            let labels: Vec<&String> = all_labels.iter().collect();
   448→
   449→            // Header row
   450→            let header = format!("  {:12} {}", "", labels.iter().map(|l| format!("{:>12}", l)).collect::<Vec<_>>().join(" "));
   451→            lines.push(header);
   452→
   453→            // Data rows
   454→            for expected in &labels {
   455→                let row = format!("  {:12} {}", expected,
   456→                    labels.iter().map(|pred| {
   457→                        let count = self.confusion.get(*expected)
   458→                            .and_then(|m| m.get(*pred))
   459→                            .copied()
   460→                            .unwrap_or(0);
   461→                        format!("{:>12}", count)
   462→                    }).collect::<Vec<_>>().join(" ")
   463→                );
   464→                lines.push(row);
   465→            }
   466→        }
   467→
   468→        // Failing examples with adapt suggestions
   469→        if !self.failures.is_empty() {
   470→            lines.push(String::new());
   471→            lines.push("  Failing examples (suggest adapt):".to_string());
   472→            for (input, expected, actual) in &self.failures {
   473→                lines.push(format!("    - {:?} -> expected {:?}, got {:?}", input, expected, actual));
   474→            }
   475→            // Generate adapt suggestions
   476→            lines.push(String::new());
   477→            lines.push("  Suggested adapt commands:".to_string());
   478→            for (input, expected, _actual) in &self.failures {
   479→                lines.push(format!("    adapt {} add_example({:?}, {:?})", self.pattern_name, input, expected));
   480→            }
   481→        }
   482→
   483→        lines.join("\n")
   484→    }
   485→}

   670→    /// ADR-0052: Emit an event to the event stream.

   672→    fn emit_event(&self, event_type: &str, source: &str, data: HashMap<String, String>, duration_ms: Option<u64>) {

   732→    /// Configure memory persistence (Phase 7.6).
   733→    /// If persist path is provided, switches to SQLite-backed stores.
   734→    /// The in-memory data is migrated to SQLite during the switch.
   735→    pub fn configure_memory(&mut self, config: &MemoryDecl) {
   736→        if let Some(ref path) = config.persist {
   737→            // Switch to SQLite backend
   738→            let db_path = std::path::PathBuf::from(path);
   739→            match SqliteStore::open(&db_path) {
   740→                Ok(sqlite_store) => {
   741→                    // Migrate existing in-memory data to SQLite
   742→                    let existing = self.memory.lock().unwrap().all_entries();
   743→                    let mut new_store: Box<dyn MemoryStore> = Box::new(sqlite_store);
   744→                    for entry in existing {
   745→                        let _ = new_store.memorize(entry);
   746→                    }
   747→                    self.memory = std::sync::Mutex::new(new_store);
   748→
   749→                    // Migrate KG edges to SQLite (sharing the same DB file)
   750→                    let existing_edges: Vec<(String, String, String, f64)> =
   751→                        self.kg.lock().unwrap().all_edges();
   752→                    if let Ok(sqlite_kg) = SqliteKg::open(&db_path) {
   753→                        let mut new_kg: Box<dyn KgStore> = Box::new(sqlite_kg);
   754→                        for (from, to, relation, weight) in existing_edges {
   755→                            let _ = new_kg.relate(&from, &to, &relation, weight);
   756→                        }
   757→                        self.kg = std::sync::Mutex::new(new_kg);
   758→                    } else {
   759→                        eprintln!("[memory] KG migration to SQLite failed; keeping in-memory KG");
   760→                    }
   761→                    eprintln!("[memory] Persistence enabled: {}", path);
   762→                    self.memory_persist_path = Some(path.clone());
   763→
   764→                    // Наряд №6 — also enable KV store SQLite persistence
   765→                    if let Err(e) = crate::builtins::init_kv_persist(path) {
   766→                        eprintln!("[kv_store] Failed to enable KV persistence: {}. KV will be in-memory only.", e);
   767→                    }
   768→
   769→                    // ADR-0056: initialize checkpoint SQLite (same DB directory)
   770→                    let cp_path = std::path::PathBuf::from(path).with_file_name("checkpoints.db");
   771→                    if let Ok(conn) = rusqlite::Connection::open(&cp_path) {
   772→                        let _ = conn.execute_batch(
   773→                            "CREATE TABLE IF NOT EXISTS checkpoints (
   774→                                flow_name TEXT NOT NULL,
   775→                                checkpoint_name TEXT NOT NULL,
   776→                                step_index INTEGER NOT NULL,
   777→                                state_json TEXT NOT NULL,
   778→                                created_at INTEGER NOT NULL,
   779→                                PRIMARY KEY (flow_name, checkpoint_name)
   780→                            )"
   781→                        );
   782→                        *self.checkpoint_db.lock().unwrap() = Some(conn);
   783→                    }
   784→                }
   785→                Err(e) => {
   786→                    eprintln!("[memory] Failed to open persistent store '{}': {}. Using in-memory.", path, e);
   787→                }
   788→            }
   789→        }
   790→        // If persist is None, keep the default InMemoryStore (already set in new())
   791→    }

   793→    /// Initialize SQLite connection for db { url: "..." } block (Наряд №7).
   794→    /// Supports "sqlite::memory:" for in-memory databases and file paths.
   795→    fn init_db_connection(&mut self, db: &DbDecl) {
   796→        let url_expr = match &db.url {
   797→            Some(expr) => expr,
   798→            None => {
   799→                eprintln!("[db] No url specified in db {{}} block — query() will be unavailable");
   800→                return;
   801→            }
   802→        };
   803→        // Evaluate the url expression (must be a string literal or variable)
   804→        let url = match self.eval_expr(url_expr) {
   805→            Ok(Value::String(s)) => s,
   806→            Ok(other) => {
   807→                eprintln!("[db] url must be a String, got {}", other.type_name());
   808→                return;
   809→            }
   810→            Err(e) => {
   811→                eprintln!("[db] Failed to evaluate url: {}", e);
   812→                return;
   813→            }
   814→        };
   815→        // Parse the URL: "sqlite::memory:" → in-memory, "sqlite:path.db" → file
   816→        let conn = if url == "sqlite::memory:" {
   817→            rusqlite::Connection::open_in_memory()
   818→        } else if url.starts_with("sqlite:") {
   819→            let path = url.trim_start_matches("sqlite:");
   820→            rusqlite::Connection::open(path)
   821→        } else {
   822→            eprintln!("[db] Unsupported URL scheme: '{}'. Use 'sqlite::memory:' or 'sqlite:path.db'", url);
   823→            return;
   824→        };
   825→        match conn {
   826→            Ok(c) => {
   827→                // Enable WAL mode for better concurrent read performance
   828→                let _ = c.execute_batch("PRAGMA journal_mode=WAL;");
   829→                let mut guard = self.db_conn.lock().unwrap();
   830→                *guard = Some(c);
   831→                // Store resolved URL for per-request interpreter reconnection
   832→                self.db_url = Some(url.clone());
   833→                eprintln!("[db] Connected: {}", url);
   834→            }
   835→            Err(e) => {
   836→                eprintln!("[db] Failed to connect to '{}': {}", url, e);
   837→            }
   838→        }
   839→    }

   841→    /// Execute a SQL query and return readable results (Наряд №7).
   842→    /// - SELECT → List of Struct (each row = struct with column names as fields)
   843→    /// - INSERT/UPDATE/DELETE → String with affected row count
   844→    /// - PRAGMA/CREATE/etc. → String "ok"
   845→    fn invoke_query(&self, args: &[Value]) -> Result<Value, String> {
   846→        let sql = match args.first() {
   847→            Some(Value::String(s)) => s.clone(),
   848→            Some(other) => return Err(format!("query() expected String SQL, got {}", other.type_name())),
   849→            None => return Err("query() requires at least 1 argument (SQL string)".to_string()),
   850→        };
   851→        let params: Vec<String> = if args.len() > 1 {
   852→            match &args[1] {
   853→                Value::List(items) => items.iter().filter_map(|v| match v {
   854→                    Value::String(s) => Some(s.clone()),
   855→                    Value::Float(n) => Some(format!("{}", n)),
   856→                    Value::Bool(b) => Some(format!("{}", b)),
   857→                    _ => None,
   858→                }).collect(),
   859→                _ => Vec::new(),
   860→            }
   861→        } else {
   862→            Vec::new()
   863→        };
   864→
   865→        let guard = self.db_conn.lock().map_err(|e| format!("db lock error: {}", e))?;
   866→        let conn = guard.as_ref().ok_or_else(|| {
   867→            "query() error: no database connection. Declare db { url: \"sqlite::memory:\" } first.".to_string()
   868→        })?;
   869→
   870→        let sql_upper = sql.trim().to_uppercase();
   871→        if sql_upper.starts_with("SELECT") || sql_upper.starts_with("PRAGMA") {
   872→            // SELECT/PRAGMA → List of Struct
   873→            let mut stmt = conn.prepare(&sql)
   874→                .map_err(|e| format!("query() SQL error: {}", e))?;
   875→            let col_names: Vec<String> = stmt.column_names().iter().map(|s| s.to_string()).collect();
   876→            let rows: Vec<Value> = stmt.query_map(rusqlite::params_from_iter(params.iter()), |row| {
   877→                let mut fields = std::collections::HashMap::new();
   878→                for (i, col) in col_names.iter().enumerate() {
   879→                    let val: Value = match row.get_ref(i) {
   880→                        Ok(rusqlite::types::ValueRef::Null) => Value::Unit,
   881→                        Ok(rusqlite::types::ValueRef::Integer(n)) => {
   882→                            // Heuristic: if the column name suggests it's an ID or count, keep as Float
   883→                            Value::Float(n as f64)
   884→                        }
   885→                        Ok(rusqlite::types::ValueRef::Real(f)) => Value::Float(f),
   886→                        Ok(rusqlite::types::ValueRef::Text(s)) => {
   887→                            Value::String(String::from_utf8_lossy(s).to_string())
   888→                        }
   889→                        Ok(rusqlite::types::ValueRef::Blob(b)) => {
   890→                            // Encode blobs as hex strings
   891→                            Value::String(b.iter().map(|byte| format!("{:02x}", byte)).collect())
   892→                        }
   893→                        Err(_) => Value::Unit,
   894→                    };
   895→                    fields.insert(col.clone(), val);
   896→                }
   897→                Ok(Value::Struct { type_name: "Row".to_string(), fields })
   898→            }).map_err(|e| format!("query() execution error: {}", e))?
   899→            .filter_map(|r| r.ok())
   900→            .collect();
   901→            Ok(Value::List(rows))
   902→        } else {
   903→            // INSERT/UPDATE/DELETE/CREATE/ALTER/etc. → affected row count as String
   904→            let affected = conn.execute(&sql, rusqlite::params_from_iter(params.iter()))
   905→                .map_err(|e| format!("query() SQL error: {}", e))?;
   906→            Ok(Value::String(affected.to_string()))
   907→        }
   908→    }

   910→    /// Execute a SQL statement via db_execute() — returns affected row count (Наряд №7).
   911→    fn invoke_db_execute(&self, args: &[Value]) -> Result<Value, String> {
   912→        let sql = match args.first() {
   913→            Some(Value::String(s)) => s.clone(),
   914→            Some(other) => return Err(format!("db_execute() expected String SQL, got {}", other.type_name())),
   915→            None => return Err("db_execute() requires at least 1 argument (SQL string)".to_string()),
   916→        };
   917→        let guard = self.db_conn.lock().map_err(|e| format!("db lock error: {}", e))?;

   922→            .map_err(|e| format!("db_execute() SQL error: {}", e))?;
   923→        Ok(Value::String(affected.to_string()))
   924→    }

   950→    /// Run a complete .mlog program.
   951→    pub fn run(&mut self, declarations: Vec<Declaration>) -> Result<Option<String>, String> {
   952→        let mut output: Option<String> = None;
   953→
   954→        for decl in declarations {
   955→            match decl {
   956→                Declaration::Import(import) => {
   957→                    self.handle_import(&import)?;
   958→                }
   959→                Declaration::EntityType(e) => {
   960→                    self.struct_types.insert(e.name.clone(), StructType {
   961→                        name: e.name.clone(),
   962→                        fields: e.fields,
   963→                    });
   964→                }
   965→                Declaration::EntityRecord(e) => {
   966→                    let value = self.instantiate_struct(&e.type_name, &e.fields)?;
   967→                    self.variables.insert(e.name.clone(), value);
   968→                }
   969→                Declaration::EntitySimple(e) => {
   970→                    let value = self.eval_expr(&e.value)?;
   971→                    self.variables.insert(e.name.clone(), value);
   972→                }
   973→                Declaration::Rule(r) => {
   974→                    self.rules.push(r);
   975→                }
   976→                Declaration::Memorize(m) => {
   977→                    let value_str = match self.eval_expr(&m.value)? {
   978→                        Value::String(s) => s,
   979→                        other => format!("{}", other),
   980→                    };
   981→                    let now = SystemTime::now()
   982→                        .duration_since(UNIX_EPOCH)
   983→                        .map(|d| d.as_secs() as i64)
   984→                        .unwrap_or(0);
   985→                    let embedding = self.embedding_manager.embed(&value_str).unwrap_or_default();
   986→                    let _ = self.memory.lock().unwrap().memorize(MemoryEntry {
   987→                        id: None,
   988→                        value: value_str.clone(),
   989→                        priority: m.priority,
   990→                        timestamp: now,
   991→                        decay_rate: 0.01,
   992→                        confidence: m.priority,
   993→                        embedding,
   994→                    });
   995→                    // ADR-0052: emit memory_store event
   996→                    let preview = if value_str.len() > 30 { &value_str[..30] } else { &value_str };
   997→                    let mut data = HashMap::new();
   998→                    data.insert("key_preview".to_string(), preview.to_string());
   999→                    data.insert("priority".to_string(), m.priority.to_string());
  1000→                    self.emit_event("memory_store", "system", data, None);
  1001→                }
  1002→                Declaration::Forget(f) => {
  1003→                    let query_str = match self.eval_expr(&f.query)? {
  1004→                        Value::String(s) => s,
  1005→                        other => format!("{}", other),
  1006→                    };
  1007→                    let now = SystemTime::now()
  1008→                        .duration_since(UNIX_EPOCH)
  1009→                        .map(|d| d.as_secs() as i64)
  1010→                        .unwrap_or(0);
  1011→                    let cutoff = now - (f.days * 86400);
  1012→                    self.memory.lock().unwrap().forget(&query_str, cutoff);

  1180→    /// Instantiate a struct from a type name and field initializers.
  1181→    fn instantiate_struct(&self, type_name: &str, inits: &[FieldInit]) -> Result<Value, String> {
  1182→        let struct_type = self.struct_types.get(type_name)
  1183→            .ok_or_else(|| format!("unknown struct type: {}", type_name))?
  1184→            .clone();
  1185→
  1186→        let mut fields = HashMap::new();
  1187→        for fd in &struct_type.fields {
  1188→            // Look for an initializer
  1189→            let init_val = inits.iter()
  1190→                .find(|fi| fi.name == fd.name)
  1191→                .map(|fi| self.eval_expr(&fi.value));
  1192→
  1193→            let value = match init_val {
  1194→                Some(Ok(v)) => v,
  1195→                Some(Err(e)) => return Err(e),
  1196→                None => {
  1197→                    // Use default value
  1198→                    match &fd.default {
  1199→                        Some(def_expr) => self.eval_expr(def_expr)?,
  1200→                        None => Value::Unit,
  1201→                    }
  1202→                }
  1203→            };
  1204→            fields.insert(fd.name.clone(), value);
  1205→        }
  1206→
  1207→        Ok(Value::Struct { type_name: type_name.to_string(), fields })
  1208→    }

  1210→    /// Handle an import declaration: load module, register namespace or merge globally.
  1211→    fn handle_import(&mut self, import: &ImportDecl) -> Result<(), String> {
  1212→        let module_path = &import.path;
  1213→
  1214→        // Register namespace mapping (alias or path itself for global merge)
  1215→        let alias = import.alias.as_ref().cloned().unwrap_or_else(|| module_path.clone());
  1216→        self.module_namespaces.insert(alias.clone(), module_path.clone());
  1217→
  1218→        // Load the module file and execute its declarations into this interpreter
  1219→        self.load_module(module_path)?;
  1220→
  1221→        Ok(())
  1222→    }

  1224→    /// Load a module by path, parsing and executing its declarations.
  1225→    /// Detects circular imports via loading_stack.
  1226→    fn load_module(&mut self, module_path: &str) -> Result<(), String> {
  1227→        // Circular import detection
  1228→        if self.loading_stack.contains(&module_path.to_string()) {
  1229→            return Err(format!(
  1230→                "circular import detected: {} -> {}",
  1231→                self.loading_stack.join(" -> "),
  1232→                module_path
  1233→            ));
  1234→        }
  1235→
  1236→        // Don't reload if already loaded (check if patterns from this module exist)
  1237→        // We track this via the loading_stack during the current load pass
  1238→        self.loading_stack.push(module_path.to_string());
  1239→
  1240→        let result = self.load_module_inner(module_path);
  1241→
  1242→        self.loading_stack.pop();
  1243→        result
  1244→    }

  1462→    /// Execute all rules in priority order (highest first, then declaration order).
  1463→    fn execute_rules(&mut self) -> Result<(), String> {
  1464→        // Sort by priority descending; stable sort preserves declaration order for ties
  1465→        let mut sorted_rules: Vec<&RuleDecl> = self.rules.iter().collect();
  1466→        sorted_rules.sort_by(|a, b| b.priority.cmp(&a.priority));
  1467→
  1468→        for rule in sorted_rules {
  1469→            let condition_met = self.eval_condition(&rule.condition, &self.variables)?;
  1470→            if condition_met {
  1471→                // Execute assignment: target.field = value
  1472→                let _target_val = self.eval_expr(&rule.target)?;
  1473→                let value_val = self.eval_expr(&rule.value)?;
  1474→
  1475→                // We need to mutate the entity in variables
  1476→                // The target is an Ident (entity name), field is the field name
  1477→                if let Expr::Ident(name) = &rule.target {
  1478→                    let entity = self.variables.get_mut(name)
  1479→                        .ok_or_else(|| format!("rule target '{}' not found", name))?;
  1480→                    entity.set_field(&rule.field, value_val)?;
  1481→                }
  1482→            }
  1483→        }
  1484→        Ok(())
  1485→    }

  1487→    /// Evaluate a rule condition.
  1488→    fn eval_condition(&self, cond: &Condition, env: &HashMap<String, Value>) -> Result<bool, String> {
  1489→        match cond {
  1490→            Condition::Contains { left, right } => {
  1491→                let lv = self.eval_expr_with_env(left, env)?;
  1492→                let rv = self.eval_expr_with_env(right, env)?;
  1493→                let ls = match &lv {
  1494→                    Value::String(s) => s.clone(),
  1495→                    other => return Err(format!("contains: left must be String, got {}", other.type_name())),
  1496→                };
  1497→                let rs = match &rv {
  1498→                    Value::String(s) => s.clone(),
  1499→                    other => return Err(format!("contains: right must be String, got {}", other.type_name())),
  1500→                };
  1501→                Ok(ls.contains(&rs))
  1502→            }
  1503→            Condition::Compare { left, op, right } => {
  1504→                let lv = self.eval_expr_with_env(left, env)?;
  1505→                let rv = self.eval_expr_with_env(right, env)?;
  1506→                let lf = lv.as_float()?;
  1507→                let rf = rv.as_float()?;
  1508→                Ok(match op {
  1509→                    CompareOp::Gt => lf > rf,
  1510→                    CompareOp::Lt => lf < rf,
  1511→                    CompareOp::Ge => lf >= rf,
  1512→                    CompareOp::Le => lf <= rf,
  1513→                    CompareOp::Eq => lf == rf,
  1514→                    CompareOp::Ne => lf != rf,
  1515→                })
  1516→            }
  1517→        }
  1518→    }

  1644→    /// Execute a flow: evaluate source, thread through pipeline steps.
  1645→    /// ADR-0056: After each step, check if a checkpoint follows. If so, save state.
  1646→    /// If resume_target is set, skip steps until we reach the checkpoint, then resume.
  1647→    fn run_flow(&mut self, flow: &FlowDecl) -> Result<String, String> {
  1648→        // Register branch definitions for this flow
  1649→        self.branch_defs.clear();
  1650→        for (step_name, branches) in &flow.branch_defs {
  1651→            self.branch_defs.insert(step_name.clone(), branches.clone());
  1652→        }
  1653→
  1654→        // ADR-0056: Determine resume start position
  1655→        let mut start_idx: usize = 0;
  1656→        let mut current: Option<Value> = None;
  1657→
  1658→        if let Some((ref target_flow, ref target_cp)) = self.resume_target {
  1659→            if target_flow == &flow.name {
  1660→                // Try to load checkpoint
  1661→                if let Some(data) = self.load_checkpoint(&flow.name, target_cp)? {
  1662→                    // Restore variables from checkpoint
  1663→                    for (k, v) in data.variables {
  1664→                        self.variables.insert(k, v);
  1665→                    }
  1666→                    // Start from the step AFTER the checkpoint
  1667→                    start_idx = data.step_index + 1;
  1668→                    current = Some(data.current_value);
  1669→                } else {
  1670→                    return Err(format!("checkpoint '{}' not found for flow '{}'", target_cp, flow.name));
  1671→                }
  1672→                // Clear resume target (one-shot)
  1673→                self.resume_target = None;
  1674→            }
  1675→        }
  1676→
  1677→        // If no resume, evaluate the source expression
  1678→        let mut current = current.unwrap_or_else(|| self.eval_expr(&flow.source).unwrap_or(Value::Unit));
  1679→
  1680→        // ADR-0056: Build reverse map: step_index -> checkpoint names at that position
  1681→        let mut checkpoint_at: HashMap<usize, Vec<String>> = HashMap::new();
  1682→        for (cp_name, &step_idx) in &flow.checkpoints {
  1683→            checkpoint_at.entry(step_idx).or_default().push(cp_name.clone());
  1684→        }
  1685→
  1686→        // Execute pipeline steps, starting from start_idx (0 for fresh run)
  1687→        for (i, step_name) in flow.pipeline.iter().enumerate() {
  1688→            if i < start_idx {
  1689→                continue; // Skip steps before resume point
  1690→            }
  1691→            current = self.run_flow_step(step_name, current)?;
  1692→
  1693→            // Check if a checkpoint follows this step
  1694→            if let Some(cp_names) = checkpoint_at.get(&i) {
  1695→                for cp_name in cp_names {
  1696→                    self.save_checkpoint(&flow.name, cp_name, i, &current)?;
  1697→                }
  1698→            }
  1699→        }
  1700→
  1701→        Ok(format!("{}", current))
  1702→    }

  1720→    /// Evaluate a branch condition: `target.field op threshold`
  1721→    fn eval_branch_condition(&self, cond: &BranchCondition, current: &Value) -> Result<bool, String> {
  1722→        // The target in branch_condition is the flow input value (current)
  1723→        let field_val = current.get_field(&cond.field)
  1724→            .map_err(|e| format!("branch condition: {}", e))?
  1725→            .clone();

  1727→        let fv = field_val.as_float()?;
  1728→        let tv = threshold.as_float()?;

  1730→            CompareOp::Gt => fv > tv,
  1731→            CompareOp::Lt => fv < tv,
  1732→            CompareOp::Ge => fv >= tv,
  1733→            CompareOp::Le => fv <= tv,
  1734→            CompareOp::Eq => fv == tv,
  1735→            CompareOp::Ne => fv != tv,

  1868→    /// ADR-0046: Build the effective system prompt for a learnable pattern.
  1869→    /// Handles all ContextMode variants:
  1870→    /// - None → return base prompt unchanged
  1871→    /// - Auto → recall(first_param_value, limit=5) + prepend context
  1872→    /// - Recall(query_expr, limit) → evaluate query, recall, prepend context
  1873→    /// - Literal(string) → prepend static text as context
  1874→    ///
  1875→    /// ADR-0055: Context compression.
  1876→    /// When context_strategy is Compress and the recalled context exceeds
  1877→    /// max_context_tokens estimated tokens, the facts are compressed via LLM
  1878→    /// summarization before being prepended to the prompt.
  1879→    fn build_effective_prompt(&self, learnable: &CompiledLearnable, args: &[Value]) -> String {
  1880→        let context_mode = match &learnable.context {
  1881→            None => return learnable.prompt.clone(),
  1882→            Some(ContextMode::None) => return learnable.prompt.clone(),
  1883→            Some(mode) => mode.clone(),
  1884→        };
  1885→
  1886→        match context_mode {
  1887→            ContextMode::None | ContextMode::Auto | ContextMode::Recall(_, _) => {
  1888→                // Determine the query string and limit for recall
  1889→                let (query, limit) = match &learnable.context {
  1890→                    Some(ContextMode::Auto) => {
  1891→                        // Use first parameter's runtime value as query
  1892→                        let query_str = if !args.is_empty() {
  1893→                            match &args[0] {
  1894→                                Value::String(s) => s.clone(),
  1895→                                other => format!("{}", other),
  1896→                            }
  1897→                        } else {
  1898→                            return learnable.prompt.clone();
  1899→                        };
  1900→                        (query_str, 5)
  1901→                    }
  1902→                    Some(ContextMode::Recall(query_expr, limit_opt)) => {
  1903→                        // Evaluate the context query expression with param names bound to args
  1904→                        let mut env: HashMap<String, Value> = HashMap::new();
  1905→                        for (i, param) in learnable.params.iter().enumerate() {
  1906→                            if i < args.len() {
  1907→                                env.insert(param.name.clone(), args[i].clone());
  1908→                            }
  1909→                        }
  1910→                        let query = match self.eval_expr_with_env(query_expr, &mut env) {
  1911→                            Ok(Value::String(s)) => s,
  1912→                            Ok(other) => format!("{}", other),
  1913→                            Err(_) => return learnable.prompt.clone(), // context eval failed → skip
  1914→                        };
  1915→                        (query, limit_opt.unwrap_or(5))
  1916→                    }
  1917→                    _ => unreachable!(),
  1918→                };
  1919→
  1920→                // Recall memories: collect up to `limit` unique results
  1921→                let mut facts = Vec::new();
  1922→                let query_embedding = self.embedding_manager.embed(&query).unwrap_or_default();
  1923→                let mut seen = std::collections::HashSet::new();
  1924→                let min_conf = 0.1_f32;
  1925→
  1926→                match self.memory.lock() {
  1927→                    Ok(store) => {
  1928→                        let all = store.all_entries();
  1929→                        let mut scored: Vec<(String, f32)> = Vec::new();
  1930→                        for entry in all {
  1931→                            if seen.contains(&entry.value) {
  1932→                                continue;
  1933→                            }
  1934→                            let score = if entry.embedding.is_empty() || query_embedding.is_empty() {
  1935→                                if entry.value.to_lowercase().contains(&query.to_lowercase()) {
  1936→                                    0.5
  1937→                                } else {
  1938→                                    0.0
  1939→                                }
  1940→                            } else {
  1941→                                cosine_similarity(&query_embedding, &entry.embedding) as f32
  1942→                                    * entry.confidence as f32
  1943→                                    * entry.priority as f32
  1944→                            };
  1945→                            if score >= min_conf {
  1946→                                seen.insert(entry.value.clone());
  1947→                                scored.push((entry.value, score));
  1948→                            }
  1949→                        }
  1950→                        scored.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal));
  1951→                        for (fact, _score) in scored.iter().take(limit) {
  1952→                            facts.push(fact.clone());
  1953→                        }
  1954→                    }
  1955→                    Err(_) => return learnable.prompt.clone(),
  1956→                }
  1957→
  1958→                if facts.is_empty() {
  1959→                    return learnable.prompt.clone();
  1960→                }
  1961→
  1962→                // Format context block
  1963→                let context_block = self.format_context_block(&facts);
  1964→
  1965→                // ADR-0055: Apply compression strategy
  1966→                if learnable.context_strategy == ContextStrategy::Compress {
  1967→                    let estimated_tokens = Self::estimate_tokens(&context_block);
  1968→                    if estimated_tokens > learnable.max_context_tokens {
  1969→                        // Compress: call LLM to summarize the context
  1970→                        let compressed = self.compress_context(&context_block);
  1971→                        return format!("{}\n{}", compressed, learnable.prompt);
  1972→                    }
  1973→                }
  1974→
  1975→                // No compression needed — inject as-is
  1976→                format!("{}\n{}", context_block, learnable.prompt)
  1977→            }
  1978→            ContextMode::Literal(literal_text) => {
  1979→                // Prepend static literal text as context
  1980→                format!("{}\n{}", literal_text, learnable.prompt)
  1981→            }
  1982→        }
  1983→    }

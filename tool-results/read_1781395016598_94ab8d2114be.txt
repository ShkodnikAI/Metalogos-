   167→fn builtin_len(args: &[Value]) -> Result<Value, String> {
   168→    match args.get(0) {
   169→        // Unicode-aware: chars().count() returns character count, not byte count.
   170→        // "Привет" (6 chars, 12 bytes) → 6.0, not 12.0.
   171→        Some(Value::String(s)) => Ok(Value::Float(s.chars().count() as f64)),
   172→        Some(Value::List(items)) => Ok(Value::Float(items.len() as f64)),
   173→        _ => Err("len() requires String or List argument".to_string()),
   174→    }
   175→}

   180→    }
   181→    Ok(Value::String(format!("{}", args[0])))

   190→fn builtin_contains(args: &[Value]) -> Result<Value, String> {
   191→    let haystack = expect_string_arg("contains", args, 0)?;
   192→    let needle = expect_string_arg("contains", args, 1)?;
   193→    let result = if haystack.contains(&needle) { 1.0 } else { 0.0 };
   194→    Ok(Value::Float(result))

   197→fn builtin_float(args: &[Value]) -> Result<Value, String> {
   198→    match args.get(0) {
   199→        Some(Value::Float(f)) => Ok(Value::Float(*f)),
   200→        Some(Value::String(s)) => s.parse::<f64>()
   201→            .map(Value::Float)
   202→            .map_err(|_| format!("float() cannot parse '{}'", s)),
   203→        _ => Err("float() requires 1 argument".to_string()),
   204→    }
   205→}

   208→    if args.is_empty() {
   209→        return Err("to_string() requires 1 argument".to_string());
   210→    }
   211→    // Use Value's Display impl — Float omits .0 for integers automatically
   212→    Ok(Value::String(format!("{}", args[0])))

   215→fn builtin_get(args: &[Value]) -> Result<Value, String> {
   216→    let list = match args.get(0) {
   217→        Some(Value::List(items)) => items,
   218→        _ => return Err("get() requires List as first argument".to_string()),
   219→    };
   220→    let index = match args.get(1) {
   221→        Some(Value::Float(f)) => *f as usize,
   222→        _ => return Err("get() requires Float index as second argument".to_string()),
   223→    };
   224→    list.get(index).cloned().ok_or_else(|| format!(
   225→        "get() index {} out of bounds (list has {} elements)",
   226→        index, list.len()
   227→    ))
   228→}

   230→fn builtin_push(args: &[Value]) -> Result<Value, String> {
   231→    let list = match args.get(0) {
   232→        Some(Value::List(items)) => items.clone(),
   233→        _ => return Err("push() requires List as first argument".to_string()),
   234→    };
   235→    let item = match args.get(1) {
   236→        Some(v) => v.clone(),
   237→        None => return Err("push() requires a second argument (item to push)".to_string()),
   238→    };
   239→    let mut new_list = list;
   240→    new_list.push(item);
   241→    Ok(Value::List(new_list))
   242→}

   244→fn builtin_index_of(args: &[Value]) -> Result<Value, String> {
   245→    let haystack = expect_string_arg("index_of", args, 0)?;
   246→    let needle = expect_string_arg("index_of", args, 1)?;
   247→    // Unicode-aware: return CHARACTER position, not byte offset.
   248→    // "Привет, мир".find("мир") byte offset = 12, char offset = 8.
   249→    // Must be consistent with substring()/char_at() which use char indices.
   250→    let char_pos = haystack
   251→        .char_indices()
   252→        .find(|(byte_idx, _)| haystack[*byte_idx..].starts_with(&needle))
   253→        .map(|(byte_idx, _)| haystack[..byte_idx].chars().count());
   254→    match char_pos {
   255→        Some(pos) => Ok(Value::Float(pos as f64)),
   256→        None => Ok(Value::Float(-1.0)),
   257→    }
   258→}

   260→fn builtin_substring(args: &[Value]) -> Result<Value, String> {
   261→    let s = expect_string_arg("substring", args, 0)?;
   262→    let start = expect_float_arg("substring", args, 1)? as usize;
   263→    let end = expect_float_arg("substring", args, 2)? as usize;
   264→    // Soft-failure: clamp to valid range, empty string if start >= len
   265→    let s_len = s.chars().count();
   266→    if start >= s_len {
   267→        return Ok(Value::String(String::new()));
   268→    }
   269→    let end = if end > s_len { s_len } else { end };
   270→    if start >= end {
   271→        return Ok(Value::String(String::new()));
   272→    }
   273→    // Convert byte indices for char-based slicing
   274→    let chars: Vec<char> = s.chars().collect();
   275→    let result: String = chars[start..end].iter().collect();
   276→    Ok(Value::String(result))
   277→}

   279→fn builtin_char_at(args: &[Value]) -> Result<Value, String> {
   280→    let s = expect_string_arg("char_at", args, 0)?;
   281→    let index = expect_float_arg("char_at", args, 1)? as usize;
   282→    // Soft-failure: return empty string on out-of-bounds
   283→    let chars: Vec<char> = s.chars().collect();
   284→    match chars.get(index) {
   285→        Some(ch) => Ok(Value::String(ch.to_string())),
   286→        None => Ok(Value::String(String::new())),
   287→    }
   288→}

   302→fn builtin_to_float(args: &[Value]) -> Result<Value, String> {
   303→    match args.get(0) {
   304→        Some(Value::Float(f)) => Ok(Value::Float(*f)),
   305→        Some(Value::String(s)) => Ok(s.parse::<f64>()
   306→            .map(Value::Float)
   307→            .unwrap_or(Value::Float(0.0))), // soft-failure: return 0.0 on parse error
   308→        Some(Value::Bool(b)) => Ok(Value::Float(if *b { 1.0 } else { 0.0 })),
   309→        _ => Ok(Value::Float(0.0)), // soft-failure
   310→    }
   311→}

   313→fn builtin_confidence(args: &[Value]) -> Result<Value, String> {
   314→    match args.get(0) {
   315→        Some(Value::Fluid(variants)) => {
   316→            let best = variants.iter().map(|v| v.confidence)
   317→                .fold(0.0_f64, f64::max);
   318→            Ok(Value::Float(best))
   319→        }
   320→        Some(_) => Ok(Value::Float(1.0)), // concrete values are fully confident
   321→        None => Err("confidence() requires 1 argument".to_string()),
   322→    }
   323→}

   327→fn expect_float_arg(fn_name: &str, args: &[Value], index: usize) -> Result<f64, String> {
   328→    if args.len() <= index {
   329→        return Err(format!("{}() requires an argument at position {}", fn_name, index));
   330→    }
   331→    match &args[index] {
   332→        Value::Float(f) => Ok(*f),
   333→        other => Err(format!(
   334→            "{}() expected Float argument, got {}",
   335→            fn_name, other.type_name()
   336→        )),
   337→    }
   338→}

   340→fn expect_string_arg(fn_name: &str, args: &[Value], index: usize) -> Result<String, String> {
   341→    if args.len() <= index {
   342→        return Err(format!("{}() requires an argument at position {}", fn_name, index));
   343→    }
   344→    match &args[index] {
   345→        Value::String(s) => Ok(s.clone()),
   346→        other => Err(format!(
   347→            "{}() expected String argument, got {}",
   348→            fn_name, other.type_name()
   349→        )),
   350→    }
   351→}

   368→fn builtin_split(args: &[Value]) -> Result<Value, String> {
   369→    let s = expect_string_arg("__split", args, 0)?;
   370→    let sep = expect_string_arg("__split", args, 1)?;
   371→    let items: Vec<Value> = if sep.is_empty() {
   372→        s.chars().map(|c| Value::String(c.to_string())).collect()
   373→    } else {
   374→        s.split(&sep).map(|part| Value::String(part.to_string())).collect()
   375→    };
   376→    Ok(Value::List(items))
   377→}

   379→fn builtin_join(args: &[Value]) -> Result<Value, String> {
   380→    let list = match args.get(0) {
   381→        Some(Value::List(items)) => items,
   382→        _ => return Err("__join() requires List as first argument".to_string()),
   383→    };
   384→    let sep = if args.len() > 1 {
   385→        match &args[1] {
   386→            Value::String(s) => s.clone(),
   387→            _ => ",".to_string(),
   388→        }
   389→    } else {
   390→        ",".to_string()
   391→    };
   392→    let parts: Vec<String> = list.iter().map(|v| format!("{}", v)).collect();
   393→    Ok(Value::String(parts.join(&sep)))
   394→}

   451→fn builtin_respond(args: &[Value]) -> Result<Value, String> {
   452→    let status_body = expect_string_arg("respond", args, 0)?;

   505→fn builtin_render(args: &[Value]) -> Result<Value, String> {
   506→    // render(template_name, key1, val1, key2, val2, ...)
   507→    // Simple {{ var }} substitution with auto-escaping
   508→    // In interpreter mode, do basic string substitution
   509→    if args.len() < 3 || (args.len() - 1) % 2 != 0 {
   510→        return Err("render() requires template name + key/value pairs (odd count)".to_string());
   511→    }
   512→    let template_name = expect_string_arg("render", args, 0)?;
   513→
   514→    // Build substitution map from remaining args (key, value pairs)
   515→    let mut vars = std::collections::HashMap::new();
   516→    let mut i = 1;
   517→    while i + 1 < args.len() {
   518→        let key = match &args[i] {
   519→            Value::String(s) => s.clone(),
   520→            other => return Err(format!("render() key must be String, got {}", other.type_name())),
   521→        };
   522→        let val = match &args[i + 1] {
   523→            Value::String(s) => s.clone(),
   524→            other => format!("{}", other),
   525→        };
   526→        vars.insert(key, val);
   527→        i += 2;
   528→    }
   529→
   530→    // In interpreter mode, generate a simple HTML string from the template name and vars
   531→    let mut html = String::from("<div class=\"template-");
   532→    html.push_str(&escape_html_chars(&template_name));
   533→    html.push_str("\">");
   534→    for (key, val) in &vars {
   535→        html.push_str(&format!("<span data-key=\"{}\">{}</span>",
   536→            escape_html_chars(key), escape_html_chars(val)));
   537→    }
   538→    html.push_str("</div>");
   539→
   540→    Ok(Value::Html(html))
   541→}

   560→    }

   636→fn builtin_encrypt(args: &[Value]) -> Result<Value, String> {
   637→    let data = expect_string_arg("encrypt", args, 0)?;
   638→    let key_str = match args.get(1) {
   639→        Some(Value::Secret(zs)) => zs.as_str(),
   640→        Some(other) => return Err(format!("encrypt() expected Secret as second arg, got {}", other.type_name())),
   641→        None => return Err("encrypt() requires 2 arguments".to_string()),
   642→    };
   643→    // Real AES-256-GCM with random 96-bit nonce (Phase 7.3)
   644→    use aes_gcm::{Aes256Gcm, AeadCore, Key};
   645→    use aes_gcm::aead::{Aead, KeyInit, OsRng};
   646→
   647→    let key_bytes = hex::decode(key_str)
   648→        .map_err(|e| format!("encrypt() invalid key format (expected hex): {}", e))?;
   649→    if key_bytes.len() != 32 {
   650→        return Err(format!("encrypt() key must be 256-bit (64 hex chars), got {} bytes", key_bytes.len()));
   651→    }
   652→    let key = Key::<Aes256Gcm>::from_slice(&key_bytes);
   653→    let cipher = Aes256Gcm::new(key);
   654→    let nonce = Aes256Gcm::generate_nonce(&mut OsRng); // 96-bit random nonce
   655→
   656→    match cipher.encrypt(&nonce, data.as_ref()) {
   657→        Ok(ciphertext) => {
   658→            // Prepend nonce to ciphertext for self-contained Encrypted value
   659→            let mut output = nonce.to_vec();
   660→            output.extend_from_slice(&ciphertext);
   661→            Ok(Value::Encrypted(output))
   662→        }
   663→        Err(e) => Err(format!("encrypt() AES-256-GCM encryption failed: {}", e)),
   664→    }
   665→}

   667→fn builtin_decrypt(args: &[Value]) -> Result<Value, String> {
   668→    let encrypted = match args.get(0) {
   669→        Some(Value::Encrypted(data)) => data.clone(),
   670→        Some(other) => return Err(format!("decrypt() expected Encrypted as first arg, got {}", other.type_name())),
   671→        None => return Err("decrypt() requires 2 arguments".to_string()),
   672→    };
   673→    let key_str = match args.get(1) {
   674→        Some(Value::Secret(zs)) => zs.as_str(),
   675→        Some(other) => return Err(format!("decrypt() expected Secret as second arg, got {}", other.type_name())),
   676→        None => return Err("decrypt() requires 2 arguments".to_string()),
   677→    };
   678→    // Real AES-256-GCM decryption (Phase 7.3)
   679→    // Encrypted format: nonce (12 bytes) || ciphertext_with_tag
   680→    use aes_gcm::{Aes256Gcm, Key, Nonce};
   681→    use aes_gcm::aead::{Aead, KeyInit};
   682→
   683→    if encrypted.len() < 13 {
   684→        // Need at least 12 (nonce) + 1 (tag minimum)
   685→        return Err("decrypt() invalid encrypted data: too short".to_string());
   686→    }
   687→
   688→    let key_bytes = hex::decode(key_str)
   689→        .map_err(|e| format!("decrypt() invalid key format (expected hex): {}", e))?;
   690→    if key_bytes.len() != 32 {
   691→        return Err(format!("decrypt() key must be 256-bit (64 hex chars), got {} bytes", key_bytes.len()));
   692→    }
   693→
   694→    let (nonce_bytes, ciphertext) = encrypted.split_at(12);
   695→    let nonce = Nonce::from_slice(nonce_bytes);
   696→    let key = Key::<Aes256Gcm>::from_slice(&key_bytes);
   697→    let cipher = Aes256Gcm::new(key);
   698→
   699→    match cipher.decrypt(nonce, ciphertext) {
   700→        Ok(plaintext) => {
   701→            match String::from_utf8(plaintext) {
   702→                Ok(s) => Ok(Value::String(s)),
   703→                Err(_) => Err("decrypt() decrypted data is not valid UTF-8".to_string()),
   704→            }
   705→        }
   706→        Err(_) => Err("decrypt() failed: incorrect key or corrupted data".to_string()),
   707→    }
   708→}

   770→/// Send an HTTP POST request. Returns the response body as String.
   771→/// Usage: http_post(url, body, content_type)
   772→/// Usage: http_post(url, body, content_type, auth_token)        — sets Authorization: Bearer <auth_token>
   773→/// Usage: http_post(url, body, content_type, headers_struct)    — sets headers from Struct fields
   774→/// Наряд №12 Bug 2: Added 4th parameter for authorization headers.
   775→fn builtin_http_post(args: &[Value]) -> Result<Value, String> {
   776→    let url = match args.get(0) {
   777→        Some(Value::String(s)) => s.clone(),
   778→        Some(other) => return Err(format!("http_post() expected String as url, got {}", other.type_name())),
   779→        None => return Err("http_post() requires at least 1 argument (url)".to_string()),
   780→    };
   781→
   782→    let body = match args.get(1) {
   783→        Some(Value::String(s)) => s.clone(),
   784→        Some(other) => return Err(format!("http_post() expected String as body, got {}", other.type_name())),
   785→        None => return Err("http_post() requires at least 2 arguments (url, body)".to_string()),
   786→    };
   787→
   788→    let content_type = match args.get(2) {
   789→        Some(Value::String(s)) => s.clone(),
   790→        _ => "application/json".to_string(),
   791→    };
   792→
   793→
   794→    let client = reqwest::blocking::Client::builder()
   795→        .timeout(std::time::Duration::from_secs(30))
   796→        .build()
   797→        .map_err(|e| format!("http_post(): failed to create client: {}", e))?;
   798→
   799→
   800→    let mut req = client
   801→        .post(&url)
   802→        .header("Content-Type", &content_type)
   803→        .body(body);
   804→
   805→    // Optional 4th argument: headers (Наряд №12 Bug 2)
   806→    // String: treat as Bearer token; Struct: set headers from fields
   807→    if let Some(headers_arg) = args.get(3) {
   808→        match headers_arg {
   809→            Value::String(auth_token) => {
   810→                if !auth_token.is_empty() {
   811→                    req = req.header("Authorization", format!("Bearer {}", auth_token));
   812→                }
   813→            }
   814→            Value::Struct { fields, .. } => {
   815→                for (key, val) in fields {
   816→                    if let Value::String(v) = val {
   817→                        req = req.header(key.as_str(), v.as_str());
   818→                    }
   819→                }
   820→            }
   821→            _ => {}
   822→        }
   823→    }
   824→
   825→    let resp = req
   826→        .send()
   827→        .map_err(|e| format!("http_post() request failed: {}", e))?;
   828→
   829→    let status = resp.status().as_u16();
   830→    let resp_body = resp.text().unwrap_or_default();
   831→
   832→    if status >= 400 {
   833→        return Err(format!("http_post() returned status {}: {}", status, resp_body));
   834→    }
   835→
   836→    Ok(Value::String(resp_body))
   837→}

   917→/// Convert serde_json::Value to METALOGOS Value (same logic as interpreter's method).
   918→fn json_value_to_mlog_value(json: &serde_json::Value) -> Value {
   919→    match json {
   920→        serde_json::Value::String(s) => Value::String(s.clone()),
   921→        serde_json::Value::Number(n) => Value::Float(n.as_f64().unwrap_or(0.0)),
   922→        serde_json::Value::Bool(b) => Value::Bool(*b),
   923→        serde_json::Value::Null => Value::Unit,
   924→        serde_json::Value::Array(arr) => {
   925→            Value::List(arr.iter().map(|v| json_value_to_mlog_value(v)).collect())
   926→        }
   927→        serde_json::Value::Object(obj) => {
   928→            let mut fields = std::collections::HashMap::new();

   934→    }

   937→/// Convert METALOGOS Value to serde_json::Value (reverse of json_value_to_mlog_value).
   938→fn mlog_value_to_json(val: &Value) -> serde_json::Value {
   939→    match val {
   940→        Value::String(s) => serde_json::Value::String(s.clone()),
   941→        Value::Float(f) => serde_json::json!(*f),
   942→        Value::Bool(b) => serde_json::Value::Bool(*b),
   943→        Value::Unit => serde_json::Value::Null,
   944→        Value::List(items) => {
   945→            serde_json::Value::Array(items.iter().map(|v| mlog_value_to_json(v)).collect())
   946→        }
   947→        Value::Struct { fields, .. } => {
   948→            let mut map = serde_json::Map::new();
   949→            for (k, v) in fields {
   950→                map.insert(k.clone(), mlog_value_to_json(v));
   951→            }
   952→            serde_json::Value::Object(map)
   953→        }
   954→        // Opaque/internal types: convert to string representation
   955→        other => serde_json::Value::String(format!("{}", other)),
   956→    }
   957→}

   972→/// Safe field access on a struct value: returns default if field missing or not a struct.
   973→/// Usage: json_get(obj, "field") -> Value (returns Unit if missing)
   974→/// Usage: json_get(obj, "field", default_value) -> Value (returns default if missing)
   975→/// Usage: json_get(obj, "nested.field.path", default) -> Value (dot-separated path)
   976→/// This is the P0 fix: prevents runtime crash when accessing optional JSON fields
   977→/// like message.voice on non-voice Telegram updates.
   978→fn builtin_json_get(args: &[Value]) -> Result<Value, String> {
   979→    let obj = match args.get(0) {
   980→        Some(v) => v,
   981→        None => return Err("json_get() requires at least 2 arguments (obj, field_path)".to_string()),
   982→    };
   983→    let path = expect_string_arg("json_get", args, 1)?;
   984→    let default_val = args.get(2).cloned().unwrap_or(Value::Unit);
   985→
   986→    // Navigate the path (dot-separated)
   987→    let mut current = obj;
   988→    for segment in path.split('.') {
   989→        match current.get_field(segment) {
   990→            Ok(val) => current = val,
   991→            Err(_) => return Ok(default_val),
   992→        }
   993→    }
   994→    Ok(current.clone())
   995→}

   997→/// Check if a struct value has a given field. Returns 1.0 (true) or 0.0 (false).
   998→/// Usage: has_field(obj, "field") -> Float
   999→/// Usage: has_field(obj, "nested.field") -> Float (dot-separated path)
  1000→fn builtin_has_field(args: &[Value]) -> Result<Value, String> {
  1001→    let obj = match args.get(0) {
  1002→        Some(v) => v,
  1003→        None => return Err("has_field() requires 2 arguments (obj, field_path)".to_string()),
  1004→    };
  1005→    let path = expect_string_arg("has_field", args, 1)?;
  1006→
  1007→    let mut current = obj;
  1008→    let segments: Vec<&str> = path.split('.').collect();
  1009→    for (i, segment) in segments.iter().enumerate() {
  1010→        match current.get_field(segment) {
  1011→            Ok(val) => {
  1012→                if i == segments.len() - 1 {
  1013→                    return Ok(Value::Float(1.0));
  1014→                }
  1015→                current = val;
  1016→            }
  1017→            Err(_) => return Ok(Value::Float(0.0)),
  1018→        }
  1019→    }
  1020→    Ok(Value::Float(0.0))
  1021→}

  1023→/// Send an HTTP GET request. Returns the response body as String.
  1024→/// Usage: http_get(url) -> String
  1025→/// Usage: http_get(url, headers_struct) -> String  — sets headers from Struct fields
  1026→fn builtin_http_get(args: &[Value]) -> Result<Value, String> {
  1027→    let url = match args.get(0) {
  1028→        Some(Value::String(s)) => s.clone(),
  1029→        Some(other) => return Err(format!("http_get() expected String as url, got {}", other.type_name())),
  1030→        None => return Err("http_get() requires 1 argument (url)".to_string()),
  1031→    };
  1032→
  1033→    let client = reqwest::blocking::Client::builder()
  1034→        .timeout(std::time::Duration::from_secs(30))
  1035→        .build()
  1036→        .map_err(|e| format!("http_get(): failed to create client: {}", e))?;
  1037→
  1038→    let mut req = client.get(&url);
  1039→
  1040→    // Optional 2nd argument: headers (String as Bearer token, Struct as header map)
  1041→    if let Some(headers_arg) = args.get(1) {
  1042→        match headers_arg {
  1043→            Value::String(auth_token) => {
  1044→                if !auth_token.is_empty() {
  1045→                    req = req.header("Authorization", format!("Bearer {}", auth_token));
  1046→                }
  1047→            }
  1048→            Value::Struct { fields, .. } => {
  1049→                for (key, val) in fields {
  1050→                    if let Value::String(v) = val {
  1051→                        req = req.header(key.as_str(), v.as_str());
  1052→                    }
  1053→                }
  1054→            }
  1055→            _ => {}
  1056→        }
  1057→    }
  1058→
  1059→    let resp = req
  1060→        .send()
  1061→        .map_err(|e| format!("http_get() request failed: {}", e))?;
  1062→
  1063→    let status = resp.status().as_u16();
  1064→    let resp_body = resp.text().unwrap_or_default();
  1065→
  1066→    if status >= 400 {
  1067→        return Err(format!("http_get() returned status {}: {}", status, resp_body));
  1068→    }
  1069→
  1070→    Ok(Value::String(resp_body))
  1071→}

  1207→/// Initialize SQLite persistence for the KV store.
  1208→/// Called by Interpreter::configure_memory() when persist path is set.
  1209→/// Creates kv_store table (key TEXT PRIMARY KEY, value TEXT) in the given database.
  1210→/// Loads existing rows into the in-memory HashMap.
  1211→pub fn init_kv_persist(db_path: &str) -> Result<(), String> {
  1212→    let conn = rusqlite::Connection::open(db_path)
  1213→        .map_err(|e| format!("[kv_store] Failed to open database '{}': {}", db_path, e))?;
  1214→    conn.execute_batch(
  1215→        "CREATE TABLE IF NOT EXISTS kv_store (key TEXT PRIMARY KEY, value TEXT NOT NULL);"
  1216→    ).map_err(|e| format!("[kv_store] Failed to create table: {}", e))?;
  1217→
  1218→    // Load existing KV pairs into in-memory HashMap (write-through cache warmup)
  1219→    {
  1220→        let mut stmt = conn.prepare("SELECT key, value FROM kv_store")
  1221→            .map_err(|e| format!("[kv_store] Failed to query: {}", e))?;
  1222→        let rows: Vec<(String, String)> = stmt.query_map([], |row| {
  1223→            Ok((row.get::<_, String>(0)?, row.get::<_, String>(1)?))
  1224→        }).map_err(|e| format!("[kv_store] Failed to iterate: {}", e))?
  1225→        .filter_map(|r| r.ok())
  1226→        .collect();
  1227→
  1228→        // Merge into in-memory store (SQLite is authoritative on init)
  1229→        if let Ok(mut store) = kv_store().lock() {
  1230→            for (key, value) in rows {
  1231→                store.insert(key, value);
  1232→            }
  1233→        }
  1234→    } // stmt is dropped here, releasing borrow on conn
  1235→
  1236→    // Store the connection globally
  1237→    let mut sqlite_guard = kv_sqlite().lock()
  1238→        .map_err(|e| format!("[kv_store] lock error: {}", e))?;
  1239→    *sqlite_guard = Some(conn);
  1240→    eprintln!("[kv_store] SQLite persistence enabled: {}", db_path);
  1241→    Ok(())
  1242→}

  1428→// Paths containing ".." or absolute paths are rejected (sandbox).

  1430→/// Validate that a path is safe (within working directory, no traversal).
  1431→fn sandbox_path(path: &str) -> Result<std::path::PathBuf, String> {
  1432→    let p = std::path::Path::new(path);
  1433→    // Reject absolute paths
  1434→    if p.is_absolute() {
  1435→        return Err(format!("file I/O sandbox: absolute paths not allowed: '{}'", path));
  1436→    }
  1437→    // Reject path traversal
  1438→    for component in p.components() {
  1439→        if let std::path::Component::ParentDir = component {
  1440→            return Err(format!("file I/O sandbox: path traversal ('..') not allowed: '{}'", path));
  1441→        }
  1442→    }
  1443→    Ok(std::path::PathBuf::from(path))
  1444→}

  1446→/// `read_file(path)` — read file contents as String.
  1447→/// Soft-failure: returns empty string on error (file not found, permission denied, etc.).
  1448→fn builtin_read_file(args: &[Value]) -> Result<Value, String> {
  1449→    let path = expect_string_arg("read_file", args, 0)?;
  1450→    let safe_path = match sandbox_path(&path) {
  1451→        Ok(p) => p,
  1452→        Err(_) => return Ok(Value::String(String::new())), // soft-failure on sandbox violation
  1453→    };
  1454→    match std::fs::read_to_string(&safe_path) {
  1455→        Ok(content) => Ok(Value::String(content)),
  1456→        Err(_) => Ok(Value::String(String::new())), // soft-failure
  1457→    }
  1458→}

  1463→    let path = expect_string_arg("write_file", args, 0)?;
  1464→    let content = match args.get(1) {
  1465→        Some(Value::String(s)) => s.clone(),

  1587→fn builtin_http_post_multipart(args: &[Value]) -> Result<Value, String> {
  1588→    let url = expect_string_arg("http_post_multipart", args, 0)?;
  1589→    let fields = match args.get(1) {
  1590→        Some(Value::Struct { fields, .. }) => fields.clone(),
  1591→        _ => return Err("http_post_multipart() requires Struct as 2nd argument (fields)".to_string()),
  1592→    };
  1593→    let files = match args.get(2) {
  1594→        Some(Value::Struct { fields, .. }) => fields.clone(),
  1595→        _ => return Err("http_post_multipart() requires Struct as 3rd argument (files)".to_string()),
  1596→    };
  1597→
  1598→    let client = reqwest::blocking::Client::builder()
  1599→        .timeout(std::time::Duration::from_secs(120))
  1600→        .build()
  1601→        .map_err(|e| format!("http_post_multipart(): client error: {}", e))?;
  1602→
  1603→    let mut form = reqwest::blocking::multipart::Form::new();
  1604→
  1605→    // Add text fields
  1606→    for (key, val) in &fields {
  1607→        if let Value::String(v) = val {
  1608→            form = form.text(key.clone(), v.clone());
  1609→        }
  1610→    }
  1611→
  1612→    // Add file fields
  1613→    for (key, val) in &files {
  1614→        if let Value::String(path) = val {
  1615→            let file_bytes = std::fs::read(path)
  1616→                .map_err(|e| format!("http_post_multipart(): cannot read file '{}': {}", path, e))?;
  1617→            let file_name = std::path::Path::new(path)
  1618→                .file_name()
  1619→                .and_then(|n| n.to_str())
  1620→                .unwrap_or("file");
  1621→            let part = reqwest::blocking::multipart::Part::bytes(file_bytes)
  1622→                .file_name(file_name.to_string());
  1623→            form = form.part(key.clone(), part);
  1624→        }
  1625→    }
  1626→
  1627→    let resp = client.post(&url).multipart(form).send()
  1628→        .map_err(|e| format!("http_post_multipart(): request failed: {}", e))?;
  1629→
  1630→    let status = resp.status().as_u16();
  1631→    let resp_body = resp.text().unwrap_or_default();
  1632→
  1633→    if status >= 400 {
  1634→        return Err(format!("http_post_multipart() returned status {}: {}", status, resp_body));
  1635→    }
  1636→
  1637→    Ok(Value::String(resp_body))
  1638→}

  1640→fn builtin_whisper_transcribe(args: &[Value]) -> Result<Value, String> {
  1641→    let file_id = expect_string_arg("whisper_transcribe", args, 0)?;
  1642→    let bot_token = expect_string_arg("whisper_transcribe", args, 1)?;
  1643→    let whisper_key = expect_string_arg("whisper_transcribe", args, 2)?;
  1644→    let provider = match args.get(3) {
  1645→        Some(Value::String(s)) => s.clone(),
  1646→        _ => "openai".to_string(),
  1647→    };
  1648→
  1649→    // Step 1: Get file path from Telegram
  1650→    let tg_client = reqwest::blocking::Client::builder()
  1651→        .timeout(std::time::Duration::from_secs(30))
  1652→        .build()
  1653→        .map_err(|e| format!("whisper_transcribe(): client error: {}", e))?;
  1654→
  1655→    let get_file_url = format!("https://api.telegram.org/bot{}/getFile?file_id={}", bot_token, file_id);
  1656→    let tg_resp = tg_client.get(&get_file_url).send()
  1657→        .map_err(|e| format!("whisper_transcribe(): Telegram getFile failed: {}", e))?;
  1658→    let tg_body: serde_json::Value = serde_json::from_str(&tg_resp.text().unwrap_or_default())
  1659→        .map_err(|e| format!("whisper_transcribe(): Telegram response parse error: {}", e))?;
  1660→
  1661→    let file_path = tg_body.get("result")
  1662→        .and_then(|r| r.get("file_path"))
  1663→        .and_then(|p| p.as_str())
  1664→        .unwrap_or("")
  1665→        .to_string();
  1666→
  1667→    if file_path.is_empty() {
  1668→        return Err("whisper_transcribe(): Telegram returned empty file_path".to_string());
  1669→    }
  1670→
  1671→    // Step 2: Download the file
  1672→    let download_url = format!("https://api.telegram.org/file/bot{}/{}", bot_token, file_path);
  1673→    let audio_bytes = tg_client.get(&download_url).send()
  1674→        .map_err(|e| format!("whisper_transcribe(): download failed: {}", e))?
  1675→        .bytes()
  1676→        .map_err(|e| format!("whisper_transcribe(): read bytes failed: {}", e))?;
  1677→
  1678→    // Step 3: Send to Whisper API
  1679→    let (api_url, auth_header, auth_value) = match provider.as_str() {
  1680→        "groq" => (
  1681→            "https://api.groq.com/openai/v1/audio/transcriptions".to_string(),
  1682→            "Authorization".to_string(),
  1683→            format!("Bearer {}", whisper_key),
  1684→        ),
  1685→        _ => ( // openai
  1686→            "https://api.openai.com/v1/audio/transcriptions".to_string(),
  1687→            "Authorization".to_string(),
  1688→            format!("Bearer {}", whisper_key),
  1689→        ),
  1690→    };
  1691→
  1692→    // Use multipart form
  1693→    let whisper_client = reqwest::blocking::Client::builder()
  1694→        .timeout(std::time::Duration::from_secs(120))
  1695→        .build()
  1696→        .map_err(|e| format!("whisper_transcribe(): whisper client error: {}", e))?;
  1697→
  1698→    let model = if provider == "groq" { "whisper-large-v3" } else { "whisper-1" };
  1699→    let mut form = reqwest::blocking::multipart::Form::new();
  1700→    form = form.text("model", model.to_string());
  1701→    let part = reqwest::blocking::multipart::Part::bytes(audio_bytes.to_vec())
  1702→        .file_name("audio.ogg");
  1703→    form = form.part("file", part);
  1704→

  1706→        .header(auth_header, auth_value)
  1707→        .multipart(form)
  1708→        .send()
  1709→        .map_err(|e| format!("whisper_transcribe(): whisper request failed: {}", e))?;
  1710→
  1711→    let status = whisper_resp.status().as_u16();
  1712→    let whisper_body = whisper_resp.text().unwrap_or_default();
  1713→
  1714→    if status >= 400 {
  1715→        return Err(format!("whisper_transcribe(): whisper API status {}: {}", status, whisper_body));
  1716→    }
  1717→
  1718→    // Parse response to extract text
  1719→    let parsed: serde_json::Value = serde_json::from_str(&whisper_body)
  1720→        .map_err(|e| format!("whisper_transcribe(): whisper response parse error: {}", e))?;
  1721→    let text = parsed.get("text").and_then(|t| t.as_str()).unwrap_or("").to_string();

  1723→    Ok(Value::String(text))
  1724→}

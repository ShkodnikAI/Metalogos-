   175→fn builtin_len(args: &[Value]) -> Result<Value, String> {
   176→    match args.get(0) {
   177→        // Unicode-aware: chars().count() returns character count, not byte count.
   178→        // "Привет" (6 chars, 12 bytes) → 6.0, not 12.0.
   179→        Some(Value::String(s)) => Ok(Value::Float(s.chars().count() as f64)),
   180→        Some(Value::List(items)) => Ok(Value::Float(items.len() as f64)),
   181→        _ => Err("len() requires String or List argument".to_string()),
   182→    }
   183→}

   187→        return Err("str() requires 1 argument".to_string());
   188→    }
   189→    Ok(Value::String(format!("{}", args[0])))
   190→}

   198→fn builtin_contains(args: &[Value]) -> Result<Value, String> {
   199→    let haystack = expect_string_arg("contains", args, 0)?;
   200→    let needle = expect_string_arg("contains", args, 1)?;
   201→    let result = if haystack.contains(&needle) { 1.0 } else { 0.0 };
   202→    Ok(Value::Float(result))
   203→}

   205→fn builtin_float(args: &[Value]) -> Result<Value, String> {
   206→    match args.get(0) {
   207→        Some(Value::Float(f)) => Ok(Value::Float(*f)),
   208→        Some(Value::String(s)) => s.parse::<f64>()
   209→            .map(Value::Float)
   210→            .map_err(|_| format!("float() cannot parse '{}'", s)),
   211→        _ => Err("float() requires 1 argument".to_string()),
   212→    }
   213→}

   215→fn builtin_to_string(args: &[Value]) -> Result<Value, String> {
   216→    if args.is_empty() {
   217→        return Err("to_string() requires 1 argument".to_string());
   218→    }
   219→    // Use Value's Display impl — Float omits .0 for integers automatically
   220→    Ok(Value::String(format!("{}", args[0])))
   221→}

   223→fn builtin_get(args: &[Value]) -> Result<Value, String> {
   224→    let list = match args.get(0) {
   225→        Some(Value::List(items)) => items,
   226→        _ => return Err("get() requires List as first argument".to_string()),
   227→    };
   228→    let index = match args.get(1) {
   229→        Some(Value::Float(f)) => *f as usize,
   230→        _ => return Err("get() requires Float index as second argument".to_string()),
   231→    };
   232→    list.get(index).cloned().ok_or_else(|| format!(
   233→        "get() index {} out of bounds (list has {} elements)",
   234→        index, list.len()
   235→    ))
   236→}

   238→fn builtin_push(args: &[Value]) -> Result<Value, String> {
   239→    let list = match args.get(0) {
   240→        Some(Value::List(items)) => items.clone(),
   241→        _ => return Err("push() requires List as first argument".to_string()),
   242→    };
   243→    let item = match args.get(1) {
   244→        Some(v) => v.clone(),
   245→        None => return Err("push() requires a second argument (item to push)".to_string()),
   246→    };
   247→    let mut new_list = list;
   248→    new_list.push(item);
   249→    Ok(Value::List(new_list))
   250→}

   252→fn builtin_index_of(args: &[Value]) -> Result<Value, String> {
   253→    let haystack = expect_string_arg("index_of", args, 0)?;
   254→    let needle = expect_string_arg("index_of", args, 1)?;
   255→    // Unicode-aware: return CHARACTER position, not byte offset.
   256→    // "Привет, мир".find("мир") byte offset = 12, char offset = 8.
   257→    // Must be consistent with substring()/char_at() which use char indices.
   258→    let char_pos = haystack
   259→        .char_indices()
   260→        .find(|(byte_idx, _)| haystack[*byte_idx..].starts_with(&needle))
   261→        .map(|(byte_idx, _)| haystack[..byte_idx].chars().count());
   262→    match char_pos {
   263→        Some(pos) => Ok(Value::Float(pos as f64)),
   264→        None => Ok(Value::Float(-1.0)),
   265→    }
   266→}

   268→fn builtin_substring(args: &[Value]) -> Result<Value, String> {
   269→    let s = expect_string_arg("substring", args, 0)?;
   270→    let start = expect_float_arg("substring", args, 1)? as usize;
   271→    let end = expect_float_arg("substring", args, 2)? as usize;
   272→    // Soft-failure: clamp to valid range, empty string if start >= len
   273→    let s_len = s.chars().count();
   274→    if start >= s_len {
   275→        return Ok(Value::String(String::new()));
   276→    }
   277→    let end = if end > s_len { s_len } else { end };
   278→    if start >= end {
   279→        return Ok(Value::String(String::new()));
   280→    }
   281→    // Convert byte indices for char-based slicing
   282→    let chars: Vec<char> = s.chars().collect();
   283→    let result: String = chars[start..end].iter().collect();
   284→    Ok(Value::String(result))
   285→}
   286→
   287→fn builtin_char_at(args: &[Value]) -> Result<Value, String> {
   288→    let s = expect_string_arg("char_at", args, 0)?;
   289→    let index = expect_float_arg("char_at", args, 1)? as usize;
   290→    // Soft-failure: return empty string on out-of-bounds
   291→    let chars: Vec<char> = s.chars().collect();
   292→    match chars.get(index) {
   293→        Some(ch) => Ok(Value::String(ch.to_string())),
   294→        None => Ok(Value::String(String::new())),
   295→    }
   296→}

   310→fn builtin_to_float(args: &[Value]) -> Result<Value, String> {
   311→    match args.get(0) {
   312→        Some(Value::Float(f)) => Ok(Value::Float(*f)),
   313→        Some(Value::String(s)) => Ok(s.parse::<f64>()
   314→            .map(Value::Float)
   315→            .unwrap_or(Value::Float(0.0))), // soft-failure: return 0.0 on parse error
   316→        Some(Value::Bool(b)) => Ok(Value::Float(if *b { 1.0 } else { 0.0 })),
   317→        _ => Ok(Value::Float(0.0)), // soft-failure
   318→    }
   319→}

   321→fn builtin_confidence(args: &[Value]) -> Result<Value, String> {
   322→    match args.get(0) {
   323→        Some(Value::Fluid(variants)) => {
   324→            let best = variants.iter().map(|v| v.confidence)
   325→                .fold(0.0_f64, f64::max);
   326→            Ok(Value::Float(best))
   327→        }
   328→        Some(_) => Ok(Value::Float(1.0)), // concrete values are fully confident
   329→        None => Err("confidence() requires 1 argument".to_string()),
   330→    }
   331→}

   335→fn expect_float_arg(fn_name: &str, args: &[Value], index: usize) -> Result<f64, String> {
   336→    if args.len() <= index {
   337→        return Err(format!("{}() requires an argument at position {}", fn_name, index));
   338→    }
   339→    match &args[index] {
   340→        Value::Float(f) => Ok(*f),
   341→        other => Err(format!(
   342→            "{}() expected Float argument, got {}",
   343→            fn_name, other.type_name()
   344→        )),
   345→    }
   346→}

   348→fn expect_string_arg(fn_name: &str, args: &[Value], index: usize) -> Result<String, String> {
   349→    if args.len() <= index {
   350→        return Err(format!("{}() requires an argument at position {}", fn_name, index));
   351→    }
   352→    match &args[index] {
   353→        Value::String(s) => Ok(s.clone()),
   354→        other => Err(format!(
   355→            "{}() expected String argument, got {}",
   356→            fn_name, other.type_name()
   357→        )),
   358→    }
   359→}

   371→    let old = expect_string_arg("__replace", args, 1)?;

   376→fn builtin_split(args: &[Value]) -> Result<Value, String> {
   377→    let s = expect_string_arg("__split", args, 0)?;
   378→    let sep = expect_string_arg("__split", args, 1)?;
   379→    let items: Vec<Value> = if sep.is_empty() {
   380→        s.chars().map(|c| Value::String(c.to_string())).collect()
   381→    } else {
   382→        s.split(&sep).map(|part| Value::String(part.to_string())).collect()
   383→    };
   384→    Ok(Value::List(items))
   385→}

   387→fn builtin_join(args: &[Value]) -> Result<Value, String> {
   388→    let list = match args.get(0) {
   389→        Some(Value::List(items)) => items,
   390→        _ => return Err("__join() requires List as first argument".to_string()),
   391→    };
   392→    let sep = if args.len() > 1 {
   393→        match &args[1] {
   394→            Value::String(s) => s.clone(),
   395→            _ => ",".to_string(),
   396→        }
   397→    } else {
   398→        ",".to_string()
   399→    };
   400→    let parts: Vec<String> = list.iter().map(|v| format!("{}", v)).collect();
   401→    Ok(Value::String(parts.join(&sep)))
   402→}

   513→fn builtin_render(args: &[Value]) -> Result<Value, String> {
   514→    // render(template_name, key1, val1, key2, val2, ...)
   515→    // Simple {{ var }} substitution with auto-escaping
   516→    // In interpreter mode, do basic string substitution
   517→    if args.len() < 3 || (args.len() - 1) % 2 != 0 {
   518→        return Err("render() requires template name + key/value pairs (odd count)".to_string());
   519→    }
   520→    let template_name = expect_string_arg("render", args, 0)?;
   521→
   522→    // Build substitution map from remaining args (key, value pairs)
   523→    let mut vars = std::collections::HashMap::new();
   524→    let mut i = 1;
   525→    while i + 1 < args.len() {
   526→        let key = match &args[i] {
   527→            Value::String(s) => s.clone(),
   528→            other => return Err(format!("render() key must be String, got {}", other.type_name())),
   529→        };
   530→        let val = match &args[i + 1] {
   531→            Value::String(s) => s.clone(),
   532→            other => format!("{}", other),
   533→        };
   534→        vars.insert(key, val);
   535→        i += 2;
   536→    }
   537→
   538→    // In interpreter mode, generate a simple HTML string from the template name and vars
   539→    let mut html = String::from("<div class=\"template-");
   540→    html.push_str(&escape_html_chars(&template_name));
   541→    html.push_str("\">");
   542→    for (key, val) in &vars {
   543→        html.push_str(&format!("<span data-key=\"{}\">{}</span>",
   544→            escape_html_chars(key), escape_html_chars(val)));
   545→    }
   546→    html.push_str("</div>");
   547→
   548→    Ok(Value::Html(html))
   549→}

   644→fn builtin_encrypt(args: &[Value]) -> Result<Value, String> {
   645→    let data = expect_string_arg("encrypt", args, 0)?;
   646→    let key_str = match args.get(1) {
   647→        Some(Value::Secret(zs)) => zs.as_str(),
   648→        Some(other) => return Err(format!("encrypt() expected Secret as second arg, got {}", other.type_name())),
   649→        None => return Err("encrypt() requires 2 arguments".to_string()),
   650→    };
   651→    // Real AES-256-GCM with random 96-bit nonce (Phase 7.3)
   652→    use aes_gcm::{Aes256Gcm, AeadCore, Key};
   653→    use aes_gcm::aead::{Aead, KeyInit, OsRng};
   654→
   655→    let key_bytes = hex::decode(key_str)
   656→        .map_err(|e| format!("encrypt() invalid key format (expected hex): {}", e))?;
   657→    if key_bytes.len() != 32 {
   658→        return Err(format!("encrypt() key must be 256-bit (64 hex chars), got {} bytes", key_bytes.len()));
   659→    }
   660→    let key = Key::<Aes256Gcm>::from_slice(&key_bytes);
   661→    let cipher = Aes256Gcm::new(key);
   662→    let nonce = Aes256Gcm::generate_nonce(&mut OsRng); // 96-bit random nonce
   663→
   664→    match cipher.encrypt(&nonce, data.as_ref()) {
   665→        Ok(ciphertext) => {
   666→            // Prepend nonce to ciphertext for self-contained Encrypted value
   667→            let mut output = nonce.to_vec();
   668→            output.extend_from_slice(&ciphertext);
   669→            Ok(Value::Encrypted(output))
   670→        }
   671→        Err(e) => Err(format!("encrypt() AES-256-GCM encryption failed: {}", e)),
   672→    }
   673→}

   675→fn builtin_decrypt(args: &[Value]) -> Result<Value, String> {
   676→    let encrypted = match args.get(0) {
   677→        Some(Value::Encrypted(data)) => data.clone(),
   678→        Some(other) => return Err(format!("decrypt() expected Encrypted as first arg, got {}", other.type_name())),
   679→        None => return Err("decrypt() requires 2 arguments".to_string()),
   680→    };
   681→    let key_str = match args.get(1) {
   682→        Some(Value::Secret(zs)) => zs.as_str(),
   683→        Some(other) => return Err(format!("decrypt() expected Secret as second arg, got {}", other.type_name())),
   684→        None => return Err("decrypt() requires 2 arguments".to_string()),
   685→    };
   686→    // Real AES-256-GCM decryption (Phase 7.3)
   687→    // Encrypted format: nonce (12 bytes) || ciphertext_with_tag
   688→    use aes_gcm::{Aes256Gcm, Key, Nonce};
   689→    use aes_gcm::aead::{Aead, KeyInit};
   690→
   691→    if encrypted.len() < 13 {
   692→        // Need at least 12 (nonce) + 1 (tag minimum)
   693→        return Err("decrypt() invalid encrypted data: too short".to_string());
   694→    }
   695→
   696→    let key_bytes = hex::decode(key_str)
   697→        .map_err(|e| format!("decrypt() invalid key format (expected hex): {}", e))?;
   698→    if key_bytes.len() != 32 {
   699→        return Err(format!("decrypt() key must be 256-bit (64 hex chars), got {} bytes", key_bytes.len()));
   700→    }
   701→
   702→    let (nonce_bytes, ciphertext) = encrypted.split_at(12);
   703→    let nonce = Nonce::from_slice(nonce_bytes);
   704→    let key = Key::<Aes256Gcm>::from_slice(&key_bytes);
   705→    let cipher = Aes256Gcm::new(key);
   706→
   707→    match cipher.decrypt(nonce, ciphertext) {
   708→        Ok(plaintext) => {
   709→            match String::from_utf8(plaintext) {
   710→                Ok(s) => Ok(Value::String(s)),
   711→                Err(_) => Err("decrypt() decrypted data is not valid UTF-8".to_string()),
   712→            }
   713→        }
   714→        Err(_) => Err("decrypt() failed: incorrect key or corrupted data".to_string()),
   715→    }
   716→}

   778→/// Send an HTTP POST request. Returns the response body as String.
   779→/// Usage: http_post(url, body, content_type)
   780→/// Usage: http_post(url, body, content_type, auth_token)        — sets Authorization: Bearer <auth_token>
   781→/// Usage: http_post(url, body, content_type, headers_struct)    — sets headers from Struct fields
   782→/// Наряд №12 Bug 2: Added 4th parameter for authorization headers.
   783→fn builtin_http_post(args: &[Value]) -> Result<Value, String> {
   784→    let url = match args.get(0) {
   785→        Some(Value::String(s)) => s.clone(),
   786→        Some(other) => return Err(format!("http_post() expected String as url, got {}", other.type_name())),
   787→        None => return Err("http_post() requires at least 1 argument (url)".to_string()),
   788→    };
   789→
   790→    let body = match args.get(1) {
   791→        Some(Value::String(s)) => s.clone(),
   792→        Some(other) => return Err(format!("http_post() expected String as body, got {}", other.type_name())),
   793→        None => return Err("http_post() requires at least 2 arguments (url, body)".to_string()),
   794→    };
   795→
   796→    let content_type = match args.get(2) {
   797→        Some(Value::String(s)) => s.clone(),
   798→        _ => "application/json".to_string(),
   799→    };
   800→
   801→
   802→    let client = reqwest::blocking::Client::builder()
   803→        .timeout(std::time::Duration::from_secs(30))
   804→        .build()
   805→        .map_err(|e| format!("http_post(): failed to create client: {}", e))?;
   806→
   807→
   808→    let mut req = client
   809→        .post(&url)
   810→        .header("Content-Type", &content_type)
   811→        .body(body);
   812→
   813→    // Optional 4th argument: headers (Наряд №12 Bug 2)
   814→    // String: treat as Bearer token; Struct: set headers from fields
   815→    if let Some(headers_arg) = args.get(3) {
   816→        match headers_arg {
   817→            Value::String(auth_token) => {
   818→                if !auth_token.is_empty() {
   819→                    req = req.header("Authorization", format!("Bearer {}", auth_token));
   820→                }
   821→            }
   822→            Value::Struct { fields, .. } => {
   823→                for (key, val) in fields {
   824→                    if let Value::String(v) = val {
   825→                        req = req.header(key.as_str(), v.as_str());
   826→                    }
   827→                }
   828→            }
   829→            _ => {}
   830→        }
   831→    }
   832→
   833→    let resp = req
   834→        .send()
   835→        .map_err(|e| format!("http_post() request failed: {}", e))?;
   836→
   837→    let status = resp.status().as_u16();
   838→    let resp_body = resp.text().unwrap_or_default();
   839→
   840→    if status >= 400 {
   841→        return Err(format!("http_post() returned status {}: {}", status, resp_body));
   842→    }
   843→
   844→    Ok(Value::String(resp_body))
   845→}

   928→        serde_json::Value::String(s) => Value::String(s.clone()),
   929→        serde_json::Value::Number(n) => Value::Float(n.as_f64().unwrap_or(0.0)),
   930→        serde_json::Value::Bool(b) => Value::Bool(*b),
   931→        serde_json::Value::Null => Value::Unit,
   932→        serde_json::Value::Array(arr) => {
   933→            Value::List(arr.iter().map(|v| json_value_to_mlog_value(v)).collect())

   935→        serde_json::Value::Object(obj) => {

   937→            for (k, v) in obj {

   940→            Value::Struct { type_name: "Json".to_string(), fields }
   941→        }
   942→    }

   980→/// Safe field access on a struct value: returns default if field missing or not a struct.
   981→/// Usage: json_get(obj, "field") -> Value (returns Unit if missing)
   982→/// Usage: json_get(obj, "field", default_value) -> Value (returns default if missing)
   983→/// Usage: json_get(obj, "nested.field.path", default) -> Value (dot-separated path)
   984→/// This is the P0 fix: prevents runtime crash when accessing optional JSON fields
   985→/// like message.voice on non-voice Telegram updates.
   986→fn builtin_json_get(args: &[Value]) -> Result<Value, String> {
   987→    let obj = match args.get(0) {
   988→        Some(v) => v,
   989→        None => return Err("json_get() requires at least 2 arguments (obj, field_path)".to_string()),
   990→    };
   991→    let path = expect_string_arg("json_get", args, 1)?;
   992→    // Bug 2.2 fix: when no default is provided, return the found value directly
   993→    // (not wrapped in Unit). The old code defaulted to Value::Unit which silently
   994→    // swallowed string values and made them unusable.
   995→    if args.len() >= 3 {
   996→        let default_val = args[2].clone();

   998→        let mut current = obj;
   999→        for segment in path.split('.') {

  1002→                Err(_) => return Ok(default_val),
  1003→            }
  1004→        }
  1005→        Ok(current.clone())
  1006→    } else {
  1007→        // 2-argument form: return the found value or Unit if not found
  1008→        let mut current = obj;
  1009→        for segment in path.split('.') {
  1010→            match current.get_field(segment) {
  1011→                Ok(val) => current = val,
  1012→                Err(_) => return Ok(Value::Unit),
  1013→            }
  1014→        }
  1015→        Ok(current.clone())
  1016→    }
  1017→}

  1045→/// Send an HTTP GET request. Returns the response body as String.
  1046→/// Usage: http_get(url) -> String
  1047→/// Usage: http_get(url, headers_struct) -> String  — sets headers from Struct fields
  1048→fn builtin_http_get(args: &[Value]) -> Result<Value, String> {
  1049→    let url = match args.get(0) {
  1050→        Some(Value::String(s)) => s.clone(),
  1051→        Some(other) => return Err(format!("http_get() expected String as url, got {}", other.type_name())),
  1052→        None => return Err("http_get() requires 1 argument (url)".to_string()),
  1053→    };
  1054→
  1055→    let client = reqwest::blocking::Client::builder()
  1056→        .timeout(std::time::Duration::from_secs(30))
  1057→        .build()
  1058→        .map_err(|e| format!("http_get(): failed to create client: {}", e))?;
  1059→
  1060→    let mut req = client.get(&url);
  1061→
  1062→    // Optional 2nd argument: headers (String as Bearer token, Struct as header map)
  1063→    if let Some(headers_arg) = args.get(1) {
  1064→        match headers_arg {
  1065→            Value::String(auth_token) => {
  1066→                if !auth_token.is_empty() {
  1067→                    req = req.header("Authorization", format!("Bearer {}", auth_token));
  1068→                }
  1069→            }
  1070→            Value::Struct { fields, .. } => {
  1071→                for (key, val) in fields {
  1072→                    if let Value::String(v) = val {
  1073→                        req = req.header(key.as_str(), v.as_str());
  1074→                    }
  1075→                }
  1076→            }
  1077→            _ => {}
  1078→        }
  1079→    }
  1080→
  1081→    let resp = req
  1082→        .send()
  1083→        .map_err(|e| format!("http_get() request failed: {}", e))?;
  1084→
  1085→    let status = resp.status().as_u16();
  1086→    let resp_body = resp.text().unwrap_or_default();
  1087→
  1088→    if status >= 400 {
  1089→        return Err(format!("http_get() returned status {}: {}", status, resp_body));
  1090→    }
  1091→
  1092→    Ok(Value::String(resp_body))
  1093→}

  1229→/// Initialize SQLite persistence for the KV store.
  1230→/// Called by Interpreter::configure_memory() when persist path is set.
  1231→/// Creates kv_store table (key TEXT PRIMARY KEY, value TEXT) in the given database.
  1232→/// Loads existing rows into the in-memory HashMap.
  1233→pub fn init_kv_persist(db_path: &str) -> Result<(), String> {
  1234→    let conn = rusqlite::Connection::open(db_path)
  1235→        .map_err(|e| format!("[kv_store] Failed to open database '{}': {}", db_path, e))?;
  1236→    conn.execute_batch(
  1237→        "CREATE TABLE IF NOT EXISTS kv_store (key TEXT PRIMARY KEY, value TEXT NOT NULL);"
  1238→    ).map_err(|e| format!("[kv_store] Failed to create table: {}", e))?;
  1239→
  1240→    // Load existing KV pairs into in-memory HashMap (write-through cache warmup)
  1241→    {
  1242→        let mut stmt = conn.prepare("SELECT key, value FROM kv_store")
  1243→            .map_err(|e| format!("[kv_store] Failed to query: {}", e))?;
  1244→        let rows: Vec<(String, String)> = stmt.query_map([], |row| {
  1245→            Ok((row.get::<_, String>(0)?, row.get::<_, String>(1)?))
  1246→        }).map_err(|e| format!("[kv_store] Failed to iterate: {}", e))?
  1247→        .filter_map(|r| r.ok())
  1248→        .collect();
  1249→
  1250→        // Merge into in-memory store (SQLite is authoritative on init)
  1251→        if let Ok(mut store) = kv_store().lock() {
  1252→            for (key, value) in rows {
  1253→                store.insert(key, value);
  1254→            }
  1255→        }
  1256→    } // stmt is dropped here, releasing borrow on conn
  1257→
  1258→    // Store the connection globally
  1259→    let mut sqlite_guard = kv_sqlite().lock()
  1260→        .map_err(|e| format!("[kv_store] lock error: {}", e))?;
  1261→    *sqlite_guard = Some(conn);
  1262→    eprintln!("[kv_store] SQLite persistence enabled: {}", db_path);
  1263→    Ok(())
  1264→}

  1376→// Resets when mlog serve restarts (by design: session data is ephemeral).
  1377→// Unlike mem_set/mem_get (global), session_* is scoped to a specific session_id.
  1378→//
  1379→// Usage:
  1380→//   session_set(session_id, key, value)   -> String (stored value)
  1381→//   session_get(session_id, key)             -> String (value or "")
  1382→//   session_clear(session_id)                -> Unit
  1383→
  1384→/// Global session store — lazy_static pattern using std::sync::OnceLock.
  1385→/// Outer key = session_id, inner key = data key, inner value = data value.
  1386→static SESSION_STORE: std::sync::OnceLock<StdMutex<std::collections::HashMap<String, std::collections::HashMap<String, String>>>> = std::sync::OnceLock::new();
  1387→
  1388→fn session_store() -> &'static StdMutex<std::collections::HashMap<String, std::collections::HashMap<String, String>>> {
  1389→    SESSION_STORE.get_or_init(|| StdMutex::new(std::collections::HashMap::new()))
  1390→}
  1391→
  1392→/// Reset the entire session store. Used by contract tests to verify restart behavior.
  1393→pub fn reset_session_store() {
  1394→    if let Ok(mut store) = session_store().lock() {
  1395→        store.clear();
  1396→    }
  1397→}
  1398→
  1399→/// Get the number of sessions in the store. Used by contract tests.
  1400→pub fn session_store_count() -> usize {
  1401→    session_store().lock().map(|s| s.len()).unwrap_or(0)
  1402→}
  1403→
  1404→/// Get the number of keys in a specific session. Used by contract tests.
  1405→pub fn session_key_count(session_id: &str) -> usize {
  1406→    session_store().lock()
  1407→        .ok()
  1408→        .and_then(|s| s.get(session_id).map(|m| m.len()))
  1409→        .unwrap_or(0)
  1410→}

  1412→/// `session_set(session_id, key, value)` — store a value scoped to a session.
  1413→/// Returns the stored value. Creates session bucket if it doesn't exist.
  1414→fn builtin_session_set(args: &[Value]) -> Result<Value, String> {
  1415→    let session_id = expect_string_arg("session_set", args, 0)?;
  1416→    let key = expect_string_arg("session_set", args, 1)?;
  1417→    let value = match args.get(2) {
  1418→        Some(Value::String(s)) => s.clone(),
  1419→        Some(other) => format!("{}", other),
  1420→        None => return Err("session_set() requires 3 arguments (session_id, key, value)".to_string()),
  1421→    };
  1422→    let mut store = session_store().lock().map_err(|e| format!("session_set() lock error: {}", e))?;
  1423→    store.entry(session_id).or_default().insert(key.clone(), value.clone());
  1424→    Ok(Value::String(value))
  1425→}

  1427→/// `session_get(session_id, key)` — retrieve a value from a session.
  1428→/// Returns empty string if session or key not found.
  1429→fn builtin_session_get(args: &[Value]) -> Result<Value, String> {
  1430→    let session_id = expect_string_arg("session_get", args, 0)?;
  1431→    let key = expect_string_arg("session_get", args, 1)?;
  1432→    let store = session_store().lock().map_err(|e| format!("session_get() lock error: {}", e))?;
  1433→    let value = store.get(&session_id)
  1434→        .and_then(|session| session.get(&key).cloned())
  1435→        .unwrap_or_default();
  1436→    Ok(Value::String(value))
  1437→}

  1450→// Paths containing ".." or absolute paths are rejected (sandbox).

  1452→/// Validate that a path is safe (within working directory, no traversal).
  1453→fn sandbox_path(path: &str) -> Result<std::path::PathBuf, String> {
  1454→    let p = std::path::Path::new(path);
  1455→    // Reject absolute paths
  1456→    if p.is_absolute() {
  1457→        return Err(format!("file I/O sandbox: absolute paths not allowed: '{}'", path));
  1458→    }
  1459→    // Reject path traversal
  1460→    for component in p.components() {
  1461→        if let std::path::Component::ParentDir = component {
  1462→            return Err(format!("file I/O sandbox: path traversal ('..') not allowed: '{}'", path));
  1463→        }
  1464→    }
  1465→    Ok(std::path::PathBuf::from(path))
  1466→}

  1468→/// `read_file(path)` — read file contents as String.
  1469→/// Soft-failure: returns empty string on error (file not found, permission denied, etc.).
  1470→fn builtin_read_file(args: &[Value]) -> Result<Value, String> {
  1471→    let path = expect_string_arg("read_file", args, 0)?;
  1472→    let safe_path = match sandbox_path(&path) {
  1473→        Ok(p) => p,
  1474→        Err(_) => return Ok(Value::String(String::new())), // soft-failure on sandbox violation
  1475→    };
  1476→    match std::fs::read_to_string(&safe_path) {
  1477→        Ok(content) => Ok(Value::String(content)),
  1478→        Err(_) => Ok(Value::String(String::new())), // soft-failure
  1479→    }
  1480→}

  1609→fn builtin_http_post_multipart(args: &[Value]) -> Result<Value, String> {
  1610→    let url = expect_string_arg("http_post_multipart", args, 0)?;
  1611→    let fields = match args.get(1) {
  1612→        Some(Value::Struct { fields, .. }) => fields.clone(),
  1613→        _ => return Err("http_post_multipart() requires Struct as 2nd argument (fields)".to_string()),
  1614→    };
  1615→    let files = match args.get(2) {
  1616→        Some(Value::Struct { fields, .. }) => fields.clone(),
  1617→        _ => return Err("http_post_multipart() requires Struct as 3rd argument (files)".to_string()),
  1618→    };
  1619→
  1620→    let client = reqwest::blocking::Client::builder()
  1621→        .timeout(std::time::Duration::from_secs(120))
  1622→        .build()
  1623→        .map_err(|e| format!("http_post_multipart(): client error: {}", e))?;
  1624→
  1625→    let mut form = reqwest::blocking::multipart::Form::new();
  1626→
  1627→    // Add text fields
  1628→    for (key, val) in &fields {
  1629→        if let Value::String(v) = val {
  1630→            form = form.text(key.clone(), v.clone());
  1631→        }
  1632→    }
  1633→
  1634→    // Add file fields
  1635→    for (key, val) in &files {
  1636→        if let Value::String(path) = val {
  1637→            let file_bytes = std::fs::read(path)
  1638→                .map_err(|e| format!("http_post_multipart(): cannot read file '{}': {}", path, e))?;
  1639→            let file_name = std::path::Path::new(path)
  1640→                .file_name()
  1641→                .and_then(|n| n.to_str())
  1642→                .unwrap_or("file");
  1643→            let part = reqwest::blocking::multipart::Part::bytes(file_bytes)
  1644→                .file_name(file_name.to_string());
  1645→            form = form.part(key.clone(), part);
  1646→        }
  1647→    }
  1648→
  1649→    let resp = client.post(&url).multipart(form).send()
  1650→        .map_err(|e| format!("http_post_multipart(): request failed: {}", e))?;
  1651→
  1652→    let status = resp.status().as_u16();
  1653→    let resp_body = resp.text().unwrap_or_default();
  1654→
  1655→    if status >= 400 {
  1656→        return Err(format!("http_post_multipart() returned status {}: {}", status, resp_body));
  1657→    }
  1658→
  1659→    Ok(Value::String(resp_body))
  1660→}

  1662→fn builtin_whisper_transcribe(args: &[Value]) -> Result<Value, String> {

  1743→    let text = parsed.get("text").and_then(|t| t.as_str()).unwrap_or("").to_string();
  1744→
  1745→    Ok(Value::String(text))
  1746→}

  1832→/// **Security**: Only available outside sandbox. Inside sandbox, always errors.
  1833→/// In server mode, command execution is disabled unless the binary is run with METALOGOS_ALLOW_EXEC=1.
  1834→fn builtin_exec(args: &[Value]) -> Result<Value, String> {
  1835→    // Security: disable in server context unless explicitly allowed
  1836→    if std::env::var("METALOGOS_ALLOW_EXEC").unwrap_or_default() != "1" {
  1837→        // Check if we're likely in server mode (has METALOGOS_PORT or METALOGOS_DB env)
  1838→        let in_server = std::env::var("METALOGOS_PORT").is_ok()
  1839→            || std::env::var("METALOGOS_DB").is_ok();
  1840→        if in_server {
  1841→            return Err("exec() is disabled in server mode. Set METALOGOS_ALLOW_EXEC=1 to enable.".to_string());
  1842→        }
  1843→    }
  1844→
  1845→    let cmd = expect_string_arg("exec", args, 0)?;
  1846→    let output = std::process::Command::new("sh")
  1847→        .arg("-c")
  1848→        .arg(&cmd)
  1849→        .output()
  1850→        .map_err(|e| format!("exec(): failed to run command: {}", e))?;
  1851→
  1852→    let stdout = String::from_utf8_lossy(&output.stdout).to_string();
  1853→    if !output.status.success() {
  1854→        let stderr = String::from_utf8_lossy(&output.stderr).to_string();
  1855→        return Err(format!("exec() command exited with {}: {}", output.status, stderr.trim()));
  1856→    }
  1857→    Ok(Value::String(stdout))
  1858→}

  1860→/// `escape_js(s) -> String` — escape a string for safe insertion into JavaScript.
  1861→/// Escapes: backslash, single quote, double quote, newline, carriage return, tab,
  1862→/// line separator, paragraph separator, and NUL.
  1863→fn builtin_escape_js(args: &[Value]) -> Result<Value, String> {
  1864→    let s = expect_string_arg("escape_js", args, 0)?;
  1865→    let mut out = String::with_capacity(s.len());
  1866→    for c in s.chars() {
  1867→        match c {
  1868→            '\\' => out.push_str("\\\\"),
  1869→            '\'' => out.push_str("\\'"),
  1870→            '"' => out.push_str("\\\""),
  1871→            '\n' => out.push_str("\\n"),
  1872→            '\r' => out.push_str("\\r"),
  1873→            '\t' => out.push_str("\\t"),
  1874→            '\u{2028}' => out.push_str("\\u2028"), // line separator
  1875→            '\u{2029}' => out.push_str("\\u2029"), // paragraph separator
  1876→            '\0' => out.push_str("\\0"),
  1877→            _ => out.push(c),
  1878→        }
  1879→    }
  1880→    Ok(Value::String(out))

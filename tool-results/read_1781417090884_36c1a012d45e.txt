    27→/// Opaque secret string with automatic memory zeroing on drop (Phase 7.3).
    28→/// Implements serde by serializing as "[SECRET]" marker — actual value is NEVER persisted.
    29→#[derive(Clone, Debug)]
    30→pub struct SecretString(pub Zeroizing<String>);
    31→
    32→impl serde::Serialize for SecretString {
    33→    fn serialize<S: serde::Serializer>(&self, s: S) -> Result<S::Ok, S::Error> {
    34→        // Never serialize the actual secret — emit a safe marker
    35→        s.serialize_str("[SECRET]")
    36→    }
    37→}
    38→
    39→impl<'de> serde::Deserialize<'de> for SecretString {
    40→    fn deserialize<D: serde::Deserializer<'de>>(d: D) -> Result<Self, D::Error> {
    41→        let inner = String::deserialize(d)?;
    42→        Ok(SecretString(Zeroizing::new(inner)))
    43→    }
    44→}

    46→impl std::ops::Deref for SecretString {
    47→    type Target = String;
    48→    fn deref(&self) -> &Self::Target {
    49→        &self.0
    50→    }
    51→}

    62→/// Runtime value.
    63→#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
    64→pub enum Value {
    65→    String(String),
    66→    Float(f64),
    67→    Bool(bool),
    68→    Struct {
    69→        type_name: String,
    70→        fields: HashMap<String, Value>,
    71→    },
    72→    /// List value: ordered collection of items.
    73→    List(Vec<Value>),
    74→    /// Fluid value: superposition of typed variants with confidence scores.
    75→    /// Collapses lazily at point of use (see `maybe_collapse`).
    76→    Fluid(Vec<FluidValueVariant>),
    77→    Unit,
    78→    /// Opaque HTML content (Phase 6.2) — cannot be concatenated, printed, or converted to String
    79→    Html(String),
    80→    /// Opaque SQL query (Phase 6.3) — only created via query() builtin
    81→    Query(String),
    82→    /// Opaque secret value (Phase 6.4) — cannot be printed or converted to String.
    83→    /// Phase 7.3: Internally uses SecretString (Zeroizing<String>) — memory is zeroed on drop.
    84→    Secret(SecretString),
    85→    /// Opaque encrypted data (Phase 6.4)
    86→    Encrypted(Vec<u8>),
    87→    /// Opaque password hash (Phase 6.4)
    88→    Hash(String),
    89→    /// Opaque session data (Phase 6.5)
    90→    Session(std::collections::HashMap<String, String>),
    91→    /// HTTP response value (Phase 6.1)
    92→    HttpResponse { status: u16, body: String },
    93→}

    95→impl std::fmt::Display for Value {
    96→    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
    97→        match self {
    98→            Value::String(s) => write!(f, "{}", s),
    99→            Value::Float(n) => write!(f, "{}", n),
   100→            Value::Bool(b) => write!(f, "{}", b),
   101→            Value::Struct { type_name, fields } => {
   102→                write!(f, "{} {{", type_name)?;
   103→                let pairs: Vec<_> = fields.iter().collect();
   104→                for (i, (k, v)) in pairs.iter().enumerate() {
   105→                    if i > 0 { write!(f, ", ")?; }
   106→                    write!(f, "{}: {}", k, v)?;
   107→                }
   108→                write!(f, "}}")
   109→            }
   110→            Value::List(items) => {
   111→                write!(f, "[")?;
   112→                for (i, item) in items.iter().enumerate() {
   113→                    if i > 0 { write!(f, ", ")?; }
   114→                    write!(f, "{}", item)?;
   115→                }
   116→                write!(f, "]")
   117→            }
   118→            Value::Fluid(variants) => {
   119→                // Display as the highest-confidence variant
   120→                let best = variants.iter().max_by(|a, b| {
   121→                    a.confidence.partial_cmp(&b.confidence).unwrap_or(std::cmp::Ordering::Equal)
   122→                });
   123→                match best {
   124→                    Some(v) => write!(f, "{}", v.value),
   125→                    None => write!(f, "()"),
   126→                }
   127→            }
   128→            Value::Unit => write!(f, "()"),
   129→            Value::Html(_) => write!(f, "[Html]"),
   130→            Value::Query(_) => write!(f, "[Query]"),
   131→            Value::Secret(_) => write!(f, "[Secret]"),
   132→            Value::Encrypted(_) => write!(f, "[Encrypted]"),
   133→            Value::Hash(_) => write!(f, "[Hash]"),
   134→            Value::Session(_) => write!(f, "[Session]"),
   135→            Value::HttpResponse { status, .. } => write!(f, "[HttpResponse {}]", status),
   136→        }
   137→    }
   138→}

   140→impl Value {
   141→    pub fn type_name(&self) -> &'static str {
   142→        match self {
   143→            Value::String(_) => "String",
   144→            Value::Float(_) => "Float",
   145→            Value::Bool(_) => "Bool",
   146→            Value::List(_) => "List",
   147→            Value::Struct { .. } => "Struct",
   148→            Value::Fluid(_) => "Fluid",
   149→            Value::Unit => "Unit",
   150→            Value::Html(_) => "Html",
   151→            Value::Query(_) => "Query",
   152→            Value::Secret(_) => "Secret",
   153→            Value::Encrypted(_) => "Encrypted",
   154→            Value::Hash(_) => "Hash",
   155→            Value::Session(_) => "Session",
   156→            Value::HttpResponse { .. } => "HttpResponse",
   157→        }
   158→    }
   159→
   160→    /// Get a field value from a struct. Returns Err if not a struct or field missing.
   161→    pub fn get_field(&self, field: &str) -> Result<&Value, String> {
   162→        match self {
   163→            Value::Struct { fields, .. } => fields.get(field)
   164→                .ok_or_else(|| format!("field '{}' not found on struct", field)),
   165→            _ => Err(format!("cannot access field '{}' on non-struct value ({})", field, self.type_name())),
   166→        }
   167→    }
   168→
   169→    /// Set a field value on a mutable struct.
   170→    pub fn set_field(&mut self, field: &str, value: Value) -> Result<(), String> {
   171→        match self {
   172→            Value::Struct { fields, .. } => {
   173→                if fields.contains_key(field) {
   174→                    fields.insert(field.to_string(), value);
   175→                    Ok(())
   176→                } else {
   177→                    Err(format!("field '{}' not found on struct", field))
   178→                }
   179→            }
   180→            _ => Err(format!("cannot set field '{}' on non-struct value", field)),
   181→        }
   182→    }
   183→
   184→    /// Convert to f64 for numeric comparisons.
   185→    pub fn as_float(&self) -> Result<f64, String> {
   186→        match self {
   187→            Value::Float(f) => Ok(*f),
   188→            Value::Bool(b) => Ok(if *b { 1.0 } else { 0.0 }),
   189→            Value::String(s) => s.parse::<f64>()
   190→                .map_err(|_| format!("cannot convert '{}' to Float", s)),
   191→            _ => Err(format!("cannot convert {} to Float", self.type_name())),
   192→        }
   193→    }
   194→
   195→    /// Convert to bool for condition checking.
   196→    pub fn as_bool(&self) -> Result<bool, String> {
   197→        match self {
   198→            Value::Bool(b) => Ok(*b),
   199→            Value::Float(f) => Ok(*f != 0.0),
   200→            Value::String(s) => Ok(!s.is_empty()),
   201→            Value::Unit => Ok(false),
   202→            _ => Err(format!("cannot convert {} to Bool", self.type_name())),
   203→        }
   204→    }
   205→
   206→    /// Check if this value is a Fluid superposition.
   207→    pub fn is_fluid(&self) -> bool {
   208→        matches!(self, Value::Fluid(_))
   209→    }
   210→}

   426→impl EvalResult {
   427→    /// Format the eval result as a human-readable report.
   428→    pub fn format_report(&self) -> String {
   429→        let mut lines = Vec::new();
   430→        lines.push(format!("Eval: {}", self.pattern_name));
   431→        lines.push(format!("  Dataset: {} examples", self.total));
   432→        lines.push(format!("  Accuracy: {:.1}% ({}/{})", self.accuracy * 100.0, self.correct, self.total));
   433→        lines.push(format!("  Threshold: {}", self.threshold));
   434→        lines.push(format!("  Result: {}", if self.passed { "PASS" } else { "FAIL (below threshold)" }));
   435→
   436→        // Confusion matrix
   437→        if !self.confusion.is_empty() {
   438→            // Collect all unique labels
   439→            let mut all_labels: std::collections::BTreeSet<String> = std::collections::BTreeSet::new();
   440→            for (expected, predictions) in &self.confusion {
   441→                all_labels.insert(expected.clone());
   442→                for pred in predictions.keys() {
   443→                    all_labels.insert(pred.clone());
   444→                }
   445→            }
   446→
   447→            let labels: Vec<&String> = all_labels.iter().collect();
   448→
   449→            // Header row
   450→            let header = format!("  {:12} {}", "", labels.iter().map(|l| format!("{:>12}", l)).collect::<Vec<_>>().join(" "));
   451→            lines.push(header);
   452→
   453→            // Data rows
   454→            for expected in &labels {
   455→                let row = format!("  {:12} {}", expected,
   456→                    labels.iter().map(|pred| {
   457→                        let count = self.confusion.get(*expected)
   458→                            .and_then(|m| m.get(*pred))
   459→                            .copied()
   460→                            .unwrap_or(0);
   461→                        format!("{:>12}", count)
   462→                    }).collect::<Vec<_>>().join(" ")
   463→                );
   464→                lines.push(row);
   465→            }
   466→        }
   467→
   468→        // Failing examples with adapt suggestions
   469→        if !self.failures.is_empty() {
   470→            lines.push(String::new());
   471→            lines.push("  Failing examples (suggest adapt):".to_string());
   472→            for (input, expected, actual) in &self.failures {
   473→                lines.push(format!("    - {:?} -> expected {:?}, got {:?}", input, expected, actual));
   474→            }
   475→            // Generate adapt suggestions
   476→            lines.push(String::new());
   477→            lines.push("  Suggested adapt commands:".to_string());
   478→            for (input, expected, _actual) in &self.failures {
   479→                lines.push(format!("    adapt {} add_example({:?}, {:?})", self.pattern_name, input, expected));
   480→            }
   481→        }
   482→
   483→        lines.join("\n")
   484→    }
   485→}

   756→    /// Configure memory persistence (Phase 7.6).
   757→    /// If persist path is provided, switches to SQLite-backed stores.
   758→    /// The in-memory data is migrated to SQLite during the switch.
   759→    pub fn configure_memory(&mut self, config: &MemoryDecl) {
   760→        if let Some(ref path) = config.persist {
   761→            // Switch to SQLite backend
   762→            let db_path = std::path::PathBuf::from(path);
   763→            match SqliteStore::open(&db_path) {
   764→                Ok(sqlite_store) => {
   765→                    // Migrate existing in-memory data to SQLite
   766→                    let existing = self.memory.lock().unwrap().all_entries();
   767→                    let mut new_store: Box<dyn MemoryStore> = Box::new(sqlite_store);
   768→                    for entry in existing {
   769→                        let _ = new_store.memorize(entry);
   770→                    }
   771→                    self.memory = std::sync::Mutex::new(new_store);
   772→
   773→                    // Migrate KG edges to SQLite (sharing the same DB file)
   774→                    let existing_edges: Vec<(String, String, String, f64)> =
   775→                        self.kg.lock().unwrap().all_edges();
   776→                    if let Ok(sqlite_kg) = SqliteKg::open(&db_path) {
   777→                        let mut new_kg: Box<dyn KgStore> = Box::new(sqlite_kg);
   778→                        for (from, to, relation, weight) in existing_edges {
   779→                            let _ = new_kg.relate(&from, &to, &relation, weight);
   780→                        }
   781→                        self.kg = std::sync::Mutex::new(new_kg);
   782→                    } else {
   783→                        eprintln!("[memory] KG migration to SQLite failed; keeping in-memory KG");
   784→                    }
   785→                    eprintln!("[memory] Persistence enabled: {}", path);
   786→                    self.memory_persist_path = Some(path.clone());
   787→
   788→                    // Наряд №6 — also enable KV store SQLite persistence
   789→                    if let Err(e) = crate::builtins::init_kv_persist(path) {
   790→                        eprintln!("[kv_store] Failed to enable KV persistence: {}. KV will be in-memory only.", e);
   791→                    }
   792→
   793→                    // ADR-0056: initialize checkpoint SQLite (same DB directory)
   794→                    let cp_path = std::path::PathBuf::from(path).with_file_name("checkpoints.db");
   795→                    if let Ok(conn) = rusqlite::Connection::open(&cp_path) {
   796→                        let _ = conn.execute_batch(
   797→                            "CREATE TABLE IF NOT EXISTS checkpoints (
   798→                                flow_name TEXT NOT NULL,
   799→                                checkpoint_name TEXT NOT NULL,
   800→                                step_index INTEGER NOT NULL,
   801→                                state_json TEXT NOT NULL,
   802→                                created_at INTEGER NOT NULL,
   803→                                PRIMARY KEY (flow_name, checkpoint_name)
   804→                            )"
   805→                        );
   806→                        *self.checkpoint_db.lock().unwrap() = Some(conn);
   807→                    }
   808→                }
   809→                Err(e) => {
   810→                    eprintln!("[memory] Failed to open persistent store '{}': {}. Using in-memory.", path, e);
   811→                }
   812→            }
   813→        }
   814→        // If persist is None, keep the default InMemoryStore (already set in new())
   815→    }

   817→    /// Initialize SQLite connection for db { url: "..." } block (Наряд №7).
   818→    /// Supports "sqlite::memory:" for in-memory databases and file paths.
   819→    fn init_db_connection(&mut self, db: &DbDecl) {
   820→        let url_expr = match &db.url {
   821→            Some(expr) => expr,
   822→            None => {
   823→                eprintln!("[db] No url specified in db {{}} block — query() will be unavailable");
   824→                return;
   825→            }
   826→        };
   827→        // Evaluate the url expression (must be a string literal or variable)
   828→        let url = match self.eval_expr(url_expr) {
   829→            Ok(Value::String(s)) => s,
   830→            Ok(other) => {
   831→                eprintln!("[db] url must be a String, got {}", other.type_name());
   832→                return;
   833→            }
   834→            Err(e) => {
   835→                eprintln!("[db] Failed to evaluate url: {}", e);
   836→                return;
   837→            }
   838→        };
   839→        // Parse the URL: "sqlite::memory:" → in-memory, "sqlite:path.db" → file
   840→        let conn = if url == "sqlite::memory:" {
   841→            rusqlite::Connection::open_in_memory()
   842→        } else if url.starts_with("sqlite:") {
   843→            let path = url.trim_start_matches("sqlite:");
   844→            rusqlite::Connection::open(path)
   845→        } else {
   846→            eprintln!("[db] Unsupported URL scheme: '{}'. Use 'sqlite::memory:' or 'sqlite:path.db'", url);
   847→            return;
   848→        };
   849→        match conn {
   850→            Ok(c) => {
   851→                // Enable WAL mode for better concurrent read performance
   852→                let _ = c.execute_batch("PRAGMA journal_mode=WAL;");
   853→                let mut guard = self.db_conn.lock().unwrap();
   854→                *guard = Some(c);
   855→                // Store resolved URL for per-request interpreter reconnection
   856→                self.db_url = Some(url.clone());
   857→                eprintln!("[db] Connected: {}", url);
   858→            }
   859→            Err(e) => {
   860→                eprintln!("[db] Failed to connect to '{}': {}", url, e);
   861→            }
   862→        }
   863→    }

   865→    /// Execute a SQL query and return readable results (Наряд №7).
   866→    /// - SELECT → List of Struct (each row = struct with column names as fields)
   867→    /// - INSERT/UPDATE/DELETE → String with affected row count
   868→    /// - PRAGMA/CREATE/etc. → String "ok"
   869→    fn invoke_query(&self, args: &[Value]) -> Result<Value, String> {
   870→        let sql = match args.first() {
   871→            Some(Value::String(s)) => s.clone(),
   872→            Some(other) => return Err(format!("query() expected String SQL, got {}", other.type_name())),
   873→            None => return Err("query() requires at least 1 argument (SQL string)".to_string()),
   874→        };
   875→        let params: Vec<String> = if args.len() > 1 {
   876→            match &args[1] {
   877→                Value::List(items) => items.iter().filter_map(|v| match v {
   878→                    Value::String(s) => Some(s.clone()),
   879→                    Value::Float(n) => Some(format!("{}", n)),
   880→                    Value::Bool(b) => Some(format!("{}", b)),
   881→                    _ => None,
   882→                }).collect(),
   883→                _ => Vec::new(),
   884→            }
   885→        } else {
   886→            Vec::new()
   887→        };
   888→
   889→        let guard = self.db_conn.lock().map_err(|e| format!("db lock error: {}", e))?;
   890→        let conn = guard.as_ref().ok_or_else(|| {
   891→            "query() error: no database connection. Declare db { url: \"sqlite::memory:\" } first.".to_string()
   892→        })?;
   893→
   894→        let sql_upper = sql.trim().to_uppercase();
   895→        if sql_upper.starts_with("SELECT") || sql_upper.starts_with("PRAGMA") {
   896→            // SELECT/PRAGMA → List of Struct
   897→            let mut stmt = conn.prepare(&sql)
   898→                .map_err(|e| format!("query() SQL error: {}", e))?;
   899→            let col_names: Vec<String> = stmt.column_names().iter().map(|s| s.to_string()).collect();
   900→            let rows: Vec<Value> = stmt.query_map(rusqlite::params_from_iter(params.iter()), |row| {
   901→                let mut fields = std::collections::HashMap::new();
   902→                for (i, col) in col_names.iter().enumerate() {
   903→                    let val: Value = match row.get_ref(i) {
   904→                        Ok(rusqlite::types::ValueRef::Null) => Value::Unit,
   905→                        Ok(rusqlite::types::ValueRef::Integer(n)) => {
   906→                            // Heuristic: if the column name suggests it's an ID or count, keep as Float
   907→                            Value::Float(n as f64)
   908→                        }
   909→                        Ok(rusqlite::types::ValueRef::Real(f)) => Value::Float(f),
   910→                        Ok(rusqlite::types::ValueRef::Text(s)) => {
   911→                            Value::String(String::from_utf8_lossy(s).to_string())
   912→                        }
   913→                        Ok(rusqlite::types::ValueRef::Blob(b)) => {
   914→                            // Encode blobs as hex strings
   915→                            Value::String(b.iter().map(|byte| format!("{:02x}", byte)).collect())
   916→                        }
   917→                        Err(_) => Value::Unit,
   918→                    };
   919→                    fields.insert(col.clone(), val);
   920→                }
   921→                Ok(Value::Struct { type_name: "Row".to_string(), fields })
   922→            }).map_err(|e| format!("query() execution error: {}", e))?
   923→            .filter_map(|r| r.ok())
   924→            .collect();
   925→            Ok(Value::List(rows))
   926→        } else {
   927→            // INSERT/UPDATE/DELETE/CREATE/ALTER/etc. → affected row count as String
   928→            let affected = conn.execute(&sql, rusqlite::params_from_iter(params.iter()))
   929→                .map_err(|e| format!("query() SQL error: {}", e))?;
   930→            Ok(Value::String(affected.to_string()))
   931→        }
   932→    }

   934→    /// Execute a SQL statement via db_execute() — returns affected row count (Наряд №7).
   935→    fn invoke_db_execute(&self, args: &[Value]) -> Result<Value, String> {
   936→        let sql = match args.first() {
   937→            Some(Value::String(s)) => s.clone(),
   938→            Some(other) => return Err(format!("db_execute() expected String SQL, got {}", other.type_name())),
   939→            None => return Err("db_execute() requires at least 1 argument (SQL string)".to_string()),

   974→    /// Run a complete .mlog program.
   975→    pub fn run(&mut self, declarations: Vec<Declaration>) -> Result<Option<String>, String> {
   976→        let mut output: Option<String> = None;
   977→
   978→        for decl in declarations {
   979→            match decl {
   980→                Declaration::Import(import) => {
   981→                    self.handle_import(&import)?;
   982→                }
   983→                Declaration::EntityType(e) => {
   984→                    self.struct_types.insert(e.name.clone(), StructType {
   985→                        name: e.name.clone(),
   986→                        fields: e.fields,
   987→                    });
   988→                }
   989→                Declaration::EntityRecord(e) => {
   990→                    let value = self.instantiate_struct(&e.type_name, &e.fields)?;
   991→                    self.variables.insert(e.name.clone(), value);
   992→                }
   993→                Declaration::EntitySimple(e) => {
   994→                    let value = self.eval_expr(&e.value)?;
   995→                    self.variables.insert(e.name.clone(), value);
   996→                }
   997→                Declaration::Rule(r) => {
   998→                    self.rules.push(r);
   999→                }
  1000→                Declaration::Memorize(m) => {
  1001→                    let value_str = match self.eval_expr(&m.value)? {
  1002→                        Value::String(s) => s,
  1003→                        other => format!("{}", other),
  1004→                    };
  1005→                    let now = SystemTime::now()
  1006→                        .duration_since(UNIX_EPOCH)
  1007→                        .map(|d| d.as_secs() as i64)
  1008→                        .unwrap_or(0);
  1009→                    let embedding = self.embedding_manager.embed(&value_str).unwrap_or_default();

  1204→    /// Instantiate a struct from a type name and field initializers.
  1205→    fn instantiate_struct(&self, type_name: &str, inits: &[FieldInit]) -> Result<Value, String> {
  1206→        let struct_type = self.struct_types.get(type_name)
  1207→            .ok_or_else(|| format!("unknown struct type: {}", type_name))?
  1208→            .clone();
  1209→
  1210→        let mut fields = HashMap::new();
  1211→        for fd in &struct_type.fields {
  1212→            // Look for an initializer
  1213→            let init_val = inits.iter()
  1214→                .find(|fi| fi.name == fd.name)
  1215→                .map(|fi| self.eval_expr(&fi.value));
  1216→
  1217→            let value = match init_val {
  1218→                Some(Ok(v)) => v,
  1219→                Some(Err(e)) => return Err(e),
  1220→                None => {
  1221→                    // Use default value
  1222→                    match &fd.default {
  1223→                        Some(def_expr) => self.eval_expr(def_expr)?,
  1224→                        None => Value::Unit,
  1225→                    }
  1226→                }
  1227→            };
  1228→            fields.insert(fd.name.clone(), value);
  1229→        }
  1230→
  1231→        Ok(Value::Struct { type_name: type_name.to_string(), fields })
  1232→    }

  1234→    /// Handle an import declaration: load module, register namespace or merge globally.
  1235→    fn handle_import(&mut self, import: &ImportDecl) -> Result<(), String> {
  1236→        let module_path = &import.path;
  1237→
  1238→        // Register namespace mapping (alias or path itself for global merge)
  1239→        let alias = import.alias.as_ref().cloned().unwrap_or_else(|| module_path.clone());
  1240→        self.module_namespaces.insert(alias.clone(), module_path.clone());
  1241→

  1248→    /// Load a module by path, parsing and executing its declarations.
  1249→    /// Detects circular imports via loading_stack.
  1250→    fn load_module(&mut self, module_path: &str) -> Result<(), String> {
  1251→        // Circular import detection
  1252→        if self.loading_stack.contains(&module_path.to_string()) {
  1253→            return Err(format!(
  1254→                "circular import detected: {} -> {}",
  1255→                self.loading_stack.join(" -> "),
  1256→                module_path
  1257→            ));
  1258→        }
  1259→
  1260→        // Don't reload if already loaded (check if patterns from this module exist)
  1261→        // We track this via the loading_stack during the current load pass
  1262→        self.loading_stack.push(module_path.to_string());
  1263→
  1264→        let result = self.load_module_inner(module_path);
  1265→
  1266→        self.loading_stack.pop();
  1267→        result
  1268→    }

  1486→    /// Execute all rules in priority order (highest first, then declaration order).
  1487→    fn execute_rules(&mut self) -> Result<(), String> {
  1488→        // Sort by priority descending; stable sort preserves declaration order for ties
  1489→        let mut sorted_rules: Vec<&RuleDecl> = self.rules.iter().collect();
  1490→        sorted_rules.sort_by(|a, b| b.priority.cmp(&a.priority));
  1491→
  1492→        for rule in sorted_rules {
  1493→            let condition_met = self.eval_condition(&rule.condition, &self.variables)?;
  1494→            if condition_met {
  1495→                // Execute assignment: target.field = value
  1496→                let _target_val = self.eval_expr(&rule.target)?;
  1497→                let value_val = self.eval_expr(&rule.value)?;
  1498→
  1499→                // We need to mutate the entity in variables
  1500→                // The target is an Ident (entity name), field is the field name
  1501→                if let Expr::Ident(name) = &rule.target {
  1502→                    let entity = self.variables.get_mut(name)
  1503→                        .ok_or_else(|| format!("rule target '{}' not found", name))?;
  1504→                    entity.set_field(&rule.field, value_val)?;
  1505→                }
  1506→            }
  1507→        }
  1508→        Ok(())
  1509→    }

  1511→    /// Evaluate a rule condition.
  1512→    fn eval_condition(&self, cond: &Condition, env: &HashMap<String, Value>) -> Result<bool, String> {
  1513→        match cond {
  1514→            Condition::Contains { left, right } => {
  1515→                let lv = self.eval_expr_with_env(left, env)?;
  1516→                let rv = self.eval_expr_with_env(right, env)?;
  1517→                let ls = match &lv {
  1518→                    Value::String(s) => s.clone(),
  1519→                    other => return Err(format!("contains: left must be String, got {}", other.type_name())),
  1520→                };
  1521→                let rs = match &rv {
  1522→                    Value::String(s) => s.clone(),
  1523→                    other => return Err(format!("contains: right must be String, got {}", other.type_name())),
  1524→                };
  1525→                Ok(ls.contains(&rs))
  1526→            }
  1527→            Condition::Compare { left, op, right } => {
  1528→                let lv = self.eval_expr_with_env(left, env)?;
  1529→                let rv = self.eval_expr_with_env(right, env)?;
  1530→                let lf = lv.as_float()?;
  1531→                let rf = rv.as_float()?;
  1532→                Ok(match op {
  1533→                    CompareOp::Gt => lf > rf,
  1534→                    CompareOp::Lt => lf < rf,
  1535→                    CompareOp::Ge => lf >= rf,
  1536→                    CompareOp::Le => lf <= rf,
  1537→                    CompareOp::Eq => lf == rf,
  1538→                    CompareOp::Ne => lf != rf,
  1539→                })
  1540→            }
  1541→        }
  1542→    }
  1543→
  1544→    /// ADR-0056: Save a checkpoint for a flow at a given pipeline step.
  1545→    fn save_checkpoint(
  1546→        &self,
  1547→        flow_name: &str,
  1548→        checkpoint_name: &str,
  1549→        step_index: usize,
  1550→        current_value: &Value,
  1551→    ) -> Result<(), String> {
  1552→        let ts = SystemTime::now()
  1553→            .duration_since(UNIX_EPOCH)
  1554→            .map(|d| d.as_millis() as i64)
  1555→            .unwrap_or(0);
  1556→
  1557→        let data = CheckpointData {
  1558→            flow_name: flow_name.to_string(),
  1559→            checkpoint_name: checkpoint_name.to_string(),
  1560→            step_index,
  1561→            current_value: current_value.clone(),
  1562→            variables: self.variables.clone(),
  1563→            created_at: ts,
  1564→        };
  1565→
  1566→        let state_json = serde_json::to_string(&data)
  1567→            .map_err(|e| format!("checkpoint serialization error: {}", e))?;
  1568→
  1569→        // Try SQLite first
  1570→        if let Some(ref conn) = *self.checkpoint_db.lock().map_err(|e| format!("checkpoint lock: {}", e))? {
  1571→            conn.execute(
  1572→                "INSERT OR REPLACE INTO checkpoints (flow_name, checkpoint_name, step_index, state_json, created_at) VALUES (?1, ?2, ?3, ?4, ?5)",
  1573→                rusqlite::params![flow_name, checkpoint_name, step_index as i64, state_json, ts],
  1574→            ).map_err(|e| format!("checkpoint save error: {}", e))?;
  1575→        } else {
  1576→            // Fallback: in-memory
  1577→            let key = format!("{}:{}", flow_name, checkpoint_name);
  1578→            self.checkpoint_mem.lock().map_err(|e| format!("checkpoint lock: {}", e))?.insert(key, data);
  1579→        }
  1580→
  1581→        Ok(())
  1582→    }

  1584→    /// ADR-0056: Load a checkpoint for a flow. Returns None if not found.
  1585→    fn load_checkpoint(&self, flow_name: &str, checkpoint_name: &str) -> Result<Option<CheckpointData>, String> {
  1586→        // Try SQLite first
  1587→        if let Some(ref conn) = *self.checkpoint_db.lock().map_err(|e| format!("checkpoint lock: {}", e))? {
  1588→            let mut stmt = conn.prepare(
  1589→                "SELECT state_json FROM checkpoints WHERE flow_name = ?1 AND checkpoint_name = ?2"
  1590→            ).map_err(|e| format!("checkpoint load error: {}", e))?;
  1591→
  1592→            let result: Result<String, _> = stmt.query_row(
  1593→                rusqlite::params![flow_name, checkpoint_name],
  1594→                |row| row.get(0),
  1595→            );
  1596→
  1597→            match result {
  1598→                Ok(state_json) => {
  1599→                    let data: CheckpointData = serde_json::from_str(&state_json)
  1600→                        .map_err(|e| format!("checkpoint deserialization error: {}", e))?;
  1601→                    Ok(Some(data))
  1602→                }
  1603→                Err(rusqlite::Error::QueryReturnedNoRows) => Ok(None),
  1604→                Err(e) => Err(format!("checkpoint load error: {}", e)),
  1605→            }
  1606→        } else {
  1607→            // Fallback: in-memory
  1608→            let key = format!("{}:{}", flow_name, checkpoint_name);
  1609→            Ok(self.checkpoint_mem.lock().map_err(|e| format!("checkpoint lock: {}", e))?.get(&key).cloned())
  1610→        }
  1611→    }

  1668→    /// Execute a flow: evaluate source, thread through pipeline steps.
  1669→    /// ADR-0056: After each step, check if a checkpoint follows. If so, save state.
  1670→    /// If resume_target is set, skip steps until we reach the checkpoint, then resume.
  1671→    fn run_flow(&mut self, flow: &FlowDecl) -> Result<String, String> {
  1672→        // Register branch definitions for this flow
  1673→        self.branch_defs.clear();
  1674→        for (step_name, branches) in &flow.branch_defs {
  1675→            self.branch_defs.insert(step_name.clone(), branches.clone());
  1676→        }
  1677→
  1678→        // ADR-0056: Determine resume start position
  1679→        let mut start_idx: usize = 0;
  1680→        let mut current: Option<Value> = None;
  1681→
  1682→        if let Some((ref target_flow, ref target_cp)) = self.resume_target {
  1683→            if target_flow == &flow.name {
  1684→                // Try to load checkpoint
  1685→                if let Some(data) = self.load_checkpoint(&flow.name, target_cp)? {
  1686→                    // Restore variables from checkpoint
  1687→                    for (k, v) in data.variables {
  1688→                        self.variables.insert(k, v);
  1689→                    }
  1690→                    // Start from the step AFTER the checkpoint
  1691→                    start_idx = data.step_index + 1;
  1692→                    current = Some(data.current_value);
  1693→                } else {
  1694→                    return Err(format!("checkpoint '{}' not found for flow '{}'", target_cp, flow.name));
  1695→                }
  1696→                // Clear resume target (one-shot)
  1697→                self.resume_target = None;
  1698→            }
  1699→        }
  1700→
  1701→        // If no resume, evaluate the source expression
  1702→        let mut current = current.unwrap_or_else(|| self.eval_expr(&flow.source).unwrap_or(Value::Unit));
  1703→
  1704→        // ADR-0056: Build reverse map: step_index -> checkpoint names at that position
  1705→        let mut checkpoint_at: HashMap<usize, Vec<String>> = HashMap::new();
  1706→        for (cp_name, &step_idx) in &flow.checkpoints {
  1707→            checkpoint_at.entry(step_idx).or_default().push(cp_name.clone());
  1708→        }
  1709→
  1710→        // Execute pipeline steps, starting from start_idx (0 for fresh run)
  1711→        for (i, step_name) in flow.pipeline.iter().enumerate() {
  1712→            if i < start_idx {
  1713→                continue; // Skip steps before resume point
  1714→            }
  1715→            current = self.run_flow_step(step_name, current)?;
  1716→
  1717→            // Check if a checkpoint follows this step
  1718→            if let Some(cp_names) = checkpoint_at.get(&i) {
  1719→                for cp_name in cp_names {
  1720→                    self.save_checkpoint(&flow.name, cp_name, i, &current)?;
  1721→                }
  1722→            }
  1723→        }
  1724→
  1725→        Ok(format!("{}", current))
  1726→    }
  1727→
  1728→    /// Execute a single flow step: check branch_defs first, else invoke as pattern/builtin.
  1729→    fn run_flow_step(&mut self, step_name: &str, current: Value) -> Result<Value, String> {
  1730→        if let Some(branches) = self.branch_defs.get(step_name).cloned() {
  1731→            // Step has branch definitions — evaluate conditions against current value
  1732→            for branch in &branches {
  1733→                if self.eval_branch_condition(&branch.condition, &current)? {
  1734→                    return self.invoke(&branch.target, vec![current]);
  1735→                }
  1736→            }
  1737→            Err(format!("no branch matched in step '{}'", step_name))
  1738→        } else {
  1739→            // No branch definitions — invoke as pattern/builtin directly
  1740→            self.invoke(step_name, vec![current])
  1741→        }
  1742→    }
  1743→
  1744→    /// Evaluate a branch condition: `target.field op threshold`
  1745→    fn eval_branch_condition(&self, cond: &BranchCondition, current: &Value) -> Result<bool, String> {
  1746→        // The target in branch_condition is the flow input value (current)
  1747→        let field_val = current.get_field(&cond.field)
  1748→            .map_err(|e| format!("branch condition: {}", e))?
  1749→            .clone();
  1750→        let threshold = self.eval_expr(&cond.threshold)?;
  1751→        let fv = field_val.as_float()?;
  1752→        let tv = threshold.as_float()?;
  1753→        Ok(match cond.op {
  1754→            CompareOp::Gt => fv > tv,
  1755→            CompareOp::Lt => fv < tv,
  1756→            CompareOp::Ge => fv >= tv,
  1757→            CompareOp::Le => fv <= tv,
  1758→            CompareOp::Eq => fv == tv,
  1759→            CompareOp::Ne => fv != tv,
  1760→        })
  1761→    }

  1763→    /// Invoke a pattern or built-in by name with given arguments.
  1764→    /// ADR-0045: Hooks (before_pattern / after_pattern) fire around pattern
  1765→    /// and learnable pattern invocations, but NOT around builtin calls.
  1766→    fn invoke(&mut self, name: &str, args: Vec<Value>) -> Result<Value, String> {
  1767→        // Check recall (memory) first — it's a built-in with memory access
  1768→        if name == "recall" {
  1769→            return self.invoke_recall(args);
  1770→        }
  1771→
  1772→        // Check find (entity store query) — needs access to interpreter state
  1773→        if name == "find" {
  1774→            return self.invoke_find(args);
  1775→        }
  1776→
  1777→        // Check learnable patterns
  1778→        if let Some(learnable) = self.learnable_patterns.get(name).cloned() {
  1779→            let collapsed_args = self.collapse_args(&learnable.params, &args);
  1780→            let learnable_clone = learnable.clone();
  1781→            return self.invoke_pattern_with_hooks(name, &collapsed_args, || {
  1782→                self.invoke_learnable_with_env(name, &learnable_clone, &collapsed_args)
  1783→            });
  1784→        }
  1785→
  1786→        // Check builtins
  1787→        if let Some(builtin_fn) = self.builtins.get(name) {
  1788→            // Phase 7.5: Sandbox enforcement — filesystem isolation
  1789→            if let Some(ref sb) = self.active_sandbox {
  1790→                if sb.forbidden.iter().any(|f| f == "filesystem") {
  1791→                    if matches!(name,
  1792→                        "read_file" | "write_file" | "append_file"
  1793→                        | "delete_file" | "file_exists" | "list_dir"
  1794→                    ) {
  1795→                        return Err(format!(
  1796→                            "filesystem access forbidden in sandbox '{}'",
  1797→                            sb.name
  1798→                        ));
  1799→                    }
  1800→                }
  1801→            }
  1802→            return builtin_fn(&args);
  1803→        }
  1804→
  1805→        // Look up compiled pattern
  1806→        let pattern = match self.patterns.get(name) {
  1807→            Some(p) => p.clone(),
  1808→            None => return Err(format!("undefined pattern or builtin: {}", name)),
  1809→        };
  1810→
  1811→        if args.len() != pattern.params.len() {
  1812→            return Err(format!(
  1813→                "pattern {} expects {} arguments, got {}",
  1814→                name,
  1815→                pattern.params.len(),
  1816→                args.len()
  1817→            ));
  1818→        }
  1819→
  1820→        // Bind parameters with Fluid collapse
  1821→        let mut local_env = self.bind_and_collapse(&pattern.params, &args)?;
  1822→
  1823→        self.invoke_pattern_with_hooks(name, &args, || {
  1824→            self.eval_statements(&pattern.body, &mut local_env)
  1825→        })
  1826→    }
  1827→
  1828→    /// ADR-0045: Execute a pattern invocation wrapped with before/after hooks.
  1829→    /// Injects hook variables: pattern_name (String), args (List),
  1830→    /// result (after only), confidence (after only).
  1831→    /// Builtins are NOT wrapped — only user-defined patterns and learnable patterns.
  1832→    fn invoke_pattern_with_hooks<F>(
  1833→        &self,
  1834→        name: &str,
  1835→        args: &[Value],
  1836→        f: F,
  1837→    ) -> Result<Value, String>
  1838→    where
  1839→        F: FnOnce() -> Result<Value, String>,
  1840→    {
  1841→        // ADR-0051: Track stats for regular (non-learnable) patterns.
  1842→        // Learnable patterns track themselves in invoke_learnable_with_env.
  1843→        if !self.learnable_patterns.contains_key(name) {
  1844→            self.record_pattern_call(name, false);
  1845→        }
  1846→
  1847→        // Execute all before_pattern hooks
  1848→        if !self.hooks_before.is_empty() {
  1849→            let mut hook_env = HashMap::new();
  1850→            hook_env.insert("pattern_name".to_string(), Value::String(name.to_string()));
  1851→            hook_env.insert("args".to_string(), Value::List(args.to_vec()));
  1852→            for hook in &self.hooks_before {
  1853→                // Ignore hook errors — hooks are advisory, not blocking
  1854→                let _ = self.eval_statements(&hook.body, &mut hook_env);
  1855→            }
  1856→        }
  1857→
  1858→        // Execute the actual pattern
  1859→        let result = f();
  1860→
  1861→        // Execute all after_pattern hooks
  1862→        if !self.hooks_after.is_empty() {
  1863→            let mut hook_env = HashMap::new();
  1864→            hook_env.insert("pattern_name".to_string(), Value::String(name.to_string()));
  1865→            hook_env.insert("args".to_string(), Value::List(args.to_vec()));
  1866→            match &result {
  1867→                Ok(val) => {
  1868→                    hook_env.insert("result".to_string(), val.clone());
  1869→                    // Extract confidence for Fluid results, default 1.0
  1870→                    let conf = match val {
  1871→                        Value::Fluid(variants) => {
  1872→                            variants.iter().map(|v| v.confidence).fold(0.0_f64, f64::max)
  1873→                        }
  1874→                        _ => 1.0,
  1875→                    };
  1876→                    hook_env.insert("confidence".to_string(), Value::Float(conf));
  1877→                }
  1878→                Err(e) => {
  1879→                    hook_env.insert("result".to_string(), Value::String(e.clone()));
  1880→                    hook_env.insert("confidence".to_string(), Value::Float(0.0));
  1881→                }
  1882→            }
  1883→            for hook in &self.hooks_after {
  1884→                // Ignore hook errors — hooks are advisory, not blocking
  1885→                let _ = self.eval_statements(&hook.body, &mut hook_env);
  1886→            }
  1887→        }
  1888→
  1889→        result
  1890→    }
  1891→
  1892→    /// ADR-0046: Build the effective system prompt for a learnable pattern.
  1893→    /// Handles all ContextMode variants:
  1894→    /// - None → return base prompt unchanged
  1895→    /// - Auto → recall(first_param_value, limit=5) + prepend context
  1896→    /// - Recall(query_expr, limit) → evaluate query, recall, prepend context
  1897→    /// - Literal(string) → prepend static text as context
  1898→    ///
  1899→    /// ADR-0055: Context compression.
  1900→    /// When context_strategy is Compress and the recalled context exceeds
  1901→    /// max_context_tokens estimated tokens, the facts are compressed via LLM
  1902→    /// summarization before being prepended to the prompt.
  1903→    fn build_effective_prompt(&self, learnable: &CompiledLearnable, args: &[Value]) -> String {
  1904→        let context_mode = match &learnable.context {
  1905→            None => return learnable.prompt.clone(),
  1906→            Some(ContextMode::None) => return learnable.prompt.clone(),
  1907→            Some(mode) => mode.clone(),
  1908→        };
  1909→
  1910→        match context_mode {
  1911→            ContextMode::None | ContextMode::Auto | ContextMode::Recall(_, _) => {
  1912→                // Determine the query string and limit for recall
  1913→                let (query, limit) = match &learnable.context {
  1914→                    Some(ContextMode::Auto) => {
  1915→                        // Use first parameter's runtime value as query
  1916→                        let query_str = if !args.is_empty() {
  1917→                            match &args[0] {
  1918→                                Value::String(s) => s.clone(),
  1919→                                other => format!("{}", other),
  1920→                            }
  1921→                        } else {
  1922→                            return learnable.prompt.clone();
  1923→                        };
  1924→                        (query_str, 5)
  1925→                    }
  1926→                    Some(ContextMode::Recall(query_expr, limit_opt)) => {
  1927→                        // Evaluate the context query expression with param names bound to args
  1928→                        let mut env: HashMap<String, Value> = HashMap::new();
  1929→                        for (i, param) in learnable.params.iter().enumerate() {
  1930→                            if i < args.len() {
  1931→                                env.insert(param.name.clone(), args[i].clone());
  1932→                            }
  1933→                        }
  1934→                        let query = match self.eval_expr_with_env(query_expr, &mut env) {
  1935→                            Ok(Value::String(s)) => s,
  1936→                            Ok(other) => format!("{}", other),
  1937→                            Err(_) => return learnable.prompt.clone(), // context eval failed → skip
  1938→                        };
  1939→                        (query, limit_opt.unwrap_or(5))
  1940→                    }
  1941→                    _ => unreachable!(),
  1942→                };
  1943→
  1944→                // Recall memories: collect up to `limit` unique results
  1945→                let mut facts = Vec::new();
  1946→                let query_embedding = self.embedding_manager.embed(&query).unwrap_or_default();
  1947→                let mut seen = std::collections::HashSet::new();
  1948→                let min_conf = 0.1_f32;
  1949→
  1950→                match self.memory.lock() {
  1951→                    Ok(store) => {
  1952→                        let all = store.all_entries();
  1953→                        let mut scored: Vec<(String, f32)> = Vec::new();
  1954→                        for entry in all {
  1955→                            if seen.contains(&entry.value) {
  1956→                                continue;
  1957→                            }
  1958→                            let score = if entry.embedding.is_empty() || query_embedding.is_empty() {
  1959→                                if entry.value.to_lowercase().contains(&query.to_lowercase()) {
  1960→                                    0.5
  1961→                                } else {
  1962→                                    0.0
  1963→                                }
  1964→                            } else {
  1965→                                cosine_similarity(&query_embedding, &entry.embedding) as f32
  1966→                                    * entry.confidence as f32
  1967→                                    * entry.priority as f32
  1968→                            };
  1969→                            if score >= min_conf {
  1970→                                seen.insert(entry.value.clone());
  1971→                                scored.push((entry.value, score));
  1972→                            }
  1973→                        }
  1974→                        scored.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal));
  1975→                        for (fact, _score) in scored.iter().take(limit) {
  1976→                            facts.push(fact.clone());
  1977→                        }
  1978→                    }
  1979→                    Err(_) => return learnable.prompt.clone(),
  1980→                }
  1981→
  1982→                if facts.is_empty() {
  1983→                    return learnable.prompt.clone();
  1984→                }
  1985→
  1986→                // Format context block
  1987→                let context_block = self.format_context_block(&facts);
  1988→
  1989→                // ADR-0055: Apply compression strategy
  1990→                if learnable.context_strategy == ContextStrategy::Compress {
  1991→                    let estimated_tokens = Self::estimate_tokens(&context_block);

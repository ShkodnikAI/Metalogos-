    27→/// Opaque secret string with automatic memory zeroing on drop (Phase 7.3).
    28→/// Implements serde by serializing as "[SECRET]" marker — actual value is NEVER persisted.
    29→#[derive(Clone, Debug)]
    30→pub struct SecretString(pub Zeroizing<String>);
    31→
    32→impl serde::Serialize for SecretString {
    33→    fn serialize<S: serde::Serializer>(&self, s: S) -> Result<S::Ok, S::Error> {
    34→        // Never serialize the actual secret — emit a safe marker
    35→        s.serialize_str("[SECRET]")
    36→    }
    37→}
    38→
    39→impl<'de> serde::Deserialize<'de> for SecretString {
    40→    fn deserialize<D: serde::Deserializer<'de>>(d: D) -> Result<Self, D::Error> {
    41→        let inner = String::deserialize(d)?;
    42→        Ok(SecretString(Zeroizing::new(inner)))
    43→    }
    44→}

    46→impl std::ops::Deref for SecretString {
    47→    type Target = String;
    48→    fn deref(&self) -> &Self::Target {
    49→        &self.0
    50→    }
    51→}

    62→/// Runtime value.
    63→#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
    64→pub enum Value {
    65→    String(String),
    66→    Float(f64),
    67→    Bool(bool),
    68→    Struct {
    69→        type_name: String,
    70→        fields: HashMap<String, Value>,
    71→    },
    72→    /// List value: ordered collection of items.
    73→    List(Vec<Value>),
    74→    /// Fluid value: superposition of typed variants with confidence scores.
    75→    /// Collapses lazily at point of use (see `maybe_collapse`).
    76→    Fluid(Vec<FluidValueVariant>),
    77→    Unit,
    78→    /// Opaque HTML content (Phase 6.2) — cannot be concatenated, printed, or converted to String
    79→    Html(String),
    80→    /// Opaque SQL query (Phase 6.3) — only created via query() builtin
    81→    Query(String),
    82→    /// Opaque secret value (Phase 6.4) — cannot be printed or converted to String.
    83→    /// Phase 7.3: Internally uses SecretString (Zeroizing<String>) — memory is zeroed on drop.
    84→    Secret(SecretString),
    85→    /// Opaque encrypted data (Phase 6.4)
    86→    Encrypted(Vec<u8>),
    87→    /// Opaque password hash (Phase 6.4)
    88→    Hash(String),
    89→    /// Opaque session data (Phase 6.5)
    90→    Session(std::collections::HashMap<String, String>),
    91→    /// HTTP response value (Phase 6.1)
    92→    HttpResponse { status: u16, body: String },
    93→}

    95→impl std::fmt::Display for Value {
    96→    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
    97→        match self {
    98→            Value::String(s) => write!(f, "{}", s),
    99→            Value::Float(n) => write!(f, "{}", n),
   100→            Value::Bool(b) => write!(f, "{}", b),
   101→            Value::Struct { type_name, fields } => {
   102→                write!(f, "{} {{", type_name)?;
   103→                let pairs: Vec<_> = fields.iter().collect();
   104→                for (i, (k, v)) in pairs.iter().enumerate() {
   105→                    if i > 0 { write!(f, ", ")?; }
   106→                    write!(f, "{}: {}", k, v)?;
   107→                }
   108→                write!(f, "}}")
   109→            }
   110→            Value::List(items) => {
   111→                write!(f, "[")?;
   112→                for (i, item) in items.iter().enumerate() {
   113→                    if i > 0 { write!(f, ", ")?; }
   114→                    write!(f, "{}", item)?;
   115→                }
   116→                write!(f, "]")
   117→            }
   118→            Value::Fluid(variants) => {
   119→                // Display as the highest-confidence variant
   120→                let best = variants.iter().max_by(|a, b| {
   121→                    a.confidence.partial_cmp(&b.confidence).unwrap_or(std::cmp::Ordering::Equal)
   122→                });
   123→                match best {
   124→                    Some(v) => write!(f, "{}", v.value),
   125→                    None => write!(f, "()"),
   126→                }
   127→            }
   128→            Value::Unit => write!(f, "()"),
   129→            Value::Html(_) => write!(f, "[Html]"),
   130→            Value::Query(_) => write!(f, "[Query]"),
   131→            Value::Secret(_) => write!(f, "[Secret]"),
   132→            Value::Encrypted(_) => write!(f, "[Encrypted]"),
   133→            Value::Hash(_) => write!(f, "[Hash]"),
   134→            Value::Session(_) => write!(f, "[Session]"),
   135→            Value::HttpResponse { status, .. } => write!(f, "[HttpResponse {}]", status),
   136→        }
   137→    }
   138→}

   140→impl Value {
   141→    pub fn type_name(&self) -> &'static str {
   142→        match self {
   143→            Value::String(_) => "String",
   144→            Value::Float(_) => "Float",
   145→            Value::Bool(_) => "Bool",
   146→            Value::List(_) => "List",
   147→            Value::Struct { .. } => "Struct",
   148→            Value::Fluid(_) => "Fluid",
   149→            Value::Unit => "Unit",
   150→            Value::Html(_) => "Html",
   151→            Value::Query(_) => "Query",
   152→            Value::Secret(_) => "Secret",
   153→            Value::Encrypted(_) => "Encrypted",
   154→            Value::Hash(_) => "Hash",
   155→            Value::Session(_) => "Session",
   156→            Value::HttpResponse { .. } => "HttpResponse",
   157→        }
   158→    }
   159→
   160→    /// Get a field value from a struct. Returns Err if not a struct or field missing.
   161→    pub fn get_field(&self, field: &str) -> Result<&Value, String> {
   162→        match self {
   163→            Value::Struct { fields, .. } => fields.get(field)
   164→                .ok_or_else(|| format!("field '{}' not found on struct", field)),
   165→            _ => Err(format!("cannot access field '{}' on non-struct value ({})", field, self.type_name())),
   166→        }
   167→    }
   168→
   169→    /// Set a field value on a mutable struct.
   170→    pub fn set_field(&mut self, field: &str, value: Value) -> Result<(), String> {
   171→        match self {
   172→            Value::Struct { fields, .. } => {
   173→                if fields.contains_key(field) {
   174→                    fields.insert(field.to_string(), value);
   175→                    Ok(())
   176→                } else {
   177→                    Err(format!("field '{}' not found on struct", field))
   178→                }
   179→            }
   180→            _ => Err(format!("cannot set field '{}' on non-struct value", field)),
   181→        }
   182→    }
   183→
   184→    /// Convert to f64 for numeric comparisons.
   185→    pub fn as_float(&self) -> Result<f64, String> {
   186→        match self {
   187→            Value::Float(f) => Ok(*f),
   188→            Value::Bool(b) => Ok(if *b { 1.0 } else { 0.0 }),
   189→            Value::String(s) => s.parse::<f64>()
   190→                .map_err(|_| format!("cannot convert '{}' to Float", s)),
   191→            _ => Err(format!("cannot convert {} to Float", self.type_name())),
   192→        }
   193→    }
   194→
   195→    /// Convert to bool for condition checking.
   196→    pub fn as_bool(&self) -> Result<bool, String> {
   197→        match self {
   198→            Value::Bool(b) => Ok(*b),
   199→            Value::Float(f) => Ok(*f != 0.0),
   200→            Value::String(s) => Ok(!s.is_empty()),
   201→            Value::Unit => Ok(false),
   202→            _ => Err(format!("cannot convert {} to Bool", self.type_name())),
   203→        }
   204→    }
   205→
   206→    /// Check if this value is a Fluid superposition.
   207→    pub fn is_fluid(&self) -> bool {
   208→        matches!(self, Value::Fluid(_))
   209→    }
   210→}

   426→impl EvalResult {
   427→    /// Format the eval result as a human-readable report.
   428→    pub fn format_report(&self) -> String {
   429→        let mut lines = Vec::new();
   430→        lines.push(format!("Eval: {}", self.pattern_name));
   431→        lines.push(format!("  Dataset: {} examples", self.total));
   432→        lines.push(format!("  Accuracy: {:.1}% ({}/{})", self.accuracy * 100.0, self.correct, self.total));
   433→        lines.push(format!("  Threshold: {}", self.threshold));
   434→        lines.push(format!("  Result: {}", if self.passed { "PASS" } else { "FAIL (below threshold)" }));
   435→
   436→        // Confusion matrix
   437→        if !self.confusion.is_empty() {
   438→            // Collect all unique labels
   439→            let mut all_labels: std::collections::BTreeSet<String> = std::collections::BTreeSet::new();
   440→            for (expected, predictions) in &self.confusion {
   441→                all_labels.insert(expected.clone());
   442→                for pred in predictions.keys() {
   443→                    all_labels.insert(pred.clone());
   444→                }
   445→            }
   446→
   447→            let labels: Vec<&String> = all_labels.iter().collect();
   448→
   449→            // Header row
   450→            let header = format!("  {:12} {}", "", labels.iter().map(|l| format!("{:>12}", l)).collect::<Vec<_>>().join(" "));
   451→            lines.push(header);
   452→
   453→            // Data rows
   454→            for expected in &labels {
   455→                let row = format!("  {:12} {}", expected,
   456→                    labels.iter().map(|pred| {
   457→                        let count = self.confusion.get(*expected)
   458→                            .and_then(|m| m.get(*pred))
   459→                            .copied()
   460→                            .unwrap_or(0);
   461→                        format!("{:>12}", count)
   462→                    }).collect::<Vec<_>>().join(" ")
   463→                );
   464→                lines.push(row);
   465→            }
   466→        }
   467→
   468→        // Failing examples with adapt suggestions
   469→        if !self.failures.is_empty() {
   470→            lines.push(String::new());
   471→            lines.push("  Failing examples (suggest adapt):".to_string());
   472→            for (input, expected, actual) in &self.failures {
   473→                lines.push(format!("    - {:?} -> expected {:?}, got {:?}", input, expected, actual));
   474→            }
   475→            // Generate adapt suggestions
   476→            lines.push(String::new());
   477→            lines.push("  Suggested adapt commands:".to_string());
   478→            for (input, expected, _actual) in &self.failures {
   479→                lines.push(format!("    adapt {} add_example({:?}, {:?})", self.pattern_name, input, expected));
   480→            }
   481→        }
   482→
   483→        lines.join("\n")
   484→    }
   485→}

   591→/// Control flow signal for loop constructs (Наряд №17).
   592→/// Break/Continue propagate through eval_statements without being confused
   593→/// with Return values. The interpreter uses Result<ControlFlow, String>
   594→/// internally for loop bodies, then converts back to Result<Value, String>
   595→/// at the public eval_statements boundary.
   596→#[derive(Debug, Clone)]
   597→pub(crate) enum ControlFlow {
   598→    /// Normal execution, optionally carrying a value (like implicit return).
   599→    ContinueNormal(Value),
   600→    /// `break` — exit the innermost loop.
   601→    Break,
   602→    /// `continue` — skip to next iteration of the innermost loop.
   603→    ContinueLoop,
   604→    /// `return expr` — early return from a pattern/function.
   605→    Return(Value),
   606→}
   607→
   608→impl ControlFlow {
   609→    fn is_break(&self) -> bool { matches!(self, ControlFlow::Break) }
   610→    fn is_continue(&self) -> bool { matches!(self, ControlFlow::ContinueLoop) }
   611→    fn is_return(&self) -> bool { matches!(self, ControlFlow::Return(_)) }
   612→    /// Extract the inner value if this is ContinueNormal or Return.
   613→    fn into_value(self) -> Value {
   614→        match self {
   615→            ControlFlow::ContinueNormal(v) | ControlFlow::Return(v) => v,
   616→            ControlFlow::Break | ControlFlow::ContinueLoop => Value::Unit,
   617→        }
   618→    }
   619→}

   786→    /// Configure memory persistence (Phase 7.6).
   787→    /// If persist path is provided, switches to SQLite-backed stores.
   788→    /// The in-memory data is migrated to SQLite during the switch.
   789→    pub fn configure_memory(&mut self, config: &MemoryDecl) {
   790→        if let Some(ref path) = config.persist {
   791→            // Switch to SQLite backend
   792→            let db_path = std::path::PathBuf::from(path);
   793→            match SqliteStore::open(&db_path) {
   794→                Ok(sqlite_store) => {
   795→                    // Migrate existing in-memory data to SQLite
   796→                    let existing = self.memory.lock().unwrap().all_entries();
   797→                    let mut new_store: Box<dyn MemoryStore> = Box::new(sqlite_store);
   798→                    for entry in existing {
   799→                        let _ = new_store.memorize(entry);
   800→                    }
   801→                    self.memory = std::sync::Mutex::new(new_store);
   802→
   803→                    // Migrate KG edges to SQLite (sharing the same DB file)
   804→                    let existing_edges: Vec<(String, String, String, f64)> =
   805→                        self.kg.lock().unwrap().all_edges();
   806→                    if let Ok(sqlite_kg) = SqliteKg::open(&db_path) {
   807→                        let mut new_kg: Box<dyn KgStore> = Box::new(sqlite_kg);
   808→                        for (from, to, relation, weight) in existing_edges {
   809→                            let _ = new_kg.relate(&from, &to, &relation, weight);
   810→                        }
   811→                        self.kg = std::sync::Mutex::new(new_kg);
   812→                    } else {
   813→                        eprintln!("[memory] KG migration to SQLite failed; keeping in-memory KG");
   814→                    }
   815→                    eprintln!("[memory] Persistence enabled: {}", path);
   816→                    self.memory_persist_path = Some(path.clone());
   817→
   818→                    // Наряд №6 — also enable KV store SQLite persistence
   819→                    if let Err(e) = crate::builtins::init_kv_persist(path) {
   820→                        eprintln!("[kv_store] Failed to enable KV persistence: {}. KV will be in-memory only.", e);
   821→                    }
   822→
   823→                    // ADR-0056: initialize checkpoint SQLite (same DB directory)
   824→                    let cp_path = std::path::PathBuf::from(path).with_file_name("checkpoints.db");
   825→                    if let Ok(conn) = rusqlite::Connection::open(&cp_path) {
   826→                        let _ = conn.execute_batch(
   827→                            "CREATE TABLE IF NOT EXISTS checkpoints (
   828→                                flow_name TEXT NOT NULL,
   829→                                checkpoint_name TEXT NOT NULL,
   830→                                step_index INTEGER NOT NULL,
   831→                                state_json TEXT NOT NULL,
   832→                                created_at INTEGER NOT NULL,
   833→                                PRIMARY KEY (flow_name, checkpoint_name)
   834→                            )"
   835→                        );
   836→                        *self.checkpoint_db.lock().unwrap() = Some(conn);
   837→                    }
   838→                }
   839→                Err(e) => {
   840→                    eprintln!("[memory] Failed to open persistent store '{}': {}. Using in-memory.", path, e);
   841→                }
   842→            }
   843→        }
   844→        // If persist is None, keep the default InMemoryStore (already set in new())
   845→    }

   847→    /// Initialize SQLite connection for db { url: "..." } block (Наряд №7).
   848→    /// Supports "sqlite::memory:" for in-memory databases and file paths.
   849→    fn init_db_connection(&mut self, db: &DbDecl) {
   850→        let url_expr = match &db.url {
   851→            Some(expr) => expr,
   852→            None => {
   853→                eprintln!("[db] No url specified in db {{}} block — query() will be unavailable");
   854→                return;
   855→            }
   856→        };
   857→        // Evaluate the url expression (must be a string literal or variable)
   858→        let url = match self.eval_expr(url_expr) {
   859→            Ok(Value::String(s)) => s,
   860→            Ok(other) => {
   861→                eprintln!("[db] url must be a String, got {}", other.type_name());
   862→                return;
   863→            }
   864→            Err(e) => {
   865→                eprintln!("[db] Failed to evaluate url: {}", e);
   866→                return;
   867→            }
   868→        };
   869→        // Parse the URL: "sqlite::memory:" → in-memory, "sqlite:path.db" → file
   870→        let conn = if url == "sqlite::memory:" {
   871→            rusqlite::Connection::open_in_memory()
   872→        } else if url.starts_with("sqlite:") {
   873→            let path = url.trim_start_matches("sqlite:");
   874→            rusqlite::Connection::open(path)
   875→        } else {
   876→            eprintln!("[db] Unsupported URL scheme: '{}'. Use 'sqlite::memory:' or 'sqlite:path.db'", url);
   877→            return;
   878→        };
   879→        match conn {
   880→            Ok(c) => {
   881→                // Enable WAL mode for better concurrent read performance
   882→                let _ = c.execute_batch("PRAGMA journal_mode=WAL;");
   883→                let mut guard = self.db_conn.lock().unwrap();
   884→                *guard = Some(c);
   885→                // Store resolved URL for per-request interpreter reconnection
   886→                self.db_url = Some(url.clone());
   887→                eprintln!("[db] Connected: {}", url);
   888→            }
   889→            Err(e) => {
   890→                eprintln!("[db] Failed to connect to '{}': {}", url, e);
   891→            }
   892→        }
   893→    }

   895→    /// Execute a SQL query and return readable results (Наряд №7).
   896→    /// - SELECT → List of Struct (each row = struct with column names as fields)
   897→    /// - INSERT/UPDATE/DELETE → String with affected row count
   898→    /// - PRAGMA/CREATE/etc. → String "ok"
   899→    fn invoke_query(&self, args: &[Value]) -> Result<Value, String> {
   900→        let sql = match args.first() {
   901→            Some(Value::String(s)) => s.clone(),
   902→            Some(other) => return Err(format!("query() expected String SQL, got {}", other.type_name())),
   903→            None => return Err("query() requires at least 1 argument (SQL string)".to_string()),
   904→        };
   905→        let params: Vec<String> = if args.len() > 1 {
   906→            match &args[1] {
   907→                Value::List(items) => items.iter().filter_map(|v| match v {
   908→                    Value::String(s) => Some(s.clone()),
   909→                    Value::Float(n) => Some(format!("{}", n)),
   910→                    Value::Bool(b) => Some(format!("{}", b)),
   911→                    _ => None,
   912→                }).collect(),
   913→                _ => Vec::new(),
   914→            }
   915→        } else {
   916→            Vec::new()
   917→        };
   918→
   919→        let guard = self.db_conn.lock().map_err(|e| format!("db lock error: {}", e))?;
   920→        let conn = guard.as_ref().ok_or_else(|| {
   921→            "query() error: no database connection. Declare db { url: \"sqlite::memory:\" } first.".to_string()
   922→        })?;
   923→
   924→        let sql_upper = sql.trim().to_uppercase();
   925→        if sql_upper.starts_with("SELECT") || sql_upper.starts_with("PRAGMA") {
   926→            // SELECT/PRAGMA → List of Struct
   927→            let mut stmt = conn.prepare(&sql)
   928→                .map_err(|e| format!("query() SQL error: {}", e))?;
   929→            let col_names: Vec<String> = stmt.column_names().iter().map(|s| s.to_string()).collect();
   930→            let rows: Vec<Value> = stmt.query_map(rusqlite::params_from_iter(params.iter()), |row| {
   931→                let mut fields = std::collections::HashMap::new();
   932→                for (i, col) in col_names.iter().enumerate() {
   933→                    let val: Value = match row.get_ref(i) {
   934→                        Ok(rusqlite::types::ValueRef::Null) => Value::Unit,
   935→                        Ok(rusqlite::types::ValueRef::Integer(n)) => {
   936→                            // Heuristic: if the column name suggests it's an ID or count, keep as Float
   937→                            Value::Float(n as f64)
   938→                        }
   939→                        Ok(rusqlite::types::ValueRef::Real(f)) => Value::Float(f),
   940→                        Ok(rusqlite::types::ValueRef::Text(s)) => {
   941→                            Value::String(String::from_utf8_lossy(s).to_string())
   942→                        }
   943→                        Ok(rusqlite::types::ValueRef::Blob(b)) => {
   944→                            // Encode blobs as hex strings
   945→                            Value::String(b.iter().map(|byte| format!("{:02x}", byte)).collect())
   946→                        }
   947→                        Err(_) => Value::Unit,
   948→                    };
   949→                    fields.insert(col.clone(), val);
   950→                }
   951→                Ok(Value::Struct { type_name: "Row".to_string(), fields })
   952→            }).map_err(|e| format!("query() execution error: {}", e))?
   953→            .filter_map(|r| r.ok())
   954→            .collect();
   955→            Ok(Value::List(rows))
   956→        } else {
   957→            // INSERT/UPDATE/DELETE/CREATE/ALTER/etc. → affected row count as String
   958→            let affected = conn.execute(&sql, rusqlite::params_from_iter(params.iter()))
   959→                .map_err(|e| format!("query() SQL error: {}", e))?;
   960→            Ok(Value::String(affected.to_string()))
   961→        }
   962→    }

   964→    /// Execute a SQL statement via db_execute() — returns affected row count (Наряд №7).
   965→    fn invoke_db_execute(&self, args: &[Value]) -> Result<Value, String> {
   966→        let sql = match args.first() {
   967→            Some(Value::String(s)) => s.clone(),
   968→            Some(other) => return Err(format!("db_execute() expected String SQL, got {}", other.type_name())),
   969→            None => return Err("db_execute() requires at least 1 argument (SQL string)".to_string()),

   980→    /// Open a new DB connection using stored db_url (Наряд №8).

   985→                rusqlite::Connection::open_in_memory()
   986→            } else if url.starts_with("sqlite:") {

  1004→    /// Run a complete .mlog program.
  1005→    pub fn run(&mut self, declarations: Vec<Declaration>) -> Result<Option<String>, String> {
  1006→        let mut output: Option<String> = None;
  1007→
  1008→        for decl in declarations {
  1009→            match decl {

  1036→                        .duration_since(UNIX_EPOCH)

  1045→                        decay_rate: 0.01,

  1052→                    data.insert("key_preview".to_string(), preview.to_string());
  1053→                    data.insert("priority".to_string(), m.priority.to_string());

  1055→                }
  1056→                Declaration::Forget(f) => {

  1058→                        Value::String(s) => s,
  1059→                        other => format!("{}", other),
  1060→                    };
  1061→                    let now = SystemTime::now()
  1062→                        .duration_since(UNIX_EPOCH)
  1063→                        .map(|d| d.as_secs() as i64)
  1064→                        .unwrap_or(0);
  1065→                    let cutoff = now - (f.days * 86400);
  1066→                    self.memory.lock().unwrap().forget(&query_str, cutoff);
  1067→                }
  1068→                Declaration::Pattern(p) => {
  1069→                    self.patterns.insert(
  1070→                        p.name.clone(),
  1071→                        CompiledPattern {
  1072→                            params: p.params.clone(),
  1073→                            body: p.body.clone(),
  1074→                        },
  1075→                    );
  1076→                }

  1082→                            prompt: lp.prompt.clone(),
  1083→                            few_shot: Vec::new(),
  1084→                            context: lp.context.clone(),

  1092→                        },
  1093→                    );
  1094→                }
  1095→                Declaration::Adapt(a) => {
  1096→                    let input_str = match self.eval_expr(&a.input_example)? {
  1097→                        Value::String(s) => s,
  1098→                        other => format!("{}", other),
  1099→                    };
  1100→                    let output_str = match self.eval_expr(&a.output_example)? {
  1101→                        Value::String(s) => s,
  1102→                        other => format!("{}", other),
  1103→                    };

  1105→                        learnable.few_shot.push((input_str.clone(), output_str.clone()));
  1106→                    } else {

  1108→                    }

  1227→                }
  1228→            }
  1229→        }

  1234→    /// Instantiate a struct from a type name and field initializers.
  1235→    fn instantiate_struct(&self, type_name: &str, inits: &[FieldInit]) -> Result<Value, String> {
  1236→        let struct_type = self.struct_types.get(type_name)
  1237→            .ok_or_else(|| format!("unknown struct type: {}", type_name))?
  1238→            .clone();
  1239→
  1240→        let mut fields = HashMap::new();
  1241→        for fd in &struct_type.fields {
  1242→            // Look for an initializer
  1243→            let init_val = inits.iter()
  1244→                .find(|fi| fi.name == fd.name)
  1245→                .map(|fi| self.eval_expr(&fi.value));
  1246→
  1247→            let value = match init_val {
  1248→                Some(Ok(v)) => v,
  1249→                Some(Err(e)) => return Err(e),
  1250→                None => {
  1251→                    // Use default value
  1252→                    match &fd.default {
  1253→                        Some(def_expr) => self.eval_expr(def_expr)?,
  1254→                        None => Value::Unit,
  1255→                    }
  1256→                }
  1257→            };
  1258→            fields.insert(fd.name.clone(), value);
  1259→        }
  1260→
  1261→        Ok(Value::Struct { type_name: type_name.to_string(), fields })
  1262→    }

  1264→    /// Handle an import declaration: load module, register namespace or merge globally.
  1265→    fn handle_import(&mut self, import: &ImportDecl) -> Result<(), String> {
  1266→        let module_path = &import.path;
  1267→
  1268→        // Register namespace mapping (alias or path itself for global merge)
  1269→        let alias = import.alias.as_ref().cloned().unwrap_or_else(|| module_path.clone());
  1270→        self.module_namespaces.insert(alias.clone(), module_path.clone());
  1271→
  1272→        // Load the module file and execute its declarations into this interpreter
  1273→        self.load_module(module_path)?;
  1274→
  1275→        Ok(())
  1276→    }

  1278→    /// Load a module by path, parsing and executing its declarations.
  1279→    /// Detects circular imports via loading_stack.
  1280→    fn load_module(&mut self, module_path: &str) -> Result<(), String> {
  1281→        // Circular import detection
  1282→        if self.loading_stack.contains(&module_path.to_string()) {
  1283→            return Err(format!(
  1284→                "circular import detected: {} -> {}",
  1285→                self.loading_stack.join(" -> "),
  1286→                module_path
  1287→            ));
  1288→        }
  1289→
  1290→        // Don't reload if already loaded (check if patterns from this module exist)
  1291→        // We track this via the loading_stack during the current load pass
  1292→        self.loading_stack.push(module_path.to_string());
  1293→
  1294→        let result = self.load_module_inner(module_path);
  1295→
  1296→        self.loading_stack.pop();
  1297→        result
  1298→    }

  1516→    /// Execute all rules in priority order (highest first, then declaration order).
  1517→    fn execute_rules(&mut self) -> Result<(), String> {
  1518→        // Sort by priority descending; stable sort preserves declaration order for ties
  1519→        let mut sorted_rules: Vec<&RuleDecl> = self.rules.iter().collect();
  1520→        sorted_rules.sort_by(|a, b| b.priority.cmp(&a.priority));
  1521→
  1522→        for rule in sorted_rules {
  1523→            let condition_met = self.eval_condition(&rule.condition, &self.variables)?;
  1524→            if condition_met {
  1525→                // Execute assignment: target.field = value
  1526→                let _target_val = self.eval_expr(&rule.target)?;
  1527→                let value_val = self.eval_expr(&rule.value)?;
  1528→
  1529→                // We need to mutate the entity in variables
  1530→                // The target is an Ident (entity name), field is the field name
  1531→                if let Expr::Ident(name) = &rule.target {
  1532→                    let entity = self.variables.get_mut(name)
  1533→                        .ok_or_else(|| format!("rule target '{}' not found", name))?;
  1534→                    entity.set_field(&rule.field, value_val)?;
  1535→                }
  1536→            }
  1537→        }
  1538→        Ok(())
  1539→    }
  1540→
  1541→    /// Evaluate a rule condition.
  1542→    fn eval_condition(&self, cond: &Condition, env: &HashMap<String, Value>) -> Result<bool, String> {
  1543→        match cond {
  1544→            Condition::Contains { left, right } => {
  1545→                let lv = self.eval_expr_with_env(left, env)?;
  1546→                let rv = self.eval_expr_with_env(right, env)?;
  1547→                let ls = match &lv {
  1548→                    Value::String(s) => s.clone(),
  1549→                    other => return Err(format!("contains: left must be String, got {}", other.type_name())),
  1550→                };
  1551→                let rs = match &rv {
  1552→                    Value::String(s) => s.clone(),
  1553→                    other => return Err(format!("contains: right must be String, got {}", other.type_name())),
  1554→                };
  1555→                Ok(ls.contains(&rs))
  1556→            }
  1557→            Condition::Compare { left, op, right } => {
  1558→                let lv = self.eval_expr_with_env(left, env)?;
  1559→                let rv = self.eval_expr_with_env(right, env)?;
  1560→                let lf = lv.as_float()?;
  1561→                let rf = rv.as_float()?;
  1562→                Ok(match op {
  1563→                    CompareOp::Gt => lf > rf,
  1564→                    CompareOp::Lt => lf < rf,
  1565→                    CompareOp::Ge => lf >= rf,
  1566→                    CompareOp::Le => lf <= rf,
  1567→                    CompareOp::Eq => lf == rf,
  1568→                    CompareOp::Ne => lf != rf,
  1569→                })
  1570→            }
  1571→        }
  1572→    }

  1574→    /// ADR-0056: Save a checkpoint for a flow at a given pipeline step.
  1575→    fn save_checkpoint(
  1576→        &self,
  1577→        flow_name: &str,
  1578→        checkpoint_name: &str,
  1579→        step_index: usize,
  1580→        current_value: &Value,
  1581→    ) -> Result<(), String> {
  1582→        let ts = SystemTime::now()
  1583→            .duration_since(UNIX_EPOCH)
  1584→            .map(|d| d.as_millis() as i64)
  1585→            .unwrap_or(0);
  1586→
  1587→        let data = CheckpointData {
  1588→            flow_name: flow_name.to_string(),
  1589→            checkpoint_name: checkpoint_name.to_string(),
  1590→            step_index,
  1591→            current_value: current_value.clone(),
  1592→            variables: self.variables.clone(),
  1593→            created_at: ts,
  1594→        };
  1595→
  1596→        let state_json = serde_json::to_string(&data)
  1597→            .map_err(|e| format!("checkpoint serialization error: {}", e))?;
  1598→
  1599→        // Try SQLite first
  1600→        if let Some(ref conn) = *self.checkpoint_db.lock().map_err(|e| format!("checkpoint lock: {}", e))? {
  1601→            conn.execute(
  1602→                "INSERT OR REPLACE INTO checkpoints (flow_name, checkpoint_name, step_index, state_json, created_at) VALUES (?1, ?2, ?3, ?4, ?5)",
  1603→                rusqlite::params![flow_name, checkpoint_name, step_index as i64, state_json, ts],
  1604→            ).map_err(|e| format!("checkpoint save error: {}", e))?;
  1605→        } else {
  1606→            // Fallback: in-memory
  1607→            let key = format!("{}:{}", flow_name, checkpoint_name);
  1608→            self.checkpoint_mem.lock().map_err(|e| format!("checkpoint lock: {}", e))?.insert(key, data);
  1609→        }
  1610→
  1611→        Ok(())
  1612→    }

  1614→    /// ADR-0056: Load a checkpoint for a flow. Returns None if not found.
  1615→    fn load_checkpoint(&self, flow_name: &str, checkpoint_name: &str) -> Result<Option<CheckpointData>, String> {
  1616→        // Try SQLite first
  1617→        if let Some(ref conn) = *self.checkpoint_db.lock().map_err(|e| format!("checkpoint lock: {}", e))? {
  1618→            let mut stmt = conn.prepare(
  1619→                "SELECT state_json FROM checkpoints WHERE flow_name = ?1 AND checkpoint_name = ?2"
  1620→            ).map_err(|e| format!("checkpoint load error: {}", e))?;
  1621→
  1622→            let result: Result<String, _> = stmt.query_row(
  1623→                rusqlite::params![flow_name, checkpoint_name],
  1624→                |row| row.get(0),
  1625→            );
  1626→
  1627→            match result {
  1628→                Ok(state_json) => {
  1629→                    let data: CheckpointData = serde_json::from_str(&state_json)
  1630→                        .map_err(|e| format!("checkpoint deserialization error: {}", e))?;
  1631→                    Ok(Some(data))

  1634→                Err(e) => Err(format!("checkpoint load error: {}", e)),

  1636→        } else {

  1698→    /// Execute a flow: evaluate source, thread through pipeline steps.
  1699→    /// ADR-0056: After each step, check if a checkpoint follows. If so, save state.
  1700→    /// If resume_target is set, skip steps until we reach the checkpoint, then resume.
  1701→    fn run_flow(&mut self, flow: &FlowDecl) -> Result<String, String> {
  1702→        // Register branch definitions for this flow
  1703→        self.branch_defs.clear();
  1704→        for (step_name, branches) in &flow.branch_defs {
  1705→            self.branch_defs.insert(step_name.clone(), branches.clone());
  1706→        }
  1707→
  1708→        // ADR-0056: Determine resume start position
  1709→        let mut start_idx: usize = 0;
  1710→        let mut current: Option<Value> = None;
  1711→
  1712→        if let Some((ref target_flow, ref target_cp)) = self.resume_target {
  1713→            if target_flow == &flow.name {
  1714→                // Try to load checkpoint
  1715→                if let Some(data) = self.load_checkpoint(&flow.name, target_cp)? {
  1716→                    // Restore variables from checkpoint
  1717→                    for (k, v) in data.variables {
  1718→                        self.variables.insert(k, v);
  1719→                    }
  1720→                    // Start from the step AFTER the checkpoint
  1721→                    start_idx = data.step_index + 1;
  1722→                    current = Some(data.current_value);
  1723→                } else {
  1724→                    return Err(format!("checkpoint '{}' not found for flow '{}'", target_cp, flow.name));
  1725→                }
  1726→                // Clear resume target (one-shot)
  1727→                self.resume_target = None;
  1728→            }
  1729→        }
  1730→
  1731→        // If no resume, evaluate the source expression
  1732→        let mut current = current.unwrap_or_else(|| self.eval_expr(&flow.source).unwrap_or(Value::Unit));
  1733→
  1734→        // ADR-0056: Build reverse map: step_index -> checkpoint names at that position
  1735→        let mut checkpoint_at: HashMap<usize, Vec<String>> = HashMap::new();
  1736→        for (cp_name, &step_idx) in &flow.checkpoints {
  1737→            checkpoint_at.entry(step_idx).or_default().push(cp_name.clone());
  1738→        }
  1739→
  1740→        // Execute pipeline steps, starting from start_idx (0 for fresh run)
  1741→        for (i, step_name) in flow.pipeline.iter().enumerate() {
  1742→            if i < start_idx {
  1743→                continue; // Skip steps before resume point
  1744→            }
  1745→            current = self.run_flow_step(step_name, current)?;
  1746→
  1747→            // Check if a checkpoint follows this step
  1748→            if let Some(cp_names) = checkpoint_at.get(&i) {
  1749→                for cp_name in cp_names {
  1750→                    self.save_checkpoint(&flow.name, cp_name, i, &current)?;
  1751→                }
  1752→            }
  1753→        }
  1754→
  1755→        Ok(format!("{}", current))
  1756→    }
  1757→
  1758→    /// Execute a single flow step: check branch_defs first, else invoke as pattern/builtin.
  1759→    fn run_flow_step(&mut self, step_name: &str, current: Value) -> Result<Value, String> {
  1760→        if let Some(branches) = self.branch_defs.get(step_name).cloned() {
  1761→            // Step has branch definitions — evaluate conditions against current value
  1762→            for branch in &branches {
  1763→                if self.eval_branch_condition(&branch.condition, &current)? {
  1764→                    return self.invoke(&branch.target, vec![current]);
  1765→                }
  1766→            }
  1767→            Err(format!("no branch matched in step '{}'", step_name))
  1768→        } else {
  1769→            // No branch definitions — invoke as pattern/builtin directly
  1770→            self.invoke(step_name, vec![current])
  1771→        }
  1772→    }
  1773→
  1774→    /// Evaluate a branch condition: `target.field op threshold`
  1775→    fn eval_branch_condition(&self, cond: &BranchCondition, current: &Value) -> Result<bool, String> {
  1776→        // The target in branch_condition is the flow input value (current)
  1777→        let field_val = current.get_field(&cond.field)
  1778→            .map_err(|e| format!("branch condition: {}", e))?
  1779→            .clone();
  1780→        let threshold = self.eval_expr(&cond.threshold)?;
  1781→        let fv = field_val.as_float()?;
  1782→        let tv = threshold.as_float()?;
  1783→        Ok(match cond.op {
  1784→            CompareOp::Gt => fv > tv,
  1785→            CompareOp::Lt => fv < tv,
  1786→            CompareOp::Ge => fv >= tv,
  1787→            CompareOp::Le => fv <= tv,
  1788→            CompareOp::Eq => fv == tv,
  1789→            CompareOp::Ne => fv != tv,
  1790→        })
  1791→    }

  1793→    /// Invoke a pattern or built-in by name with given arguments.
  1794→    /// ADR-0045: Hooks (before_pattern / after_pattern) fire around pattern
  1795→    /// and learnable pattern invocations, but NOT around builtin calls.
  1796→    fn invoke(&mut self, name: &str, args: Vec<Value>) -> Result<Value, String> {
  1797→        // Check recall (memory) first — it's a built-in with memory access
  1798→        if name == "recall" {
  1799→            return self.invoke_recall(args);
  1800→        }
  1801→
  1802→        // Check find (entity store query) — needs access to interpreter state
  1803→        if name == "find" {
  1804→            return self.invoke_find(args);
  1805→        }
  1806→
  1807→        // Check learnable patterns
  1808→        if let Some(learnable) = self.learnable_patterns.get(name).cloned() {
  1809→            let collapsed_args = self.collapse_args(&learnable.params, &args);
  1810→            let learnable_clone = learnable.clone();
  1811→            return self.invoke_pattern_with_hooks(name, &collapsed_args, || {
  1812→                self.invoke_learnable_with_env(name, &learnable_clone, &collapsed_args)
  1813→            });
  1814→        }
  1815→
  1816→        // Check builtins
  1817→        if let Some(builtin_fn) = self.builtins.get(name) {
  1818→            // Phase 7.5: Sandbox enforcement — filesystem isolation
  1819→            if let Some(ref sb) = self.active_sandbox {
  1820→                if sb.forbidden.iter().any(|f| f == "filesystem") {
  1821→                    if matches!(name,
  1822→                        "read_file" | "write_file" | "append_file"
  1823→                        | "delete_file" | "file_exists" | "list_dir"
  1824→                    ) {
  1825→                        return Err(format!(
  1826→                            "filesystem access forbidden in sandbox '{}'",
  1827→                            sb.name
  1828→                        ));
  1829→                    }
  1830→                }
  1831→            }
  1832→            return builtin_fn(&args);
  1833→        }
  1834→
  1835→        // Look up compiled pattern
  1836→        let pattern = match self.patterns.get(name) {
  1837→            Some(p) => p.clone(),
  1838→            None => return Err(format!("undefined pattern or builtin: {}", name)),
  1839→        };
  1840→
  1841→        if args.len() != pattern.params.len() {
  1842→            return Err(format!(
  1843→                "pattern {} expects {} arguments, got {}",
  1844→                name,
  1845→                pattern.params.len(),
  1846→                args.len()
  1847→            ));
  1848→        }
  1849→
  1850→        // Bind parameters with Fluid collapse
  1851→        let mut local_env = self.bind_and_collapse(&pattern.params, &args)?;
  1852→
  1853→        self.invoke_pattern_with_hooks(name, &args, || {
  1854→            self.eval_statements(&pattern.body, &mut local_env)
  1855→        })
  1856→    }
  1857→
  1858→    /// ADR-0045: Execute a pattern invocation wrapped with before/after hooks.
  1859→    /// Injects hook variables: pattern_name (String), args (List),
  1860→    /// result (after only), confidence (after only).
  1861→    /// Builtins are NOT wrapped — only user-defined patterns and learnable patterns.
  1862→    fn invoke_pattern_with_hooks<F>(
  1863→        &self,
  1864→        name: &str,
  1865→        args: &[Value],
  1866→        f: F,
  1867→    ) -> Result<Value, String>
  1868→    where
  1869→        F: FnOnce() -> Result<Value, String>,
  1870→    {
  1871→        // ADR-0051: Track stats for regular (non-learnable) patterns.
  1872→        // Learnable patterns track themselves in invoke_learnable_with_env.
  1873→        if !self.learnable_patterns.contains_key(name) {
  1874→            self.record_pattern_call(name, false);
  1875→        }
  1876→
  1877→        // Execute all before_pattern hooks
  1878→        if !self.hooks_before.is_empty() {
  1879→            let mut hook_env = HashMap::new();
  1880→            hook_env.insert("pattern_name".to_string(), Value::String(name.to_string()));
  1881→            hook_env.insert("args".to_string(), Value::List(args.to_vec()));
  1882→            for hook in &self.hooks_before {
  1883→                // Ignore hook errors — hooks are advisory, not blocking
  1884→                let _ = self.eval_statements(&hook.body, &mut hook_env);
  1885→            }
  1886→        }
  1887→
  1888→        // Execute the actual pattern
  1889→        let result = f();
  1890→
  1891→        // Execute all after_pattern hooks
  1892→        if !self.hooks_after.is_empty() {
  1893→            let mut hook_env = HashMap::new();
  1894→            hook_env.insert("pattern_name".to_string(), Value::String(name.to_string()));
  1895→            hook_env.insert("args".to_string(), Value::List(args.to_vec()));
  1896→            match &result {
  1897→                Ok(val) => {
  1898→                    hook_env.insert("result".to_string(), val.clone());
  1899→                    // Extract confidence for Fluid results, default 1.0
  1900→                    let conf = match val {
  1901→                        Value::Fluid(variants) => {
  1902→                            variants.iter().map(|v| v.confidence).fold(0.0_f64, f64::max)
  1903→                        }
  1904→                        _ => 1.0,
  1905→                    };
  1906→                    hook_env.insert("confidence".to_string(), Value::Float(conf));
  1907→                }
  1908→                Err(e) => {
  1909→                    hook_env.insert("result".to_string(), Value::String(e.clone()));
  1910→                    hook_env.insert("confidence".to_string(), Value::Float(0.0));
  1911→                }
  1912→            }
  1913→            for hook in &self.hooks_after {
  1914→                // Ignore hook errors — hooks are advisory, not blocking
  1915→                let _ = self.eval_statements(&hook.body, &mut hook_env);
  1916→            }
  1917→        }
  1918→
  1919→        result
  1920→    }

  1922→    /// ADR-0046: Build the effective system prompt for a learnable pattern.
  1923→    /// Handles all ContextMode variants:
  1924→    /// - None → return base prompt unchanged
  1925→    /// - Auto → recall(first_param_value, limit=5) + prepend context
  1926→    /// - Recall(query_expr, limit) → evaluate query, recall, prepend context
  1927→    /// - Literal(string) → prepend static text as context
  1928→    ///
  1929→    /// ADR-0055: Context compression.
  1930→    /// When context_strategy is Compress and the recalled context exceeds
  1931→    /// max_context_tokens estimated tokens, the facts are compressed via LLM
  1932→    /// summarization before being prepended to the prompt.
  1933→    fn build_effective_prompt(&self, learnable: &CompiledLearnable, args: &[Value]) -> String {
  1934→        let context_mode = match &learnable.context {
  1935→            None => return learnable.prompt.clone(),
  1936→            Some(ContextMode::None) => return learnable.prompt.clone(),
  1937→            Some(mode) => mode.clone(),
  1938→        };
  1939→
  1940→        match context_mode {
  1941→            ContextMode::None | ContextMode::Auto | ContextMode::Recall(_, _) => {
  1942→                // Determine the query string and limit for recall
  1943→                let (query, limit) = match &learnable.context {
  1944→                    Some(ContextMode::Auto) => {
  1945→                        // Use first parameter's runtime value as query
  1946→                        let query_str = if !args.is_empty() {
  1947→                            match &args[0] {
  1948→                                Value::String(s) => s.clone(),
  1949→                                other => format!("{}", other),
  1950→                            }
  1951→                        } else {
  1952→                            return learnable.prompt.clone();
  1953→                        };
  1954→                        (query_str, 5)
  1955→                    }
  1956→                    Some(ContextMode::Recall(query_expr, limit_opt)) => {
  1957→                        // Evaluate the context query expression with param names bound to args
  1958→                        let mut env: HashMap<String, Value> = HashMap::new();
  1959→                        for (i, param) in learnable.params.iter().enumerate() {
  1960→                            if i < args.len() {
  1961→                                env.insert(param.name.clone(), args[i].clone());
  1962→                            }
  1963→                        }
  1964→                        let query = match self.eval_expr_with_env(query_expr, &mut env) {
  1965→                            Ok(Value::String(s)) => s,
  1966→                            Ok(other) => format!("{}", other),
  1967→                            Err(_) => return learnable.prompt.clone(), // context eval failed → skip
  1968→                        };
  1969→                        (query, limit_opt.unwrap_or(5))
  1970→                    }
  1971→                    _ => unreachable!(),
  1972→                };
  1973→
  1974→                // Recall memories: collect up to `limit` unique results
  1975→                let mut facts = Vec::new();
  1976→                let query_embedding = self.embedding_manager.embed(&query).unwrap_or_default();
  1977→                let mut seen = std::collections::HashSet::new();
  1978→                let min_conf = 0.1_f32;
  1979→
  1980→                match self.memory.lock() {
  1981→                    Ok(store) => {
  1982→                        let all = store.all_entries();
  1983→                        let mut scored: Vec<(String, f32)> = Vec::new();
  1984→                        for entry in all {
  1985→                            if seen.contains(&entry.value) {
  1986→                                continue;
  1987→                            }
  1988→                            let score = if entry.embedding.is_empty() || query_embedding.is_empty() {
  1989→                                if entry.value.to_lowercase().contains(&query.to_lowercase()) {

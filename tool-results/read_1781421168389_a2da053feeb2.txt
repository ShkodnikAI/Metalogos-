     1→// ── METALOGOS Stack VM — Phase 4.1 ─────────────────────────────
     2→//
     3→// Executes a compiled bytecode Program. The VM maintains:
     4→//   - A value stack (operands for instructions)
     5→//   - A call stack (frames for pattern invocations)
     6→//   - Global variables (slots)
     7→//   - Pattern table (compiled user-defined functions)
     8→//   - Learnable table (LLM-backed patterns)
     9→//   - Memory store (memorize/recall)
    10→//   - Knowledge graph relations
    11→//   - Mutate log (messages from mutate declarations)
    12→//
    13→// Design: a single main loop that dispatches on the current instruction.
    14→// Function calls push a new frame; Return pops back.
    15→
    16→use std::collections::HashMap;
    17→use std::time::{SystemTime, UNIX_EPOCH};
    18→
    19→use crate::ast::CompareOp as AstCompareOp;
    20→use crate::bytecode::*;
    21→use crate::builtins::Builtins;
    22→use crate::interpreter::{
    23→    FluidValueVariant, Value,
    24→};
    25→use crate::llm;
    26→
    27→/// The METALOGOS stack-based virtual machine.
    28→pub struct Vm {
    29→    /// Global variable slots.
    30→    globals: Vec<Value>,
    31→    /// Global variable names (index = slot).
    32→    global_names: Vec<String>,
    33→    /// Pattern table: index → CompiledFn (modified at runtime by RegisterPattern).
    34→    patterns: Vec<CompiledFn>,
    35→    /// Learnable pattern table: index → (info, few_shot, original_few_shot).
    36→    learnables: Vec<(CompiledLearnableInfo, Vec<(String, String)>)>,
    37→    /// Built-in function registry.
    38→    builtins: Builtins,
    39→    /// Builtin name lookup table (index → name).
    40→    builtin_names: Vec<String>,
    41→    /// Memory store.
    42→    memory: Vec<VmMemoryEntry>,
    43→    /// Knowledge graph relations.
    44→    relations: Vec<VmRelation>,
    45→    /// Rule table (from program.rules).
    46→    rules: Vec<CompiledRule>,
    47→    /// Mutate log messages.
    48→    mutate_log: Vec<String>,
    49→    /// Collections loaded flag (for map/filter/reduce).
    50→    collections_loaded: bool,
    51→}
    52→
    53→/// Collapse threshold for Fluid values (matches interpreter).
    54→const COLLAPSE_THRESHOLD: f64 = 0.1;
    55→
    56→impl Vm {
    57→    /// Create a new VM with empty state.
    58→    pub fn new() -> Self {
    59→        let builtin_names = vec![
    60→            "upper".to_string(),        // 0
    61→            "lower".to_string(),        // 1
    62→            "len".to_string(),          // 2
    63→            "str".to_string(),          // 3
    64→            "print".to_string(),        // 4
    65→            "contains".to_string(),     // 5
    66→            "float".to_string(),        // 6
    67→            "__trim".to_string(),       // 7
    68→            "__replace".to_string(),    // 8
    69→            "__split".to_string(),      // 9
    70→            "__join".to_string(),       // 10
    71→            "__abs".to_string(),        // 11
    72→            "__min".to_string(),        // 12
    73→            "__max".to_string(),        // 13
    74→            "__clamp".to_string(),      // 14
    75→            "__round".to_string(),      // 15
    76→            "__first".to_string(),      // 16
    77→            "__last".to_string(),       // 17
    78→            "__push".to_string(),       // 18
    79→            "__list_len".to_string(),   // 19
    80→            "recall".to_string(),       // 20
    81→            "stdin".to_string(),        // 21
    82→            "split_tokens".to_string(), // 22
    83→            "if_eq".to_string(),        // 23
    84→            "newline".to_string(),      // 24
    85→            "is_string_token".to_string(), // 25
    86→        ];
    87→
    88→        Vm {
    89→            globals: Vec::new(),
    90→            global_names: Vec::new(),
    91→            patterns: Vec::new(),
    92→            learnables: Vec::new(),
    93→            builtins: Builtins::new(),
    94→            builtin_names,
    95→            memory: Vec::new(),
    96→            relations: Vec::new(),
    97→            rules: Vec::new(),
    98→            mutate_log: Vec::new(),
    99→            collections_loaded: false,
   100→        }
   101→    }
   102→
   103→    /// Execute a compiled program. Returns the flow output (if any),
   104→    /// with mutate log messages prepended if present.
   105→    pub fn run(&mut self, program: Program) -> Result<Option<String>, String> {
   106→        // Initialize globals
   107→        self.globals = vec![Value::Unit; program.globals.len()];
   108→        self.global_names = program.globals.clone();
   109→        self.collections_loaded = program.collections_loaded;
   110→
   111→        // Sort rules by priority descending (matches interpreter semantics)
   112→        let mut rules = program.rules.clone();
   113→        rules.sort_by(|a, b| b.priority.cmp(&a.priority));
   114→        self.rules = rules;
   115→
   116→        // Execute main_code
   117→        let mut stack: Vec<Value> = Vec::new();
   118→        let mut call_stack: Vec<CallFrame> = Vec::new();
   119→        let mut ip = 0;
   120→        let code = &program.main_code;
   121→        let mut flow_output: Option<String> = None;
   122→
   123→        while ip < code.len() {
   124→            let instr = &code[ip];
   125→            match instr {
   126→                // ── Constants & Variables ─────────────────────
   127→                Instruction::Const(v) => {
   128→                    stack.push(v.clone());
   129→                    ip += 1;
   130→                }
   131→                Instruction::LoadGlobal(slot) => {
   132→                    let val = self.globals.get(*slot)
   133→                        .cloned()
   134→                        .unwrap_or(Value::Unit);
   135→                    stack.push(val);
   136→                    ip += 1;
   137→                }
   138→                Instruction::LoadGlobalByName(name) => {
   139→                    // Search globals by name
   140→                    let val = program.globals.iter().position(|n| n == name)
   141→                        .and_then(|slot| self.globals.get(slot).cloned())
   142→                        .unwrap_or(Value::Unit);
   143→                    stack.push(val);
   144→                    ip += 1;
   145→                }
   146→                Instruction::StoreGlobal(slot) => {
   147→                    let val = stack.pop().unwrap_or(Value::Unit);
   148→                    if *slot < self.globals.len() {
   149→                        self.globals[*slot] = val;
   150→                    }
   151→                    ip += 1;
   152→                }
   153→                Instruction::LoadLocal(slot) => {
   154→                    // Local variables are stored on the stack, below the base pointer
   155→                    let bp = call_stack.last().map(|f| f.base_bp).unwrap_or(0);
   156→                    let idx = bp + slot;
   157→                    let val = stack.get(idx).cloned().unwrap_or(Value::Unit);
   158→                    stack.push(val);
   159→                    ip += 1;
   160→                }
   161→                Instruction::StoreLocal(slot) => {
   162→                    let bp = call_stack.last().map(|f| f.base_bp).unwrap_or(0);
   163→                    let idx = bp + slot;
   164→                    let val = stack.pop().unwrap_or(Value::Unit);
   165→                    if idx >= stack.len() {
   166→                        stack.resize(idx + 1, Value::Unit);
   167→                    }
   168→                    stack[idx] = val;
   169→                    ip += 1;
   170→                }
   171→
   172→                // ── Registration ──────────────────────────────
   173→                Instruction::RegisterPattern(fn_def) => {
   174→                    self.patterns.push(fn_def.clone());
   175→                    ip += 1;
   176→                }
   177→                Instruction::RegisterLearnable(info) => {
   178→                    self.learnables.push((info.clone(), Vec::new()));
   179→                    ip += 1;
   180→                }
   181→
   182→                // ── Function Calls ────────────────────────────
   183→                Instruction::CallBuiltin(idx, arity) => {
   184→                    let name = self.builtin_names.get(*idx)
   185→                        .map(|s| s.as_str())
   186→                        .unwrap_or("unknown");
   187→                    let mut args = Vec::new();
   188→                    for _ in 0..*arity {
   189→                        args.insert(0, stack.pop().unwrap_or(Value::Unit));
   190→                    }
   191→                    let result = self.call_builtin(name, &args)?;
   192→                    stack.push(result);
   193→                    ip += 1;
   194→                }
   195→                Instruction::CallPattern(idx, arity) => {
   196→                    let pattern = self.patterns.get(*idx)
   197→                        .ok_or_else(|| format!("VM: pattern index {} not found", idx))?
   198→                        .clone();
   199→
   200→                    // Check arity
   201→                    if *arity != pattern.param_count {
   202→                        return Err(format!(
   203→                            "VM: pattern {} expects {} args, got {}",
   204→                            pattern.name, pattern.param_count, arity
   205→                        ));
   206→                    }
   207→
   208→                    // ── VM bytecode path ────────────────────────
   209→                    // Pop arguments (in reverse) and bind as locals
   210→                    let mut locals = Vec::new();
   211→                    for _ in 0..*arity {
   212→                        locals.insert(0, stack.pop().unwrap_or(Value::Unit));
   213→                    }
   214→
   215→                    // Push call frame
   216→                    let return_ip = ip + 1;
   217→                    let base_bp = stack.len();
   218→                    call_stack.push(CallFrame { return_ip, base_bp });
   219→
   220→                    // Push locals onto stack
   221→                    for local in locals {
   222→                        stack.push(local);
   223→                    }
   224→
   225→                    // Switch to pattern code
   226→                    let result = self.execute_code(&pattern.code, &mut stack, &mut call_stack, &program)?;
   227→                    stack.push(result);
   228→                    // IP already advanced by 1 (return_ip)
   229→                    ip = return_ip;
   230→                }
   231→                Instruction::Return => {
   232→                    let _return_val = stack.pop().unwrap_or(Value::Unit);
   233→                    // Pop call frame and restore stack
   234→                    if let Some(frame) = call_stack.pop() {
   235→                        // Remove locals from stack
   236→                        stack.truncate(frame.base_bp);
   237→                        // We'll push the return value after returning
   238→                        // (handled by the caller)
   239→                    }
   240→                    // Return the value to the execute_code caller
   241→                    // Actually, this is tricky with the current design...
   242→                    // Let me use a different approach — see execute_code below
   243→                    ip += 1;
   244→                }
   245→                Instruction::LlmCall(idx, arity) => {
   246→                    let mut args = Vec::new();
   247→                    for _ in 0..*arity {
   248→                        args.insert(0, stack.pop().unwrap_or(Value::Unit));
   249→                    }
   250→                    let result = self.call_llm(*idx, &args)?;
   251→                    stack.push(result);
   252→                    ip += 1;
   253→                }
   254→
   255→                // ── Binary Operations ──────────────────────────
   256→                Instruction::Add => {
   257→                    let right = stack.pop().unwrap_or(Value::Unit);
   258→                    let left = stack.pop().unwrap_or(Value::Unit);
   259→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Add, right)?);
   260→                    ip += 1;
   261→                }
   262→                Instruction::Sub => {
   263→                    let right = stack.pop().unwrap_or(Value::Unit);
   264→                    let left = stack.pop().unwrap_or(Value::Unit);
   265→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Sub, right)?);
   266→                    ip += 1;
   267→                }
   268→                Instruction::Mul => {
   269→                    let right = stack.pop().unwrap_or(Value::Unit);
   270→                    let left = stack.pop().unwrap_or(Value::Unit);
   271→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Mul, right)?);
   272→                    ip += 1;
   273→                }
   274→                Instruction::Div => {
   275→                    let right = stack.pop().unwrap_or(Value::Unit);
   276→                    let left = stack.pop().unwrap_or(Value::Unit);
   277→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Div, right)?);
   278→                    ip += 1;
   279→                }
   280→                Instruction::Contains => {
   281→                    let right = stack.pop().unwrap_or(Value::Unit);
   282→                    let left = stack.pop().unwrap_or(Value::Unit);
   283→                    let result = self.eval_contains(left, right)?;
   284→                    stack.push(result);
   285→                    ip += 1;
   286→                }
   287→                Instruction::CmpGt => {
   288→                    let right = stack.pop().unwrap_or(Value::Unit);
   289→                    let left = stack.pop().unwrap_or(Value::Unit);
   290→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Gt));
   291→                    ip += 1;
   292→                }
   293→                Instruction::CmpLt => {
   294→                    let right = stack.pop().unwrap_or(Value::Unit);
   295→                    let left = stack.pop().unwrap_or(Value::Unit);
   296→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Lt));
   297→                    ip += 1;
   298→                }
   299→                Instruction::CmpGe => {
   300→                    let right = stack.pop().unwrap_or(Value::Unit);
   301→                    let left = stack.pop().unwrap_or(Value::Unit);
   302→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Ge));
   303→                    ip += 1;
   304→                }
   305→                Instruction::CmpLe => {
   306→                    let right = stack.pop().unwrap_or(Value::Unit);
   307→                    let left = stack.pop().unwrap_or(Value::Unit);
   308→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Le));
   309→                    ip += 1;
   310→                }
   311→                Instruction::CmpEq => {
   312→                    let right = stack.pop().unwrap_or(Value::Unit);
   313→                    let left = stack.pop().unwrap_or(Value::Unit);
   314→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Eq));
   315→                    ip += 1;
   316→                }
   317→                Instruction::CmpNe => {
   318→                    let right = stack.pop().unwrap_or(Value::Unit);
   319→                    let left = stack.pop().unwrap_or(Value::Unit);
   320→                    let eq_result = self.eval_cmp(left, right, AstCompareOp::Eq);
   321→                    match eq_result {
   322→                        Value::Float(f) => stack.push(Value::Float(if f == 1.0 { 0.0 } else { 1.0 })),
   323→                        _ => stack.push(eq_result),
   324→                    }
   325→                    ip += 1;
   326→                }
   327→
   328→                // ── Struct Operations ─────────────────────────
   329→                Instruction::MakeStruct(type_name, field_names) => {
   330→                    let mut fields = HashMap::new();
   331→                    // Values are on stack in field order (first pushed = bottom)
   332→                    // Pop in reverse to get correct order
   333→                    let mut values = Vec::new();
   334→                    for _ in 0..field_names.len() {
   335→                        values.insert(0, stack.pop().unwrap_or(Value::Unit));
   336→                    }
   337→                    for (name, val) in field_names.iter().zip(values.iter()) {
   338→                        fields.insert(name.clone(), val.clone());
   339→                    }
   340→                    stack.push(Value::Struct {
   341→                        type_name: type_name.clone(),
   342→                        fields,
   343→                    });
   344→                    ip += 1;
   345→                }
   346→                Instruction::GetField(field) => {
   347→                    let val = stack.pop().unwrap_or(Value::Unit);
   348→                    let result = val.get_field(field)
   349→                        .cloned()
   350→                        .unwrap_or(Value::Unit);
   351→                    stack.push(result);
   352→                    ip += 1;
   353→                }
   354→                Instruction::IndexAccess => {
   355→                    let idx = stack.pop().unwrap_or(Value::Unit);
   356→                    let base = stack.pop().unwrap_or(Value::Unit);
   357→                    let result = match (&base, &idx) {
   358→                        (Value::List(items), Value::Float(f)) => {
   359→                            let i = *f as isize;
   360→                            if i < 0 {
   361→                                let abs_i = items.len().wrapping_sub((-i) as usize);
   362→                                items.get(abs_i).cloned().unwrap_or(Value::Unit)
   363→                            } else {
   364→                                items.get(i as usize).cloned().unwrap_or(Value::Unit)
   365→                            }
   366→                        }
   367→                        (Value::Struct { fields, .. }, Value::String(key)) => {
   368→                            fields.get(key).cloned().unwrap_or(Value::Unit)
   369→                        }
   370→                        _ => Value::Unit,
   371→                    };
   372→                    stack.push(result);
   373→                    ip += 1;
   374→                }
   375→
   376→                // ── Fluid Types ───────────────────────────────
   377→                Instruction::MakeFluid(count) => {
   378→                    let mut variants: Vec<FluidValueVariant> = Vec::new();
   379→                    // Stack has pairs: value, confidence (bottom to top)
   380→                    // Pop count pairs
   381→                    let mut pairs: Vec<(Value, Value)> = Vec::new();
   382→                    for _ in 0..*count {
   383→                        let confidence = stack.pop().unwrap_or(Value::Float(0.0));
   384→                        let value = stack.pop().unwrap_or(Value::Unit);
   385→                        pairs.insert(0, (value, confidence));
   386→                    }
   387→                    for (value, confidence) in pairs {
   388→                        let conf_f = confidence.as_float().unwrap_or(0.0);
   389→                        // Determine type name from value
   390→                        let type_name = match &value {
   391→                            Value::Float(_) => "Float",
   392→                            Value::String(_) => "String",
   393→                            _ => "Unit",
   394→                        }.to_string();
   395→                        variants.push(FluidValueVariant {
   396→                            type_name,
   397→                            value,
   398→                            confidence: conf_f,
   399→                        });
   400→                    }
   401→                    stack.push(Value::Fluid(variants));
   402→                    ip += 1;
   403→                }
   404→
   405→                // ── Control Flow ───────────────────────────────
   406→                Instruction::Jump(target) => {
   407→                    ip = *target;
   408→                }
   409→                Instruction::JumpIfNot(target) => {
   410→                    let val = stack.pop().unwrap_or(Value::Unit);
   411→                    if !is_truthy(&val) {
   412→                        ip = *target;
   413→                    } else {
   414→                        ip += 1;
   415→                    }
   416→                }
   417→                Instruction::JumpIfLow(threshold, target) => {
   418→                    let val = stack.pop().unwrap_or(Value::Unit);
   419→                    let below = match &val {
   420→                        Value::Float(f) => *f < *threshold,
   421→                        Value::Fluid(variants) => {
   422→                            // Use the maximum confidence
   423→                            let max_conf = variants.iter()
   424→                                .map(|v| v.confidence)
   425→                                .fold(0.0_f64, f64::max);
   426→                            max_conf < *threshold
   427→                        }
   428→                        _ => false,
   429→                    };
   430→                    if below {
   431→                        ip = *target;
   432→                    } else {
   433→                        ip += 1;
   434→                    }
   435→                }
   436→
   437→                // ── METALOGOS Memory ────────────────────────────
   438→                Instruction::Collapse(required_type) => {
   439→                    let val = stack.pop().unwrap_or(Value::Unit);
   440→                    let collapsed = self.maybe_collapse(&val, required_type);
   441→                    stack.push(collapsed);
   442→                    ip += 1;
   443→                }
   444→                Instruction::Memorize(priority) => {
   445→                    let val = stack.pop().unwrap_or(Value::Unit);
   446→                    let value_str = match val {
   447→                        Value::String(s) => s,
   448→                        other => format!("{}", other),
   449→                    };
   450→                    let now = SystemTime::now()
   451→                        .duration_since(UNIX_EPOCH)
   452→                        .map(|d| d.as_secs() as i64)
   453→                        .unwrap_or(0);
   454→                    self.memory.push(VmMemoryEntry {
   455→                        value: value_str,
   456→                        priority: *priority,
   457→                        timestamp: now,
   458→                        decay_rate: 0.01,
   459→                    });
   460→                    ip += 1;
   461→                }
   462→                Instruction::Recall => {
   463→                    let query = stack.pop().unwrap_or(Value::Unit);
   464→                    let query_str = match query {
   465→                        Value::String(s) => s,
   466→                        other => format!("{}", other),
   467→                    };
   468→                    let result = self.recall(&query_str, 0.0);
   469→                    stack.push(Value::String(result));
   470→                    ip += 1;
   471→                }
   472→                Instruction::Forget(days) => {
   473→                    let query = stack.pop().unwrap_or(Value::Unit);
   474→                    let query_str = match query {
   475→                        Value::String(s) => s,
   476→                        other => format!("{}", other),
   477→                    };
   478→                    let now = SystemTime::now()
   479→                        .duration_since(UNIX_EPOCH)
   480→                        .map(|d| d.as_secs() as i64)
   481→                        .unwrap_or(0);
   482→                    let cutoff = now - (days * 86400);
   483→                    self.memory.retain(|entry| {
   484→                        !(entry.value.contains(&query_str) && entry.timestamp < cutoff)
   485→                    });
   486→                    ip += 1;
   487→                }
   488→
   489→                // ── Adapt / Relate / Mutate ─────────────────────
   490→                Instruction::Adapt(pattern_name) => {
   491→                    let output_str = match stack.pop().unwrap_or(Value::Unit) {
   492→                        Value::String(s) => s,
   493→                        other => format!("{}", other),
   494→                    };
   495→                    let input_str = match stack.pop().unwrap_or(Value::Unit) {
   496→                        Value::String(s) => s,
   497→                        other => format!("{}", other),
   498→                    };
   499→                    // Find the learnable by name and add the example
   500→                    let mut found = false;
   501→                    for (info, few_shot) in &mut self.learnables {
   502→                        if info.name == *pattern_name {
   503→                            few_shot.push((input_str, output_str));
   504→                            found = true;
   505→                            break;
   506→                        }
   507→                    }
   508→                    if !found {
   509→                        return Err(format!("VM adapt: learnable pattern '{}' not found", pattern_name));
   510→                    }
   511→                    ip += 1;
   512→                }
   513→                Instruction::Relate => {
   514→                    let relation = match stack.pop().unwrap_or(Value::Unit) {
   515→                        Value::String(s) => s,
   516→                        other => format!("{}", other),
   517→                    };
   518→                    let to = match stack.pop().unwrap_or(Value::Unit) {
   519→                        Value::String(s) => s,
   520→                        other => format!("{}", other),
   521→                    };
   522→                    let from = match stack.pop().unwrap_or(Value::Unit) {
   523→                        Value::String(s) => s,
   524→                        other => format!("{}", other),
   525→                    };
   526→                    self.relations.push(VmRelation { from, to, relation });
   527→                    ip += 1;
   528→                }
   529→                Instruction::Mutate { pattern_name, example_count, rollback_threshold, rollback_op } => {
   530→                    // Pop example_count pairs of (input, output)
   531→                    let mut new_examples = Vec::new();
   532→                    for _ in 0..*example_count {
   533→                        let output_str = match stack.pop().unwrap_or(Value::Unit) {
   534→                            Value::String(s) => s,
   535→                            other => format!("{}", other),
   536→                        };
   537→                        let input_str = match stack.pop().unwrap_or(Value::Unit) {
   538→                            Value::String(s) => s,
   539→                            other => format!("{}", other),
   540→                        };
   541→                        new_examples.push((input_str, output_str));
   542→                    }
   543→
   544→                    // Find the learnable and mutate
   545→                    let msg = self.handle_mutate(pattern_name, new_examples, *rollback_threshold, rollback_op.clone())?;
   546→                    self.mutate_log.push(msg);
   547→                    ip += 1;
   548→                }
   549→
   550→                // ── Pipeline ───────────────────────────────────
   551→                Instruction::FlowExec { source_expr, pipeline, branch_defs } => {
   552→                    // Load the source value
   553→                    let source_val = match source_expr {
   554→                        FlowExpr::GlobalSlot(slot) => {
   555→                            self.globals.get(*slot).cloned().unwrap_or(Value::Unit)
   556→                        }
   557→                        FlowExpr::Ident(name) => {
   558→                            program.globals.iter().position(|n| n == name)
   559→                                .and_then(|slot| self.globals.get(slot).cloned())
   560→                                .unwrap_or(Value::Unit)
   561→                        }
   562→                        FlowExpr::Const(v) => v.clone(),
   563→                    };
   564→
   565→                    // Execute pipeline steps
   566→                    let mut current = source_val;
   567→                    for step_name in pipeline {
   568→                        current = self.run_flow_step(step_name, current, branch_defs)?;
   569→                    }
   570→
   571→                    // The pipeline result is the flow output
   572→                    let output_str = format!("{}", current);
   573→                    flow_output = Some(output_str);
   574→                    ip += 1;
   575→                }
   576→
   577→                // ── Rule Engine ─────────────────────────────────
   578→                Instruction::ExecuteRules => {
   579→                    self.execute_rules()?;
   580→                    ip += 1;
   581→                }
   582→
   583→                // ── Meta ───────────────────────────────────────
   584→                Instruction::Halt => {
   585→                    break;
   586→                }
   587→            }
   588→        }
   589→
   590→        // Build final output with mutate log
   591→        let mutate_log = std::mem::take(&mut self.mutate_log);
   592→        if mutate_log.is_empty() {
   593→            Ok(flow_output)
   594→        } else {
   595→            match flow_output {
   596→                Some(flow) => {
   597→                    let mut result = mutate_log.join("\n");
   598→                    result.push_str("\n");
   599→                    result.push_str(&flow);
   600→                    Ok(Some(result))
   601→                }
   602→                None => Ok(Some(mutate_log.join("\n"))),
   603→            }
   604→        }
   605→    }
   606→
   607→    /// Execute a block of code (e.g., pattern body) and return the result.
   608→    /// This handles the call stack and Return instructions internally.
   609→    fn execute_code(
   610→        &self,
   611→        code: &[Instruction],
   612→        stack: &mut Vec<Value>,
   613→        call_stack: &mut Vec<CallFrame>,
   614→        program: &Program,
   615→    ) -> Result<Value, String> {
   616→        let mut ip = 0;
   617→        // Safety: prevent infinite loops (max iterations per 100 instructions)
   618→        let max_iterations = code.len().max(1) * 100_000;
   619→        let mut iterations = 0;
   620→        while ip < code.len() {
   621→            iterations += 1;
   622→            if iterations > max_iterations {
   623→                return Err(format!("VM execute_code: possible infinite loop ({} iterations, {} instructions)", iterations, code.len()));
   624→            }
   625→            let instr = &code[ip];
   626→            match instr {
   627→                Instruction::LoadLocal(slot) => {
   628→                    let bp = call_stack.last().map(|f| f.base_bp).unwrap_or(0);
   629→                    let idx = bp + slot;
   630→                    let val = stack.get(idx).cloned().unwrap_or(Value::Unit);
   631→                    stack.push(val);
   632→                    ip += 1;
   633→                }
   634→                Instruction::StoreLocal(slot) => {
   635→                    let bp = call_stack.last().map(|f| f.base_bp).unwrap_or(0);
   636→                    let idx = bp + slot;
   637→                    let val = stack.pop().unwrap_or(Value::Unit);
   638→                    // Ensure stack has space at idx
   639→                    if idx >= stack.len() {
   640→                        stack.resize(idx + 1, Value::Unit);
   641→                    }
   642→                    stack[idx] = val;
   643→                    ip += 1;
   644→                }
   645→                Instruction::Const(v) => {
   646→                    stack.push(v.clone());
   647→                    ip += 1;
   648→                }
   649→                Instruction::LoadGlobal(slot) => {
   650→                    let val = self.globals.get(*slot).cloned().unwrap_or(Value::Unit);
   651→                    stack.push(val);
   652→                    ip += 1;
   653→                }
   654→                Instruction::CallBuiltin(idx, arity) => {
   655→                    let name = self.builtin_names.get(*idx)
   656→                        .map(|s| s.as_str())
   657→                        .unwrap_or("unknown");
   658→                    let mut args = Vec::new();
   659→                    for _ in 0..*arity {
   660→                        args.insert(0, stack.pop().unwrap_or(Value::Unit));
   661→                    }
   662→                    let result = self.call_builtin(name, &args)?;
   663→                    stack.push(result);
   664→                    ip += 1;
   665→                }
   666→                Instruction::CallPattern(pidx, arity) => {
   667→                    let pattern = self.patterns.get(*pidx)
   668→                        .ok_or_else(|| format!("VM: pattern index {} not found", pidx))?
   669→                        .clone();
   670→                    if *arity != pattern.param_count {
   671→                        return Err(format!(
   672→                            "VM: pattern {} expects {} args, got {}",
   673→                            pattern.name, pattern.param_count, arity
   674→                        ));
   675→                    }
   676→                    let mut locals = Vec::new();
   677→                    for _ in 0..*arity {
   678→                        locals.insert(0, stack.pop().unwrap_or(Value::Unit));
   679→                    }
   680→                    let base_bp = stack.len();
   681→                    call_stack.push(CallFrame { return_ip: ip + 1, base_bp });
   682→                    for local in locals {
   683→                        stack.push(local);
   684→                    }
   685→                    let result = self.execute_code(&pattern.code, stack, call_stack, program)?;
   686→                    call_stack.pop();
   687→                    stack.push(result);
   688→                    ip += 1;
   689→                }
   690→                Instruction::Add => {
   691→                    let right = stack.pop().unwrap_or(Value::Unit);
   692→                    let left = stack.pop().unwrap_or(Value::Unit);
   693→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Add, right)?);
   694→                    ip += 1;
   695→                }
   696→                Instruction::Sub => {
   697→                    let right = stack.pop().unwrap_or(Value::Unit);
   698→                    let left = stack.pop().unwrap_or(Value::Unit);
   699→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Sub, right)?);
   700→                    ip += 1;
   701→                }
   702→                Instruction::Mul => {
   703→                    let right = stack.pop().unwrap_or(Value::Unit);
   704→                    let left = stack.pop().unwrap_or(Value::Unit);
   705→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Mul, right)?);
   706→                    ip += 1;
   707→                }
   708→                Instruction::Div => {
   709→                    let right = stack.pop().unwrap_or(Value::Unit);
   710→                    let left = stack.pop().unwrap_or(Value::Unit);
   711→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Div, right)?);
   712→                    ip += 1;
   713→                }
   714→                Instruction::Return => {
   715→                    return Ok(stack.pop().unwrap_or(Value::Unit));
   716→                }
   717→                // Phase 5.1: control flow in pattern bodies
   718→                Instruction::Jump(target) => {
   719→                    ip = *target;
   720→                }
   721→                Instruction::JumpIfNot(target) => {
   722→                    let val = stack.pop().unwrap_or(Value::Unit);
   723→                    if !is_truthy(&val) {
   724→                        ip = *target;
   725→                    } else {
   726→                        ip += 1;
   727→                    }
   728→                }
   729→                // Phase 5.1: comparison operators
   730→                Instruction::CmpGt => {
   731→                    let right = stack.pop().unwrap_or(Value::Unit);
   732→                    let left = stack.pop().unwrap_or(Value::Unit);
   733→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Gt));
   734→                    ip += 1;
   735→                }
   736→                Instruction::CmpLt => {
   737→                    let right = stack.pop().unwrap_or(Value::Unit);
   738→                    let left = stack.pop().unwrap_or(Value::Unit);
   739→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Lt));
   740→                    ip += 1;
   741→                }
   742→                Instruction::CmpGe => {
   743→                    let right = stack.pop().unwrap_or(Value::Unit);
   744→                    let left = stack.pop().unwrap_or(Value::Unit);
   745→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Ge));
   746→                    ip += 1;
   747→                }
   748→                Instruction::CmpLe => {
   749→                    let right = stack.pop().unwrap_or(Value::Unit);
   750→                    let left = stack.pop().unwrap_or(Value::Unit);
   751→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Le));
   752→                    ip += 1;
   753→                }
   754→                Instruction::CmpEq => {
   755→                    let right = stack.pop().unwrap_or(Value::Unit);
   756→                    let left = stack.pop().unwrap_or(Value::Unit);
   757→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Eq));
   758→                    ip += 1;
   759→                }
   760→                Instruction::CmpNe => {
   761→                    let right = stack.pop().unwrap_or(Value::Unit);
   762→                    let left = stack.pop().unwrap_or(Value::Unit);
   763→                    let eq_result = self.eval_cmp(left, right, AstCompareOp::Eq);
   764→                    match eq_result {
   765→                        Value::Float(f) => stack.push(Value::Float(if f == 1.0 { 0.0 } else { 1.0 })),
   766→                        _ => stack.push(eq_result),
   767→                    }
   768→                    ip += 1;
   769→                }
   770→                Instruction::GetField(field) => {
   771→                    let val = stack.pop().unwrap_or(Value::Unit);
   772→                    let result = val.get_field(field).cloned().unwrap_or(Value::Unit);
   773→                    stack.push(result);
   774→                    ip += 1;
   775→                }
   776→                Instruction::LoadGlobalByName(name) => {
   777→                    let val = program.globals.iter().position(|n| n == name)
   778→                        .and_then(|slot| self.globals.get(slot).cloned())
   779→                        .unwrap_or(Value::Unit);
   780→                    stack.push(val);
   781→                    ip += 1;
   782→                }
   783→                Instruction::Collapse(required_type) => {
   784→                    let val = stack.pop().unwrap_or(Value::Unit);
   785→                    let collapsed = self.maybe_collapse(&val, required_type);
   786→                    stack.push(collapsed);
   787→                    ip += 1;
   788→                }
   789→                Instruction::LlmCall(idx, arity) => {
   790→                    let mut args = Vec::new();
   791→                    for _ in 0..*arity {
   792→                        args.insert(0, stack.pop().unwrap_or(Value::Unit));
   793→                    }
   794→                    let result = self.call_llm(*idx, &args)?;
   795→                    stack.push(result);
   796→                    ip += 1;
   797→                }
   798→                Instruction::Recall => {
   799→                    let query = stack.pop().unwrap_or(Value::Unit);
   800→                    let query_str = match query {
   801→                        Value::String(s) => s,
   802→                        other => format!("{}", other),
   803→                    };
   804→                    let result = self.recall(&query_str, 0.0);
   805→                    stack.push(Value::String(result));
   806→                    ip += 1;
   807→                }
   808→                // For any unhandled instruction, skip
   809→                _ => { ip += 1; }
   810→            }
   811→        }
   812→        Ok(stack.pop().unwrap_or(Value::Unit))
   813→    }

   815→    /// Call a built-in function by name.
   816→    fn call_builtin(&self, name: &str, args: &[Value]) -> Result<Value, String> {
   817→        if name == "recall" {
   818→            let query = match args.get(0) {
   819→                Some(Value::String(s)) => s.clone(),
   820→                other => return Err(format!("recall() expected String, got {:?}", other)),
   821→            };
   822→            let min_conf = if args.len() > 1 {
   823→                args[1].as_float().unwrap_or(0.0)
   824→            } else {
   825→                0.0
   826→            };
   827→            return Ok(Value::String(self.recall(&query, min_conf)));
   828→        }
   829→
   830→        // Collections operations
   831→        if self.collections_loaded {
   832→            match name {
   833→                "map" | "filter" | "reduce" => {
   834→                    return Err(format!("VM: collection op '{}' not yet implemented in VM path", name));
   835→                }
   836→                _ => {}
   837→            }
   838→        }
   839→
   840→        if let Some(builtin_fn) = self.builtins.get(name) {
   841→            return builtin_fn(args);
   842→        }
   843→        Err(format!("VM: undefined builtin: {}", name))
   844→    }

   846→    /// Call an LLM-backed learnable pattern.
   847→    fn call_llm(&self, idx: usize, args: &[Value]) -> Result<Value, String> {
   848→        let (info, few_shot) = self.learnables.get(idx)
   849→            .ok_or_else(|| format!("VM: learnable index {} not found", idx))?;
   850→
   851→        // Build input string
   852→        let input_parts: Vec<String> = args.iter().map(|a| format!("{}", a)).collect();
   853→        let input = input_parts.join(", ");
   854→
   855→        // Check few-shot cache first
   856→        for (example_input, example_output) in few_shot {
   857→            if input == *example_input {
   858→                return Ok(Value::String(example_output.clone()));
   859→            }
   860→        }
   861→
   862→        // Call LLM backend
   863→        let backend = llm::create_llm_backend();
   864→        let response = backend.call(&info.prompt, &input)?;
   865→
   866→        // Try to parse JSON response into Value::Struct
   867→        if let Ok(json) = serde_json::from_str::<serde_json::Value>(&response) {
   868→            if let Some(obj) = json.as_object() {
   869→                let mut fields = std::collections::HashMap::new();
   870→                for (k, v) in obj {
   871→                    fields.insert(k.clone(), Vm::json_to_value(v));
   872→                }
   873→                return Ok(Value::Struct {
   874→                    type_name: "LlmResponse".to_string(),
   875→                    fields,
   876→                });
   877→            }
   878→        }
   879→
   880→        Ok(Value::String(response))
   881→    }

   883→    /// Convert serde_json::Value to METALOGOS Value (for VM).
   884→    fn json_to_value(json: &serde_json::Value) -> Value {
   885→        match json {
   886→            serde_json::Value::String(s) => Value::String(s.clone()),
   887→            serde_json::Value::Number(n) => Value::Float(n.as_f64().unwrap_or(0.0)),
   888→            serde_json::Value::Bool(b) => Value::Bool(*b),
   889→            serde_json::Value::Null => Value::Unit,
   890→            serde_json::Value::Array(arr) => {
   891→                Value::List(arr.iter().map(|v| Vm::json_to_value(v)).collect())
   892→            }
   893→            serde_json::Value::Object(obj) => {
   894→                let mut fields = std::collections::HashMap::new();
   895→                for (k, v) in obj {
   896→                    fields.insert(k.clone(), Vm::json_to_value(v));
   897→                }
   898→                Value::Struct { type_name: "Json".to_string(), fields }
   899→            }
   900→        }
   901→    }

   903→    /// Execute a flow step: check branches, then invoke pattern/builtin.
   904→    fn run_flow_step(
   905→        &mut self,
   906→        step_name: &str,
   907→        current: Value,
   908→        branch_defs: &[(String, Vec<BranchDef>)],
   909→    ) -> Result<Value, String> {
   910→        // Check if step has branch definitions
   911→        for (bd_step, branches) in branch_defs {
   912→            if bd_step == step_name {
   913→                for branch in branches {
   914→                    if self.eval_branch_condition(branch, &current)? {
   915→                        return self.invoke_step(&branch.target, vec![current]);
   916→                    }
   917→                }
   918→                return Err(format!("VM: no branch matched in step '{}'", step_name));
   919→            }
   920→        }
   921→
   922→        // No branch definitions — invoke as pattern/builtin
   923→        self.invoke_step(step_name, vec![current])
   924→    }
   925→
   926→    /// Evaluate a branch condition against a value.
   927→    fn eval_branch_condition(&self, branch: &BranchDef, current: &Value) -> Result<bool, String> {
   928→        let field_val = current.get_field(&branch.condition_field)
   929→            .map_err(|e| format!("branch condition: {}", e))?
   930→            .clone();
   931→        let fv = field_val.as_float()?;
   932→        let tv = branch.condition_threshold.as_float()?;
   933→        Ok(match branch.condition_op {
   934→            ConditionOp::Gt => fv > tv,
   935→            ConditionOp::Lt => fv < tv,
   936→            ConditionOp::Ge => fv >= tv,
   937→            ConditionOp::Le => fv <= tv,
   938→            ConditionOp::Eq => fv == tv,
   939→        })
   940→    }

   942→    /// Invoke a pattern or builtin by name (used in flow pipeline).
   943→    fn invoke_step(&mut self, name: &str, args: Vec<Value>) -> Result<Value, String> {
   944→        // Check learnables first
   945→        for (i, (info, _)) in self.learnables.iter().enumerate() {
   946→            if info.name == name {
   947→                return self.call_llm(i, &args);
   948→            }
   949→        }
   950→
   951→        // Check patterns
   952→        for pattern in self.patterns.iter() {
   953→            if pattern.name == name {
   954→                if args.len() != pattern.param_count {
   955→                    return Err(format!(
   956→                        "VM: pattern {} expects {} args, got {}",
   957→                        name, pattern.param_count, args.len()
   958→                    ));
   959→                }
   960→
   961→                // ── VM bytecode path ───────────────────────────
   962→                // Collapse Fluid arguments to parameter types
   963→                let collapsed_args: Vec<Value> = args.iter()
   964→                    .zip(pattern.param_types.iter())
   965→                    .map(|(arg, param_type)| self.maybe_collapse(arg, param_type))
   966→                    .collect();
   967→                // Execute pattern body as bytecode
   968→                let mut stack: Vec<Value> = Vec::new();
   969→                let mut call_stack: Vec<CallFrame> = Vec::new();
   970→                let program = Program {
   971→                    globals: Vec::new(),
   972→                    patterns: Vec::new(),
   973→                    learnables: Vec::new(),
   974→                    rules: Vec::new(),
   975→                    main_code: Vec::new(),
   976→                    collections_loaded: false,
   977→                };
   978→                for arg in collapsed_args {
   979→                    stack.push(arg);
   980→                }
   981→                return self.execute_code(&pattern.code, &mut stack, &mut call_stack, &program);
   982→            }
   983→        }
   984→
   985→        // Check builtins
   986→        self.call_builtin(name, &args)
   987→    }

   989→    /// Execute all registered rules (already sorted by priority descending).
   990→    fn execute_rules(&mut self) -> Result<(), String> {
   991→        for rule in &self.rules {
   992→            let condition_met = self.eval_rule_condition(&rule.condition)?;
   993→            if condition_met {
   994→                // Find the target entity by name in globals
   995→                let target_slot = self.global_names.iter().position(|n| n == &rule.target_name);
   996→                if let Some(slot) = target_slot {
   997→                    let value = self.eval_rule_value(&rule.value_expr)?;
   998→                    // Set field on the struct
   999→                    if let Some(entity) = self.globals.get_mut(slot) {
  1000→                        let _ = entity.set_field(&rule.field, value);
  1001→                    }
  1002→                }
  1003→            }
  1004→        }
  1005→        Ok(())
  1006→    }

  1008→    /// Evaluate a rule condition.
  1009→    fn eval_rule_condition(&self, cond: &RuleCondition) -> Result<bool, String> {
  1010→        match cond {
  1011→            RuleCondition::Contains { left, right } => {
  1012→                let ls = self.eval_rule_value(left)?;
  1013→                let rs = self.eval_rule_value(right)?;
  1014→                match (&ls, &rs) {
  1015→                    (Value::String(l), Value::String(r)) => Ok(l.contains(r)),
  1016→                    _ => Err("contains: both sides must be String".to_string()),
  1017→                }
  1018→            }
  1019→            RuleCondition::Compare { left, op, right } => {
  1020→                let lv = self.eval_rule_value(left)?;
  1021→                let rv = self.eval_rule_value(right)?;
  1022→                let lf = lv.as_float()?;
  1023→                let rf = rv.as_float()?;
  1024→                Ok(match op {
  1025→                    ConditionOp::Gt => lf > rf,
  1026→                    ConditionOp::Lt => lf < rf,
  1027→                    ConditionOp::Ge => lf >= rf,
  1028→                    ConditionOp::Le => lf <= rf,
  1029→                    ConditionOp::Eq => lf == rf,
  1030→                })
  1031→            }
  1032→        }
  1033→    }
  1034→
  1035→    /// Evaluate a rule value expression to a runtime value.
  1036→    fn eval_rule_value(&self, expr: &RuleValueExpr) -> Result<Value, String> {
  1037→        match expr {
  1038→            RuleValueExpr::StringLit(s) => Ok(Value::String(s.clone())),
  1039→            RuleValueExpr::FloatLit(f) => Ok(Value::Float(*f)),
  1040→            RuleValueExpr::Ident(name) => {
  1041→                // Search globals by name
  1042→                let slot = self.global_names.iter().position(|n| n == name);
  1043→                match slot {
  1044→                    Some(s) => Ok(self.globals.get(s).cloned().unwrap_or(Value::Unit)),
  1045→                    None => Err(format!("VM rule: undefined variable '{}'", name)),
  1046→                }
  1047→            }
  1048→            RuleValueExpr::FieldAccess(entity_name, field) => {
  1049→                let slot = self.global_names.iter().position(|n| n == entity_name);
  1050→                match slot {
  1051→                    Some(s) => {
  1052→                        let entity = self.globals.get(s).cloned().unwrap_or(Value::Unit);
  1053→                        entity.get_field(field).cloned()
  1054→                            .map_err(|e| format!("VM rule field access: {}", e))
  1055→                    }
  1056→                    None => Err(format!("VM rule: undefined entity '{}'", entity_name)),
  1057→                }
  1058→            }
  1059→        }
  1060→    }

  1062→    /// Handle a mutate declaration.
  1063→    fn handle_mutate(
  1064→        &mut self,
  1065→        pattern_name: &str,
  1066→        new_examples: Vec<(String, String)>,
  1067→        rollback_threshold: Option<f64>,
  1068→        rollback_op: Option<ConditionOp>,
  1069→    ) -> Result<String, String> {
  1070→        // Find the learnable
  1071→        for (_, (info, few_shot)) in self.learnables.iter_mut().enumerate() {
  1072→            if info.name == pattern_name {
  1073→                let original = few_shot.clone();
  1074→                *few_shot = new_examples;
  1075→
  1076→                // Mock accuracy (always 0.95 for MockLlm)
  1077→                let accuracy: f64 = 0.95;
  1078→
  1079→                let kept = match (&rollback_op, &rollback_threshold) {
  1080→                    (Some(ConditionOp::Lt), Some(threshold)) => accuracy >= *threshold,
  1081→                    (Some(ConditionOp::Le), Some(threshold)) => accuracy > *threshold,
  1082→                    (Some(ConditionOp::Gt), Some(_)) |
  1083→                    (Some(ConditionOp::Ge), Some(_)) => false,
  1084→                    (Some(ConditionOp::Eq), Some(threshold)) => (accuracy - threshold).abs() < 1e-9,
  1085→                    _ => true,
  1086→                };
  1087→
  1088→                if kept {
  1089→                    return Ok(format!("[MUTATE] {}: accuracy={}, kept (>= {:.1})",
  1090→                        pattern_name, accuracy,
  1091→                        rollback_threshold.unwrap_or(0.0)));
  1092→                } else {
  1093→                    *few_shot = original;
  1094→                    return Ok(format!("[MUTATE] {}: accuracy={}, rolled back (below {:.1})",
  1095→                        pattern_name, accuracy,
  1096→                        rollback_threshold.unwrap_or(0.0)));
  1097→                }
  1098→            }
  1099→        }
  1100→        Err(format!("VM mutate: learnable pattern '{}' not found", pattern_name))
  1101→    }

  1103→    /// Recall from memory: find best matching entry by substring + decay.
  1104→    fn recall(&self, query: &str, min_confidence: f64) -> String {
  1105→        let now = SystemTime::now()
  1106→            .duration_since(UNIX_EPOCH)
  1107→            .map(|d| d.as_secs() as i64)
  1108→            .unwrap_or(0);
  1109→
  1110→        let mut best_match: Option<&VmMemoryEntry> = None;
  1111→        let mut best_activation: f64 = 0.0;
  1112→
  1113→        for entry in &self.memory {
  1114→            if !entry.value.contains(query) {
  1115→                continue;
  1116→            }
  1117→            let age_days = ((now - entry.timestamp).max(0) as f64) / 86400.0;
  1118→            let activation = entry.priority * (-entry.decay_rate * age_days).exp();
  1119→            if activation > best_activation && activation >= min_confidence {
  1120→                best_activation = activation;
  1121→                best_match = Some(entry);
  1122→            }
  1123→        }
  1124→
  1125→        match best_match {
  1126→            Some(entry) => {
  1127→                let mut result = entry.value.clone();
  1128→                // Walk knowledge graph for related memories
  1129→                for rel in &self.relations {
  1130→                    if rel.from == entry.value {
  1131→                        result.push_str("\n");
  1132→                        result.push_str(&format!("[GRAPH] {} -> {}", rel.relation, rel.to));
  1133→                    } else if rel.to == entry.value {
  1134→                        result.push_str("\n");
  1135→                        result.push_str(&format!("[GRAPH] {} -> {}", rel.relation, rel.from));
  1136→                    }
  1137→                }
  1138→                result
  1139→            }
  1140→            None => String::new(),
  1141→        }
  1142→    }

  1144→    /// Collapse a Fluid value to a concrete type.
  1145→    fn maybe_collapse(&self, value: &Value, required_type: &str) -> Value {
  1146→        match value {
  1147→            Value::Fluid(variants) => {
  1148→                let best = variants.iter()
  1149→                    .filter(|v| v.type_name == required_type)
  1150→                    .max_by(|a, b| {
  1151→                        a.confidence.partial_cmp(&b.confidence)
  1152→                            .unwrap_or(std::cmp::Ordering::Equal)
  1153→                    });
  1154→                match best {
  1155→                    Some(variant) if variant.confidence >= COLLAPSE_THRESHOLD => {
  1156→                        variant.value.clone()
  1157→                    }
  1158→                    _ => Value::Unit,
  1159→                }
  1160→            }
  1161→            other => other.clone(),
  1162→        }
  1163→    }

  1165→    /// Evaluate a binary operation.
  1166→    fn eval_binop(&self, left: Value, op: crate::ast::BinOp, right: Value) -> Result<Value, String> {
  1167→        match (left, right) {
  1168→            (Value::String(a), Value::String(b)) => match op {
  1169→                crate::ast::BinOp::Add => Ok(Value::String(format!("{}{}", a, b))),
  1170→                crate::ast::BinOp::Eq => Ok(Value::Float(if a == b { 1.0 } else { 0.0 })),
  1171→                crate::ast::BinOp::Ne => Ok(Value::Float(if a != b { 1.0 } else { 0.0 })),
  1172→                _ => Err(format!("cannot apply {:?} to two Strings", op)),
  1173→            },
  1174→            (Value::Float(a), Value::Float(b)) => match op {
  1175→                crate::ast::BinOp::Add => Ok(Value::Float(a + b)),
  1176→                crate::ast::BinOp::Sub => Ok(Value::Float(a - b)),
  1177→                crate::ast::BinOp::Mul => Ok(Value::Float(a * b)),
  1178→                crate::ast::BinOp::Div => {
  1179→                    if b == 0.0 {
  1180→                        Err("division by zero".to_string())
  1181→                    } else {
  1182→                        Ok(Value::Float(a / b))
  1183→                    }
  1184→                }
  1185→                crate::ast::BinOp::Gt => Ok(Value::Float(if a > b { 1.0 } else { 0.0 })),
  1186→                crate::ast::BinOp::Lt => Ok(Value::Float(if a < b { 1.0 } else { 0.0 })),
  1187→                crate::ast::BinOp::Ge => Ok(Value::Float(if a >= b { 1.0 } else { 0.0 })),
  1188→                crate::ast::BinOp::Le => Ok(Value::Float(if a <= b { 1.0 } else { 0.0 })),
  1189→                crate::ast::BinOp::Eq => Ok(Value::Float(if a == b { 1.0 } else { 0.0 })),
  1190→                crate::ast::BinOp::Ne => Ok(Value::Float(if a != b { 1.0 } else { 0.0 })),
  1191→            },
  1192→            (l, r) => Err(format!(
  1193→                "type mismatch: {} {:?} {}",
  1194→                l.type_name(), op, r.type_name()
  1195→            )),
  1196→        }
  1197→    }

  1199→    /// Evaluate contains(left, right).
  1200→    fn eval_contains(&self, left: Value, right: Value) -> Result<Value, String> {
  1201→        let ls = match left {
  1202→            Value::String(s) => s,
  1203→            other => return Err(format!("contains: left must be String, got {}", other.type_name())),
  1204→        };
  1205→        let rs = match right {
  1206→            Value::String(s) => s,
  1207→            other => return Err(format!("contains: right must be String, got {}", other.type_name())),
  1208→        };
  1209→        Ok(Value::Float(if ls.contains(&rs) { 1.0 } else { 0.0 }))
  1210→    }

  1212→    /// Evaluate a comparison: push 1.0 (true) or 0.0 (false).
  1213→    fn eval_cmp(&self, left: Value, right: Value, op: AstCompareOp) -> Value {
  1214→        let result = match (left.as_float(), right.as_float()) {
  1215→            (Ok(lf), Ok(rf)) => match op {
  1216→                AstCompareOp::Gt => lf > rf,
  1217→                AstCompareOp::Lt => lf < rf,
  1218→                AstCompareOp::Ge => lf >= rf,
  1219→                AstCompareOp::Le => lf <= rf,
  1220→                AstCompareOp::Eq => lf == rf,
  1221→                _ => false,
  1222→            },
  1223→            _ => false,
  1224→        };
  1225→        Value::Float(if result { 1.0 } else { 0.0 })
  1226→    }
  1227→}

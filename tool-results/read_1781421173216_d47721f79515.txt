    14→    14→    14→/// The compiler translates declarations into a bytecode Program.
    15→    15→    15→pub struct Compiler {
    16→    16→    16→    /// Global variable name -> slot index.
    17→    17→    17→    global_slots: HashMap<String, usize>,
    18→    18→    18→    /// Next available global slot.
    19→    19→    19→    next_global: usize,
    20→    20→    20→    /// Pattern name -> index in program.patterns.
    21→    21→    21→    pattern_indices: HashMap<String, usize>,
    22→    22→    22→    /// Learnable pattern name -> index in program.learnables.
    23→    23→    23→    learnable_indices: HashMap<String, usize>,
    24→    24→    24→    /// Builtin name -> index in builtin table.
    25→    25→    25→    builtin_indices: HashMap<String, usize>,
    26→    26→    26→    /// Struct type name -> field names (in declaration order).
    27→    27→    27→    struct_fields: HashMap<String, Vec<String>>,
    28→    28→    28→    /// Rule declarations to compile.
    29→    29→    29→    rules: Vec<CompiledRule>,
    30→    30→    30→    /// Root directory for import resolution.
    31→    31→    31→    std_root: PathBuf,
    32→    32→    32→    /// Already-imported modules.
    33→    33→    33→    imported_modules: HashSet<String>,
    34→    34→    34→    /// Collections loaded flag.
    35→    35→    35→    collections_loaded: bool,
    36→    36→    36→    /// Sandbox declarations (recorded).
    37→    37→    37→    sandboxes: HashMap<String, SandboxDecl>,
    38→    38→    38→}

    50→    50→    50→    /// Create a compiler with a custom std root directory for import resolution.
    51→    51→    51→    pub fn with_std_root(std_root: PathBuf) -> Self {
    52→    52→    52→        let mut builtin_indices = HashMap::new();
    53→    53→    53→        let builtins = [
    54→    54→    54→            "upper", "lower", "len", "str", "print", "contains", "float",
    55→    55→    55→            "__trim", "__replace", "__split", "__join",
    56→    56→    56→            "__abs", "__min", "__max", "__clamp", "__round",
    57→    57→    57→            "__first", "__last", "__push", "__list_len",
    58→    58→    58→            "recall",
    59→    59→    59→            // Phase 4.4 self-hosting
    60→    60→    60→            "stdin", "split_tokens", "if_eq", "newline", "is_string_token",
    61→    61→    61→        ];
    62→    62→    62→        for (i, name) in builtins.iter().enumerate() {
    63→    63→    63→            builtin_indices.insert(name.to_string(), i);
    64→    64→    64→        }
    65→    65→    65→
    66→    66→    66→        Compiler {
    67→    67→    67→            global_slots: HashMap::new(),
    68→    68→    68→            next_global: 0,
    69→    69→    69→            pattern_indices: HashMap::new(),
    70→    70→    70→            learnable_indices: HashMap::new(),
    71→    71→    71→            builtin_indices,
    72→    72→    72→            struct_fields: HashMap::new(),
    73→    73→    73→            rules: Vec::new(),
    74→    74→    74→            std_root,
    75→    75→    75→            imported_modules: HashSet::new(),
    76→    76→    76→            collections_loaded: false,
    77→    77→    77→            sandboxes: HashMap::new(),
    78→    78→    78→        }
    79→    79→    79→    }

    81→    81→    81→    /// Compile a list of declarations into a bytecode Program.
    82→    82→    82→    pub fn compile(&mut self, declarations: Vec<Declaration>) -> Result<Program, String> {
    83→    83→    83→        // Phase 1: resolve imports
    84→    84→    84→        let mut all_decls = Vec::new();
    85→    85→    85→        for decl in declarations {
    86→    86→    86→            if let Declaration::Import(import) = &decl {
    87→    87→    87→                // Fix 2: use `import.path` instead of `import.module_path`
    88→    88→    88→                if !self.imported_modules.contains(&import.path) {
    89→    89→    89→                    let imported = self.resolve_import(&import.path)?;
    90→    90→    90→                    all_decls.extend(imported);
    91→    91→    91→                }
    92→    92→    92→            } else {
    93→    93→    93→                all_decls.push(decl);
    94→    94→    94→            }
    95→    95→    95→        }
    96→    96→    96→
    97→    97→    97→        // Phase 2: two-pass compilation
    98→    98→    98→        // Pass 1: collect struct types, pattern names, learnable names, global slots
    99→    99→    99→        self.pass1(&all_decls)?;
   100→   100→   100→
   101→   101→   101→        // Pass 2: generate main_code
   102→   102→   102→        let main_code = self.pass2(&all_decls)?;
   103→   103→   103→
   104→   104→   104→        let program = Program {
   105→   105→   105→            // Build globals list with actual names (indexed by slot)
   106→   106→   106→            globals: (0..self.next_global)
   107→   107→   107→                .map(|i| self.global_slots.iter()
   108→   108→   108→                    .find(|(_, &slot)| slot == i)
   109→   109→   109→                    .map(|(name, _)| name.clone())
   110→   110→   110→                    .unwrap_or_default())
   111→   111→   111→                .collect(),
   112→   112→   112→            patterns: Vec::new(), // will be filled from pass1 data
   113→   113→   113→            learnables: Vec::new(),
   114→   114→   114→            rules: std::mem::take(&mut self.rules),
   115→   115→   115→            main_code,
   116→   116→   116→            collections_loaded: self.collections_loaded,
   117→   117→   117→        };
   118→   118→   118→
   119→   119→   119→        Ok(program)
   120→   120→   120→    }
   121→   121→   121→
   122→   122→   122→    /// Pass 1: collect type info, register patterns, assign global slots.
   123→   123→   123→    fn pass1(&mut self, decls: &[Declaration]) -> Result<(), String> {
   124→   124→   124→        for decl in decls {
   125→   125→   125→            match decl {
   126→   126→   126→                Declaration::EntityType(e) => {
   127→   127→   127→                    let fields: Vec<String> = e.fields.iter().map(|f| f.name.clone()).collect();
   128→   128→   128→                    self.struct_fields.insert(e.name.clone(), fields);
   129→   129→   129→                }
   130→   130→   130→                Declaration::EntityRecord(e) => {
   131→   131→   131→                    self.ensure_global(&e.name);
   132→   132→   132→                }
   133→   133→   133→                Declaration::EntitySimple(e) => {
   134→   134→   134→                    self.ensure_global(&e.name);
   135→   135→   135→                }
   136→   136→   136→                Declaration::Pattern(p) => {
   137→   137→   137→                    let idx = self.pattern_indices.len();
   138→   138→   138→                    self.pattern_indices.insert(p.name.clone(), idx);
   139→   139→   139→                }
   140→   140→   140→                Declaration::LearnablePattern(lp) => {
   141→   141→   141→                    let idx = self.learnable_indices.len();
   142→   142→   142→                    self.learnable_indices.insert(lp.name.clone(), idx);
   143→   143→   143→                }
   144→   144→   144→                Declaration::Fluid(fl) => {
   145→   145→   145→                    self.ensure_global(&fl.name);
   146→   146→   146→                }
   147→   147→   147→                Declaration::Rule(r) => {
   148→   148→   148→                    self.rules.push(self.compile_rule(r)?);
   149→   149→   149→                }
   150→   150→   150→                Declaration::Memorize(_) => {
   151→   151→   151→                    // Handled in pass2
   152→   152→   152→                }
   153→   153→   153→                Declaration::Sandbox(s) => {
   154→   154→   154→                    self.sandboxes.insert(s.name.clone(), s.clone());
   155→   155→   155→                }
   156→   156→   156→                Declaration::Adapt(_) | Declaration::Relate(_) | Declaration::Mutate(_) | Declaration::Forget(_) => {
   157→   157→   157→                    // Handled in pass2
   158→   158→   158→                }
   159→   159→   159→                Declaration::MlogServer(_) | Declaration::Template(_) | Declaration::Db(_) | Declaration::Memory(_) | Declaration::Conversation(_) | Declaration::Tool(_) | Declaration::LlmConfig(_) => {
   160→   160→   160→                    // Phase 6: handled elsewhere
   161→   161→   161→                }
   162→   162→   162→                _ => {}
   163→   163→   163→            }
   164→   164→   164→        }
   165→   165→   165→        Ok(())
   166→   166→   166→    }
   167→   167→   167→
   168→   168→   168→    /// Pass 2: generate main_code instructions.
   169→   169→   169→    fn pass2(&mut self, decls: &[Declaration]) -> Result<Vec<Instruction>, String> {
   170→   170→   170→        let mut code = Vec::new();
   171→   171→   171→
   172→   172→   172→        for decl in decls {
   173→   173→   173→            match decl {
   174→   174→   174→                Declaration::EntityType(e) => {
   175→   175→   175→                    // Struct type already registered in pass1. No runtime instruction needed.
   176→   176→   176→                    // (The VM will need to know about struct types for MakeStruct.)
   177→   177→   177→                    // We emit a special marker or nothing — the VM handles this via metadata.
   178→   178→   178→                    let _ = e;
   179→   179→   179→                }
   180→   180→   180→                Declaration::EntityRecord(e) => {
   181→   181→   181→                    // Evaluate each field initializer, create struct, store globally
   182→   182→   182→                    let field_names = self.struct_fields.get(&e.type_name)
   183→   183→   183→                        .ok_or_else(|| format!("compile: unknown struct type: {}", e.type_name))?;
   184→   184→   184→
   185→   185→   185→                    // Push field values in field order
   186→   186→   186→                    for fd_name in field_names {
   187→   187→   187→                        // Find initializer for this field
   188→   188→   188→                        let init = e.fields.iter().find(|fi| fi.name == *fd_name);
   189→   189→   189→                        match init {
   190→   190→   190→                            Some(fi) => self.compile_expr(&fi.value, &mut code)?,
   191→   191→   191→                            None => code.push(Instruction::Const(Value::Unit)),
   192→   192→   192→                        }
   193→   193→   193→                    }
   194→   194→   194→
   195→   195→   195→                    let slot = self.global_slots[&e.name];
   196→   196→   196→                    code.push(Instruction::MakeStruct(e.type_name.clone(), field_names.clone()));
   197→   197→   197→                    code.push(Instruction::StoreGlobal(slot));
   198→   198→   198→                }
   199→   199→   199→                Declaration::EntitySimple(e) => {
   200→   200→   200→                    self.compile_expr(&e.value, &mut code)?;
   201→   201→   201→                    let slot = self.global_slots[&e.name];
   202→   202→   202→                    code.push(Instruction::StoreGlobal(slot));
   203→   203→   203→                }
   204→   204→   204→                Declaration::Pattern(p) => {
   205→   205→   205→                    // Compile pattern body with parameter names as locals
   206→   206→   206→                    let mut locals: HashMap<String, usize> = p.params.iter()
   207→   207→   207→                        .enumerate()
   208→   208→   208→                        .map(|(i, param)| (param.name.clone(), i))
   209→   209→   209→                        .collect();
   210→   210→   210→                    let fn_code = self.compile_pattern_body_with_locals(&p.body, &mut locals)?;
   211→   211→   211→                    let is_pure = Self::analyze_purity(&fn_code, &p.params);
   212→   212→   212→                    let compiled = CompiledFn {
   213→   213→   213→                        name: p.name.clone(),
   214→   214→   214→                        param_count: p.params.len(),
   215→   215→   215→                        param_types: p.params.iter().map(|param| param.type_name.clone()).collect(),
   216→   216→   216→                        code: fn_code,
   217→   217→   217→                        is_pure,
   218→   218→   218→                    };
   219→   219→   219→                    // We'll add this to the program's patterns list
   220→   220→   220→                    // For now, store in a side-channel; we'll fix this below.
   221→   221→   221→                    // Actually, let's store directly. The problem is that we're building
   222→   222→   222→                    // the program in compile(), not here. Let's use a different approach.
   223→   223→   223→                    // We'll add a pseudo-instruction to register the pattern.
   224→   224→   224→                    code.push(Instruction::RegisterPattern(compiled));
   225→   225→   225→                }
   226→   226→   226→                Declaration::LearnablePattern(lp) => {
   227→   227→   227→                    code.push(Instruction::RegisterLearnable(CompiledLearnableInfo {
   228→   228→   228→                        name: lp.name.clone(),
   229→   229→   229→                        param_count: lp.params.len(),
   230→   230→   230→                        prompt: lp.prompt.clone(),
   231→   231→   231→                        few_shot: Vec::new(),
   232→   232→   232→                    }));
   233→   233→   233→                }
   234→   234→   234→                Declaration::Rule(_) => {
   235→   235→   235→                    // Rules are already compiled. Emit ExecuteRules before any flow.
   236→   236→   236→                    // We'll insert this before the flow instruction.
   237→   237→   237→                }
   238→   238→   238→                Declaration::Memorize(m) => {
   239→   239→   239→                    self.compile_expr(&m.value, &mut code)?;
   240→   240→   240→                    code.push(Instruction::Memorize(m.priority));
   241→   241→   241→                }
   242→   242→   242→                Declaration::Forget(f) => {
   243→   243→   243→                    self.compile_expr(&f.query, &mut code)?;
   244→   244→   244→                    // We need a Forget instruction — let's add it to the bytecode
   245→   245→   245→                    // For now, emit as a special instruction
   246→   246→   246→                    code.push(Instruction::Forget(f.days));
   247→   247→   247→                }
   248→   248→   248→                Declaration::Fluid(fl) => {
   249→   249→   249→                    // Compile Fluid value construction: push value, confidence pairs
   250→   250→   250→                    for v in &fl.variants {
   251→   251→   251→                        self.compile_expr(&v.value, &mut code)?;
   252→   252→   252→                        // Push confidence
   253→   253→   253→                        code.push(Instruction::Const(Value::Float(v.confidence)));
   254→   254→   254→                    }
   255→   255→   255→                    let slot = self.global_slots[&fl.name];
   256→   256→   256→                    code.push(Instruction::MakeFluid(fl.variants.len()));
   257→   257→   257→                    code.push(Instruction::StoreGlobal(slot));
   258→   258→   258→                }
   259→   259→   259→                Declaration::Adapt(a) => {
   260→   260→   260→                    // Evaluate input/output examples (try to resolve literals)
   261→   261→   261→                    // For now, just emit an adapt instruction
   262→   262→   262→                    self.compile_expr(&a.input_example, &mut code)?;
   263→   263→   263→                    self.compile_expr(&a.output_example, &mut code)?;
   264→   264→   264→                    code.push(Instruction::Adapt(a.pattern_name.clone()));
   265→   265→   265→                }
   266→   266→   266→                Declaration::Relate(r) => {
   267→   267→   267→                    self.compile_expr(&r.from, &mut code)?;
   268→   268→   268→                    self.compile_expr(&r.to, &mut code)?;
   269→   269→   269→                    code.push(Instruction::Const(Value::String(r.relation.clone())));
   270→   270→   270→                    code.push(Instruction::Relate);
   271→   271→   271→                }
   272→   272→   272→                Declaration::Sandbox(_) => {
   273→   273→   273→                    // No runtime instruction needed
   274→   274→   274→                }
   275→   275→   275→                Declaration::Mutate(m) => {
   276→   276→   276→                    // Compile new examples and rollback info
   277→   277→   277→                    let mut examples = Vec::new();
   278→   278→   278→                    for (inp, out) in &m.new_examples {
   279→   279→   279→                        self.compile_expr(inp, &mut code)?;
   280→   280→   280→                        self.compile_expr(out, &mut code)?;
   281→   281→   281→                        examples.push((String::new(), String::new())); // placeholder
   282→   282→   282→                    }
   283→   283→   283→                    let rollback_op = m.rollback_op.map(|op| match op {
   284→   284→   284→                        AstCompareOp::Gt => ConditionOp::Gt,
   285→   285→   285→                        AstCompareOp::Lt => ConditionOp::Lt,
   286→   286→   286→                        AstCompareOp::Ge => ConditionOp::Ge,
   287→   287→   287→                        AstCompareOp::Le => ConditionOp::Le,
   288→   288→   288→                        AstCompareOp::Eq => ConditionOp::Eq,
   289→   289→   289→                        _ => ConditionOp::Eq, // Ne and others fall back to Eq
   290→   290→   290→                    });
   291→   291→   291→                    code.push(Instruction::Mutate {
   292→   292→   292→                        pattern_name: m.pattern_name.clone(),
   293→   293→   293→                        example_count: m.new_examples.len(),
   294→   294→   294→                        rollback_threshold: m.rollback_threshold,
   295→   295→   295→                        rollback_op,
   296→   296→   296→                    });
   297→   297→   297→                }
   298→   298→   298→                Declaration::Flow(f) => {
   299→   299→   299→                    // Emit ExecuteRules before the flow (if any rules exist)
   300→   300→   300→                    if !self.rules.is_empty() {
   301→   301→   301→                        code.push(Instruction::ExecuteRules);
   302→   302→   302→                    }
   303→   303→   303→                    // Compile the flow source expression
   304→   304→   304→                    let source_expr = self.compile_flow_expr(&f.source);
   305→   305→   305→                    let mut branch_defs = Vec::new();
   306→   306→   306→                    for (step_name, branches) in &f.branch_defs {
   307→   307→   307→                        let compiled_branches: Vec<BranchDef> = branches.iter()
   308→   308→   308→                            .map(|b| {
   309→   309→   309→                                let op = match b.condition.op {
   310→   310→   310→                                    AstCompareOp::Gt => ConditionOp::Gt,
   311→   311→   311→                                    AstCompareOp::Lt => ConditionOp::Lt,
   312→   312→   312→                                    AstCompareOp::Ge => ConditionOp::Ge,
   313→   313→   313→                                    AstCompareOp::Le => ConditionOp::Le,
   314→   314→   314→                                    AstCompareOp::Eq => ConditionOp::Eq,
   315→   315→   315→                                    _ => ConditionOp::Eq, // Ne and others fall back to Eq
   316→   316→   316→                                };
   317→   317→   317→                                // Compile the threshold expression to a constant if possible
   318→   318→   318→                                let threshold_val = self.eval_const_expr(&b.condition.threshold);
   319→   319→   319→                                BranchDef {
   320→   320→   320→                                    label: b.label.clone(),
   321→   321→   321→                                    condition_field: b.condition.field.clone(),
   322→   322→   322→                                    condition_op: op,
   323→   323→   323→                                    condition_threshold: threshold_val,
   324→   324→   324→                                    target: b.target.clone(),
   325→   325→   325→                                }
   326→   326→   326→                            })
   327→   327→   327→                            .collect();
   328→   328→   328→                        branch_defs.push((step_name.clone(), compiled_branches));
   329→   329→   329→                    }
   330→   330→   330→                    code.push(Instruction::FlowExec {
   331→   331→   331→                        source_expr,
   332→   332→   332→                        pipeline: f.pipeline.clone(),
   333→   333→   333→                        branch_defs,
   334→   334→   334→                    });
   335→   335→   335→                }
   336→   336→   336→                Declaration::Import(_) => {
   337→   337→   337→                    // Already resolved in import preprocessing
   338→   338→   338→                }
   339→   339→   339→                Declaration::MlogServer(_) | Declaration::Template(_) | Declaration::Db(_) | Declaration::Memory(_)
   340→   340→   340→                | Declaration::Hook(_) | Declaration::Eval(_) | Declaration::Conversation(_) | Declaration::Tool(_) | Declaration::LlmConfig(_) => {
   341→   341→   341→                    // Phase 6+: no bytecode instruction needed
   342→   342→   342→                }
   343→   343→   343→            }
   344→   344→   344→        }
   345→   345→   345→
   346→   346→   346→        code.push(Instruction::Halt);
   347→   347→   347→        Ok(code)
   348→   348→   348→    }

   350→   350→   350→    /// Compile an AST expression into stack instructions (no locals context).
   351→   351→   351→    fn compile_expr(&self, expr: &Expr, code: &mut Vec<Instruction>) -> Result<(), String> {
   352→   352→   352→        self.compile_expr_with_locals(expr, code, &HashMap::new())
   353→   353→   353→    }
   354→   354→   354→
   355→   355→   355→    /// Compile an AST expression into stack instructions with a locals map.
   356→   356→   356→    fn compile_expr_with_locals(&self, expr: &Expr, code: &mut Vec<Instruction>, locals: &HashMap<String, usize>) -> Result<(), String> {
   357→   357→   357→        match expr {
   358→   358→   358→            Expr::StringLit(s) => {
   359→   359→   359→                code.push(Instruction::Const(Value::String(s.clone())));
   360→   360→   360→            }
   361→   361→   361→            Expr::FloatLit(f) => {
   362→   362→   362→                code.push(Instruction::Const(Value::Float(*f)));
   363→   363→   363→            }
   364→   364→   364→            Expr::Ident(name) => {
   365→   365→   365→                // Check if it's a local (parameter or let binding) first
   366→   366→   366→                if let Some(&slot) = locals.get(name) {
   367→   367→   367→                    code.push(Instruction::LoadLocal(slot));
   368→   368→   368→                } else if let Some(&slot) = self.global_slots.get(name) {
   369→   369→   369→                    code.push(Instruction::LoadGlobal(slot));
   370→   370→   370→                } else {
   371→   371→   371→                    code.push(Instruction::LoadGlobalByName(name.clone()));
   372→   372→   372→                }
   373→   373→   373→            }
   374→   374→   374→            Expr::FieldAccess(base, field) => {
   375→   375→   375→                self.compile_expr_with_locals(base, code, locals)?;
   376→   376→   376→                code.push(Instruction::GetField(field.clone()));
   377→   377→   377→            }
   378→   378→   378→            Expr::FnCall(name, args) => {
   379→   379→   379→                for arg in args {
   380→   380→   380→                    self.compile_expr_with_locals(arg, code, locals)?;
   381→   381→   381→                }
   382→   382→   382→                let arity = args.len();
   383→   383→   383→                // Check if it's a builtin
   384→   384→   384→                if let Some(&idx) = self.builtin_indices.get(name) {

   479→   479→   479→    fn compile_pattern_body_with_locals(&self, body: &[Statement], locals: &mut HashMap<String, usize>) -> Result<Vec<Instruction>, String> {

   486→   486→   486→                    let slot = next_slot;

   507→   507→   507→                Statement::EachWithIndex { index_var, item_var, iterable, body } => {

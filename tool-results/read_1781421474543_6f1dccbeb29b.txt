     1→// ── METALOGOS Bytecode Compiler — Phase 4.1 ────────────────────
     2→// Translates AST (Vec<Declaration>) into a bytecode Program.
     3→// Contract: the emitted Program, when executed by the VM, produces
     4→// the same output as the tree-walking interpreter.
     5→
     6→use std::collections::{HashMap, HashSet};
     7→use std::path::PathBuf;
     8→
     9→use crate::ast::*;
    10→use crate::ast::CompareOp as AstCompareOp;
    11→use crate::bytecode::*;
    12→use crate::interpreter::Value;
    13→
    14→/// The compiler translates declarations into a bytecode Program.
    15→pub struct Compiler {
    16→    /// Global variable name -> slot index.
    17→    global_slots: HashMap<String, usize>,
    18→    /// Next available global slot.
    19→    next_global: usize,
    20→    /// Pattern name -> index in program.patterns.
    21→    pattern_indices: HashMap<String, usize>,
    22→    /// Learnable pattern name -> index in program.learnables.
    23→    learnable_indices: HashMap<String, usize>,
    24→    /// Builtin name -> index in builtin table.
    25→    builtin_indices: HashMap<String, usize>,
    26→    /// Struct type name -> field names (in declaration order).
    27→    struct_fields: HashMap<String, Vec<String>>,
    28→    /// Rule declarations to compile.
    29→    rules: Vec<CompiledRule>,
    30→    /// Root directory for import resolution.
    31→    std_root: PathBuf,
    32→    /// Already-imported modules.
    33→    imported_modules: HashSet<String>,
    34→    /// Collections loaded flag.
    35→    collections_loaded: bool,
    36→    /// Sandbox declarations (recorded).
    37→    sandboxes: HashMap<String, SandboxDecl>,
    38→}
    39→
    40→
    41→
    42→impl Compiler {
    43→    /// Create a new compiler with default settings.
    44→    pub fn new() -> Self {
    45→        Self::with_std_root(
    46→            std::env::current_dir().unwrap_or_else(|_| PathBuf::from("."))
    47→        )
    48→    }
    49→
    50→    /// Create a compiler with a custom std root directory for import resolution.
    51→    pub fn with_std_root(std_root: PathBuf) -> Self {
    52→        let mut builtin_indices = HashMap::new();
    53→        let builtins = [
    54→            "upper", "lower", "len", "str", "print", "contains", "float",
    55→            "__trim", "__replace", "__split", "__join",
    56→            "__abs", "__min", "__max", "__clamp", "__round",
    57→            "__first", "__last", "__push", "__list_len",
    58→            "recall",
    59→            // Phase 4.4 self-hosting
    60→            "stdin", "split_tokens", "if_eq", "newline", "is_string_token",
    61→        ];
    62→        for (i, name) in builtins.iter().enumerate() {
    63→            builtin_indices.insert(name.to_string(), i);
    64→        }
    65→
    66→        Compiler {
    67→            global_slots: HashMap::new(),
    68→            next_global: 0,
    69→            pattern_indices: HashMap::new(),
    70→            learnable_indices: HashMap::new(),
    71→            builtin_indices,
    72→            struct_fields: HashMap::new(),
    73→            rules: Vec::new(),
    74→            std_root,
    75→            imported_modules: HashSet::new(),
    76→            collections_loaded: false,
    77→            sandboxes: HashMap::new(),
    78→        }
    79→    }
    80→
    81→    /// Compile a list of declarations into a bytecode Program.
    82→    pub fn compile(&mut self, declarations: Vec<Declaration>) -> Result<Program, String> {
    83→        // Phase 1: resolve imports
    84→        let mut all_decls = Vec::new();
    85→        for decl in declarations {
    86→            if let Declaration::Import(import) = &decl {
    87→                // Fix 2: use `import.path` instead of `import.module_path`
    88→                if !self.imported_modules.contains(&import.path) {
    89→                    let imported = self.resolve_import(&import.path)?;
    90→                    all_decls.extend(imported);
    91→                }
    92→            } else {
    93→                all_decls.push(decl);
    94→            }
    95→        }
    96→
    97→        // Phase 2: two-pass compilation
    98→        // Pass 1: collect struct types, pattern names, learnable names, global slots
    99→        self.pass1(&all_decls)?;
   100→
   101→        // Pass 2: generate main_code
   102→        let main_code = self.pass2(&all_decls)?;
   103→
   104→        let program = Program {
   105→            // Build globals list with actual names (indexed by slot)
   106→            globals: (0..self.next_global)
   107→                .map(|i| self.global_slots.iter()
   108→                    .find(|(_, &slot)| slot == i)
   109→                    .map(|(name, _)| name.clone())
   110→                    .unwrap_or_default())
   111→                .collect(),
   112→            patterns: Vec::new(), // will be filled from pass1 data
   113→            learnables: Vec::new(),
   114→            rules: std::mem::take(&mut self.rules),
   115→            main_code,
   116→            collections_loaded: self.collections_loaded,
   117→        };
   118→
   119→        Ok(program)
   120→    }
   121→
   122→    /// Pass 1: collect type info, register patterns, assign global slots.
   123→    fn pass1(&mut self, decls: &[Declaration]) -> Result<(), String> {
   124→        for decl in decls {
   125→            match decl {
   126→                Declaration::EntityType(e) => {
   127→                    let fields: Vec<String> = e.fields.iter().map(|f| f.name.clone()).collect();
   128→                    self.struct_fields.insert(e.name.clone(), fields);
   129→                }
   130→                Declaration::EntityRecord(e) => {
   131→                    self.ensure_global(&e.name);
   132→                }
   133→                Declaration::EntitySimple(e) => {
   134→                    self.ensure_global(&e.name);
   135→                }
   136→                Declaration::Pattern(p) => {
   137→                    let idx = self.pattern_indices.len();
   138→                    self.pattern_indices.insert(p.name.clone(), idx);
   139→                }
   140→                Declaration::LearnablePattern(lp) => {
   141→                    let idx = self.learnable_indices.len();
   142→                    self.learnable_indices.insert(lp.name.clone(), idx);
   143→                }
   144→                Declaration::Fluid(fl) => {
   145→                    self.ensure_global(&fl.name);
   146→                }
   147→                Declaration::Rule(r) => {
   148→                    self.rules.push(self.compile_rule(r)?);
   149→                }
   150→                Declaration::Memorize(_) => {
   151→                    // Handled in pass2
   152→                }
   153→                Declaration::Sandbox(s) => {
   154→                    self.sandboxes.insert(s.name.clone(), s.clone());
   155→                }
   156→                Declaration::Adapt(_) | Declaration::Relate(_) | Declaration::Mutate(_) | Declaration::Forget(_) => {
   157→                    // Handled in pass2
   158→                }
   159→                Declaration::MlogServer(_) | Declaration::Template(_) | Declaration::Db(_) | Declaration::Memory(_) | Declaration::Conversation(_) | Declaration::Tool(_) | Declaration::LlmConfig(_) => {
   160→                    // Phase 6: handled elsewhere
   161→                }
   162→                _ => {}
   163→            }
   164→        }
   165→        Ok(())
   166→    }
   167→
   168→    /// Pass 2: generate main_code instructions.
   169→    fn pass2(&mut self, decls: &[Declaration]) -> Result<Vec<Instruction>, String> {
   170→        let mut code = Vec::new();
   171→
   172→        for decl in decls {
   173→            match decl {
   174→                Declaration::EntityType(e) => {
   175→                    // Struct type already registered in pass1. No runtime instruction needed.
   176→                    // (The VM will need to know about struct types for MakeStruct.)
   177→                    // We emit a special marker or nothing — the VM handles this via metadata.
   178→                    let _ = e;
   179→                }
   180→                Declaration::EntityRecord(e) => {
   181→                    // Evaluate each field initializer, create struct, store globally
   182→                    let field_names = self.struct_fields.get(&e.type_name)
   183→                        .ok_or_else(|| format!("compile: unknown struct type: {}", e.type_name))?;
   184→
   185→                    // Push field values in field order
   186→                    for fd_name in field_names {
   187→                        // Find initializer for this field
   188→                        let init = e.fields.iter().find(|fi| fi.name == *fd_name);
   189→                        match init {
   190→                            Some(fi) => self.compile_expr(&fi.value, &mut code)?,
   191→                            None => code.push(Instruction::Const(Value::Unit)),
   192→                        }
   193→                    }
   194→
   195→                    let slot = self.global_slots[&e.name];
   196→                    code.push(Instruction::MakeStruct(e.type_name.clone(), field_names.clone()));
   197→                    code.push(Instruction::StoreGlobal(slot));
   198→                }
   199→                Declaration::EntitySimple(e) => {
   200→                    self.compile_expr(&e.value, &mut code)?;
   201→                    let slot = self.global_slots[&e.name];
   202→                    code.push(Instruction::StoreGlobal(slot));
   203→                }
   204→                Declaration::Pattern(p) => {
   205→                    // Compile pattern body with parameter names as locals
   206→                    let mut locals: HashMap<String, usize> = p.params.iter()
   207→                        .enumerate()
   208→                        .map(|(i, param)| (param.name.clone(), i))
   209→                        .collect();
   210→                    let fn_code = self.compile_pattern_body_with_locals(&p.body, &mut locals)?;
   211→                    let is_pure = Self::analyze_purity(&fn_code, &p.params);
   212→                    let compiled = CompiledFn {
   213→                        name: p.name.clone(),
   214→                        param_count: p.params.len(),
   215→                        param_types: p.params.iter().map(|param| param.type_name.clone()).collect(),
   216→                        code: fn_code,
   217→                        is_pure,
   218→                    };
   219→                    // We'll add this to the program's patterns list
   220→                    // For now, store in a side-channel; we'll fix this below.
   221→                    // Actually, let's store directly. The problem is that we're building
   222→                    // the program in compile(), not here. Let's use a different approach.
   223→                    // We'll add a pseudo-instruction to register the pattern.
   224→                    code.push(Instruction::RegisterPattern(compiled));
   225→                }
   226→                Declaration::LearnablePattern(lp) => {
   227→                    code.push(Instruction::RegisterLearnable(CompiledLearnableInfo {
   228→                        name: lp.name.clone(),
   229→                        param_count: lp.params.len(),
   230→                        prompt: lp.prompt.clone(),
   231→                        few_shot: Vec::new(),
   232→                    }));
   233→                }
   234→                Declaration::Rule(_) => {
   235→                    // Rules are already compiled. Emit ExecuteRules before any flow.
   236→                    // We'll insert this before the flow instruction.
   237→                }
   238→                Declaration::Memorize(m) => {
   239→                    self.compile_expr(&m.value, &mut code)?;
   240→                    code.push(Instruction::Memorize(m.priority));
   241→                }
   242→                Declaration::Forget(f) => {
   243→                    self.compile_expr(&f.query, &mut code)?;
   244→                    // We need a Forget instruction — let's add it to the bytecode
   245→                    // For now, emit as a special instruction
   246→                    code.push(Instruction::Forget(f.days));
   247→                }
   248→                Declaration::Fluid(fl) => {
   249→                    // Compile Fluid value construction: push value, confidence pairs
   250→                    for v in &fl.variants {
   251→                        self.compile_expr(&v.value, &mut code)?;
   252→                        // Push confidence
   253→                        code.push(Instruction::Const(Value::Float(v.confidence)));
   254→                    }
   255→                    let slot = self.global_slots[&fl.name];
   256→                    code.push(Instruction::MakeFluid(fl.variants.len()));
   257→                    code.push(Instruction::StoreGlobal(slot));
   258→                }
   259→                Declaration::Adapt(a) => {
   260→                    // Evaluate input/output examples (try to resolve literals)
   261→                    // For now, just emit an adapt instruction
   262→                    self.compile_expr(&a.input_example, &mut code)?;
   263→                    self.compile_expr(&a.output_example, &mut code)?;
   264→                    code.push(Instruction::Adapt(a.pattern_name.clone()));
   265→                }
   266→                Declaration::Relate(r) => {
   267→                    self.compile_expr(&r.from, &mut code)?;
   268→                    self.compile_expr(&r.to, &mut code)?;
   269→                    code.push(Instruction::Const(Value::String(r.relation.clone())));
   270→                    code.push(Instruction::Relate);
   271→                }
   272→                Declaration::Sandbox(_) => {
   273→                    // No runtime instruction needed
   274→                }
   275→                Declaration::Mutate(m) => {
   276→                    // Compile new examples and rollback info
   277→                    let mut examples = Vec::new();
   278→                    for (inp, out) in &m.new_examples {
   279→                        self.compile_expr(inp, &mut code)?;
   280→                        self.compile_expr(out, &mut code)?;
   281→                        examples.push((String::new(), String::new())); // placeholder
   282→                    }
   283→                    let rollback_op = m.rollback_op.map(|op| match op {
   284→                        AstCompareOp::Gt => ConditionOp::Gt,
   285→                        AstCompareOp::Lt => ConditionOp::Lt,
   286→                        AstCompareOp::Ge => ConditionOp::Ge,
   287→                        AstCompareOp::Le => ConditionOp::Le,
   288→                        AstCompareOp::Eq => ConditionOp::Eq,
   289→                        _ => ConditionOp::Eq, // Ne and others fall back to Eq
   290→                    });
   291→                    code.push(Instruction::Mutate {
   292→                        pattern_name: m.pattern_name.clone(),
   293→                        example_count: m.new_examples.len(),
   294→                        rollback_threshold: m.rollback_threshold,
   295→                        rollback_op,
   296→                    });
   297→                }
   298→                Declaration::Flow(f) => {
   299→                    // Emit ExecuteRules before the flow (if any rules exist)
   300→                    if !self.rules.is_empty() {
   301→                        code.push(Instruction::ExecuteRules);
   302→                    }
   303→                    // Compile the flow source expression
   304→                    let source_expr = self.compile_flow_expr(&f.source);
   305→                    let mut branch_defs = Vec::new();
   306→                    for (step_name, branches) in &f.branch_defs {
   307→                        let compiled_branches: Vec<BranchDef> = branches.iter()
   308→                            .map(|b| {
   309→                                let op = match b.condition.op {
   310→                                    AstCompareOp::Gt => ConditionOp::Gt,
   311→                                    AstCompareOp::Lt => ConditionOp::Lt,
   312→                                    AstCompareOp::Ge => ConditionOp::Ge,
   313→                                    AstCompareOp::Le => ConditionOp::Le,
   314→                                    AstCompareOp::Eq => ConditionOp::Eq,
   315→                                    _ => ConditionOp::Eq, // Ne and others fall back to Eq
   316→                                };
   317→                                // Compile the threshold expression to a constant if possible
   318→                                let threshold_val = self.eval_const_expr(&b.condition.threshold);
   319→                                BranchDef {
   320→                                    label: b.label.clone(),
   321→                                    condition_field: b.condition.field.clone(),
   322→                                    condition_op: op,
   323→                                    condition_threshold: threshold_val,
   324→                                    target: b.target.clone(),
   325→                                }
   326→                            })
   327→                            .collect();
   328→                        branch_defs.push((step_name.clone(), compiled_branches));
   329→                    }
   330→                    code.push(Instruction::FlowExec {
   331→                        source_expr,
   332→                        pipeline: f.pipeline.clone(),
   333→                        branch_defs,
   334→                    });
   335→                }
   336→                Declaration::Import(_) => {
   337→                    // Already resolved in import preprocessing
   338→                }
   339→                Declaration::MlogServer(_) | Declaration::Template(_) | Declaration::Db(_) | Declaration::Memory(_)
   340→                | Declaration::Hook(_) | Declaration::Eval(_) | Declaration::Conversation(_) | Declaration::Tool(_) | Declaration::LlmConfig(_) => {
   341→                    // Phase 6+: no bytecode instruction needed
   342→                }
   343→            }
   344→        }
   345→
   346→        code.push(Instruction::Halt);
   347→        Ok(code)
   348→    }

   350→    /// Compile an AST expression into stack instructions (no locals context).
   351→    fn compile_expr(&self, expr: &Expr, code: &mut Vec<Instruction>) -> Result<(), String> {
   352→        self.compile_expr_with_locals(expr, code, &HashMap::new())
   353→    }
   354→
   355→    /// Compile an AST expression into stack instructions with a locals map.
   356→    fn compile_expr_with_locals(&self, expr: &Expr, code: &mut Vec<Instruction>, locals: &HashMap<String, usize>) -> Result<(), String> {
   357→        match expr {
   358→            Expr::StringLit(s) => {
   359→                code.push(Instruction::Const(Value::String(s.clone())));
   360→            }
   361→            Expr::FloatLit(f) => {
   362→                code.push(Instruction::Const(Value::Float(*f)));
   363→            }
   364→            Expr::Ident(name) => {
   365→                // Check if it's a local (parameter or let binding) first
   366→                if let Some(&slot) = locals.get(name) {
   367→                    code.push(Instruction::LoadLocal(slot));
   368→                } else if let Some(&slot) = self.global_slots.get(name) {
   369→                    code.push(Instruction::LoadGlobal(slot));
   370→                } else {
   371→                    code.push(Instruction::LoadGlobalByName(name.clone()));
   372→                }
   373→            }
   374→            Expr::FieldAccess(base, field) => {
   375→                self.compile_expr_with_locals(base, code, locals)?;
   376→                code.push(Instruction::GetField(field.clone()));
   377→            }
   378→            Expr::FnCall(name, args) => {
   379→                for arg in args {
   380→                    self.compile_expr_with_locals(arg, code, locals)?;
   381→                }
   382→                let arity = args.len();
   383→                // Check if it's a builtin
   384→                if let Some(&idx) = self.builtin_indices.get(name) {
   385→                    code.push(Instruction::CallBuiltin(idx, arity));
   386→                } else if let Some(&idx) = self.pattern_indices.get(name) {
   387→                    code.push(Instruction::CallPattern(idx, arity));
   388→                } else if let Some(&idx) = self.learnable_indices.get(name) {
   389→                    code.push(Instruction::LlmCall(idx, arity));
   390→                } else {
   391→                    return Err(format!("compile: undefined function: {}", name));
   392→                }
   393→            }
   394→            Expr::BinaryOp(left, op, right) => {
   395→                self.compile_expr_with_locals(left, code, locals)?;
   396→                self.compile_expr_with_locals(right, code, locals)?;
   397→                match op {
   398→                    BinOp::Add => code.push(Instruction::Add),
   399→                    BinOp::Sub => code.push(Instruction::Sub),
   400→                    BinOp::Mul => code.push(Instruction::Mul),
   401→                    BinOp::Div => code.push(Instruction::Div),
   402→                    // Phase 5.1: comparison operators
   403→                    BinOp::Gt => code.push(Instruction::CmpGt),
   404→                    BinOp::Lt => code.push(Instruction::CmpLt),
   405→                    BinOp::Ge => code.push(Instruction::CmpGe),
   406→                    BinOp::Le => code.push(Instruction::CmpLe),
   407→                    BinOp::Eq => code.push(Instruction::CmpEq),
   408→                    BinOp::Ne => code.push(Instruction::CmpNe),
   409→                }
   410→            }
   411→            Expr::IfElse(cond, then_expr, else_expr) => {
   412→                // Compile condition
   413→                self.compile_expr_with_locals(cond, code, locals)?;
   414→                // Jump to else branch if falsy
   415→                let jump_to_else = code.len();
   416→                code.push(Instruction::JumpIfNot(0)); // placeholder
   417→                // Compile then branch
   418→                self.compile_expr_with_locals(then_expr, code, locals)?;
   419→                // Jump past else branch
   420→                let jump_to_end = code.len();
   421→                code.push(Instruction::Jump(0)); // placeholder
   422→                // Patch: else branch starts here
   423→                let else_start = code.len();
   424→                if let Some(Instruction::JumpIfNot(ref mut target)) = code.get_mut(jump_to_else) {
   425→                    *target = else_start;
   426→                }
   427→                // Compile else branch
   428→                self.compile_expr_with_locals(else_expr, code, locals)?;
   429→                // Patch: end jump target
   430→                let end = code.len();
   431→                if let Some(Instruction::Jump(ref mut target)) = code.get_mut(jump_to_end) {
   432→                    *target = end;
   433→                }
   434→            }
   435→            // Additional expression forms in Metalogos- AST
   436→            Expr::BoolLit(b) => {
   437→                code.push(Instruction::Const(Value::Float(if *b { 1.0 } else { 0.0 })));
   438→            }
   439→            Expr::QualifiedCall { module: _, function: _, args: _ } => {
   440→                return Err("compile: qualified calls not yet supported in bytecode".to_string());
   441→            }
   442→            Expr::List(items) => {
   443→                // Push each item, then construct via MakeStruct-like approach
   444→                // For now, emit as individual pushes (list construction not fully supported)
   445→                for item in items {
   446→                    self.compile_expr_with_locals(item, code, locals)?;
   447→                }
   448→                // We'll use a special const for list length tracking (Phase 5)
   449→                // For now, lists are pushed as individual values
   450→                code.push(Instruction::Const(Value::Float(items.len() as f64)));
   451→            }
   452→            Expr::IndexAccess(base, index) => {
   453→                self.compile_expr_with_locals(base, code, locals)?;
   454→                self.compile_expr_with_locals(index, code, locals)?;
   455→                code.push(Instruction::IndexAccess);
   456→            }
   457→            Expr::StructLit(fields) => {
   458→                let field_names: Vec<String> = fields.keys().cloned().collect();
   459→                for (_, val_expr) in fields {
   460→                    self.compile_expr_with_locals(val_expr, code, locals)?;
   461→                }
   462→                code.push(Instruction::MakeStruct("Struct".to_string(), field_names));
   463→            }
   464→            // Наряд №14 P0-3: block if/else expression — deferred to tree-walking
   465→            Expr::BlockIfElse { .. } => {
   466→                // Compiled as Unit placeholder; tree-walking interpreter handles it
   467→                code.push(Instruction::Const(Value::Unit));
   468→            }
   469→            // Наряд №14 P1-4: try expression — deferred to tree-walking
   470→            Expr::Try(_) => {
   471→                code.push(Instruction::Const(Value::Unit));
   472→            }
   473→        }
   474→        Ok(())
   475→    }

   477→    /// Compile a pattern body with parameter names as locals.
   478→    /// Phase 5.1: also handles let bindings, assigning additional local slots.
   479→    fn compile_pattern_body_with_locals(&self, body: &[Statement], locals: &mut HashMap<String, usize>) -> Result<Vec<Instruction>, String> {
   480→        let mut code = Vec::new();
   481→        let mut next_slot = locals.len();
   482→        for stmt in body {
   483→            match stmt {
   484→                // Fix 1: use Statement::LetBinding { name, value } instead of Statement::Let(name, expr)
   485→                Statement::LetBinding { name, value, mutable: _ } => {
   486→                    let slot = next_slot;
   487→                    next_slot += 1;
   488→                    locals.insert(name.clone(), slot);
   489→                    self.compile_expr_with_locals(value, &mut code, locals)?;
   490→                    code.push(Instruction::StoreLocal(slot));
   491→                }
   492→                Statement::Return(expr) => {
   493→                    self.compile_expr_with_locals(expr, &mut code, locals)?;
   494→                    code.push(Instruction::Return);
   495→                }
   496→                // Наряд №14: match — compiled as chained if-jump-else (simplified)
   497→                Statement::Match { scrutinee, arms, else_body } => {
   498→                    self.compile_expr_with_locals(scrutinee, &mut code, locals)?;
   499→                    // For now, store scrutinee and evaluate arms via tree-walking fallback.
   500→                    // Full bytecode match compilation is deferred — the VM path uses
   501→                    // the tree-walking interpreter for match statements.
   502→                    let _ = (arms, else_body);
   503→                }
   504→                // Наряд №17: break/continue — not meaningful in non-loop context
   505→                Statement::Break | Statement::Continue => {}
   506→                // Наряд №17.3: each with index — compile as indexed loop
   507→                Statement::EachWithIndex { index_var, item_var, iterable, body } => {
   508→                    // Compile: evaluate iterable, then for each item push index + item to locals
   509→                    // For now, emit as Unit (full compilation deferred to Наряд №18)
   510→                    let _ = (index_var, item_var, iterable, body);
   511→                }
   512→                _ => {}
   513→            }
   514→        }
   515→        Ok(code)
   516→    }

   518→    /// Compile a flow source expression into a FlowExpr.
   519→    fn compile_flow_expr(&self, expr: &Expr) -> FlowExpr {
   520→        match expr {
   521→            Expr::Ident(name) => {
   522→                if let Some(&slot) = self.global_slots.get(name) {
   523→                    FlowExpr::GlobalSlot(slot)
   524→                } else {
   525→                    FlowExpr::Ident(name.clone())
   526→                }
   527→            }
   528→            Expr::StringLit(s) => FlowExpr::Const(Value::String(s.clone())),
   529→            Expr::FloatLit(f) => FlowExpr::Const(Value::Float(*f)),
   530→            _ => FlowExpr::Ident(format!("{:?}", expr)),
   531→        }
   532→    }

   543→    /// Compile a rule into CompiledRule.
   544→    fn compile_rule(&self, rule: &RuleDecl) -> Result<CompiledRule, String> {
   545→        let condition = match &rule.condition {
   546→            Condition::Contains { left, right } => {
   547→                RuleCondition::Contains {
   548→                    left: self.rule_value_expr(left),
   549→                    right: self.rule_value_expr(right),
   550→                }
   551→            }
   552→            Condition::Compare { left, op, right } => {
   553→                RuleCondition::Compare {
   554→                    left: self.rule_value_expr(left),
   555→                    op: match op {
   556→                        AstCompareOp::Gt => ConditionOp::Gt,
   557→                        AstCompareOp::Lt => ConditionOp::Lt,
   558→                        AstCompareOp::Ge => ConditionOp::Ge,
   559→                        AstCompareOp::Le => ConditionOp::Le,
   560→                        AstCompareOp::Eq => ConditionOp::Eq,
   561→                        _ => ConditionOp::Eq, // Ne and others fall back to Eq
   562→                    },
   563→                    right: self.rule_value_expr(right),
   564→                }
   565→            }
   566→        };
   567→
   568→        let target_name = match &rule.target {
   569→            Expr::Ident(name) => name.clone(),
   570→            _ => return Err("rule target must be an identifier".to_string()),
   571→        };
   572→
   573→        Ok(CompiledRule {
   574→            condition,
   575→            target_name,
   576→            field: rule.field.clone(),
   577→            value_expr: self.rule_value_expr(&rule.value),
   578→            priority: rule.priority,
   579→        })
   580→    }

   582→    /// Convert an AST expression to a simplified rule value expression.
   583→    fn rule_value_expr(&self, expr: &Expr) -> RuleValueExpr {
   584→        match expr {
   585→            Expr::Ident(name) => RuleValueExpr::Ident(name.clone()),
   586→            Expr::StringLit(s) => RuleValueExpr::StringLit(s.clone()),
   587→            Expr::FloatLit(f) => RuleValueExpr::FloatLit(*f),
   588→            Expr::FieldAccess(base, field) => {
   589→                if let Expr::Ident(name) = base.as_ref() {
   590→                    RuleValueExpr::FieldAccess(name.clone(), field.clone())
   591→                } else {
   592→                    RuleValueExpr::Ident(format!("{:?}", expr))
   593→                }
   594→            }
   595→            _ => RuleValueExpr::Ident(format!("{:?}", expr)),
   596→        }
   597→    }

   599→    /// Analyze a compiled pattern body for purity.
   600→    /// A pattern is pure if it contains only:
   601→    ///   - LoadLocal, Const (any type), Add, Sub, Mul, Div
   602→    ///   - CmpGt/CmpLt/CmpGe/CmpLe/CmpEq (allowed in purity, but NOT JIT-compiled)
   603→    ///   - Return
   604→    /// And ALL parameter types are "Float".
   605→    /// No globals, no builtins, no LLM calls, no struct operations, no memory.
   606→    /// Note: is_pure is a broader check than JIT-eligibility. JIT additionally
   607→    /// requires only arithmetic (no Cmp*). See JitCompiler::is_jit_eligible().
   608→    fn analyze_purity(code: &[Instruction], params: &[crate::ast::Param]) -> bool {
   609→        // All params must be Float
   610→        if params.iter().any(|p| p.type_name != "Float") {
   611→            return false;
   612→        }
   613→        for instr in code {
   614→            match instr {
   615→                Instruction::LoadLocal(_) |
   616→                Instruction::Const(_) |
   617→                Instruction::Add |
   618→                Instruction::Sub |
   619→                Instruction::Mul |
   620→                Instruction::Div |
   621→                Instruction::CmpGt |
   622→                Instruction::CmpLt |
   623→                Instruction::CmpGe |
   624→                Instruction::CmpLe |
   625→                Instruction::CmpEq |
   626→                Instruction::CmpNe |
   627→                Instruction::Return => {}
   628→                _ => return false,
   629→            }
   630→        }
   631→        true
   632→    }

   634→    /// Ensure a global slot exists for the given name.
   635→    fn ensure_global(&mut self, name: &str) {
   636→        if !self.global_slots.contains_key(name) {
   637→            let slot = self.next_global;
   638→            self.global_slots.insert(name.to_string(), slot);
   639→            self.next_global += 1;
   640→        }

   643→    /// Resolve an import: find file, parse, recursively resolve sub-imports.
   644→    fn resolve_import(&mut self, module_path: &str) -> Result<Vec<Declaration>, String> {
   645→        if self.imported_modules.contains(module_path) {
   646→            return Ok(Vec::new());
   647→        }
   648→        self.imported_modules.insert(module_path.to_string());
   649→
   650→        if module_path == "std/collections" {
   651→            self.collections_loaded = true;
   652→        }
   653→
   654→        let file_path = self.std_root.join(module_path).with_extension("mlog");
   655→        let source = std::fs::read_to_string(&file_path)
   656→            .map_err(|e| format!(
   657→                "import '{}': cannot read {:?}: {}",
   658→                module_path, file_path, e
   659→            ))?;
   660→
   661→        let mut declarations = crate::parser::parse(&source)
   662→            .map_err(|e| format!("import '{}': parse error: {}", module_path, e))?;
   663→
   664→        let mut resolved = Vec::new();
   665→        for decl in declarations.drain(..) {
   666→            if let Declaration::Import(sub_import) = &decl {
   667→                // Fix 2: use `sub_import.path` instead of `sub_import.module_path`
   668→                if !self.imported_modules.contains(&sub_import.path) {
   669→                    let sub_decls = self.resolve_import(&sub_import.path)?;
   670→                    resolved.extend(sub_decls);
   671→                }
   672→            } else {
   673→                resolved.push(decl);
   674→            }
   675→        }
   676→
   677→        Ok(resolved)
   678→    }

    17→/// Parse a .mlog source string into a list of declarations.
    18→/// Templates with `}` in their body (HTML, CSS, JS) are handled via
    19→/// pre-processing: template bodies are extracted with balanced brace counting,
    20→/// replaced with placeholders for Pest parsing, then restored.
    21→pub fn parse(source: &str) -> Result<Vec<Declaration>, ParseError> {
    22→    // Наряд №14 P1-5: Use a thread with 8MB stack for large files.
    23→    // Pest's recursive descent can overflow the default 2-8MB stack
    24→    // on deeply nested expressions or very long string literals.
    25→    let source_owned = source.to_string();
    26→    let handle = std::thread::Builder::new()
    27→        .stack_size(8 * 1024 * 1024)
    28→        .spawn(move || parse_inner(&source_owned))
    29→        .expect("failed to spawn parser thread");
    30→    handle.join().expect("parser thread panicked")
    31→}
    32→
    33→fn parse_inner(source: &str) -> Result<Vec<Declaration>, ParseError> {
    34→    // Pre-process: extract template bodies that contain } characters.
    35→    // Replace with unique placeholders to avoid Pest's "stop at first }" limitation.
    36→    let (preprocessed, template_bodies) = preprocess_templates(source);
    37→
    38→    let pairs = MlogParser::parse(Rule::program, &preprocessed)?;
    39→    let mut declarations = Vec::new();
    40→
    41→    for pair in pairs {
    42→        for inner_pair in pair.into_inner() {
    43→            match inner_pair.as_rule() {
    44→                Rule::mlogserver_decl => declarations.push(parse_mlogserver_decl(inner_pair)),
    45→                Rule::template_decl => declarations.push(parse_template_decl_with_body(inner_pair, &template_bodies)),
    46→                Rule::db_decl => declarations.push(parse_db_decl(inner_pair)),
    47→                Rule::memory_decl => declarations.push(parse_memory_decl(inner_pair)),
    48→                Rule::import_decl => declarations.push(parse_import_decl(inner_pair)),
    49→                Rule::entity_type_decl => declarations.push(parse_entity_type_decl(inner_pair)),
    50→                Rule::entity_record_decl => declarations.push(parse_entity_record_decl(inner_pair)),
    51→                Rule::entity_simple_decl => declarations.push(parse_entity_simple_decl(inner_pair)),
    52→                Rule::rule_decl => declarations.push(parse_rule_decl(inner_pair)),
    53→                Rule::memorize_decl => declarations.push(parse_memorize_decl(inner_pair)),
    54→                Rule::forget_decl => declarations.push(parse_forget_decl(inner_pair)),
    55→                Rule::if_block_stmt => declarations.push(Declaration::Pattern(PatternDecl { name: "_top_level_if".to_string(), params: vec![], return_type: "Unit".to_string(), body: vec![parse_if_block_stmt(inner_pair)] })),
    56→                Rule::fluid_decl => declarations.push(parse_fluid_decl(inner_pair)),
    57→                Rule::adapt_decl => declarations.push(parse_adapt_decl(inner_pair)),
    58→                Rule::relate_decl => declarations.push(parse_relate_decl(inner_pair)),
    59→                Rule::sandbox_decl => declarations.push(parse_sandbox_decl(inner_pair)),
    60→                Rule::hook_decl => declarations.push(parse_hook_decl(inner_pair)),
    61→                Rule::mutate_decl => declarations.push(parse_mutate_decl(inner_pair)),
    62→                Rule::eval_decl => declarations.push(parse_eval_decl(inner_pair)),
    63→                Rule::conversation_decl => declarations.push(parse_conversation_decl(inner_pair)),
    64→                Rule::llm_decl => declarations.push(parse_llm_decl(inner_pair)),
    65→                Rule::tool_decl => declarations.push(parse_tool_decl(inner_pair)),
    66→                Rule::learnable_pattern_decl => declarations.push(parse_learnable_pattern_decl(inner_pair)),
    67→                Rule::pattern_decl => declarations.push(parse_pattern_decl(inner_pair)),
    68→                Rule::flow_decl => declarations.push(parse_flow_decl(inner_pair)),
    69→                _ => {}
    70→            }
    71→        }
    72→    }
    73→
    74→    Ok(declarations)
    75→}

   101→fn parse_mlogserver_decl(pair: Pair<Rule>) -> Declaration {
   102→    let _children = children_of(&pair);
   103→    let body_children: Vec<Pair<Rule>> = pair.clone().into_inner()
   104→        .filter(|c| c.as_rule() == Rule::mlogserver_body)
   105→        .flat_map(|c| c.into_inner())
   106→        .collect();
   107→
   108→    let port: u16 = body_children.iter()
   109→        .find(|c| c.as_rule() == Rule::mlogserver_port)
   110→        .and_then(|c| find_child_str(&children_of(c), Rule::INT))
   111→        .and_then(|s| s.parse().ok())
   112→        .unwrap_or(8080);
   113→
   114→    let mut middleware = Vec::new();
   115→    for child in &body_children {
   116→        if child.as_rule() == Rule::mlogserver_middleware {
   117→            if let Some(il) = child.clone().into_inner().find(|c| c.as_rule() == Rule::ident_list) {
   118→                middleware = il.clone().into_inner()
   119→                    .filter(|c| c.as_rule() == Rule::IDENT)
   120→                    .map(|c| pair_str(&c))
   121→                    .collect();
   122→            }
   123→        }
   124→    }
   125→
   126→    let routes: Vec<RouteDecl> = body_children.iter()
   127→        .filter(|c| c.as_rule() == Rule::route_decl)
   128→        .map(|c| parse_route_decl(c.clone()))
   129→        .collect();
   130→
   131→    Declaration::MlogServer(MlogServerDecl { port, middleware, routes })
   132→}

   134→fn parse_route_decl(pair: Pair<Rule>) -> RouteDecl {
   135→    let children: Vec<Pair<Rule>> = pair.clone().into_inner().collect();
   136→    let path = children.iter()
   137→        .find(|c| c.as_rule() == Rule::STRING_LITERAL)
   138→        .map(|c| {
   139→            let s = c.as_str();
   140→            s[1..s.len()-1].to_string()
   141→        })
   142→        .unwrap_or_default();
   143→
   144→    let method = children.iter()
   145→        .filter(|c| c.as_rule() == Rule::IDENT)
   146→        .map(|c| pair_str(c))
   147→        .next() // first IDENT after STRING_LITERAL is the HTTP method
   148→        .unwrap_or_else(|| "GET".to_string());
   149→
   150→    let mut requires = Vec::new();
   151→    for child in &children {
   152→        if child.as_rule() == Rule::route_requires {
   153→            if let Some(il) = child.clone().into_inner().find(|c| c.as_rule() == Rule::ident_list) {
   154→                requires = il.clone().into_inner()
   155→                    .filter(|c| c.as_rule() == Rule::IDENT)
   156→                    .map(|c| pair_str(&c))
   157→                    .collect();
   158→            }
   159→        }
   160→    }
   161→
   162→    let body: Vec<Statement> = pair.clone().into_inner()
   163→        .filter(|c| c.as_rule() == Rule::statement)
   164→        .map(|c| parse_single_statement(c))
   165→        .collect();
   166→
   167→    RouteDecl { path, method, requires, body }
   168→}

   172→/// Pre-process source to handle template bodies containing `}` (HTML, CSS, JS).
   173→/// Extracts template bodies using balanced brace counting, replaces with safe placeholders,
   174→/// and returns a mapping of placeholder -> actual body content.
   175→/// Uses char_indices() for Unicode-safe byte positioning.
   176→fn preprocess_templates(source: &str) -> (String, HashMap<String, String>) {
   177→    let mut result = source.to_string();
   178→    let mut bodies = HashMap::new();
   179→    let mut counter = 0u32;
   180→
   181→    // Find template declarations and extract balanced brace bodies
   182→    let mut search_from = 0;
   183→    while search_from < result.len() {
   184→        // Find "template" keyword (ASCII-only, find() is safe)
   185→        if let Some(start) = result[search_from..].find("template") {
   186→            let abs_start = search_from + start;
   187→            // Skip if this is part of a longer identifier (check preceding char)
   188→            if abs_start > 0 && result.as_bytes().get(abs_start - 1).map(|&b| b.is_ascii_alphanumeric()).unwrap_or(false) {
   189→                search_from = abs_start + 1;
   190→                continue;
   191→            }
   192→
   193→            // Find the opening { of the template body (after type_name)
   194→            // '{' is ASCII, find() on ASCII patterns is char-boundary-safe
   195→            if let Some(brace_pos) = result[abs_start..].find('{') {
   196→                let abs_brace = abs_start + brace_pos;
   197→                // Find the matching closing } using balanced brace counting.
   198→                // MUST use char_indices() to get correct BYTE offsets for Unicode-safe slicing.
   199→                let mut depth = 0;
   200→                let mut end_byte_pos = None;
   201→                for (byte_offset, ch) in result[abs_brace..].char_indices() {
   202→                    match ch {
   203→                        '{' => depth += 1,
   204→                        '}' => {
   205→                            depth -= 1;
   206→                            if depth == 0 {
   207→                                end_byte_pos = Some(abs_brace + byte_offset);
   208→                                break;
   209→                            }
   210→                        }
   211→                        _ => {}
   212→                    }
   213→                }
   214→
   215→                if let Some(close_pos) = end_byte_pos {
   216→                    // Extract the body between the braces (byte offsets are char-boundary-safe)
   217→                    let body = result[abs_brace + 1..close_pos].to_string();
   218→                    let placeholder = format!("__TEMPLATE_BODY_{}__", counter);
   219→                    counter += 1;
   220→
   221→                    // Replace the body content with a safe placeholder (no })
   222→                    let replacement = format!("{{{}}}", placeholder);
   223→                    result.replace_range(abs_brace..=close_pos, &replacement);
   224→                    bodies.insert(placeholder, body);
   225→
   226→                    search_from = abs_brace + replacement.len();
   227→                    continue;
   228→                }
   229→            }
   230→            search_from = abs_start + 1;
   231→        } else {
   232→            break;
   233→        }
   234→    }
   235→
   236→    (result, bodies)
   237→}

   239→/// Parse a template declaration, restoring the pre-processed body.
   240→fn parse_template_decl_with_body(pair: Pair<Rule>, bodies: &HashMap<String, String>) -> Declaration {
   241→    let children = children_of(&pair);
   242→    let name = find_child_str(&children, Rule::IDENT).unwrap_or_default();
   243→    let params = find_child(&children, Rule::params)
   244→        .map(|p| parse_params(p))
   245→        .unwrap_or_default();
   246→    let return_type = find_child_str(&children, Rule::type_name).unwrap_or_default();
   247→
   248→    // The grammar captured the placeholder in template_body_raw
   249→    let body = pair.clone().into_inner()
   250→        .find(|c| c.as_rule() == Rule::template_body_raw)
   251→        .map(|c| {
   252→            let placeholder = c.as_str().trim().to_string();
   253→            // Look up the placeholder in our pre-processed bodies map
   254→            if let Some(actual_body) = bodies.get(&placeholder) {
   255→                actual_body.clone()
   256→            } else {
   257→                // Fallback: use the raw text (for templates without } in body)
   258→                placeholder
   259→            }
   260→        })
   261→        .unwrap_or_default();
   262→
   263→    Declaration::Template(TemplateDecl { name, params, return_type, body })
   264→}

   266→/// Extract content between balanced braces from a string like "{ content } }".
   267→/// Handles nested braces by counting depth.
   268→fn extract_balanced_braces(s: &str) -> String {
   269→    let chars: Vec<char> = s.chars().collect();
   270→    if chars.is_empty() || chars[0] != '{' {
   271→        return String::new();
   272→    }
   273→    let mut depth = 0;
   274→    let mut end = 0;
   275→    for (i, &ch) in chars.iter().enumerate() {
   276→        match ch {
   277→            '{' => depth += 1,
   278→            '}' => {
   279→                depth -= 1;
   280→                if depth == 0 {
   281→                    end = i;
   282→                    break;
   283→                }
   284→            }
   285→            _ => {}
   286→        }
   287→    }
   288→    if end > 0 {
   289→        chars[1..end].iter().collect()
   290→    } else {
   291→        String::new()
   292→    }
   293→}

   297→fn parse_db_decl(pair: Pair<Rule>) -> Declaration {
   298→    let children: Vec<Pair<Rule>> = pair.into_inner()
   299→        .filter(|c| c.as_rule() == Rule::db_body)
   300→        .flat_map(|c| c.into_inner())
   301→        .collect();
   302→
   303→    let url = children.iter()
   304→        .find(|c| c.as_rule() == Rule::db_url)
   305→        .and_then(|c| {
   306→            let c_children = children_of(c);
   307→            find_child(&c_children, Rule::expression).map(|e| parse_expression(e))
   308→        });
   309→
   310→    let pool_size = children.iter()
   311→        .find(|c| c.as_rule() == Rule::db_pool)
   312→        .and_then(|c| find_child_str(&children_of(c), Rule::INT))
   313→        .and_then(|s| s.parse().ok());
   314→
   315→    let migrate = children.iter()
   316→        .find(|c| c.as_rule() == Rule::db_migrate)
   317→        .and_then(|c| find_child_str(&children_of(c), Rule::STRING_LITERAL))
   318→        .map(|s| s[1..s.len()-1].to_string());
   319→
   320→    Declaration::Db(DbDecl { url, pool_size, migrate })
   321→}

   371→/// Process escape sequences in a string literal (without outer quotes).
   372→fn unescape_string(s: &str) -> String {
   373→    let trimmed = &s[1..s.len()-1]; // strip outer quotes
   374→    let mut result = String::with_capacity(trimmed.len());
   375→    let mut chars = trimmed.chars().peekable();
   376→    while let Some(c) = chars.next() {
   377→        if c == '\\' {
   378→            match chars.peek() {
   379→                Some('\"') => { result.push('\"'); chars.next(); }
   380→                Some('\\') => { result.push('\\'); chars.next(); }
   381→                Some('n')  => { result.push('\n'); chars.next(); }
   382→                Some('t')  => { result.push('\t'); chars.next(); }
   383→                Some('r')  => { result.push('\r'); chars.next(); }
   384→                _ => { result.push(c); }
   385→            }
   386→        } else {
   387→            result.push(c);
   388→        }
   389→    }
   390→    result
   391→}

   393→/// Convert a literal pair (STRING_LITERAL, FLOAT_LITERAL, or IDENT) to an Expr.
   394→fn parse_literal_to_expr(pair: &Pair<Rule>) -> Expr {
   395→    let inner = pair.clone().into_inner().next().unwrap();
   396→    match inner.as_rule() {
   397→        Rule::STRING_LITERAL => Expr::StringLit(unescape_string(inner.as_str())),
   398→        Rule::FLOAT_LITERAL => Expr::FloatLit(inner.as_str().parse().unwrap_or(0.0)),
   399→        Rule::IDENT => Expr::Ident(inner.as_str().to_string()),
   400→        _ => Expr::StringLit(pair.as_str().to_string()),
   401→    }
   402→}

   441→fn parse_rule_decl(pair: Pair<Rule>) -> Declaration {
   442→    let children = children_of(&pair);
   443→    // Children: condition (contains/compare), assignment, [INT]
   444→    let condition_pair = &children[0];
   445→    let condition = parse_condition(condition_pair.clone());
   446→
   447→    // assignment = { IDENT ~ "." ~ IDENT ~ "=" ~ expression }
   448→    // Children: [IDENT(target), IDENT(field), expression(value)]
   449→    let assignment_children = children_of(&children[1]);
   450→    let target = Expr::Ident(pair_str(&assignment_children[0]));
   451→    let field = pair_str(&assignment_children[1]);
   452→    let value = parse_expression(assignment_children[2].clone());
   453→
   454→    let priority = find_child_str(&children, Rule::INT)
   455→        .and_then(|s| s.parse::<i32>().ok())
   456→        .unwrap_or(0);
   457→
   458→    Declaration::Rule(RuleDecl { condition, target, field, value, priority })
   459→}
   460→
   461→fn parse_condition(pair: Pair<Rule>) -> Condition {
   462→    match pair.as_rule() {
   463→        Rule::contains_condition => {
   464→            let children = children_of(&pair);
   465→            // Children: expression, CONTAINS_KW, expression
   466→            let left = parse_expression(children[0].clone());
   467→            let right = parse_expression(children[2].clone());
   468→            Condition::Contains { left, right }
   469→        }
   470→        Rule::compare_condition => {
   471→            let children = children_of(&pair);
   472→            // Children: expression, compare_op, expression
   473→            Condition::Compare {
   474→                left: parse_expression(children[0].clone()),
   475→                op: parse_compare_op(&children[1]),
   476→                right: parse_expression(children[2].clone()),
   477→            }
   478→        }
   479→        _ => unreachable!("expected condition, got {:?}", pair.as_rule()),
   480→    }
   481→}
   482→
   483→fn parse_compare_op(pair: &Pair<Rule>) -> CompareOp {
   484→    match pair.as_str().trim() {
   485→        ">" => CompareOp::Gt,
   486→        "<" => CompareOp::Lt,
   487→        ">=" => CompareOp::Ge,
   488→        "<=" => CompareOp::Le,
   489→        "==" => CompareOp::Eq,
   490→        _ => unreachable!("unexpected compare op: {}", pair.as_str()),
   491→    }
   492→}

   494→// ── Fluid Types (Phase 1) ──────────────────────────────────────────
   495→
   496→fn parse_fluid_decl(pair: Pair<Rule>) -> Declaration {
   497→    let children = children_of(&pair);
   498→    // fluid_decl = { FLUID_KW ~ IDENT ~ "=" ~ fluid_branch ~ ("or" ~ fluid_branch)* }
   499→    let name = find_child_str(&children, Rule::IDENT).unwrap_or_default();
   500→
   501→    let variants: Vec<FluidVariant> = children.iter()
   502→        .filter(|c| c.as_rule() == Rule::fluid_branch)
   503→        .map(|c| parse_fluid_branch(c.clone()))
   504→        .collect();
   505→
   506→    Declaration::Fluid(FluidDecl { name, variants })
   507→}

   509→fn parse_fluid_branch(pair: Pair<Rule>) -> FluidVariant {
   510→    let children = children_of(&pair);
   511→    // fluid_branch = { type_name ~ LBRACKET ~ expression ~ RBRACKET ~ LBRACKET ~ FLOAT_LITERAL ~ RBRACKET }
   512→    let type_name = find_child_str(&children, Rule::type_name).unwrap_or_default();
   513→
   514→    let exprs: Vec<Pair<Rule>> = children.iter()
   515→        .filter(|c| c.as_rule() == Rule::expression)
   516→        .cloned()
   517→        .collect();
   518→    let value = if !exprs.is_empty() { parse_expression(exprs[0].clone()) } else { Expr::StringLit(String::new()) };
   519→
   520→    let floats: Vec<&Pair<Rule>> = children.iter()
   521→        .filter(|c| c.as_rule() == Rule::FLOAT_LITERAL)
   522→        .collect();
   523→    // The last FLOAT_LITERAL is the confidence (value may also be a float)
   524→    let confidence = floats.last()
   525→        .map(|f| f.as_str().parse().unwrap_or(0.0))
   526→        .unwrap_or(0.0);
   527→
   528→    FluidVariant { type_name, value, confidence }
   529→}

   596→        .collect();

   603→fn parse_sandbox_decl(pair: Pair<Rule>) -> Declaration {
   604→    let children = children_of(&pair);
   605→    // sandbox_decl = { SANDBOX_KW ~ IDENT ~ "{" ~ sandbox_body "}" }
   606→    let name = find_child_str(&children, Rule::IDENT).unwrap_or_default();
   607→
   608→    let mut allowed = Vec::new();
   609→    let mut forbidden = Vec::new();
   610→    let mut timeout: i64 = 30;
   611→
   612→    if let Some(body_pair) = find_child(&children, Rule::sandbox_body) {
   613→        let body_children = children_of(&body_pair);
   614→        // Extract allowed list
   615→        if let Some(al_pair) = body_children.iter().find(|c| c.as_rule() == Rule::sandbox_allowed) {
   616→            let al_children = children_of(al_pair);
   617→            if let Some(il_pair) = al_children.iter().find(|c| c.as_rule() == Rule::ident_list) {
   618→                allowed = il_pair.clone().into_inner()
   619→                    .filter(|c| c.as_rule() == Rule::IDENT)
   620→                    .map(|c| pair_str(&c))
   621→                    .collect();
   622→            }
   623→        }
   624→        // Extract forbidden list
   625→        if let Some(fb_pair) = body_children.iter().find(|c| c.as_rule() == Rule::sandbox_forbidden) {
   626→            let fb_children = children_of(fb_pair);
   627→            if let Some(il_pair) = fb_children.iter().find(|c| c.as_rule() == Rule::ident_list) {
   628→                forbidden = il_pair.clone().into_inner()
   629→                    .filter(|c| c.as_rule() == Rule::IDENT)
   630→                    .map(|c| pair_str(&c))
   631→                    .collect();
   632→            }
   633→        }
   634→        // Extract timeout
   635→        if let Some(to_pair) = body_children.iter().find(|c| c.as_rule() == Rule::sandbox_timeout) {
   636→            if let Some(int_val) = find_child_str(&children_of(to_pair), Rule::INT) {
   637→                timeout = int_val.parse().unwrap_or(30);
   638→            }
   639→        }
   640→    }
   641→
   642→    Declaration::Sandbox(SandboxDecl { name, allowed, forbidden, timeout })
   643→}

   647→fn parse_mutate_decl(pair: Pair<Rule>) -> Declaration {
   648→    let children = children_of(&pair);
   649→    // mutate_decl = { MUTATE_KW ~ IDENT ~ "{" ~ mutate_body "}" }
   650→    let pattern_name = find_child_str(&children, Rule::IDENT).unwrap_or_default();
   651→
   652→    let mut new_examples: Vec<(Expr, Expr)> = Vec::new();
   653→    let mut rollback_threshold: Option<f64> = None;
   654→    let mut rollback_op: Option<CompareOp> = None;
   655→
   656→    if let Some(body_pair) = find_child(&children, Rule::mutate_body) {
   657→        let body_children = children_of(&body_pair);
   658→        // Extract add_example pairs
   659→        for ae_pair in body_children.iter().filter(|c| c.as_rule() == Rule::mutate_add_example) {
   660→            let ae_children = children_of(ae_pair);
   661→            let exprs: Vec<Expr> = ae_children.iter()
   662→                .filter(|c| c.as_rule() == Rule::expression)
   663→                .map(|c| parse_expression(c.clone()))
   664→                .collect();
   665→            if exprs.len() >= 2 {
   666→                new_examples.push((exprs[0].clone(), exprs[1].clone()));
   667→            }
   668→        }
   669→        // Extract rollback_if condition
   670→        if let Some(rb_pair) = body_children.iter().find(|c| c.as_rule() == Rule::mutate_rollback) {
   671→            let rb_children = children_of(rb_pair);
   672→            // Find compare_op and FLOAT_LITERAL
   673→            if let Some(op_pair) = rb_children.iter().find(|c| c.as_rule() == Rule::compare_op) {
   674→                rollback_op = Some(parse_compare_op(op_pair));
   675→            }
   676→            if let Some(float_pair) = rb_children.iter().find(|c| c.as_rule() == Rule::FLOAT_LITERAL) {
   677→                rollback_threshold = Some(float_pair.as_str().parse().unwrap_or(0.0));
   678→            }
   679→        }
   680→    }
   681→
   682→    Declaration::Mutate(MutateDecl {
   683→        pattern_name,
   684→        new_examples,
   685→        rollback_threshold,
   686→        rollback_op,
   687→    })
   688→}

   719→// ── LLM Config (Наряд №4: Smart LLM Routing) ──────────────────────────
   720→
   721→fn parse_llm_decl(pair: Pair<Rule>) -> Declaration {
   722→    let children: Vec<Pair<Rule>> = pair.into_inner()
   723→        .filter(|c| c.as_rule() == Rule::llm_body)
   724→        .flat_map(|c| c.into_inner())
   725→        .collect();
   726→
   727→    // Parse providers list
   728→    let mut providers = Vec::new();
   729→    if let Some(pl_pair) = children.iter().find(|c| c.as_rule() == Rule::llm_providers) {
   730→        // Flatten: llm_provider_list -> llm_provider_entry*
   731→        let entries: Vec<Pair<Rule>> = pl_pair.clone().into_inner()
   732→            .flat_map(|c| {
   733→                if c.as_rule() == Rule::llm_provider_entry {
   734→                    vec![c]
   735→                } else if c.as_rule() == Rule::llm_provider_list {
   736→                    c.into_inner()
   737→                        .filter(|inner| inner.as_rule() == Rule::llm_provider_entry)
   738→                        .collect()
   739→                } else {
   740→                    vec![]
   741→                }
   742→            })
   743→            .collect();
   744→
   745→        for entry_pair in entries {
   746→            let entry_children = children_of(&entry_pair);
   747→
   748→            let alias = entry_children.iter()
   749→                .find(|c| c.as_rule() == Rule::llm_provider_alias)
   750→                .and_then(|c| find_child_str(&children_of(c), Rule::IDENT))
   751→                .unwrap_or_default();
   752→
   753→            let provider = entry_children.iter()
   754→                .find(|c| c.as_rule() == Rule::llm_provider_name)
   755→                .and_then(|c| find_child_str(&children_of(c), Rule::IDENT))
   756→                .unwrap_or_default();
   757→
   758→            let key = entry_children.iter()
   759→                .find(|c| c.as_rule() == Rule::llm_provider_key)
   760→                .and_then(|c| {
   761→                    find_child(&children_of(c), Rule::expression).map(|e| parse_expression(e))
   762→                });
   763→
   764→            let url = entry_children.iter()
   765→                .find(|c| c.as_rule() == Rule::llm_provider_url)
   766→                .and_then(|c| find_child_str(&children_of(c), Rule::STRING_LITERAL))
   767→                .map(|s| s[1..s.len()-1].to_string());
   768→
   769→            providers.push(LlmProviderEntry { alias, provider, key, url });
   770→        }
   771→    }
   772→
   773→    // Parse default_model
   774→    let default_model = children.iter()
   775→        .find(|c| c.as_rule() == Rule::llm_default_model)
   776→        .and_then(|c| {
   777→            find_child(&children_of(c), Rule::expression).and_then(|e| {
   778→                if let Expr::StringLit(s) = parse_expression(e) {
   779→                    Some(s)
   780→                } else {
   781→                    None
   782→                }
   783→            })
   784→        });
   785→
   786→    // Parse failover mode
   787→    let failover = children.iter()
   788→        .find(|c| c.as_rule() == Rule::llm_failover)
   789→        .and_then(|c| find_child_str(&children_of(c), Rule::IDENT));
   790→
   791→    // Parse circuit_breaker
   792→    let circuit_breaker = children.iter()
   793→        .find(|c| c.as_rule() == Rule::llm_circuit_breaker)
   794→        .and_then(|c| find_child_str(&children_of(c), Rule::INT))
   795→        .and_then(|s| s.parse().ok())
   796→        .unwrap_or(3);
   797→
   798→    // Parse timeout
   799→    let timeout = children.iter()
   800→        .find(|c| c.as_rule() == Rule::llm_timeout)
   801→        .and_then(|c| find_child_str(&children_of(c), Rule::INT))
   802→        .and_then(|s| s.parse().ok())
   803→        .unwrap_or(30);
   804→
   805→    Declaration::LlmConfig(LlmConfigDecl {
   806→        providers,
   807→        default_model,
   808→        failover,
   809→        circuit_breaker,
   810→        timeout,
   811→    })
   812→}

   845→// ── Eval Harness (ADR-0050) ──────────────────────────────────────────

   847→fn parse_eval_decl(pair: Pair<Rule>) -> Declaration {
   848→    let children = children_of(&pair);
   849→    // eval_decl = { EVAL_KW ~ IDENT ~ "{" ~ eval_body ~ "}" }
   850→    let pattern_name = find_child_str(&children, Rule::IDENT).unwrap_or_default();
   851→
   852→    let mut dataset: Vec<(String, String)> = Vec::new();
   853→    let mut metric = "accuracy".to_string();
   854→    let mut threshold: f64 = 0.8;
   855→
   856→    if let Some(body_pair) = find_child(&children, Rule::eval_body) {
   857→        let body_children = children_of(&body_pair);
   858→
   859→        // Extract dataset: [("input", "expected"), ...]
   860→        if let Some(ds_pair) = body_children.iter().find(|c| c.as_rule() == Rule::eval_dataset) {
   861→            // eval_example is nested inside eval_example_list (if present)
   862→            // or directly in eval_dataset when there's a single example.
   863→            // Flatten both levels to collect all eval_example pairs.
   864→            let examples: Vec<Pair<Rule>> = ds_pair.clone().into_inner()
   865→                .flat_map(|c| {
   866→                    if c.as_rule() == Rule::eval_example {
   867→                        vec![c]
   868→                    } else if c.as_rule() == Rule::eval_example_list {
   869→                        c.into_inner()
   870→                            .filter(|inner| inner.as_rule() == Rule::eval_example)
   871→                            .collect()
   872→                    } else {
   873→                        vec![]
   874→                    }
   875→                })
   876→                .collect();
   877→            for ex_pair in examples {
   878→                let strings: Vec<Pair<Rule>> = ex_pair.clone().into_inner()
   879→                    .filter(|c| c.as_rule() == Rule::STRING_LITERAL)
   880→                    .collect();
   881→                let input = if strings.len() >= 1 {
   882→                    unescape_string(strings[0].as_str())
   883→                } else {
   884→                    String::new()
   885→                };
   886→                let expected = if strings.len() >= 2 {
   887→                    unescape_string(strings[1].as_str())
   888→                } else {
   889→                    String::new()
   890→                };
   891→                dataset.push((input, expected));
   892→            }
   893→        }
   894→
   895→        // Extract metric: accuracy (or future metrics)
   896→        if let Some(m_pair) = body_children.iter().find(|c| c.as_rule() == Rule::eval_metric) {
   897→            let m_children = children_of(m_pair);
   898→            metric = find_child_str(&m_children, Rule::IDENT).unwrap_or_else(|| "accuracy".to_string());
   899→        }
   900→
   901→        // Extract threshold: 0.8
   902→        if let Some(t_pair) = body_children.iter().find(|c| c.as_rule() == Rule::eval_threshold) {
   903→            let t_children = children_of(t_pair);
   904→            threshold = find_child_str(&t_children, Rule::FLOAT_LITERAL)
   905→                .and_then(|s| s.parse().ok())
   906→                .unwrap_or(0.8);
   907→        }
   908→    }
   909→
   910→    Declaration::Eval(EvalDecl {
   911→        pattern_name,
   912→        dataset,
   913→        metric,
   914→        threshold,
   915→    })
   916→}

   946→// ── Learnable Pattern (M3) ────────────────────────────────────────────

   948→fn parse_learnable_pattern_decl(pair: Pair<Rule>) -> Declaration {
   949→    let children = children_of(&pair);
   950→    // Children: IDENT, [params], ARROW, type_name, learnable_body
   951→    let name = find_child_str(&children, Rule::IDENT).unwrap_or_default();
   952→    let params = find_child(&children, Rule::params)
   953→        .map(|p| parse_params(p))
   954→        .unwrap_or_default();
   955→    let return_type = find_child_str(&children, Rule::type_name).unwrap_or_default();
   956→
   957→    // Extract prompt, context, model, max_tokens, cache, cache_ttl from learnable_body
   958→    let mut prompt = String::new();
   959→    let mut context: Option<ContextMode> = None;
   960→    let mut model: Option<String> = None;
   961→    let mut max_tokens: Option<u32> = None;
   962→    let mut cache = false;
   963→    let mut cache_ttl: u64 = 3600; // default 1 hour
   964→    let mut conversation: Option<String> = None;
   965→    let mut context_strategy: ContextStrategy = ContextStrategy::None;
   966→    let mut max_context_tokens: usize = 2000; // default 2000
   967→
   968→    if let Some(body_pair) = find_child(&children, Rule::learnable_body) {
   969→        let body_children = children_of(&body_pair);
   970→
   971→        // Extract prompt from prompt_line -> expression
   972→        if let Some(pl_pair) = body_children.iter().find(|c| c.as_rule() == Rule::prompt_line) {
   973→            let pl_children = children_of(pl_pair);
   974→            if let Some(expr_pair) = pl_children.iter().find(|c| c.as_rule() == Rule::expression) {
   975→                if let Expr::StringLit(s) = parse_expression(expr_pair.clone()) {
   976→                    prompt = s;
   977→                }
   978→            }
   979→        }
   980→
   981→        // Extract context: supports recall(...), auto, none, or string literal
   982→        // NOTE: context_line uses _{ } (silent choice), so the inner rule appears
   983→        // directly in body_children, NOT wrapped in a context_line node.
   984→        let ctx_pair = body_children.iter().find(|c| {
   985→            matches!(c.as_rule(),
   986→                Rule::context_recall_line |
   987→                Rule::context_auto_line |
   988→                Rule::context_none_line |
   989→                Rule::context_literal_line)
   990→        });
   991→        if let Some(inner_pair) = ctx_pair {
   992→            match inner_pair.as_rule() {
   993→                Rule::context_recall_line => {
   994→                    // context: recall(query_expr, limit=N)
   995→                    let inner_children = children_of(inner_pair);
   996→                    let exprs: Vec<Expr> = inner_children.iter()
   997→                        .filter(|c| c.as_rule() == Rule::expression)
   998→                        .cloned()
   999→                        .map(|p| parse_expression(p))
  1000→                        .collect();
  1001→                    if !exprs.is_empty() {
  1002→                        let limit = if exprs.len() >= 2 {
  1003→                            if let Expr::FloatLit(n) = exprs[1].clone() {
  1004→                                Some(n as usize)
  1005→                            } else {
  1006→                                None
  1007→                            }
  1008→                        } else {
  1009→                            None
  1010→                        };
  1011→                        context = Some(ContextMode::Recall(exprs[0].clone(), limit));
  1012→                    }
  1013→                }
  1014→                Rule::context_auto_line => {
  1015→                    // context: auto → recall(first_param, limit=5)
  1016→                    context = Some(ContextMode::Auto);
  1017→                }
  1018→                Rule::context_none_line => {
  1019→                    // context: none → explicitly no context
  1020→                    context = Some(ContextMode::None);
  1021→                }
  1022→                Rule::context_literal_line => {
  1023→                    // context: "some string" or context: expr
  1024→                    let inner_children = children_of(inner_pair);
  1025→                    if let Some(expr_pair) = inner_children.iter().find(|c| c.as_rule() == Rule::expression) {
  1026→                        if let Expr::StringLit(s) = parse_expression(expr_pair.clone()) {
  1027→                            context = Some(ContextMode::Literal(s));
  1028→                        }
  1029→                    }
  1030→                }
  1031→                _ => {}
  1032→            }
  1033→        }
  1034→
  1035→        // Extract conversation: "current" or conversation: <expr> (ADR-0053)
  1036→        if let Some(conv_pair) = body_children.iter().find(|c| c.as_rule() == Rule::conversation_line) {
  1037→            let conv_children = children_of(conv_pair);
  1038→            if let Some(expr_pair) = conv_children.iter().find(|c| c.as_rule() == Rule::expression) {
  1039→                if let Expr::StringLit(s) = parse_expression(expr_pair.clone()) {
  1040→                    conversation = Some(s);
  1041→                }
  1042→            }
  1043→        }
  1044→
  1045→        // Extract model: "haiku" (ADR-0048)
  1046→        if let Some(m_pair) = body_children.iter().find(|c| c.as_rule() == Rule::model_line) {
  1047→            let m_children = children_of(m_pair);
  1048→            if let Some(expr_pair) = m_children.iter().find(|c| c.as_rule() == Rule::expression) {
  1049→                if let Expr::StringLit(s) = parse_expression(expr_pair.clone()) {
  1050→                    model = Some(s);
  1051→                }
  1052→            }
  1053→        }
  1054→
  1055→        // Extract max_tokens: N
  1056→        if let Some(mt_pair) = body_children.iter().find(|c| c.as_rule() == Rule::max_tokens_line) {
  1057→            let mt_children = children_of(mt_pair);
  1058→            if let Some(expr_pair) = mt_children.iter().find(|c| c.as_rule() == Rule::expression) {
  1059→                if let Expr::FloatLit(n) = parse_expression(expr_pair.clone()) {
  1060→                    max_tokens = Some(n as u32);
  1061→                }
  1062→            }
  1063→        }
  1064→
  1065→        // Extract cache: true/false
  1066→        if let Some(c_pair) = body_children.iter().find(|c| c.as_rule() == Rule::cache_line) {
  1067→            let c_str = c_pair.as_str().trim();
  1068→            // cache_line = { "cache" ~ ":" ~ BOOL_LITERAL }
  1069→            // Extract the boolean value after the colon
  1070→            if let Some(colon_pos) = c_str.find(':') {
  1071→                let val_str = c_str[colon_pos + 1..].trim();
  1072→                cache = val_str == "true";
  1073→            }
  1074→        }
  1075→
  1076→        // Extract cache_ttl: N.minutes
  1077→        if let Some(ttl_pair) = body_children.iter().find(|c| c.as_rule() == Rule::cache_ttl_line) {
  1078→            let ttl_children = children_of(ttl_pair);
  1079→            // cache_ttl_line = { "cache_ttl" ~ ":" ~ expression ~ "." ~ IDENT }
  1080→            let exprs: Vec<Expr> = ttl_children.iter()
  1081→                .filter(|c| c.as_rule() == Rule::expression)
  1082→                .cloned()
  1083→                .map(|p| parse_expression(p))
  1084→                .collect();
  1085→            let unit_ident = ttl_children.iter()
  1086→                .filter(|c| c.as_rule() == Rule::IDENT)
  1087→                .map(|c| pair_str(c))
  1088→                .next()
  1089→                .unwrap_or_default();
  1090→            if let Some(Expr::FloatLit(n)) = exprs.first() {
  1091→                let n_val = *n as u64;
  1092→                cache_ttl = match unit_ident.as_str() {
  1093→                    "seconds" | "second" => n_val,
  1094→                    "minutes" | "minute" => n_val * 60,
  1095→                    "hours" | "hour" => n_val * 3600,
  1096→                    "days" | "day" => n_val * 86400,
  1097→                    _ => n_val * 60, // default to minutes
  1098→                };
  1099→            }
  1100→        }
  1101→
  1102→        // Extract context_strategy: none | auto | compress (ADR-0055)
  1103→        if let Some(cs_pair) = body_children.iter().find(|c| c.as_rule() == Rule::context_strategy_line) {
  1104→            let cs_str = cs_pair.as_str().trim();
  1105→            // context_strategy_line = { "context_strategy" ~ ":" ~ context_strategy_value }
  1106→            if let Some(colon_pos) = cs_str.find(':') {
  1107→                let val_str = cs_str[colon_pos + 1..].trim();
  1108→                context_strategy = match val_str {
  1109→                    "auto" => ContextStrategy::Auto,
  1110→                    "compress" => ContextStrategy::Compress,
  1111→                    _ => ContextStrategy::None,
  1112→                };
  1113→            }
  1114→        }
  1115→
  1116→        // Extract max_context_tokens: N (ADR-0055)
  1117→        if let Some(mct_pair) = body_children.iter().find(|c| c.as_rule() == Rule::max_context_tokens_line) {
  1118→            if let Some(int_val) = find_child_str(&children_of(mct_pair), Rule::INT) {
  1119→                max_context_tokens = int_val.parse().unwrap_or(2000);
  1120→            }
  1121→        }
  1122→    }
  1123→
  1124→    Declaration::LearnablePattern(LearnablePatternDecl {
  1125→        name,
  1126→        params,
  1127→        return_type,
  1128→        prompt,
  1129→        context,
  1130→        context_strategy,
  1131→        max_context_tokens,
  1132→        model,
  1133→        max_tokens,
  1134→        cache,
  1135→        cache_ttl,
  1136→        conversation,
  1137→    })
  1138→}

  1142→fn parse_pattern_decl(pair: Pair<Rule>) -> Declaration {
  1143→    let children = children_of(&pair);
  1144→    // Children: IDENT, params, ARROW, type_name, pattern_body
  1145→    let name = find_child_str(&children, Rule::IDENT).unwrap_or_default();
  1146→    let params = find_child(&children, Rule::params)
  1147→        .map(|p| parse_params(p))
  1148→        .unwrap_or_default();
  1149→    let return_type = find_child_str(&children, Rule::type_name).unwrap_or_default();
  1150→    let body = find_child(&children, Rule::pattern_body)
  1151→        .map(|p| parse_pattern_body(p))
  1152→        .unwrap_or_default();
  1153→    Declaration::Pattern(PatternDecl { name, params, return_type, body })
  1154→}

  1156→fn parse_params(pair: Pair<Rule>) -> Vec<Param> {
  1157→    pair.into_inner()

  1159→        .map(|p| {
  1160→            let children = children_of(&p);
  1161→            Param {
  1162→                name: find_child_str(&children, Rule::IDENT).unwrap_or_default(),
  1163→                type_name: find_child_str(&children, Rule::type_name).unwrap_or_default(),
  1164→            }
  1165→        })
  1166→        .collect()
  1167→}

  1170→    pair.into_inner()
  1171→        .filter(|s| s.as_rule() == Rule::statement)
  1172→        .map(|s| parse_single_statement(s))
  1173→        .collect()
  1174→}

  1176→/// Parse a single statement from its rule pair.
  1177→fn parse_single_statement(pair: Pair<Rule>) -> Statement {
  1178→    let children = children_of(&pair);
  1179→    // statement = { match_stmt | if_block_stmt | each_stmt | ... }
  1180→    // Наряд №14: match_stmt is now a proper AST statement
  1181→    if let Some(m_pair) = children.iter().find(|c| c.as_rule() == Rule::match_stmt) {
  1182→        return parse_match_stmt(m_pair.clone());
  1183→    }
  1184→    // NOTE: match_stmt previously was parsed as a regular expression — now has full AST support.
  1185→    if let Some(ib_pair) = children.iter().find(|c| c.as_rule() == Rule::if_block_stmt) {
  1186→        parse_if_block_stmt(ib_pair.clone())
  1187→    } else if let Some(each_pair) = children.iter().find(|c| c.as_rule() == Rule::each_stmt) {
  1188→        let each_children: Vec<Pair<Rule>> = each_pair.clone().into_inner().collect();
  1189→        // each_stmt: IDENT [COMMA IDENT] "in" expression { body }
  1190→        // Наряд №17.3: support `each i, item in list { ... }`
  1191→        let idents: Vec<String> = each_children.iter()
  1192→            .filter(|c| c.as_rule() == Rule::IDENT)
  1193→            .map(|c| pair_str(c))
  1194→            .collect();
  1195→        let in_idx = each_children.iter().position(|c| c.as_str().trim() == "in").expect("each_stmt must have 'in'");
  1196→        let iterable = parse_expression(each_children[in_idx + 1].clone());
  1197→        let body: Vec<Statement> = each_children[in_idx + 2..].iter()
  1198→            .filter(|c| c.as_rule() == Rule::statement)
  1199→            .map(|c| parse_single_statement(c.clone()))
  1200→            .collect();
  1201→        if idents.len() == 2 {
  1202→            // each i, item in list { ... } — index + value
  1203→            // Desugar into: let _each_list = <iterable>; let i = 0; while i < len(_each_list) { let item = _each_list[i]; <body>; i = i + 1 }
  1204→            // We create a synthetic Each that the interpreter will handle
  1205→            let index_var = idents[0].clone();
  1206→            let item_var = idents[1].clone();
  1207→            Statement::EachWithIndex { index_var, item_var, iterable, body }
  1208→        } else {
  1209→            let variable = idents[0].clone();
  1210→            Statement::Each { variable, iterable, body }
  1211→        }
  1212→    } else if let Some(while_pair) = children.iter().find(|c| c.as_rule() == Rule::while_stmt) {
  1213→        let while_children: Vec<Pair<Rule>> = while_pair.clone().into_inner().collect();
  1214→        // children: expression(condition), statement*(body)
  1215→        let condition = parse_expression(while_children[0].clone());
  1216→        let body: Vec<Statement> = while_children[1..].iter()
  1217→            .filter(|c| c.as_rule() == Rule::statement)
  1218→            .map(|c| parse_single_statement(c.clone()))
  1219→            .collect();
  1220→        Statement::While { condition, body }
  1221→    } else if let Some(lb_pair) = children.iter().find(|c| c.as_rule() == Rule::let_binding) {
  1222→        let lb_children = children_of(lb_pair);
  1223→        let name = find_child_str(&lb_children, Rule::IDENT).unwrap_or_default();
  1224→        let expr = find_child(&lb_children, Rule::expression).unwrap();
  1225→        let mutable = lb_children.iter().any(|c| c.as_str().trim() == "mut");
  1226→        Statement::LetBinding { name, value: parse_expression(expr), mutable }
  1227→    } else if let Some(ae_pair) = children.iter().find(|c| c.as_rule() == Rule::assign_or_expr) {
  1228→        // assign_or_expr = { IDENT ~ ASSIGN ~ expression | expression }
  1229→        let ae_children: Vec<Pair<Rule>> = ae_pair.clone().into_inner().collect();
  1230→        // Check if this is an assignment (has IDENT + expression) or expression
  1231→        let has_assign = ae_children.iter().any(|c| c.as_rule() == Rule::ASSIGN);
  1232→        if has_assign {
  1233→            let name = pair_str(&ae_children[0]); // IDENT is first
  1234→            let expr = ae_children.iter().find(|c| c.as_rule() == Rule::expression)
  1235→                .cloned().expect("assign_or_expr assignment must have expression");
  1236→            Statement::Assign { name, value: parse_expression(expr) }
  1237→        } else {
  1238→            // Expression statement (function call, etc.)
  1239→            let expr = ae_children.iter().find(|c| c.as_rule() == Rule::expression)
  1240→                .cloned().expect("assign_or_expr expression must have expression");
  1241→            Statement::ExprStmt(parse_expression(expr))
  1242→        }
  1243→    } else if let Some(rs_pair) = children.iter().find(|c| c.as_rule() == Rule::return_stmt) {
  1244→        let rs_children = children_of(rs_pair);
  1245→        let expr = find_child(&rs_children, Rule::expression).unwrap();
  1246→        Statement::Return(parse_expression(expr))
  1247→    } else if let Some(br_pair) = children.iter().find(|c| c.as_rule() == Rule::break_stmt) {
  1248→        Statement::Break
  1249→    } else if let Some(co_pair) = children.iter().find(|c| c.as_rule() == Rule::continue_stmt) {
  1250→        Statement::Continue
  1251→    } else if let Some(it_pair) = children.iter().find(|c| c.as_rule() == Rule::if_then_stmt) {
  1252→        // if_then_stmt with optional else: "if expr then { ... } [else if expr then { ... }]* [else { ... }]"
  1253→        let it_children: Vec<Pair<Rule>> = it_pair.clone().into_inner().collect();
  1254→        let condition = parse_expression(it_children.iter().find(|c| c.as_rule() == Rule::expression).cloned().unwrap());
  1255→        let body: Vec<Statement> = it_children.iter()
  1256→            .filter(|c| c.as_rule() == Rule::statement)
  1257→            .map(|c| parse_single_statement(c.clone()))
  1258→            .collect();
  1259→
  1260→        // Check for else_if_then_block and else blocks
  1261→        let mut else_ifs = Vec::new();
  1262→        let mut else_body: Option<Vec<Statement>> = None;
  1263→        let mut in_else = false;
  1264→
  1265→        for child in &it_children {
  1266→            match child.as_rule() {
  1267→                Rule::else_if_then_block => {
  1268→                    let ei_children = children_of(child);
  1269→                    let ei_condition = ei_children.iter()
  1270→                        .find(|c| c.as_rule() == Rule::expression)
  1271→                        .map(|c| parse_expression(c.clone()))
  1272→                        .unwrap_or_else(|| Expr::BoolLit(true));
  1273→                    let ei_body: Vec<Statement> = ei_children.iter()
  1274→                        .filter(|c| c.as_rule() == Rule::statement)
  1275→                        .map(|c| parse_single_statement(c.clone()))
  1276→                        .collect();
  1277→                    else_ifs.push((ei_condition, ei_body));
  1278→                }
  1279→                Rule::statement => {
  1280→                    if in_else {
  1281→                        if else_body.is_none() {
  1282→                            else_body = Some(Vec::new());
  1283→                        }
  1284→                        if let Some(ref mut eb) = else_body {
  1285→                            eb.push(parse_single_statement(child.clone()));
  1286→                        }
  1287→                    }
  1288→                }
  1289→                _ => {
  1290→                    if child.as_str().trim() == "else" {
  1291→                        in_else = true;
  1292→                    }
  1293→                }
  1294→            }
  1295→        }
  1296→
  1297→        if else_ifs.is_empty() && else_body.is_none() {
  1298→            Statement::IfThen(Box::new(condition), body)
  1299→        } else {
  1300→            Statement::IfElseBlock { condition, then_body: body, else_ifs, else_body }
  1301→        }
  1302→    } else if let Some(es_pair) = children.iter().find(|c| c.as_rule() == Rule::expr_stmt) {
  1303→        // Legacy expr_stmt fallback (shouldn't normally be reached with assign_or_expr)
  1304→        let expr = es_pair.clone().into_inner().find(|c| c.as_rule() == Rule::expression)
  1305→            .expect("expr_stmt must contain expression");
  1306→        Statement::ExprStmt(parse_expression(expr))
  1307→    } else if let Some(as_pair) = children.iter().find(|c| c.as_rule() == Rule::assign_stmt) {
  1308→        // Legacy assign_stmt fallback
  1309→        let as_children = children_of(as_pair);
  1310→        let name = find_child_str(&as_children, Rule::IDENT).unwrap_or_default();
  1311→        let expr = find_child(&as_children, Rule::expression).unwrap();
  1312→        Statement::Assign { name, value: parse_expression(expr) }
  1313→    } else {
  1314→        // Fallback: direct expression child (legacy)
  1315→        let expr = find_child(&children, Rule::expression)
  1316→            .expect(&format!("unrecognized statement: '{}'", pair.as_str()));
  1317→        Statement::Return(parse_expression(expr))
  1318→    }
  1319→}

  1321→/// Parse a match statement: `match expr { "val" then { stmts } ... else { stmts } }` (Наряд №14)
  1322→fn parse_match_stmt(pair: Pair<Rule>) -> Statement {
  1323→    use crate::ast::{MatchArm, CompareOp as AstCmp};
  1324→    let children: Vec<Pair<Rule>> = pair.into_inner().collect();
  1325→
  1326→    // First child is the scrutinee expression
  1327→    let scrutinee = children.iter()
  1328→        .find(|c| c.as_rule() == Rule::expression)
  1329→        .map(|c| parse_expression(c.clone()))
  1330→        .unwrap_or_else(|| Expr::StringLit(String::new()));
  1331→
  1332→    // Parse match arms
  1333→    let mut arms = Vec::new();
  1334→    let mut else_body: Option<Vec<Statement>> = None;
  1335→
  1336→    for child in &children {
  1337→        match child.as_rule() {
  1338→            Rule::match_arm_exact => {
  1339→                let arm_children: Vec<Pair<Rule>> = child.clone().into_inner().collect();
  1340→                let value = arm_children.iter()
  1341→                    .find(|c| c.as_rule() == Rule::STRING_LITERAL)
  1342→                    .map(|c| unescape_string(c.as_str()))
  1343→                    .unwrap_or_default();
  1344→                let body = arm_children.iter()
  1345→                    .filter(|c| c.as_rule() == Rule::statement)
  1346→                    .map(|c| parse_single_statement(c.clone()))
  1347→                    .collect();
  1348→                arms.push(MatchArm::Exact(value, body));
  1349→            }
  1350→            Rule::match_arm_starts => {
  1351→                let arm_children: Vec<Pair<Rule>> = child.clone().into_inner().collect();
  1352→                let prefix = arm_children.iter()
  1353→                    .find(|c| c.as_rule() == Rule::STRING_LITERAL)
  1354→                    .map(|c| unescape_string(c.as_str()))
  1355→                    .unwrap_or_default();
  1356→                let body = arm_children.iter()
  1357→                    .filter(|c| c.as_rule() == Rule::statement)
  1358→                    .map(|c| parse_single_statement(c.clone()))
  1359→                    .collect();
  1360→                arms.push(MatchArm::StartsWith(prefix, body));
  1361→            }
  1362→            Rule::match_arm_contains => {
  1363→                let arm_children: Vec<Pair<Rule>> = child.clone().into_inner().collect();
  1364→                let substr = arm_children.iter()
  1365→                    .find(|c| c.as_rule() == Rule::STRING_LITERAL)
  1366→                    .map(|c| unescape_string(c.as_str()))
  1367→                    .unwrap_or_default();
  1368→                let body = arm_children.iter()
  1369→                    .filter(|c| c.as_rule() == Rule::statement)
  1370→                    .map(|c| parse_single_statement(c.clone()))
  1371→                    .collect();
  1372→                arms.push(MatchArm::Contains(substr, body));
  1373→            }
  1374→            Rule::match_arm_compare => {
  1375→                let arm_children: Vec<Pair<Rule>> = child.clone().into_inner().collect();
  1376→                // First non-statement child is the operator (from compare_op rule)
  1377→                let op_str = arm_children.iter()
  1378→                    .find(|c| c.as_rule() == Rule::compare_op)
  1379→                    .map(|c| c.as_str().trim())
  1380→                    .unwrap_or("==");
  1381→                let op = match op_str {
  1382→                    ">=" => AstCmp::Ge,
  1383→                    "<=" => AstCmp::Le,
  1384→                    "!=" => AstCmp::Ne,
  1385→                    ">" => AstCmp::Gt,
  1386→                    "<" => AstCmp::Lt,
  1387→                    "==" => AstCmp::Eq,
  1388→                    _ => AstCmp::Eq,
  1389→                };
  1390→                let expr = arm_children.iter()
  1391→                    .find(|c| c.as_rule() == Rule::expression)
  1392→                    .map(|c| parse_expression(c.clone()))
  1393→                    .unwrap_or_else(|| Expr::FloatLit(0.0));
  1394→                let body = arm_children.iter()
  1395→                    .filter(|c| c.as_rule() == Rule::statement)
  1396→                    .map(|c| parse_single_statement(c.clone()))
  1397→                    .collect();
  1398→                arms.push(MatchArm::Compare(op, expr, body));
  1399→            }
  1400→            Rule::match_else => {
  1401→                let else_children: Vec<Pair<Rule>> = child.clone().into_inner().collect();
  1402→                else_body = Some(else_children.iter()
  1403→                    .filter(|c| c.as_rule() == Rule::statement)
  1404→                    .map(|c| parse_single_statement(c.clone()))
  1405→                    .collect());
  1406→            }
  1407→            _ => {}
  1408→        }
  1409→    }
  1410→
  1411→    Statement::Match { scrutinee, arms, else_body }
  1412→}
  1413→
  1414→/// Наряд №14 P0-3: Parse block if/else as expression.
  1415→/// `if cond { stmts } else if cond { stmts } else { stmts }` → Expr::BlockIfElse
  1416→fn parse_block_if_else_expr(pair: Pair<Rule>) -> Expr {
  1417→    let children: Vec<Pair<Rule>> = pair.into_inner().collect();
  1418→    let condition = children.iter()
  1419→        .find(|c| c.as_rule() == Rule::expression)
  1420→        .map(|c| parse_expression(c.clone()))
  1421→        .unwrap_or_else(|| Expr::BoolLit(true));
  1422→
  1423→    let mut then_body = Vec::new();
  1424→    let mut else_ifs = Vec::new();
  1425→    let mut else_body: Option<Vec<Statement>> = None;
  1426→    let mut in_then = false;
  1427→    let mut in_else = false;
  1428→
  1429→    for child in &children {
  1430→        match child.as_rule() {
  1431→            Rule::statement => {
  1432→                if in_else {
  1433→                    if else_body.is_none() { else_body = Some(Vec::new()); }
  1434→                    if let Some(ref mut eb) = else_body {
  1435→                        eb.push(parse_single_statement(child.clone()));
  1436→                    }
  1437→                } else {
  1438→                    in_then = true;
  1439→                    then_body.push(parse_single_statement(child.clone()));
  1440→                }
  1441→            }
  1442→            Rule::else_if_block => {
  1443→                in_then = false;
  1444→                in_else = false;
  1445→                let ei_children = children_of(child);
  1446→                let ei_condition = ei_children.iter()
  1447→                    .find(|c| c.as_rule() == Rule::expression)
  1448→                    .map(|c| parse_expression(c.clone()))
  1449→                    .unwrap_or_else(|| Expr::BoolLit(true));
  1450→                let ei_body: Vec<Statement> = ei_children.iter()
  1451→                    .filter(|c| c.as_rule() == Rule::statement)
  1452→                    .map(|c| parse_single_statement(c.clone()))
  1453→                    .collect();
  1454→                else_ifs.push((ei_condition, ei_body));
  1455→            }
  1456→            _ => {
  1457→                if child.as_str().trim() == "else" {
  1458→                    in_then = false;
  1459→                    in_else = true;
  1460→                }
  1461→            }
  1462→        }
  1463→    }
  1464→
  1465→    Expr::BlockIfElse {
  1466→        condition: Box::new(condition),
  1467→        then_body,
  1468→        else_ifs: else_ifs.into_iter().map(|(c, b)| (Box::new(c), b)).collect(),
  1469→        else_body,
  1470→    }
  1471→}
  1472→
  1473→/// Parse a block-style if statement: `if expr { stmts } else if expr { stmts } else { stmts }`
  1474→fn parse_if_block_stmt(pair: Pair<Rule>) -> Statement {
  1475→    let children = children_of(&pair);
  1476→    // if_block_stmt = { "if" ~ expression "{" ~ statement* ~ "}" ~ else_if_block* ~ ("else" "{" ~ statement* ~ "}")? }
  1477→    // The pair directly contains the whole if_block_stmt; its children are:
  1478→    // [expression, "{", statement*, "}", else_if_block*, ("else", "{", statement*, "}")?]
  1479→    let mut expr_idx = 0;
  1480→    // Skip any non-expression, non-statement children to find the condition expression
  1481→    let condition = children.iter()
  1482→        .find(|c| c.as_rule() == Rule::expression)
  1483→        .map(|c| parse_expression(c.clone()))
  1484→        .unwrap_or_else(|| Expr::BoolLit(true));
  1485→
  1486→    // Collect then_body statements (between first "{" and matching "}")
  1487→    let mut then_body = Vec::new();
  1488→    let mut else_ifs = Vec::new();
  1489→    let mut else_body: Option<Vec<Statement>> = None;
  1490→
  1491→    let mut in_then = false;
  1492→    let mut in_else = false;
  1493→    for child in &children {
  1494→        match child.as_rule() {
  1495→            Rule::statement => {
  1496→                if in_else {
  1497→                    if else_body.is_none() {
  1498→                        else_body = Some(Vec::new());
  1499→                    }
  1500→                    if let Some(ref mut eb) = else_body {
  1501→                        eb.push(parse_single_statement(child.clone()));
  1502→                    }
  1503→                } else if !in_then {
  1504→                    in_then = true;
  1505→                    then_body.push(parse_single_statement(child.clone()));
  1506→                } else {
  1507→                    then_body.push(parse_single_statement(child.clone()));
  1508→                }
  1509→            }
  1510→            Rule::else_if_block => {
  1511→                in_then = false;
  1512→                in_else = false;
  1513→                // Parse else if block: condition + body
  1514→                let ei_children = children_of(child);
  1515→                let ei_condition = ei_children.iter()
  1516→                    .find(|c| c.as_rule() == Rule::expression)
  1517→                    .map(|c| parse_expression(c.clone()))
  1518→                    .unwrap_or_else(|| Expr::BoolLit(true));
  1519→                let ei_body: Vec<Statement> = ei_children.iter()
  1520→                    .filter(|c| c.as_rule() == Rule::statement)
  1521→                    .map(|c| parse_single_statement(c.clone()))
  1522→                    .collect();
  1523→                else_ifs.push((ei_condition, ei_body));
  1524→            }
  1525→            _ => {
  1526→                // Skip non-statement children (braces, keywords, etc.)
  1527→                if child.as_str().trim() == "else" {
  1528→                    in_then = false;
  1529→                    in_else = true;
  1530→                }
  1531→            }
  1532→        }
  1533→    }
  1534→
  1535→    Statement::IfElseBlock { condition, then_body, else_ifs, else_body }
  1536→}
  1537→
  1538→// ── Flow ──────────────────────────────────────────────────────────────
  1539→// flow_decl = { "flow" ~ IDENT ~ "{" ~ flow_pipeline ~ branch_def* ~ "}" }
  1540→// flow_pipeline = { "input" ":" type_name "=" expression ~ flow_step* ~ ARROW ~ "output" }
  1541→// flow_step     = { ARROW ~ (checkpoint_call | step_ident) }
  1542→// checkpoint_call = { "checkpoint" ~ "(" ~ STRING_LITERAL ~ ")" }
  1543→// branch_def    = { step_ident ~ "{" ~ branch* ~ "}" }
  1544→
  1545→fn parse_flow_decl(pair: Pair<Rule>) -> Declaration {
  1546→    let children = children_of(&pair);
  1547→    let name = find_child_str(&children, Rule::IDENT).unwrap_or_default();
  1548→
  1549→    // children: IDENT, flow_pipeline, [branch_def, ...]
  1550→    let pipeline_pair = find_child(&children, Rule::flow_pipeline).unwrap();
  1551→    let pipeline_children = children_of(&pipeline_pair);
  1552→
  1553→    let mut input_type = String::new();
  1554→    let mut source: Option<Expr> = None;
  1555→    let mut pipeline_steps: Vec<String> = Vec::new();
  1556→    let mut checkpoints: HashMap<String, usize> = HashMap::new();
  1557→
  1558→    // Walk pipeline children: type_name, expression, flow_step*, ARROW
  1559→    let mut i = 0;
  1560→    // First: type_name
  1561→    if i < pipeline_children.len() && pipeline_children[i].as_rule() == Rule::type_name {
  1562→        let tn_inner = children_of(&pipeline_children[i]);
  1563→        input_type = pair_str(&tn_inner[0]);
  1564→        i += 1;
  1565→    }
  1566→    // Second: expression (source)
  1567→    if i < pipeline_children.len() && pipeline_children[i].as_rule() == Rule::expression {
  1568→        source = Some(parse_expression(pipeline_children[i].clone()));
  1569→        i += 1;
  1570→    }
  1571→    // Remaining: flow_step* then final ARROW -> output
  1572→    while i < pipeline_children.len() {
  1573→        if pipeline_children[i].as_rule() == Rule::flow_step {
  1574→            let step_children = children_of(&pipeline_children[i]);
  1575→            // flow_step = { ARROW ~ (checkpoint_call | step_ident) }
  1576→            for sc in &step_children {
  1577→                if sc.as_rule() == Rule::step_ident {
  1578→                    pipeline_steps.push(pair_str(sc));
  1579→                } else if sc.as_rule() == Rule::checkpoint_call {
  1580→                    // checkpoint("name") — maps to the PRECEDING step index
  1581→                    let full = sc.as_str(); // e.g., 'checkpoint("mid")'
  1582→                    // Extract the checkpoint name between quotes
  1583→                    let cp_name = if let Some(s) = full.find('"') {
  1584→                        if let Some(e) = full.rfind('"') {
  1585→                            if e > s + 1 {
  1586→                                full[s+1..e].to_string()
  1587→                            } else {
  1588→                                String::new()
  1589→                            }
  1590→                        } else {
  1591→                            String::new()
  1592→                        }
  1593→                    } else {
  1594→                        String::new()
  1595→                    };
  1596→                    // Checkpoint fires AFTER the last pipeline step added
  1597→                    let step_idx = if pipeline_steps.is_empty() { 0 } else { pipeline_steps.len() - 1 };
  1598→                    checkpoints.insert(cp_name, step_idx);
  1599→                }
  1600→            }
  1601→            i += 1;
  1602→        } else if pipeline_children[i].as_rule() == Rule::ARROW {
  1603→            // Final ARROW before "output" — skip
  1604→            i += 1;
  1605→        } else {
  1606→            i += 1;
  1607→        }
  1608→    }
  1609→
  1610→    // Parse branch_def blocks after the pipeline
  1611→    let mut branch_defs: Vec<(String, Vec<Branch>)> = Vec::new();
  1612→    for child in &children {
  1613→        if child.as_rule() == Rule::branch_def {
  1614→            let bd_children = children_of(child);
  1615→            let step_name = pair_str(&bd_children[0]); // step_ident
  1616→            let branches: Vec<Branch> = bd_children.iter()
  1617→                .filter(|c| c.as_rule() == Rule::branch)
  1618→                .map(|c| parse_branch(c.clone()))
  1619→                .collect();
  1620→            branch_defs.push((step_name, branches));
  1621→        }
  1622→    }
  1623→
  1624→    Declaration::Flow(FlowDecl {
  1625→        name: name.clone(),
  1626→        input_type,
  1627→        source: source.unwrap(),
  1628→        pipeline: pipeline_steps.clone(),
  1629→        branch_defs,
  1630→        checkpoints: checkpoints.clone(),
  1631→    })
  1632→}
  1633→
  1634→fn parse_branch(pair: Pair<Rule>) -> Branch {
  1635→    let children = children_of(&pair);
  1636→    // branch = { IDENT ~ "(" ~ branch_condition ~ ")" ~ ARROW ~ step_ident }
  1637→    let label = pair_str(&children[0]);
  1638→    let cond_pair = find_child(&children, Rule::branch_condition).unwrap();
  1639→    let target = pair_str(&children.last().unwrap()); // step_ident is last
  1640→    Branch {
  1641→        label,
  1642→        condition: parse_branch_condition(cond_pair),
  1643→        target,
  1644→    }
  1645→}
  1646→
  1647→fn parse_branch_condition(pair: Pair<Rule>) -> BranchCondition {
  1648→    let children = children_of(&pair);
  1649→    // branch_condition = { IDENT ~ "." ~ IDENT ~ compare_op ~ expression }
  1650→    // Children: [IDENT(target), IDENT(field), compare_op, expression(threshold)]
  1651→    BranchCondition {
  1652→        target: Expr::Ident(pair_str(&children[0])),
  1653→        field: pair_str(&children[1]),
  1654→        op: parse_compare_op(&children[2]),
  1655→        threshold: parse_expression(children[3].clone()),
  1656→    }
  1657→}

  1659→// ── Expressions ─────────────────────────────────────────────────────
  1660→
  1661→fn parse_binop(pair: &Pair<Rule>) -> BinOp {
  1662→    match pair.as_str().trim() {
  1663→        "+" => BinOp::Add,
  1664→        "-" => BinOp::Sub,
  1665→        "*" => BinOp::Mul,
  1666→        "/" => BinOp::Div,
  1667→        ">=" => BinOp::Ge,
  1668→        "<=" => BinOp::Le,
  1669→        ">" => BinOp::Gt,
  1670→        "<" => BinOp::Lt,
  1671→        "==" => BinOp::Eq,
  1672→        "!=" => BinOp::Ne,
  1673→        _ => unreachable!("unexpected binop: {}", pair.as_str()),
  1674→    }
  1675→}
  1676→
  1677→fn parse_expression(pair: Pair<Rule>) -> Expr {
  1678→    match pair.as_rule() {
  1679→        Rule::expression => {
  1680→            let inner = pair.into_inner().next().unwrap();
  1681→            parse_expression(inner)
  1682→        }
  1683→        Rule::binary_expr => {
  1684→            let children = children_of(&pair);
  1685→            if children.is_empty() { return Expr::StringLit(String::new()); }
  1686→            let mut left = parse_expression(children[0].clone());
  1687→            let mut i = 1;
  1688→            while i + 1 < children.len() {
  1689→                if children[i].as_rule() == Rule::binop {
  1690→                    let op = parse_binop(&children[i]);
  1691→                    left = Expr::BinaryOp(
  1692→                        Box::new(left), op,
  1693→                        Box::new(parse_expression(children[i + 1].clone())),
  1694→                    );
  1695→                    i += 2;
  1696→                } else {
  1697→                    i += 1;
  1698→                }
  1699→            }
  1700→            left
  1701→        }
  1702→        Rule::unary_expr => {
  1703→            parse_expression(pair.into_inner().next().unwrap())
  1704→        }
  1705→        // Наряд №14 P1-4: try expression
  1706→        Rule::try_expr => {
  1707→            let inner = pair.into_inner().next().unwrap();
  1708→            let expr = parse_expression(inner);
  1709→            Expr::Try(Box::new(expr))
  1710→        }
  1711→        Rule::if_else_expr => {
  1712→            let children = children_of(&pair);
  1713→            // if_else_expr = { "if" ~ expression ~ "then" ~ expression ~ "else" ~ expression }
  1714→            let exprs: Vec<Expr> = children.iter()
  1715→                .filter(|c| c.as_rule() == Rule::expression)
  1716→                .map(|c| parse_expression(c.clone()))
  1717→                .collect();
  1718→            // Should have exactly 3 expression children: cond, then, else
  1719→            let cond = exprs[0].clone();
  1720→            let then_br = exprs[1].clone();
  1721→            let else_br = exprs[2].clone();
  1722→            Expr::IfElse(Box::new(cond), Box::new(then_br), Box::new(else_br))
  1723→        }
  1724→        Rule::access_expr => {
  1725→            let children = children_of(&pair);
  1726→            // access_expr = { primary_expr ~ postfix_op* }
  1727→            // postfix_op = { "." ~ IDENT | "[" ~ expression "]" }
  1728→            let mut base = parse_expression(children[0].clone()); // primary_expr
  1729→            let mut i = 1;
  1730→            while i < children.len() {
  1731→                let child = &children[i];
  1732→                let inner = children_of(child);
  1733→                // Check if it's a field access (contains IDENT)
  1734→                if let Some(ident) = inner.iter().find(|c| c.as_rule() == Rule::IDENT) {
  1735→                    base = Expr::FieldAccess(Box::new(base), pair_str(ident));
  1736→                } else if let Some(expr_pair) = inner.iter().find(|c| c.as_rule() == Rule::expression) {
  1737→                    // Index access
  1738→                    base = Expr::IndexAccess(Box::new(base), Box::new(parse_expression(expr_pair.clone())));
  1739→                }
  1740→                i += 1;
  1741→            }
  1742→            base
  1743→        }
  1744→        Rule::qualified_call_expr => {
  1745→            let children = children_of(&pair);
  1746→            // qualified_call_expr = { IDENT ~ "." ~ IDENT ~ "(" ~ expression_list? ~ ")" }
  1747→            let idents: Vec<String> = children.iter()
  1748→                .filter(|c| c.as_rule() == Rule::IDENT)
  1749→                .map(|c| pair_str(c))
  1750→                .collect();
  1751→            let module = idents.get(0).cloned().unwrap_or_default();
  1752→            let function = idents.get(1).cloned().unwrap_or_default();
  1753→            let mut args = Vec::new();
  1754→            for child in children.iter() {
  1755→                if child.as_rule() == Rule::expression_list {
  1756→                    for ap in child.clone().into_inner() {
  1757→                        if ap.as_rule() == Rule::expression {
  1758→                            args.push(parse_expression(ap));
  1759→                        }
  1760→                    }
  1761→                }
  1762→            }
  1763→            Expr::QualifiedCall { module, function, args }
  1764→        }
  1765→        Rule::call_expr => {
  1766→            let children = children_of(&pair);
  1767→            let fname = pair_str(&children[0]);
  1768→            // Handle integer literals as function names (e.g., if INT is matched as call_expr)
  1769→            if fname == "true" || fname == "false" {
  1770→                // Bool literal parsed as call_expr — shouldn't happen but handle gracefully
  1771→                return Expr::BoolLit(fname == "true");
  1772→            }
  1773→            let mut args = Vec::new();
  1774→            for child in children.iter().skip(1) {
  1775→                match child.as_rule() {
  1776→                    Rule::expression_list => {
  1777→                        for ap in child.clone().into_inner() {
  1778→                            if ap.as_rule() == Rule::expression {
  1779→                                args.push(parse_expression(ap));
  1780→                            }
  1781→                        }
  1782→                    }
  1783→                    Rule::expression => args.push(parse_expression(child.clone())),
  1784→                    _ => {}
  1785→                }
  1786→            }
  1787→            Expr::FnCall(fname, args)
  1788→        }
  1789→        Rule::unary_minus => {
  1790→            let children: Vec<_> = pair.clone().into_inner().collect();
  1791→            let inner_expr = if !children.is_empty() {
  1792→                parse_expression(children[0].clone())
  1793→            } else {
  1794→                Expr::FloatLit(0.0)
  1795→            };
  1796→            // Negate: 0.0 - val
  1797→            Expr::BinaryOp(
  1798→                Box::new(Expr::FloatLit(0.0)),
  1799→                BinOp::Sub,
  1800→                Box::new(inner_expr),
  1801→            )
  1802→        }
  1803→        Rule::primary_expr => {
  1804→            let inner = pair.into_inner().next().unwrap();
  1805→            match inner.as_rule() {
  1806→                Rule::block_if_else_expr => {
  1807→                    // Наряд №14 P0-3: block if/else as expression
  1808→                    parse_block_if_else_expr(inner)
  1809→                }
  1810→                Rule::struct_literal => {
  1811→                    let mut fields = std::collections::HashMap::new();
  1812→                    for child in inner.clone().into_inner() {
  1813→                        if child.as_rule() == Rule::struct_field_init {
  1814→                            let field_children: Vec<_> = child.into_inner().collect();
  1815→                            // struct_field_init = { IDENT ~ COLON ~ expression }
  1816→                            let name = pair_str(&field_children[0]);
  1817→                            let value = parse_expression(field_children.last().unwrap().clone());
  1818→                            fields.insert(name, value);
  1819→                        }
  1820→                    }
  1821→                    Expr::StructLit(fields)
  1822→                }
  1823→                Rule::list_literal => {
  1824→                    let mut items = Vec::new();
  1825→                    for child in inner.clone().into_inner() {
  1826→                        match child.as_rule() {
  1827→                            Rule::expression_list => {
  1828→                                // expression_list contains nested expression pairs
  1829→                                for expr_pair in child.into_inner() {
  1830→                                    if expr_pair.as_rule() == Rule::expression {
  1831→                                        items.push(parse_expression(expr_pair));
  1832→                                    }
  1833→                                }
  1834→                            }
  1835→                            Rule::expression => {
  1836→                                items.push(parse_expression(child));
  1837→                            }
  1838→                            _ => {}
  1839→                        }
  1840→                    }
  1841→                    Expr::List(items)
  1842→                }
  1843→                Rule::BOOL_LITERAL => {
  1844→                    Expr::BoolLit(inner.as_str() == "true")
  1845→                }
  1846→                Rule::STRING_LITERAL => {
  1847→                    Expr::StringLit(unescape_string(inner.as_str()))
  1848→                }
  1849→                Rule::FLOAT_LITERAL => {
  1850→                    Expr::FloatLit(inner.as_str().parse().unwrap_or(0.0))
  1851→                }
  1852→                Rule::INT => {
  1853→                    Expr::FloatLit(inner.as_str().parse().unwrap_or(0.0))
  1854→                }
  1855→                Rule::IDENT => Expr::Ident(inner.as_str().to_string()),
  1856→                _ => Expr::Ident(inner.as_str().to_string()),
  1857→            }
  1858→        }
  1859→        _ => Expr::StringLit(pair.as_str().to_string()),
  1860→    }
  1861→}
  1862→

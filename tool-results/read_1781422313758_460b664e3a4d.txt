     1→// ── METALOGOS Stack VM — Phase 4.1 ─────────────────────────────
     2→//
     3→// Executes a compiled bytecode Program. The VM maintains:
     4→//   - A value stack (operands for instructions)
     5→//   - A call stack (frames for pattern invocations)
     6→//   - Global variables (slots)
     7→//   - Pattern table (compiled user-defined functions)
     8→//   - Learnable table (LLM-backed patterns)
     9→//   - Memory store (memorize/recall)
    10→//   - Knowledge graph relations
    11→//   - Mutate log (messages from mutate declarations)
    12→//
    13→// Design: a single main loop that dispatches on the current instruction.
    14→// Function calls push a new frame; Return pops back.
    15→

   204→    /// Execute a compiled program. Returns the flow output (if any),
   205→    /// with mutate log messages prepended if present.
   206→    pub fn run(&mut self, program: Program) -> Result<Option<String>, String> {
   207→        // Initialize globals
   208→        self.globals = vec![Value::Unit; program.globals.len()];
   209→        self.global_names = program.globals.clone();
   210→        self.collections_loaded = program.collections_loaded;
   211→
   212→        // Sort rules by priority descending (matches interpreter semantics)
   213→        let mut rules = program.rules.clone();
   214→        rules.sort_by(|a, b| b.priority.cmp(&a.priority));
   215→        self.rules = rules;
   216→
   217→        // Execute main_code
   218→        let mut stack: Vec<Value> = Vec::new();
   219→        let mut call_stack: Vec<CallFrame> = Vec::new();
   220→        let mut ip = 0;
   221→        let code = &program.main_code;
   222→        let mut flow_output: Option<String> = None;
   223→
   224→        while ip < code.len() {
   225→            let instr = &code[ip];
   226→            match instr {
   227→                // ── Constants & Variables ─────────────────────
   228→                Instruction::Const(v) => {
   229→                    stack.push(v.clone());
   230→                    ip += 1;
   231→                }
   232→                Instruction::LoadGlobal(slot) => {
   233→                    let val = self.globals.get(*slot)
   234→                        .cloned()
   235→                        .unwrap_or(Value::Unit);
   236→                    stack.push(val);
   237→                    ip += 1;
   238→                }
   239→                Instruction::LoadGlobalByName(name) => {
   240→                    // Search globals by name
   241→                    let val = program.globals.iter().position(|n| n == name)
   242→                        .and_then(|slot| self.globals.get(slot).cloned())
   243→                        .unwrap_or(Value::Unit);
   244→                    stack.push(val);
   245→                    ip += 1;
   246→                }
   247→                Instruction::StoreGlobal(slot) => {
   248→                    let val = stack.pop().unwrap_or(Value::Unit);
   249→                    if *slot < self.globals.len() {
   250→                        self.globals[*slot] = val;
   251→                    }
   252→                    ip += 1;
   253→                }
   254→                Instruction::LoadLocal(slot) => {
   255→                    // Local variables are stored on the stack, below the base pointer
   256→                    let bp = call_stack.last().map(|f| f.base_bp).unwrap_or(0);
   257→                    let idx = bp + slot;
   258→                    let val = stack.get(idx).cloned().unwrap_or(Value::Unit);
   259→                    stack.push(val);
   260→                    ip += 1;
   261→                }
   262→                Instruction::StoreLocal(slot) => {
   263→                    let bp = call_stack.last().map(|f| f.base_bp).unwrap_or(0);
   264→                    let idx = bp + slot;
   265→                    let val = stack.pop().unwrap_or(Value::Unit);
   266→                    if idx >= stack.len() {
   267→                        stack.resize(idx + 1, Value::Unit);
   268→                    }
   269→                    stack[idx] = val;
   270→                    ip += 1;
   271→                }
   272→
   273→                // ── Registration ──────────────────────────────
   274→                Instruction::RegisterPattern(fn_def) => {
   275→                    self.patterns.push(fn_def.clone());
   276→                    ip += 1;
   277→                }
   278→                Instruction::RegisterLearnable(info) => {
   279→                    self.learnables.push((info.clone(), Vec::new()));
   280→                    ip += 1;
   281→                }
   282→
   283→                // ── Function Calls ────────────────────────────
   284→                Instruction::CallBuiltin(idx, arity) => {
   285→                    let name = self.builtin_names.get(*idx)
   286→                        .map(|s| s.as_str())
   287→                        .unwrap_or("unknown");
   288→                    let mut args = Vec::new();
   289→                    for _ in 0..*arity {
   290→                        args.insert(0, stack.pop().unwrap_or(Value::Unit));
   291→                    }
   292→                    let result = self.call_builtin(name, &args)?;
   293→                    stack.push(result);
   294→                    ip += 1;
   295→                }
   296→                Instruction::CallPattern(idx, arity) => {
   297→                    let pattern = self.patterns.get(*idx)
   298→                        .ok_or_else(|| format!("VM: pattern index {} not found", idx))?
   299→                        .clone();
   300→
   301→                    // Check arity
   302→                    if *arity != pattern.param_count {
   303→                        return Err(format!(
   304→                            "VM: pattern {} expects {} args, got {}",
   305→                            pattern.name, pattern.param_count, arity
   306→                        ));
   307→                    }
   308→
   309→                    // ── VM bytecode path ────────────────────────
   310→                    // Pop arguments (in reverse) and bind as locals
   311→                    let mut locals = Vec::new();
   312→                    for _ in 0..*arity {
   313→                        locals.insert(0, stack.pop().unwrap_or(Value::Unit));
   314→                    }
   315→
   316→                    // Push call frame
   317→                    let return_ip = ip + 1;
   318→                    let base_bp = stack.len();
   319→                    call_stack.push(CallFrame { return_ip, base_bp });
   320→
   321→                    // Push locals onto stack
   322→                    for local in locals {
   323→                        stack.push(local);
   324→                    }
   325→
   326→                    // Switch to pattern code
   327→                    let result = self.execute_code(&pattern.code, &mut stack, &mut call_stack, &program)?;
   328→                    stack.push(result);
   329→                    // IP already advanced by 1 (return_ip)
   330→                    ip = return_ip;
   331→                }
   332→                Instruction::Return => {
   333→                    let _return_val = stack.pop().unwrap_or(Value::Unit);
   334→                    // Pop call frame and restore stack
   335→                    if let Some(frame) = call_stack.pop() {
   336→                        // Remove locals from stack
   337→                        stack.truncate(frame.base_bp);
   338→                        // We'll push the return value after returning
   339→                        // (handled by the caller)
   340→                    }
   341→                    // Return the value to the execute_code caller
   342→                    // Actually, this is tricky with the current design...
   343→                    // Let me use a different approach — see execute_code below
   344→                    ip += 1;
   345→                }
   346→                Instruction::LlmCall(idx, arity) => {
   347→                    let mut args = Vec::new();
   348→                    for _ in 0..*arity {
   349→                        args.insert(0, stack.pop().unwrap_or(Value::Unit));
   350→                    }
   351→                    let result = self.call_llm(*idx, &args)?;
   352→                    stack.push(result);
   353→                    ip += 1;
   354→                }
   355→
   356→                // ── Binary Operations ──────────────────────────
   357→                Instruction::Add => {
   358→                    let right = stack.pop().unwrap_or(Value::Unit);
   359→                    let left = stack.pop().unwrap_or(Value::Unit);
   360→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Add, right)?);
   361→                    ip += 1;
   362→                }
   363→                Instruction::Sub => {
   364→                    let right = stack.pop().unwrap_or(Value::Unit);
   365→                    let left = stack.pop().unwrap_or(Value::Unit);
   366→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Sub, right)?);
   367→                    ip += 1;
   368→                }
   369→                Instruction::Mul => {
   370→                    let right = stack.pop().unwrap_or(Value::Unit);
   371→                    let left = stack.pop().unwrap_or(Value::Unit);
   372→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Mul, right)?);
   373→                    ip += 1;
   374→                }
   375→                Instruction::Div => {
   376→                    let right = stack.pop().unwrap_or(Value::Unit);
   377→                    let left = stack.pop().unwrap_or(Value::Unit);
   378→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Div, right)?);
   379→                    ip += 1;
   380→                }
   381→                Instruction::Contains => {
   382→                    let right = stack.pop().unwrap_or(Value::Unit);
   383→                    let left = stack.pop().unwrap_or(Value::Unit);
   384→                    let result = self.eval_contains(left, right)?;
   385→                    stack.push(result);
   386→                    ip += 1;
   387→                }
   388→                Instruction::CmpGt => {
   389→                    let right = stack.pop().unwrap_or(Value::Unit);
   390→                    let left = stack.pop().unwrap_or(Value::Unit);
   391→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Gt));
   392→                    ip += 1;
   393→                }
   394→                Instruction::CmpLt => {
   395→                    let right = stack.pop().unwrap_or(Value::Unit);
   396→                    let left = stack.pop().unwrap_or(Value::Unit);
   397→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Lt));
   398→                    ip += 1;
   399→                }
   400→                Instruction::CmpGe => {
   401→                    let right = stack.pop().unwrap_or(Value::Unit);
   402→                    let left = stack.pop().unwrap_or(Value::Unit);
   403→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Ge));
   404→                    ip += 1;
   405→                }
   406→                Instruction::CmpLe => {
   407→                    let right = stack.pop().unwrap_or(Value::Unit);
   408→                    let left = stack.pop().unwrap_or(Value::Unit);
   409→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Le));
   410→                    ip += 1;
   411→                }
   412→                Instruction::CmpEq => {
   413→                    let right = stack.pop().unwrap_or(Value::Unit);
   414→                    let left = stack.pop().unwrap_or(Value::Unit);
   415→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Eq));
   416→                    ip += 1;
   417→                }
   418→                Instruction::CmpNe => {
   419→                    let right = stack.pop().unwrap_or(Value::Unit);
   420→                    let left = stack.pop().unwrap_or(Value::Unit);
   421→                    let eq_result = self.eval_cmp(left, right, AstCompareOp::Eq);
   422→                    match eq_result {
   423→                        Value::Float(f) => stack.push(Value::Float(if f == 1.0 { 0.0 } else { 1.0 })),
   424→                        _ => stack.push(eq_result),
   425→                    }
   426→                    ip += 1;
   427→                }
   428→
   429→                // ── Struct Operations ─────────────────────────
   430→                Instruction::MakeStruct(type_name, field_names) => {
   431→                    let mut fields = HashMap::new();
   432→                    // Values are on stack in field order (first pushed = bottom)
   433→                    // Pop in reverse to get correct order
   434→                    let mut values = Vec::new();
   435→                    for _ in 0..field_names.len() {
   436→                        values.insert(0, stack.pop().unwrap_or(Value::Unit));
   437→                    }
   438→                    for (name, val) in field_names.iter().zip(values.iter()) {
   439→                        fields.insert(name.clone(), val.clone());
   440→                    }
   441→                    stack.push(Value::Struct {
   442→                        type_name: type_name.clone(),
   443→                        fields,
   444→                    });
   445→                    ip += 1;
   446→                }
   447→                Instruction::GetField(field) => {
   448→                    let val = stack.pop().unwrap_or(Value::Unit);
   449→                    let result = val.get_field(field)
   450→                        .cloned()
   451→                        .unwrap_or(Value::Unit);
   452→                    stack.push(result);
   453→                    ip += 1;
   454→                }
   455→                Instruction::IndexAccess => {
   456→                    let idx = stack.pop().unwrap_or(Value::Unit);
   457→                    let base = stack.pop().unwrap_or(Value::Unit);
   458→                    let result = match (&base, &idx) {
   459→                        (Value::List(items), Value::Float(f)) => {
   460→                            let i = *f as isize;
   461→                            if i < 0 {
   462→                                let abs_i = items.len().wrapping_sub((-i) as usize);
   463→                                items.get(abs_i).cloned().unwrap_or(Value::Unit)
   464→                            } else {
   465→                                items.get(i as usize).cloned().unwrap_or(Value::Unit)
   466→                            }
   467→                        }
   468→                        (Value::Struct { fields, .. }, Value::String(key)) => {
   469→                            fields.get(key).cloned().unwrap_or(Value::Unit)
   470→                        }
   471→                        _ => Value::Unit,
   472→                    };
   473→                    stack.push(result);
   474→                    ip += 1;
   475→                }
   476→
   477→                // ── List Operations (Наряд №18) ───────────────
   478→                Instruction::MakeList(n) => {
   479→                    let mut items: Vec<Value> = Vec::with_capacity(*n);
   480→                    for _ in 0..*n {
   481→                        items.push(stack.pop().unwrap_or(Value::Unit));
   482→                    }
   483→                    items.reverse(); // Restore original order
   484→                    stack.push(Value::List(items));
   485→                    ip += 1;
   486→                }
   487→                Instruction::ListLen => {
   488→                    let val = stack.pop().unwrap_or(Value::Unit);
   489→                    let len = match &val {
   490→                        Value::List(items) => items.len() as f64,
   491→                        Value::String(s) => s.len() as f64,
   492→                        _ => 0.0,
   493→                    };
   494→                    stack.push(Value::Float(len));
   495→                    ip += 1;
   496→                }
   497→                Instruction::Pop => {
   498→                    stack.pop();
   499→                    ip += 1;
   500→                }
   501→
   502→                // ── Fluid Types ───────────────────────────────
   503→                Instruction::MakeFluid(count) => {
   504→                    let mut variants: Vec<FluidValueVariant> = Vec::new();
   505→                    // Stack has pairs: value, confidence (bottom to top)
   506→                    // Pop count pairs
   507→                    let mut pairs: Vec<(Value, Value)> = Vec::new();
   508→                    for _ in 0..*count {
   509→                        let confidence = stack.pop().unwrap_or(Value::Float(0.0));
   510→                        let value = stack.pop().unwrap_or(Value::Unit);
   511→                        pairs.insert(0, (value, confidence));
   512→                    }
   513→                    for (value, confidence) in pairs {
   514→                        let conf_f = confidence.as_float().unwrap_or(0.0);
   515→                        // Determine type name from value
   516→                        let type_name = match &value {
   517→                            Value::Float(_) => "Float",
   518→                            Value::String(_) => "String",
   519→                            _ => "Unit",
   520→                        }.to_string();
   521→                        variants.push(FluidValueVariant {
   522→                            type_name,
   523→                            value,
   524→                            confidence: conf_f,
   525→                        });
   526→                    }
   527→                    stack.push(Value::Fluid(variants));
   528→                    ip += 1;
   529→                }
   530→
   531→                // ── Control Flow ───────────────────────────────
   532→                Instruction::Jump(target) => {
   533→                    ip = *target;
   534→                }
   535→                Instruction::JumpIfNot(target) => {
   536→                    let val = stack.pop().unwrap_or(Value::Unit);
   537→                    if !is_truthy(&val) {
   538→                        ip = *target;
   539→                    } else {
   540→                        ip += 1;
   541→                    }
   542→                }
   543→                Instruction::JumpIfLow(threshold, target) => {
   544→                    let val = stack.pop().unwrap_or(Value::Unit);
   545→                    let below = match &val {
   546→                        Value::Float(f) => *f < *threshold,
   547→                        Value::Fluid(variants) => {
   548→                            // Use the maximum confidence
   549→                            let max_conf = variants.iter()
   550→                                .map(|v| v.confidence)
   551→                                .fold(0.0_f64, f64::max);
   552→                            max_conf < *threshold
   553→                        }
   554→                        _ => false,
   555→                    };
   556→                    if below {
   557→                        ip = *target;
   558→                    } else {
   559→                        ip += 1;
   560→                    }
   561→                }
   562→
   563→                // ── METALOGOS Memory ────────────────────────────
   564→                Instruction::Collapse(required_type) => {
   565→                    let val = stack.pop().unwrap_or(Value::Unit);
   566→                    let collapsed = self.maybe_collapse(&val, required_type);
   567→                    stack.push(collapsed);
   568→                    ip += 1;
   569→                }
   570→                Instruction::Memorize(priority) => {
   571→                    let val = stack.pop().unwrap_or(Value::Unit);
   572→                    let value_str = match val {
   573→                        Value::String(s) => s,
   574→                        other => format!("{}", other),
   575→                    };
   576→                    let now = SystemTime::now()
   577→                        .duration_since(UNIX_EPOCH)
   578→                        .map(|d| d.as_secs() as i64)
   579→                        .unwrap_or(0);
   580→                    self.memory.push(VmMemoryEntry {
   581→                        value: value_str,
   582→                        priority: *priority,
   583→                        timestamp: now,
   584→                        decay_rate: 0.01,
   585→                    });
   586→                    ip += 1;
   587→                }
   588→                Instruction::Recall => {
   589→                    let query = stack.pop().unwrap_or(Value::Unit);
   590→                    let query_str = match query {
   591→                        Value::String(s) => s,
   592→                        other => format!("{}", other),
   593→                    };
   594→                    let result = self.recall(&query_str, 0.0);
   595→                    stack.push(Value::String(result));
   596→                    ip += 1;
   597→                }
   598→                Instruction::Forget(days) => {
   599→                    let query = stack.pop().unwrap_or(Value::Unit);
   600→                    let query_str = match query {
   601→                        Value::String(s) => s,
   602→                        other => format!("{}", other),
   603→                    };
   604→                    let now = SystemTime::now()
   605→                        .duration_since(UNIX_EPOCH)
   606→                        .map(|d| d.as_secs() as i64)
   607→                        .unwrap_or(0);
   608→                    let cutoff = now - (days * 86400);
   609→                    self.memory.retain(|entry| {
   610→                        !(entry.value.contains(&query_str) && entry.timestamp < cutoff)
   611→                    });
   612→                    ip += 1;
   613→                }
   614→
   615→                // ── Adapt / Relate / Mutate ─────────────────────
   616→                Instruction::Adapt(pattern_name) => {
   617→                    let output_str = match stack.pop().unwrap_or(Value::Unit) {
   618→                        Value::String(s) => s,
   619→                        other => format!("{}", other),
   620→                    };
   621→                    let input_str = match stack.pop().unwrap_or(Value::Unit) {
   622→                        Value::String(s) => s,
   623→                        other => format!("{}", other),
   624→                    };
   625→                    // Find the learnable by name and add the example
   626→                    let mut found = false;
   627→                    for (info, few_shot) in &mut self.learnables {
   628→                        if info.name == *pattern_name {
   629→                            few_shot.push((input_str, output_str));
   630→                            found = true;
   631→                            break;
   632→                        }
   633→                    }
   634→                    if !found {
   635→                        return Err(format!("VM adapt: learnable pattern '{}' not found", pattern_name));
   636→                    }
   637→                    ip += 1;
   638→                }
   639→                Instruction::Relate => {
   640→                    let relation = match stack.pop().unwrap_or(Value::Unit) {
   641→                        Value::String(s) => s,
   642→                        other => format!("{}", other),
   643→                    };
   644→                    let to = match stack.pop().unwrap_or(Value::Unit) {
   645→                        Value::String(s) => s,
   646→                        other => format!("{}", other),
   647→                    };
   648→                    let from = match stack.pop().unwrap_or(Value::Unit) {
   649→                        Value::String(s) => s,
   650→                        other => format!("{}", other),
   651→                    };
   652→                    self.relations.push(VmRelation { from, to, relation });
   653→                    ip += 1;
   654→                }
   655→                Instruction::Mutate { pattern_name, example_count, rollback_threshold, rollback_op } => {
   656→                    // Pop example_count pairs of (input, output)
   657→                    let mut new_examples = Vec::new();
   658→                    for _ in 0..*example_count {
   659→                        let output_str = match stack.pop().unwrap_or(Value::Unit) {
   660→                            Value::String(s) => s,
   661→                            other => format!("{}", other),
   662→                        };
   663→                        let input_str = match stack.pop().unwrap_or(Value::Unit) {
   664→                            Value::String(s) => s,
   665→                            other => format!("{}", other),
   666→                        };
   667→                        new_examples.push((input_str, output_str));
   668→                    }
   669→
   670→                    // Find the learnable and mutate
   671→                    let msg = self.handle_mutate(pattern_name, new_examples, *rollback_threshold, rollback_op.clone())?;
   672→                    self.mutate_log.push(msg);
   673→                    ip += 1;
   674→                }
   675→
   676→                // ── Pipeline ───────────────────────────────────
   677→                Instruction::FlowExec { source_expr, pipeline, branch_defs } => {
   678→                    // Load the source value
   679→                    let source_val = match source_expr {
   680→                        FlowExpr::GlobalSlot(slot) => {
   681→                            self.globals.get(*slot).cloned().unwrap_or(Value::Unit)
   682→                        }
   683→                        FlowExpr::Ident(name) => {
   684→                            program.globals.iter().position(|n| n == name)
   685→                                .and_then(|slot| self.globals.get(slot).cloned())
   686→                                .unwrap_or(Value::Unit)
   687→                        }
   688→                        FlowExpr::Const(v) => v.clone(),
   689→                    };
   690→
   691→                    // Execute pipeline steps
   692→                    let mut current = source_val;
   693→                    for step_name in pipeline {
   694→                        current = self.run_flow_step(step_name, current, branch_defs)?;
   695→                    }
   696→
   697→                    // The pipeline result is the flow output
   698→                    let output_str = format!("{}", current);
   699→                    flow_output = Some(output_str);
   700→                    ip += 1;
   701→                }
   702→
   703→                // ── Rule Engine ─────────────────────────────────
   704→                Instruction::ExecuteRules => {
   705→                    self.execute_rules()?;
   706→                    ip += 1;
   707→                }
   708→
   709→                // ── Meta ───────────────────────────────────────
   710→                Instruction::Halt => {
   711→                    break;
   712→                }
   713→            }
   714→        }
   715→
   716→        // Build final output with mutate log
   717→        let mutate_log = std::mem::take(&mut self.mutate_log);
   718→        if mutate_log.is_empty() {
   719→            Ok(flow_output)
   720→        } else {
   721→            match flow_output {
   722→                Some(flow) => {
   723→                    let mut result = mutate_log.join("\n");
   724→                    result.push_str("\n");
   725→                    result.push_str(&flow);
   726→                    Ok(Some(result))
   727→                }
   728→                None => Ok(Some(mutate_log.join("\n"))),
   729→            }
   730→        }
   731→    }
   732→
   733→    /// Execute a block of code (e.g., pattern body) and return the result.
   734→    /// This handles the call stack and Return instructions internally.
   735→    fn execute_code(
   736→        &self,
   737→        code: &[Instruction],
   738→        stack: &mut Vec<Value>,
   739→        call_stack: &mut Vec<CallFrame>,
   740→        program: &Program,
   741→    ) -> Result<Value, String> {
   742→        let mut ip = 0;
   743→        // Safety: prevent infinite loops (max iterations per 100 instructions)
   744→        let max_iterations = code.len().max(1) * 100_000;
   745→        let mut iterations = 0;
   746→        while ip < code.len() {
   747→            iterations += 1;
   748→            if iterations > max_iterations {
   749→                return Err(format!("VM execute_code: possible infinite loop ({} iterations, {} instructions)", iterations, code.len()));
   750→            }
   751→            let instr = &code[ip];
   752→            match instr {
   753→                Instruction::LoadLocal(slot) => {
   754→                    let bp = call_stack.last().map(|f| f.base_bp).unwrap_or(0);
   755→                    let idx = bp + slot;
   756→                    let val = stack.get(idx).cloned().unwrap_or(Value::Unit);
   757→                    stack.push(val);
   758→                    ip += 1;
   759→                }
   760→                Instruction::StoreLocal(slot) => {
   761→                    let bp = call_stack.last().map(|f| f.base_bp).unwrap_or(0);
   762→                    let idx = bp + slot;
   763→                    let val = stack.pop().unwrap_or(Value::Unit);
   764→                    // Ensure stack has space at idx
   765→                    if idx >= stack.len() {
   766→                        stack.resize(idx + 1, Value::Unit);
   767→                    }
   768→                    stack[idx] = val;
   769→                    ip += 1;
   770→                }
   771→                Instruction::Const(v) => {
   772→                    stack.push(v.clone());
   773→                    ip += 1;
   774→                }
   775→                Instruction::LoadGlobal(slot) => {
   776→                    let val = self.globals.get(*slot).cloned().unwrap_or(Value::Unit);
   777→                    stack.push(val);
   778→                    ip += 1;
   779→                }
   780→                Instruction::CallBuiltin(idx, arity) => {
   781→                    let name = self.builtin_names.get(*idx)
   782→                        .map(|s| s.as_str())
   783→                        .unwrap_or("unknown");
   784→                    let mut args = Vec::new();
   785→                    for _ in 0..*arity {
   786→                        args.insert(0, stack.pop().unwrap_or(Value::Unit));
   787→                    }
   788→                    let result = self.call_builtin(name, &args)?;
   789→                    stack.push(result);
   790→                    ip += 1;
   791→                }
   792→                Instruction::CallPattern(pidx, arity) => {
   793→                    let pattern = self.patterns.get(*pidx)
   794→                        .ok_or_else(|| format!("VM: pattern index {} not found", pidx))?
   795→                        .clone();
   796→                    if *arity != pattern.param_count {
   797→                        return Err(format!(
   798→                            "VM: pattern {} expects {} args, got {}",
   799→                            pattern.name, pattern.param_count, arity
   800→                        ));
   801→                    }
   802→                    let mut locals = Vec::new();
   803→                    for _ in 0..*arity {
   804→                        locals.insert(0, stack.pop().unwrap_or(Value::Unit));
   805→                    }
   806→                    let base_bp = stack.len();
   807→                    call_stack.push(CallFrame { return_ip: ip + 1, base_bp });
   808→                    for local in locals {
   809→                        stack.push(local);
   810→                    }
   811→                    let result = self.execute_code(&pattern.code, stack, call_stack, program)?;
   812→                    call_stack.pop();
   813→                    stack.push(result);
   814→                    ip += 1;
   815→                }
   816→                Instruction::Add => {
   817→                    let right = stack.pop().unwrap_or(Value::Unit);
   818→                    let left = stack.pop().unwrap_or(Value::Unit);
   819→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Add, right)?);
   820→                    ip += 1;
   821→                }
   822→                Instruction::Sub => {
   823→                    let right = stack.pop().unwrap_or(Value::Unit);
   824→                    let left = stack.pop().unwrap_or(Value::Unit);
   825→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Sub, right)?);
   826→                    ip += 1;
   827→                }
   828→                Instruction::Mul => {
   829→                    let right = stack.pop().unwrap_or(Value::Unit);
   830→                    let left = stack.pop().unwrap_or(Value::Unit);
   831→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Mul, right)?);
   832→                    ip += 1;
   833→                }
   834→                Instruction::Div => {
   835→                    let right = stack.pop().unwrap_or(Value::Unit);
   836→                    let left = stack.pop().unwrap_or(Value::Unit);
   837→                    stack.push(self.eval_binop(left, crate::ast::BinOp::Div, right)?);
   838→                    ip += 1;
   839→                }
   840→                Instruction::Return => {
   841→                    return Ok(stack.pop().unwrap_or(Value::Unit));
   842→                }
   843→                // Phase 5.1: control flow in pattern bodies
   844→                Instruction::Jump(target) => {
   845→                    ip = *target;
   846→                }
   847→                Instruction::JumpIfNot(target) => {
   848→                    let val = stack.pop().unwrap_or(Value::Unit);
   849→                    if !is_truthy(&val) {
   850→                        ip = *target;
   851→                    } else {
   852→                        ip += 1;
   853→                    }
   854→                }
   855→                // Phase 5.1: comparison operators
   856→                Instruction::CmpGt => {
   857→                    let right = stack.pop().unwrap_or(Value::Unit);
   858→                    let left = stack.pop().unwrap_or(Value::Unit);
   859→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Gt));
   860→                    ip += 1;
   861→                }
   862→                Instruction::CmpLt => {
   863→                    let right = stack.pop().unwrap_or(Value::Unit);
   864→                    let left = stack.pop().unwrap_or(Value::Unit);
   865→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Lt));
   866→                    ip += 1;
   867→                }
   868→                Instruction::CmpGe => {
   869→                    let right = stack.pop().unwrap_or(Value::Unit);
   870→                    let left = stack.pop().unwrap_or(Value::Unit);
   871→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Ge));
   872→                    ip += 1;
   873→                }
   874→                Instruction::CmpLe => {
   875→                    let right = stack.pop().unwrap_or(Value::Unit);
   876→                    let left = stack.pop().unwrap_or(Value::Unit);
   877→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Le));
   878→                    ip += 1;
   879→                }
   880→                Instruction::CmpEq => {
   881→                    let right = stack.pop().unwrap_or(Value::Unit);
   882→                    let left = stack.pop().unwrap_or(Value::Unit);
   883→                    stack.push(self.eval_cmp(left, right, AstCompareOp::Eq));
   884→                    ip += 1;
   885→                }
   886→                Instruction::CmpNe => {
   887→                    let right = stack.pop().unwrap_or(Value::Unit);
   888→                    let left = stack.pop().unwrap_or(Value::Unit);
   889→                    let eq_result = self.eval_cmp(left, right, AstCompareOp::Eq);
   890→                    match eq_result {
   891→                        Value::Float(f) => stack.push(Value::Float(if f == 1.0 { 0.0 } else { 1.0 })),
   892→                        _ => stack.push(eq_result),
   893→                    }
   894→                    ip += 1;
   895→                }
   896→                Instruction::GetField(field) => {
   897→                    let val = stack.pop().unwrap_or(Value::Unit);
   898→                    let result = val.get_field(field).cloned().unwrap_or(Value::Unit);
   899→                    stack.push(result);
   900→                    ip += 1;
   901→                }
   902→                Instruction::LoadGlobalByName(name) => {
   903→                    let val = program.globals.iter().position(|n| n == name)
   904→                        .and_then(|slot| self.globals.get(slot).cloned())
   905→                        .unwrap_or(Value::Unit);
   906→                    stack.push(val);
   907→                    ip += 1;
   908→                }
   909→                Instruction::Collapse(required_type) => {
   910→                    let val = stack.pop().unwrap_or(Value::Unit);
   911→                    let collapsed = self.maybe_collapse(&val, required_type);
   912→                    stack.push(collapsed);
   913→                    ip += 1;
   914→                }
   915→                Instruction::LlmCall(idx, arity) => {
   916→                    let mut args = Vec::new();
   917→                    for _ in 0..*arity {
   918→                        args.insert(0, stack.pop().unwrap_or(Value::Unit));
   919→                    }
   920→                    let result = self.call_llm(*idx, &args)?;
   921→                    stack.push(result);
   922→                    ip += 1;
   923→                }
   924→                Instruction::Recall => {
   925→                    let query = stack.pop().unwrap_or(Value::Unit);
   926→                    let query_str = match query {
   927→                        Value::String(s) => s,
   928→                        other => format!("{}", other),
   929→                    };
   930→                    let result = self.recall(&query_str, 0.0);
   931→                    stack.push(Value::String(result));
   932→                    ip += 1;
   933→                }
   934→                // For any unhandled instruction, skip
   935→                _ => { ip += 1; }
   936→            }
   937→        }
   938→        Ok(stack.pop().unwrap_or(Value::Unit))
   939→    }

   941→    /// Call a built-in function by name.
   942→    fn call_builtin(&self, name: &str, args: &[Value]) -> Result<Value, String> {
   943→        if name == "recall" {
   944→            let query = match args.get(0) {
   945→                Some(Value::String(s)) => s.clone(),
   946→                other => return Err(format!("recall() expected String, got {:?}", other)),
   947→            };
   948→            let min_conf = if args.len() > 1 {
   949→                args[1].as_float().unwrap_or(0.0)
   950→            } else {
   951→                0.0
   952→            };
   953→            return Ok(Value::String(self.recall(&query, min_conf)));
   954→        }
   955→
   956→        // Collections operations
   957→        if self.collections_loaded {
   958→            match name {
   959→                "map" | "filter" | "reduce" => {
   960→                    return Err(format!("VM: collection op '{}' not yet implemented in VM path", name));
   961→                }
   962→                _ => {}
   963→            }
   964→        }
   965→
   966→        if let Some(builtin_fn) = self.builtins.get(name) {
   967→            return builtin_fn(args);
   968→        }
   969→        Err(format!("VM: undefined builtin: {}", name))
   970→    }

   972→    /// Call an LLM-backed learnable pattern.
   973→    fn call_llm(&self, idx: usize, args: &[Value]) -> Result<Value, String> {
   974→        let (info, few_shot) = self.learnables.get(idx)
   975→            .ok_or_else(|| format!("VM: learnable index {} not found", idx))?;
   976→
   977→        // Build input string
   978→        let input_parts: Vec<String> = args.iter().map(|a| format!("{}", a)).collect();
   979→        let input = input_parts.join(", ");
   980→
   981→        // Check few-shot cache first
   982→        for (example_input, example_output) in few_shot {
   983→            if input == *example_input {
   984→                return Ok(Value::String(example_output.clone()));
   985→            }
   986→        }
   987→
   988→        // Call LLM backend
   989→        let backend = llm::create_llm_backend();
   990→        let response = backend.call(&info.prompt, &input)?;
   991→
   992→        // Try to parse JSON response into Value::Struct
   993→        if let Ok(json) = serde_json::from_str::<serde_json::Value>(&response) {
   994→            if let Some(obj) = json.as_object() {
   995→                let mut fields = std::collections::HashMap::new();
   996→                for (k, v) in obj {
   997→                    fields.insert(k.clone(), Vm::json_to_value(v));
   998→                }
   999→                return Ok(Value::Struct {
  1000→                    type_name: "LlmResponse".to_string(),
  1001→                    fields,
  1002→                });
  1003→            }
  1004→        }
  1005→
  1006→        Ok(Value::String(response))
  1007→    }

  1009→    /// Convert serde_json::Value to METALOGOS Value (for VM).
  1010→    fn json_to_value(json: &serde_json::Value) -> Value {
  1011→        match json {
  1012→            serde_json::Value::String(s) => Value::String(s.clone()),
  1013→            serde_json::Value::Number(n) => Value::Float(n.as_f64().unwrap_or(0.0)),
  1014→            serde_json::Value::Bool(b) => Value::Bool(*b),
  1015→            serde_json::Value::Null => Value::Unit,
  1016→            serde_json::Value::Array(arr) => {
  1017→                Value::List(arr.iter().map(|v| Vm::json_to_value(v)).collect())
  1018→            }
  1019→            serde_json::Value::Object(obj) => {
  1020→                let mut fields = std::collections::HashMap::new();
  1021→                for (k, v) in obj {
  1022→                    fields.insert(k.clone(), Vm::json_to_value(v));
  1023→                }
  1024→                Value::Struct { type_name: "Json".to_string(), fields }
  1025→            }
  1026→        }
  1027→    }

  1029→    /// Execute a flow step: check branches, then invoke pattern/builtin.
  1030→    fn run_flow_step(
  1031→        &mut self,
  1032→        step_name: &str,
  1033→        current: Value,
  1034→        branch_defs: &[(String, Vec<BranchDef>)],
  1035→    ) -> Result<Value, String> {
  1036→        // Check if step has branch definitions
  1037→        for (bd_step, branches) in branch_defs {
  1038→            if bd_step == step_name {
  1039→                for branch in branches {
  1040→                    if self.eval_branch_condition(branch, &current)? {
  1041→                        return self.invoke_step(&branch.target, vec![current]);
  1042→                    }
  1043→                }
  1044→                return Err(format!("VM: no branch matched in step '{}'", step_name));
  1045→            }
  1046→        }
  1047→
  1048→        // No branch definitions — invoke as pattern/builtin
  1049→        self.invoke_step(step_name, vec![current])
  1050→    }
  1051→
  1052→    /// Evaluate a branch condition against a value.
  1053→    fn eval_branch_condition(&self, branch: &BranchDef, current: &Value) -> Result<bool, String> {
  1054→        let field_val = current.get_field(&branch.condition_field)
  1055→            .map_err(|e| format!("branch condition: {}", e))?
  1056→            .clone();
  1057→        let fv = field_val.as_float()?;
  1058→        let tv = branch.condition_threshold.as_float()?;
  1059→        Ok(match branch.condition_op {
  1060→            ConditionOp::Gt => fv > tv,
  1061→            ConditionOp::Lt => fv < tv,
  1062→            ConditionOp::Ge => fv >= tv,
  1063→            ConditionOp::Le => fv <= tv,
  1064→            ConditionOp::Eq => fv == tv,
  1065→        })
  1066→    }
  1067→
  1068→    /// Invoke a pattern or builtin by name (used in flow pipeline).
  1069→    fn invoke_step(&mut self, name: &str, args: Vec<Value>) -> Result<Value, String> {
  1070→        // Check learnables first
  1071→        for (i, (info, _)) in self.learnables.iter().enumerate() {
  1072→            if info.name == name {
  1073→                return self.call_llm(i, &args);
  1074→            }
  1075→        }
  1076→
  1077→        // Check patterns
  1078→        for pattern in self.patterns.iter() {
  1079→            if pattern.name == name {
  1080→                if args.len() != pattern.param_count {
  1081→                    return Err(format!(
  1082→                        "VM: pattern {} expects {} args, got {}",
  1083→                        name, pattern.param_count, args.len()
  1084→                    ));
  1085→                }
  1086→
  1087→                // ── VM bytecode path ───────────────────────────
  1088→                // Collapse Fluid arguments to parameter types
  1089→                let collapsed_args: Vec<Value> = args.iter()
  1090→                    .zip(pattern.param_types.iter())
  1091→                    .map(|(arg, param_type)| self.maybe_collapse(arg, param_type))
  1092→                    .collect();
  1093→                // Execute pattern body as bytecode
  1094→                let mut stack: Vec<Value> = Vec::new();
  1095→                let mut call_stack: Vec<CallFrame> = Vec::new();
  1096→                let program = Program {
  1097→                    globals: Vec::new(),
  1098→                    patterns: Vec::new(),
  1099→                    learnables: Vec::new(),
  1100→                    rules: Vec::new(),
  1101→                    main_code: Vec::new(),
  1102→                    collections_loaded: false,
  1103→                };
  1104→                for arg in collapsed_args {
  1105→                    stack.push(arg);
  1106→                }
  1107→                return self.execute_code(&pattern.code, &mut stack, &mut call_stack, &program);
  1108→            }
  1109→        }
  1110→
  1111→        // Check builtins
  1112→        self.call_builtin(name, &args)
  1113→    }

  1115→    /// Execute all registered rules (already sorted by priority descending).
  1116→    fn execute_rules(&mut self) -> Result<(), String> {
  1117→        for rule in &self.rules {
  1118→            let condition_met = self.eval_rule_condition(&rule.condition)?;
  1119→            if condition_met {
  1120→                // Find the target entity by name in globals
  1121→                let target_slot = self.global_names.iter().position(|n| n == &rule.target_name);
  1122→                if let Some(slot) = target_slot {
  1123→                    let value = self.eval_rule_value(&rule.value_expr)?;
  1124→                    // Set field on the struct
  1125→                    if let Some(entity) = self.globals.get_mut(slot) {
  1126→                        let _ = entity.set_field(&rule.field, value);
  1127→                    }
  1128→                }
  1129→            }
  1130→        }
  1131→        Ok(())
  1132→    }

  1134→    /// Evaluate a rule condition.
  1135→    fn eval_rule_condition(&self, cond: &RuleCondition) -> Result<bool, String> {
  1136→        match cond {
  1137→            RuleCondition::Contains { left, right } => {
  1138→                let ls = self.eval_rule_value(left)?;
  1139→                let rs = self.eval_rule_value(right)?;
  1140→                match (&ls, &rs) {
  1141→                    (Value::String(l), Value::String(r)) => Ok(l.contains(r)),
  1142→                    _ => Err("contains: both sides must be String".to_string()),
  1143→                }
  1144→            }
  1145→            RuleCondition::Compare { left, op, right } => {
  1146→                let lv = self.eval_rule_value(left)?;
  1147→                let rv = self.eval_rule_value(right)?;
  1148→                let lf = lv.as_float()?;
  1149→                let rf = rv.as_float()?;
  1150→                Ok(match op {
  1151→                    ConditionOp::Gt => lf > rf,
  1152→                    ConditionOp::Lt => lf < rf,
  1153→                    ConditionOp::Ge => lf >= rf,
  1154→                    ConditionOp::Le => lf <= rf,
  1155→                    ConditionOp::Eq => lf == rf,
  1156→                })
  1157→            }
  1158→        }
  1159→    }

  1161→    /// Evaluate a rule value expression to a runtime value.
  1162→    fn eval_rule_value(&self, expr: &RuleValueExpr) -> Result<Value, String> {
  1163→        match expr {
  1164→            RuleValueExpr::StringLit(s) => Ok(Value::String(s.clone())),
  1165→            RuleValueExpr::FloatLit(f) => Ok(Value::Float(*f)),
  1166→            RuleValueExpr::Ident(name) => {
  1167→                // Search globals by name
  1168→                let slot = self.global_names.iter().position(|n| n == name);
  1169→                match slot {
  1170→                    Some(s) => Ok(self.globals.get(s).cloned().unwrap_or(Value::Unit)),
  1171→                    None => Err(format!("VM rule: undefined variable '{}'", name)),
  1172→                }
  1173→            }
  1174→            RuleValueExpr::FieldAccess(entity_name, field) => {
  1175→                let slot = self.global_names.iter().position(|n| n == entity_name);
  1176→                match slot {
  1177→                    Some(s) => {
  1178→                        let entity = self.globals.get(s).cloned().unwrap_or(Value::Unit);
  1179→                        entity.get_field(field).cloned()
  1180→                            .map_err(|e| format!("VM rule field access: {}", e))
  1181→                    }
  1182→                    None => Err(format!("VM rule: undefined entity '{}'", entity_name)),
  1183→                }
  1184→            }
  1185→        }
  1186→    }

  1188→    /// Handle a mutate declaration.
  1189→    fn handle_mutate(
  1190→        &mut self,
  1191→        pattern_name: &str,
  1192→        new_examples: Vec<(String, String)>,
  1193→        rollback_threshold: Option<f64>,
  1194→        rollback_op: Option<ConditionOp>,
  1195→    ) -> Result<String, String> {
  1196→        // Find the learnable
  1197→        for (_, (info, few_shot)) in self.learnables.iter_mut().enumerate() {
  1198→            if info.name == pattern_name {
  1199→                let original = few_shot.clone();
  1200→                *few_shot = new_examples;
  1201→
  1202→                // Mock accuracy (always 0.95 for MockLlm)
  1203→                let accuracy: f64 = 0.95;
  1204→
  1205→                let kept = match (&rollback_op, &rollback_threshold) {
  1206→                    (Some(ConditionOp::Lt), Some(threshold)) => accuracy >= *threshold,
  1207→                    (Some(ConditionOp::Le), Some(threshold)) => accuracy > *threshold,
  1208→                    (Some(ConditionOp::Gt), Some(_)) |
  1209→                    (Some(ConditionOp::Ge), Some(_)) => false,
  1210→                    (Some(ConditionOp::Eq), Some(threshold)) => (accuracy - threshold).abs() < 1e-9,
  1211→                    _ => true,
  1212→                };
  1213→
  1214→                if kept {
  1215→                    return Ok(format!("[MUTATE] {}: accuracy={}, kept (>= {:.1})",
  1216→                        pattern_name, accuracy,
  1217→                        rollback_threshold.unwrap_or(0.0)));
  1218→                } else {
  1219→                    *few_shot = original;
  1220→                    return Ok(format!("[MUTATE] {}: accuracy={}, rolled back (below {:.1})",
  1221→                        pattern_name, accuracy,
  1222→                        rollback_threshold.unwrap_or(0.0)));
  1223→                }
  1224→            }
  1225→        }
  1226→        Err(format!("VM mutate: learnable pattern '{}' not found", pattern_name))
  1227→    }

  1229→    /// Recall from memory: find best matching entry by substring + decay.
  1230→    fn recall(&self, query: &str, min_confidence: f64) -> String {
  1231→        let now = SystemTime::now()
  1232→            .duration_since(UNIX_EPOCH)
  1233→            .map(|d| d.as_secs() as i64)
  1234→            .unwrap_or(0);
  1235→
  1236→        let mut best_match: Option<&VmMemoryEntry> = None;
  1237→        let mut best_activation: f64 = 0.0;
  1238→
  1239→        for entry in &self.memory {
  1240→            if !entry.value.contains(query) {
  1241→                continue;
  1242→            }
  1243→            let age_days = ((now - entry.timestamp).max(0) as f64) / 86400.0;
  1244→            let activation = entry.priority * (-entry.decay_rate * age_days).exp();
  1245→            if activation > best_activation && activation >= min_confidence {
  1246→                best_activation = activation;
  1247→                best_match = Some(entry);
  1248→            }
  1249→        }
  1250→
  1251→        match best_match {
  1252→            Some(entry) => {
  1253→                let mut result = entry.value.clone();
  1254→                // Walk knowledge graph for related memories
  1255→                for rel in &self.relations {
  1256→                    if rel.from == entry.value {
  1257→                        result.push_str("\n");
  1258→                        result.push_str(&format!("[GRAPH] {} -> {}", rel.relation, rel.to));
  1259→                    } else if rel.to == entry.value {
  1260→                        result.push_str("\n");
  1261→                        result.push_str(&format!("[GRAPH] {} -> {}", rel.relation, rel.from));
  1262→                    }
  1263→                }
  1264→                result
  1265→            }
  1266→            None => String::new(),
  1267→        }
  1268→    }

  1270→    /// Collapse a Fluid value to a concrete type.
  1271→    fn maybe_collapse(&self, value: &Value, required_type: &str) -> Value {
  1272→        match value {
  1273→            Value::Fluid(variants) => {
  1274→                let best = variants.iter()
  1275→                    .filter(|v| v.type_name == required_type)
  1276→                    .max_by(|a, b| {
  1277→                        a.confidence.partial_cmp(&b.confidence)
  1278→                            .unwrap_or(std::cmp::Ordering::Equal)
  1279→                    });
  1280→                match best {
  1281→                    Some(variant) if variant.confidence >= COLLAPSE_THRESHOLD => {
  1282→                        variant.value.clone()
  1283→                    }
  1284→                    _ => Value::Unit,
  1285→                }
  1286→            }
  1287→            other => other.clone(),
  1288→        }
  1289→    }

  1291→    /// Evaluate a binary operation.
  1292→    fn eval_binop(&self, left: Value, op: crate::ast::BinOp, right: Value) -> Result<Value, String> {
  1293→        match (left, right) {
  1294→            (Value::String(a), Value::String(b)) => match op {
  1295→                crate::ast::BinOp::Add => Ok(Value::String(format!("{}{}", a, b))),
  1296→                crate::ast::BinOp::Eq => Ok(Value::Float(if a == b { 1.0 } else { 0.0 })),
  1297→                crate::ast::BinOp::Ne => Ok(Value::Float(if a != b { 1.0 } else { 0.0 })),
  1298→                _ => Err(format!("cannot apply {:?} to two Strings", op)),
  1299→            },
  1300→            (Value::Float(a), Value::Float(b)) => match op {
  1301→                crate::ast::BinOp::Add => Ok(Value::Float(a + b)),
  1302→                crate::ast::BinOp::Sub => Ok(Value::Float(a - b)),
  1303→                crate::ast::BinOp::Mul => Ok(Value::Float(a * b)),
  1304→                crate::ast::BinOp::Div => {
  1305→                    if b == 0.0 {
  1306→                        Err("division by zero".to_string())
  1307→                    } else {
  1308→                        Ok(Value::Float(a / b))
  1309→                    }
  1310→                }
  1311→                crate::ast::BinOp::Gt => Ok(Value::Float(if a > b { 1.0 } else { 0.0 })),
  1312→                crate::ast::BinOp::Lt => Ok(Value::Float(if a < b { 1.0 } else { 0.0 })),
  1313→                crate::ast::BinOp::Ge => Ok(Value::Float(if a >= b { 1.0 } else { 0.0 })),
  1314→                crate::ast::BinOp::Le => Ok(Value::Float(if a <= b { 1.0 } else { 0.0 })),
  1315→                crate::ast::BinOp::Eq => Ok(Value::Float(if a == b { 1.0 } else { 0.0 })),
  1316→                crate::ast::BinOp::Ne => Ok(Value::Float(if a != b { 1.0 } else { 0.0 })),
  1317→            },
  1318→            (l, r) => Err(format!(
  1319→                "type mismatch: {} {:?} {}",
  1320→                l.type_name(), op, r.type_name()
  1321→            )),
  1322→        }
  1323→    }

  1325→    /// Evaluate contains(left, right).
  1326→    fn eval_contains(&self, left: Value, right: Value) -> Result<Value, String> {
  1327→        let ls = match left {
  1328→            Value::String(s) => s,
  1329→            other => return Err(format!("contains: left must be String, got {}", other.type_name())),
  1330→        };
  1331→        let rs = match right {
  1332→            Value::String(s) => s,
  1333→            other => return Err(format!("contains: right must be String, got {}", other.type_name())),
  1334→        };

  1336→    }

  1338→    /// Evaluate a comparison: push 1.0 (true) or 0.0 (false).
  1339→    fn eval_cmp(&self, left: Value, right: Value, op: AstCompareOp) -> Value {
  1340→        let result = match (left.as_float(), right.as_float()) {
  1341→            (Ok(lf), Ok(rf)) => match op {
  1342→                AstCompareOp::Gt => lf > rf,
  1343→                AstCompareOp::Lt => lf < rf,
  1344→                AstCompareOp::Ge => lf >= rf,
  1345→                AstCompareOp::Le => lf <= rf,
  1346→                AstCompareOp::Eq => lf == rf,
  1347→                _ => false,
  1348→            },
  1349→            _ => false,
  1350→        };
  1351→        Value::Float(if result { 1.0 } else { 0.0 })
  1352→    }

     3→// Contract: the emitted Program, when executed by the VM, produces
     4→// the same output as the tree-walking interpreter.

    14→/// The compiler translates declarations into a bytecode Program.
    15→pub struct Compiler {
    16→    /// Global variable name -> slot index.
    17→    global_slots: HashMap<String, usize>,
    18→    /// Next available global slot.
    19→    next_global: usize,
    20→    /// Pattern name -> index in program.patterns.
    21→    pattern_indices: HashMap<String, usize>,
    22→    /// Learnable pattern name -> index in program.learnables.
    23→    learnable_indices: HashMap<String, usize>,
    24→    /// Builtin name -> index in builtin table.
    25→    builtin_indices: HashMap<String, usize>,
    26→    /// Struct type name -> field names (in declaration order).
    27→    struct_fields: HashMap<String, Vec<String>>,
    28→    /// Rule declarations to compile.
    29→    rules: Vec<CompiledRule>,
    30→    /// Root directory for import resolution.
    31→    std_root: PathBuf,
    32→    /// Already-imported modules.
    33→    imported_modules: HashSet<String>,
    34→    /// Collections loaded flag.
    35→    collections_loaded: bool,
    36→    /// Sandbox declarations (recorded).
    37→    sandboxes: HashMap<String, SandboxDecl>,
    38→}

   136→    /// Compile a list of declarations into a bytecode Program.
   137→    pub fn compile(&mut self, declarations: Vec<Declaration>) -> Result<Program, String> {
   138→        // Phase 1: resolve imports
   139→        let mut all_decls = Vec::new();
   140→        for decl in declarations {
   141→            if let Declaration::Import(import) = &decl {
   142→                // Fix 2: use `import.path` instead of `import.module_path`
   143→                if !self.imported_modules.contains(&import.path) {
   144→                    let imported = self.resolve_import(&import.path)?;
   145→                    all_decls.extend(imported);
   146→                }
   147→            } else {
   148→                all_decls.push(decl);
   149→            }
   150→        }
   151→
   152→        // Phase 2: two-pass compilation
   153→        // Pass 1: collect struct types, pattern names, learnable names, global slots
   154→        self.pass1(&all_decls)?;
   155→
   156→        // Pass 2: generate main_code
   157→        let main_code = self.pass2(&all_decls)?;
   158→
   159→        let program = Program {
   160→            // Build globals list with actual names (indexed by slot)
   161→            globals: (0..self.next_global)
   162→                .map(|i| self.global_slots.iter()
   163→                    .find(|(_, &slot)| slot == i)
   164→                    .map(|(name, _)| name.clone())
   165→                    .unwrap_or_default())
   166→                .collect(),
   167→            patterns: Vec::new(), // will be filled from pass1 data
   168→            learnables: Vec::new(),
   169→            rules: std::mem::take(&mut self.rules),
   170→            main_code,
   171→            collections_loaded: self.collections_loaded,
   172→        };
   173→
   174→        Ok(program)
   175→    }
   176→
   177→    /// Pass 1: collect type info, register patterns, assign global slots.
   178→    fn pass1(&mut self, decls: &[Declaration]) -> Result<(), String> {
   179→        for decl in decls {
   180→            match decl {
   181→                Declaration::EntityType(e) => {
   182→                    let fields: Vec<String> = e.fields.iter().map(|f| f.name.clone()).collect();
   183→                    self.struct_fields.insert(e.name.clone(), fields);
   184→                }
   185→                Declaration::EntityRecord(e) => {
   186→                    self.ensure_global(&e.name);
   187→                }
   188→                Declaration::EntitySimple(e) => {
   189→                    self.ensure_global(&e.name);
   190→                }
   191→                Declaration::Pattern(p) => {
   192→                    let idx = self.pattern_indices.len();
   193→                    self.pattern_indices.insert(p.name.clone(), idx);
   194→                }
   195→                Declaration::LearnablePattern(lp) => {
   196→                    let idx = self.learnable_indices.len();
   197→                    self.learnable_indices.insert(lp.name.clone(), idx);
   198→                }
   199→                Declaration::Fluid(fl) => {
   200→                    self.ensure_global(&fl.name);
   201→                }
   202→                Declaration::Rule(r) => {
   203→                    self.rules.push(self.compile_rule(r)?);
   204→                }
   205→                Declaration::Memorize(_) => {
   206→                    // Handled in pass2
   207→                }
   208→                Declaration::Sandbox(s) => {
   209→                    self.sandboxes.insert(s.name.clone(), s.clone());
   210→                }
   211→                Declaration::Adapt(_) | Declaration::Relate(_) | Declaration::Mutate(_) | Declaration::Forget(_) => {
   212→                    // Handled in pass2
   213→                }
   214→                Declaration::MlogServer(_) | Declaration::Template(_) | Declaration::Db(_) | Declaration::Memory(_) | Declaration::Conversation(_) | Declaration::Tool(_) | Declaration::LlmConfig(_) => {
   215→                    // Phase 6: handled elsewhere
   216→                }
   217→                _ => {}
   218→            }
   219→        }
   220→        Ok(())
   221→    }
   222→
   223→    /// Pass 2: generate main_code instructions.
   224→    fn pass2(&mut self, decls: &[Declaration]) -> Result<Vec<Instruction>, String> {
   225→        let mut code = Vec::new();
   226→
   227→        for decl in decls {
   228→            match decl {
   229→                Declaration::EntityType(e) => {
   230→                    // Struct type already registered in pass1. No runtime instruction needed.
   231→                    // (The VM will need to know about struct types for MakeStruct.)
   232→                    // We emit a special marker or nothing — the VM handles this via metadata.
   233→                    let _ = e;
   234→                }
   235→                Declaration::EntityRecord(e) => {
   236→                    // Evaluate each field initializer, create struct, store globally
   237→                    let field_names = self.struct_fields.get(&e.type_name)
   238→                        .ok_or_else(|| format!("compile: unknown struct type: {}", e.type_name))?;
   239→
   240→                    // Push field values in field order
   241→                    for fd_name in field_names {
   242→                        // Find initializer for this field
   243→                        let init = e.fields.iter().find(|fi| fi.name == *fd_name);
   244→                        match init {
   245→                            Some(fi) => self.compile_expr(&fi.value, &mut code)?,
   246→                            None => code.push(Instruction::Const(Value::Unit)),
   247→                        }
   248→                    }
   249→
   250→                    let slot = self.global_slots[&e.name];
   251→                    code.push(Instruction::MakeStruct(e.type_name.clone(), field_names.clone()));
   252→                    code.push(Instruction::StoreGlobal(slot));
   253→                }
   254→                Declaration::EntitySimple(e) => {
   255→                    self.compile_expr(&e.value, &mut code)?;
   256→                    let slot = self.global_slots[&e.name];
   257→                    code.push(Instruction::StoreGlobal(slot));
   258→                }
   259→                Declaration::Pattern(p) => {
   260→                    // Compile pattern body with parameter names as locals
   261→                    let mut locals: HashMap<String, usize> = p.params.iter()
   262→                        .enumerate()
   263→                        .map(|(i, param)| (param.name.clone(), i))
   264→                        .collect();
   265→                    let fn_code = self.compile_pattern_body_with_locals(&p.body, &mut locals)?;
   266→                    let is_pure = Self::analyze_purity(&fn_code, &p.params);
   267→                    let compiled = CompiledFn {
   268→                        name: p.name.clone(),
   269→                        param_count: p.params.len(),
   270→                        param_types: p.params.iter().map(|param| param.type_name.clone()).collect(),
   271→                        code: fn_code,
   272→                        is_pure,
   273→                    };
   274→                    // We'll add this to the program's patterns list
   275→                    // For now, store in a side-channel; we'll fix this below.
   276→                    // Actually, let's store directly. The problem is that we're building
   277→                    // the program in compile(), not here. Let's use a different approach.
   278→                    // We'll add a pseudo-instruction to register the pattern.
   279→                    code.push(Instruction::RegisterPattern(compiled));
   280→                }
   281→                Declaration::LearnablePattern(lp) => {
   282→                    code.push(Instruction::RegisterLearnable(CompiledLearnableInfo {
   283→                        name: lp.name.clone(),
   284→                        param_count: lp.params.len(),
   285→                        prompt: lp.prompt.clone(),
   286→                        few_shot: Vec::new(),
   287→                    }));
   288→                }
   289→                Declaration::Rule(_) => {
   290→                    // Rules are already compiled. Emit ExecuteRules before any flow.
   291→                    // We'll insert this before the flow instruction.
   292→                }
   293→                Declaration::Memorize(m) => {
   294→                    self.compile_expr(&m.value, &mut code)?;
   295→                    code.push(Instruction::Memorize(m.priority));
   296→                }
   297→                Declaration::Forget(f) => {
   298→                    self.compile_expr(&f.query, &mut code)?;
   299→                    // We need a Forget instruction — let's add it to the bytecode
   300→                    // For now, emit as a special instruction
   301→                    code.push(Instruction::Forget(f.days));
   302→                }
   303→                Declaration::Fluid(fl) => {
   304→                    // Compile Fluid value construction: push value, confidence pairs
   305→                    for v in &fl.variants {
   306→                        self.compile_expr(&v.value, &mut code)?;
   307→                        // Push confidence
   308→                        code.push(Instruction::Const(Value::Float(v.confidence)));
   309→                    }
   310→                    let slot = self.global_slots[&fl.name];
   311→                    code.push(Instruction::MakeFluid(fl.variants.len()));
   312→                    code.push(Instruction::StoreGlobal(slot));
   313→                }
   314→                Declaration::Adapt(a) => {
   315→                    // Evaluate input/output examples (try to resolve literals)
   316→                    // For now, just emit an adapt instruction
   317→                    self.compile_expr(&a.input_example, &mut code)?;
   318→                    self.compile_expr(&a.output_example, &mut code)?;
   319→                    code.push(Instruction::Adapt(a.pattern_name.clone()));
   320→                }
   321→                Declaration::Relate(r) => {
   322→                    self.compile_expr(&r.from, &mut code)?;
   323→                    self.compile_expr(&r.to, &mut code)?;
   324→                    code.push(Instruction::Const(Value::String(r.relation.clone())));
   325→                    code.push(Instruction::Relate);
   326→                }
   327→                Declaration::Sandbox(_) => {
   328→                    // No runtime instruction needed
   329→                }
   330→                Declaration::Mutate(m) => {
   331→                    // Compile new examples and rollback info
   332→                    let mut examples = Vec::new();
   333→                    for (inp, out) in &m.new_examples {
   334→                        self.compile_expr(inp, &mut code)?;
   335→                        self.compile_expr(out, &mut code)?;
   336→                        examples.push((String::new(), String::new())); // placeholder
   337→                    }
   338→                    let rollback_op = m.rollback_op.map(|op| match op {
   339→                        AstCompareOp::Gt => ConditionOp::Gt,
   340→                        AstCompareOp::Lt => ConditionOp::Lt,
   341→                        AstCompareOp::Ge => ConditionOp::Ge,
   342→                        AstCompareOp::Le => ConditionOp::Le,
   343→                        AstCompareOp::Eq => ConditionOp::Eq,
   344→                        _ => ConditionOp::Eq, // Ne and others fall back to Eq
   345→                    });
   346→                    code.push(Instruction::Mutate {
   347→                        pattern_name: m.pattern_name.clone(),
   348→                        example_count: m.new_examples.len(),
   349→                        rollback_threshold: m.rollback_threshold,
   350→                        rollback_op,
   351→                    });
   352→                }
   353→                Declaration::Flow(f) => {
   354→                    // Emit ExecuteRules before the flow (if any rules exist)
   355→                    if !self.rules.is_empty() {
   356→                        code.push(Instruction::ExecuteRules);
   357→                    }
   358→                    // Compile the flow source expression
   359→                    let source_expr = self.compile_flow_expr(&f.source);
   360→                    let mut branch_defs = Vec::new();
   361→                    for (step_name, branches) in &f.branch_defs {
   362→                        let compiled_branches: Vec<BranchDef> = branches.iter()
   363→                            .map(|b| {
   364→                                let op = match b.condition.op {
   365→                                    AstCompareOp::Gt => ConditionOp::Gt,
   366→                                    AstCompareOp::Lt => ConditionOp::Lt,
   367→                                    AstCompareOp::Ge => ConditionOp::Ge,
   368→                                    AstCompareOp::Le => ConditionOp::Le,
   369→                                    AstCompareOp::Eq => ConditionOp::Eq,
   370→                                    _ => ConditionOp::Eq, // Ne and others fall back to Eq
   371→                                };
   372→                                // Compile the threshold expression to a constant if possible
   373→                                let threshold_val = self.eval_const_expr(&b.condition.threshold);
   374→                                BranchDef {
   375→                                    label: b.label.clone(),
   376→                                    condition_field: b.condition.field.clone(),
   377→                                    condition_op: op,
   378→                                    condition_threshold: threshold_val,
   379→                                    target: b.target.clone(),
   380→                                }
   381→                            })
   382→                            .collect();
   383→                        branch_defs.push((step_name.clone(), compiled_branches));
   384→                    }
   385→                    code.push(Instruction::FlowExec {
   386→                        source_expr,
   387→                        pipeline: f.pipeline.clone(),
   388→                        branch_defs,
   389→                    });
   390→                }
   391→                Declaration::Import(_) => {
   392→                    // Already resolved in import preprocessing
   393→                }
   394→                Declaration::MlogServer(_) | Declaration::Template(_) | Declaration::Db(_) | Declaration::Memory(_)
   395→                | Declaration::Hook(_) | Declaration::Eval(_) | Declaration::Conversation(_) | Declaration::Tool(_) | Declaration::LlmConfig(_) => {
   396→                    // Phase 6+: no bytecode instruction needed
   397→                }
   398→            }
   399→        }
   400→
   401→        code.push(Instruction::Halt);
   402→        Ok(code)
   403→    }

   405→    /// Compile an AST expression into stack instructions (no locals context).
   406→    fn compile_expr(&self, expr: &Expr, code: &mut Vec<Instruction>) -> Result<(), String> {
   407→        self.compile_expr_with_locals(expr, code, &HashMap::new())
   408→    }
   409→
   410→    /// Compile an AST expression into stack instructions with a locals map.
   411→    fn compile_expr_with_locals(&self, expr: &Expr, code: &mut Vec<Instruction>, locals: &HashMap<String, usize>) -> Result<(), String> {
   412→        match expr {
   413→            Expr::StringLit(s) => {
   414→                code.push(Instruction::Const(Value::String(s.clone())));
   415→            }
   416→            Expr::FloatLit(f) => {
   417→                code.push(Instruction::Const(Value::Float(*f)));
   418→            }
   419→            Expr::Ident(name) => {
   420→                // Check if it's a local (parameter or let binding) first
   421→                if let Some(&slot) = locals.get(name) {
   422→                    code.push(Instruction::LoadLocal(slot));
   423→                } else if let Some(&slot) = self.global_slots.get(name) {
   424→                    code.push(Instruction::LoadGlobal(slot));
   425→                } else {
   426→                    code.push(Instruction::LoadGlobalByName(name.clone()));
   427→                }
   428→            }
   429→            Expr::FieldAccess(base, field) => {
   430→                self.compile_expr_with_locals(base, code, locals)?;
   431→                code.push(Instruction::GetField(field.clone()));
   432→            }
   433→            Expr::FnCall(name, args) => {
   434→                for arg in args {
   435→                    self.compile_expr_with_locals(arg, code, locals)?;
   436→                }
   437→                let arity = args.len();
   438→                // Check if it's a builtin
   439→                if let Some(&idx) = self.builtin_indices.get(name) {
   440→                    code.push(Instruction::CallBuiltin(idx, arity));
   441→                } else if let Some(&idx) = self.pattern_indices.get(name) {
   442→                    code.push(Instruction::CallPattern(idx, arity));
   443→                } else if let Some(&idx) = self.learnable_indices.get(name) {
   444→                    code.push(Instruction::LlmCall(idx, arity));
   445→                } else {
   446→                    return Err(format!("compile: undefined function: {}", name));
   447→                }
   448→            }
   449→            Expr::BinaryOp(left, op, right) => {
   450→                self.compile_expr_with_locals(left, code, locals)?;
   451→                self.compile_expr_with_locals(right, code, locals)?;
   452→                match op {
   453→                    BinOp::Add => code.push(Instruction::Add),
   454→                    BinOp::Sub => code.push(Instruction::Sub),
   455→                    BinOp::Mul => code.push(Instruction::Mul),
   456→                    BinOp::Div => code.push(Instruction::Div),
   457→                    // Phase 5.1: comparison operators
   458→                    BinOp::Gt => code.push(Instruction::CmpGt),
   459→                    BinOp::Lt => code.push(Instruction::CmpLt),
   460→                    BinOp::Ge => code.push(Instruction::CmpGe),
   461→                    BinOp::Le => code.push(Instruction::CmpLe),
   462→                    BinOp::Eq => code.push(Instruction::CmpEq),
   463→                    BinOp::Ne => code.push(Instruction::CmpNe),
   464→                }
   465→            }
   466→            Expr::IfElse(cond, then_expr, else_expr) => {
   467→                // Compile condition
   468→                self.compile_expr_with_locals(cond, code, locals)?;
   469→                // Jump to else branch if falsy
   470→                let jump_to_else = code.len();
   471→                code.push(Instruction::JumpIfNot(0)); // placeholder
   472→                // Compile then branch
   473→                self.compile_expr_with_locals(then_expr, code, locals)?;
   474→                // Jump past else branch
   475→                let jump_to_end = code.len();
   476→                code.push(Instruction::Jump(0)); // placeholder
   477→                // Patch: else branch starts here
   478→                let else_start = code.len();
   479→                if let Some(Instruction::JumpIfNot(ref mut target)) = code.get_mut(jump_to_else) {
   480→                    *target = else_start;
   481→                }
   482→                // Compile else branch
   483→                self.compile_expr_with_locals(else_expr, code, locals)?;
   484→                // Patch: end jump target
   485→                let end = code.len();
   486→                if let Some(Instruction::Jump(ref mut target)) = code.get_mut(jump_to_end) {
   487→                    *target = end;
   488→                }
   489→            }
   490→            // Additional expression forms in Metalogos- AST
   491→            Expr::BoolLit(b) => {
   492→                code.push(Instruction::Const(Value::Float(if *b { 1.0 } else { 0.0 })));
   493→            }
   494→            // Наряд №18: QualifiedCall — resolve function part (ignore module prefix)
   495→            Expr::QualifiedCall { module: _, function, args } => {
   496→                for arg in args {
   497→                    self.compile_expr_with_locals(arg, code, locals)?;
   498→                }
   499→                let arity = args.len();
   500→                if let Some(&idx) = self.builtin_indices.get(function) {
   501→                    code.push(Instruction::CallBuiltin(idx, arity));
   502→                } else if let Some(&idx) = self.pattern_indices.get(function) {
   503→                    code.push(Instruction::CallPattern(idx, arity));
   504→                } else if let Some(&idx) = self.learnable_indices.get(function) {
   505→                    code.push(Instruction::LlmCall(idx, arity));
   506→                } else {
   507→                    return Err(format!("compile: undefined function in qualified call: {}", function));
   508→                }
   509→            }
   510→            // Наряд №18: proper list construction via MakeList
   511→            Expr::List(items) => {
   512→                for item in items {
   513→                    self.compile_expr_with_locals(item, code, locals)?;
   514→                }
   515→                code.push(Instruction::MakeList(items.len()));
   516→            }
   517→            Expr::IndexAccess(base, index) => {
   518→                self.compile_expr_with_locals(base, code, locals)?;
   519→                self.compile_expr_with_locals(index, code, locals)?;
   520→                code.push(Instruction::IndexAccess);
   521→            }
   522→            Expr::StructLit(fields) => {
   523→                let field_names: Vec<String> = fields.keys().cloned().collect();
   524→                for (_, val_expr) in fields {
   525→                    self.compile_expr_with_locals(val_expr, code, locals)?;
   526→                }
   527→                code.push(Instruction::MakeStruct("Struct".to_string(), field_names));
   528→            }
   529→            // Наряд №14 P0-3: block if/else expression — deferred to tree-walking
   530→            Expr::BlockIfElse { .. } => {
   531→                // Compiled as Unit placeholder; tree-walking interpreter handles it
   532→                code.push(Instruction::Const(Value::Unit));
   533→            }
   534→            // Наряд №14 P1-4: try expression — deferred to tree-walking
   535→            Expr::Try(_) => {
   536→                code.push(Instruction::Const(Value::Unit));
   537→            }
   538→        }
   539→        Ok(())
   540→    }
   541→
   542→    /// Loop context for break/continue jump patching. (Наряд №18)
   543→    struct LoopCtx {
   544→        /// Address of the condition check (for continue to jump back to).
   545→        continue_addr: usize,
   546→        /// Locations of Jump(0) placeholders emitted by Break that need patching.
   547→        break_patches: Vec<usize>,
   548→    }

   550→    /// Compile a pattern body with parameter names as locals.
   551→    /// Phase 5.1: also handles let bindings, assigning additional local slots.
   552→    /// Наряд №18: full statement compilation — IfElseBlock, Each, EachWithIndex,
   553→    /// While, Assign, Match, ExprStmt, IfThen, Break, Continue.
   554→    fn compile_pattern_body_with_locals(&self, body: &[Statement], locals: &mut HashMap<String, usize>) -> Result<Vec<Instruction>, String> {
   555→        let mut code = Vec::new();
   556→        let mut next_slot = locals.len();
   557→        self.compile_stmts(body, &mut code, locals, &mut next_slot, None)?;
   558→        Ok(code)
   559→    }
   560→
   561→    /// Compile a list of statements into the code buffer.
   562→    /// Наряд №18: the core statement compilation engine.
   563→    fn compile_stmts(
   564→        &self,
   565→        stmts: &[Statement],
   566→        code: &mut Vec<Instruction>,
   567→        locals: &mut HashMap<String, usize>,
   568→        next_slot: &mut usize,
   569→        loop_ctx: Option<&mut LoopCtx>,
   570→    ) -> Result<(), String> {
   571→        for stmt in stmts {
   572→            self.compile_stmt(stmt, code, locals, next_slot, loop_ctx)?;
   573→        }
   574→        Ok(())
   575→    }
   576→
   577→    /// Compile a single statement into the code buffer.
   578→    fn compile_stmt(
   579→        &self,
   580→        stmt: &Statement,
   581→        code: &mut Vec<Instruction>,
   582→        locals: &mut HashMap<String, usize>,
   583→        next_slot: &mut usize,
   584→        loop_ctx: Option<&mut LoopCtx>,
   585→    ) -> Result<(), String> {
   586→        match stmt {
   587→            Statement::LetBinding { name, value, mutable: _ } => {
   588→                let slot = *next_slot;
   589→                *next_slot += 1;
   590→                locals.insert(name.clone(), slot);
   591→                self.compile_expr_with_locals(value, code, locals)?;
   592→                code.push(Instruction::StoreLocal(slot));
   593→            }
   594→            Statement::Return(expr) => {
   595→                self.compile_expr_with_locals(expr, code, locals)?;
   596→                code.push(Instruction::Return);
   597→            }
   598→            // Наряд №18: Assign — store to local if known, else global
   599→            Statement::Assign { name, value } => {
   600→                self.compile_expr_with_locals(value, code, locals)?;
   601→                if let Some(&slot) = locals.get(name) {
   602→                    code.push(Instruction::StoreLocal(slot));
   603→                } else if let Some(&slot) = self.global_slots.get(name) {
   604→                    code.push(Instruction::StoreGlobal(slot));
   605→                } else {
   606→                    // Fallback: store by name
   607→                    code.push(Instruction::Pop);
   608→                }
   609→            }
   610→            // Наряд №18: ExprStmt — compile expression, discard result
   611→            Statement::ExprStmt(expr) => {
   612→                self.compile_expr_with_locals(expr, code, locals)?;
   613→                code.push(Instruction::Pop);
   614→            }
   615→            // Наряд №18: IfElseBlock — full JumpIfNot/Jump chain
   616→            Statement::IfElseBlock { condition, then_body, else_ifs, else_body } => {
   617→                self.compile_expr_with_locals(condition, code, locals)?;
   618→                let jump_to_else = code.len();
   619→                code.push(Instruction::JumpIfNot(0)); // placeholder
   620→                // Compile then_body
   621→                self.compile_stmts(then_body, code, locals, next_slot, loop_ctx)?;
   622→                let jump_to_end = code.len();
   623→                code.push(Instruction::Jump(0)); // placeholder
   624→                // Patch: else branch starts here
   625→                let else_start = code.len();
   626→                if let Some(Instruction::JumpIfNot(ref mut target)) = code.get_mut(jump_to_else) {
   627→                    *target = else_start;
   628→                }
   629→                // Compile else-if chain — track each end-jump for patching
   630→                let mut ei_end_jumps: Vec<usize> = Vec::new();
   631→                for (ei_cond, ei_body) in else_ifs {
   632→                    self.compile_expr_with_locals(ei_cond, code, locals)?;
   633→                    let ei_jump = code.len();
   634→                    code.push(Instruction::JumpIfNot(0)); // placeholder
   635→                    self.compile_stmts(ei_body, code, locals, next_slot, loop_ctx)?;
   636→                    let ei_jump_end = code.len();
   637→                    code.push(Instruction::Jump(0)); // placeholder
   638→                    ei_end_jumps.push(ei_jump_end);
   639→                    // Patch else-if condition jump to next arm
   640→                    let ei_next = code.len();
   641→                    if let Some(Instruction::JumpIfNot(ref mut target)) = code.get_mut(ei_jump) {
   642→                        *target = ei_next;
   643→                    }
   644→                }
   645→                // Compile else body
   646→                if let Some(eb) = else_body {
   647→                    self.compile_stmts(eb, code, locals, next_slot, loop_ctx)?;
   648→                }
   649→                // Patch all end jumps to here
   650→                let end = code.len();
   651→                if let Some(Instruction::Jump(ref mut target)) = code.get_mut(jump_to_end) {
   652→                    *target = end;
   653→                }
   654→                for &patch_addr in &ei_end_jumps {
   655→                    if let Some(Instruction::Jump(ref mut target)) = code.get_mut(patch_addr) {
   656→                        *target = end;
   657→                    }
   658→                }
   659→            }
   660→            // Наряд №18: IfThen — single-branch conditional
   661→            Statement::IfThen(cond, body) => {
   662→                self.compile_expr_with_locals(cond, code, locals)?;
   663→                let jump_past = code.len();
   664→                code.push(Instruction::JumpIfNot(0)); // placeholder
   665→                self.compile_stmts(body, code, locals, next_slot, loop_ctx)?;
   666→                let after = code.len();
   667→                if let Some(Instruction::JumpIfNot(ref mut target)) = code.get_mut(jump_past) {
   668→                    *target = after;
   669→                }
   670→            }
   671→            // Наряд №18: Each — indexed iteration over list
   672→            Statement::Each { variable, iterable, body } => {
   673→                // Evaluate iterable → list on stack
   674→                self.compile_expr_with_locals(iterable, code, locals)?;
   675→                // Store list in hidden local
   676→                let list_slot = *next_slot;
   677→                *next_slot += 1;
   678→                code.push(Instruction::StoreLocal(list_slot));
   679→                // Initialize index = 0
   680→                let idx_slot = *next_slot;
   681→                *next_slot += 1;
   682→                code.push(Instruction::Const(Value::Float(0.0)));
   683→                code.push(Instruction::StoreLocal(idx_slot));
   684→                // Set up loop context
   685→                let mut inner_loop = LoopCtx {
   686→                    continue_addr: 0,
   687→                    break_patches: Vec::new(),
   688→                };
   689→                // Condition check: idx < len(list)
   690→                let cond_addr = code.len();
   691→                inner_loop.continue_addr = cond_addr;
   692→                code.push(Instruction::LoadLocal(idx_slot));
   693→                code.push(Instruction::LoadLocal(list_slot));
   694→                code.push(Instruction::ListLen);
   695→                code.push(Instruction::CmpLt);
   696→                let exit_jump = code.len();
   697→                code.push(Instruction::JumpIfNot(0)); // placeholder
   698→                // Load item: list[idx]
   699→                code.push(Instruction::LoadLocal(list_slot));
   700→                code.push(Instruction::LoadLocal(idx_slot));
   701→                code.push(Instruction::IndexAccess);
   702→                // Bind item variable
   703→                let item_slot = if let Some(&existing) = locals.get(variable) {
   704→                    existing
   705→                } else {
   706→                    let s = *next_slot;
   707→                    *next_slot += 1;
   708→                    locals.insert(variable.clone(), s);
   709→                    s
   710→                };
   711→                code.push(Instruction::StoreLocal(item_slot));
   712→                // Compile body with loop context
   713→                self.compile_stmts(body, code, locals, next_slot, Some(&mut inner_loop))?;
   714→                // Increment index
   715→                code.push(Instruction::LoadLocal(idx_slot));
   716→                code.push(Instruction::Const(Value::Float(1.0)));
   717→                code.push(Instruction::Add);
   718→                code.push(Instruction::StoreLocal(idx_slot));
   719→                // Jump back to condition
   720→                code.push(Instruction::Jump(cond_addr));
   721→                // Loop end
   722→                let loop_end = code.len();
   723→                // Patch exit jump
   724→                if let Some(Instruction::JumpIfNot(ref mut target)) = code.get_mut(exit_jump) {
   725→                    *target = loop_end;
   726→                }
   727→                // Patch break jumps
   728→                for &patch_addr in &inner_loop.break_patches {
   729→                    if let Some(Instruction::Jump(ref mut target)) = code.get_mut(patch_addr) {
   730→                        *target = loop_end;
   731→                    }
   732→                }
   733→            }
   734→            // Наряд №18: EachWithIndex — two-variable iteration
   735→            Statement::EachWithIndex { index_var, item_var, iterable, body } => {
   736→                self.compile_expr_with_locals(iterable, code, locals)?;
   737→                let list_slot = *next_slot;
   738→                *next_slot += 1;
   739→                code.push(Instruction::StoreLocal(list_slot));
   740→                let idx_slot = *next_slot;
   741→                *next_slot += 1;
   742→                code.push(Instruction::Const(Value::Float(0.0)));
   743→                code.push(Instruction::StoreLocal(idx_slot));
   744→                let mut inner_loop = LoopCtx {
   745→                    continue_addr: 0,
   746→                    break_patches: Vec::new(),
   747→                };
   748→                let cond_addr = code.len();
   749→                inner_loop.continue_addr = cond_addr;
   750→                code.push(Instruction::LoadLocal(idx_slot));
   751→                code.push(Instruction::LoadLocal(list_slot));
   752→                code.push(Instruction::ListLen);
   753→                code.push(Instruction::CmpLt);
   754→                let exit_jump = code.len();
   755→                code.push(Instruction::JumpIfNot(0));
   756→                // Bind index var
   757→                let idx_bind = if let Some(&existing) = locals.get(index_var) {
   758→                    existing
   759→                } else {
   760→                    let s = *next_slot;
   761→                    *next_slot += 1;
   762→                    locals.insert(index_var.clone(), s);
   763→                    s
   764→                };
   765→                code.push(Instruction::LoadLocal(idx_slot));
   766→                code.push(Instruction::StoreLocal(idx_bind));
   767→                // Load and bind item var
   768→                code.push(Instruction::LoadLocal(list_slot));
   769→                code.push(Instruction::LoadLocal(idx_slot));
   770→                code.push(Instruction::IndexAccess);
   771→                let item_slot = if let Some(&existing) = locals.get(item_var) {
   772→                    existing
   773→                } else {
   774→                    let s = *next_slot;
   775→                    *next_slot += 1;
   776→                    locals.insert(item_var.clone(), s);
   777→                    s
   778→                };
   779→                code.push(Instruction::StoreLocal(item_slot));
   780→                // Compile body
   781→                self.compile_stmts(body, code, locals, next_slot, Some(&mut inner_loop))?;
   782→                // Increment index
   783→                code.push(Instruction::LoadLocal(idx_slot));
   784→                code.push(Instruction::Const(Value::Float(1.0)));
   785→                code.push(Instruction::Add);
   786→                code.push(Instruction::StoreLocal(idx_slot));
   787→                code.push(Instruction::Jump(cond_addr));
   788→                let loop_end = code.len();
   789→                if let Some(Instruction::JumpIfNot(ref mut target)) = code.get_mut(exit_jump) {
   790→                    *target = loop_end;
   791→                }
   792→                for &patch_addr in &inner_loop.break_patches {
   793→                    if let Some(Instruction::Jump(ref mut target)) = code.get_mut(patch_addr) {
   794→                        *target = loop_end;
   795→                    }
   796→                }
   797→            }
   798→            // Наряд №18: While — condition + body loop
   799→            Statement::While { condition, body } => {
   800→                let mut inner_loop = LoopCtx {
   801→                    continue_addr: 0,
   802→                    break_patches: Vec::new(),
   803→                };
   804→                let cond_addr = code.len();
   805→                inner_loop.continue_addr = cond_addr;
   806→                self.compile_expr_with_locals(condition, code, locals)?;
   807→                let exit_jump = code.len();
   808→                code.push(Instruction::JumpIfNot(0));
   809→                self.compile_stmts(body, code, locals, next_slot, Some(&mut inner_loop))?;
   810→                code.push(Instruction::Jump(cond_addr));
   811→                let loop_end = code.len();
   812→                if let Some(Instruction::JumpIfNot(ref mut target)) = code.get_mut(exit_jump) {
   813→                    *target = loop_end;
   814→                }
   815→                for &patch_addr in &inner_loop.break_patches {
   816→                    if let Some(Instruction::Jump(ref mut target)) = code.get_mut(patch_addr) {
   817→                        *target = loop_end;
   818→                    }
   819→                }
   820→            }
   821→            // Наряд №18: Match — dispatch via chained comparisons
   822→            Statement::Match { scrutinee, arms, else_body } => {
   823→                // Store scrutinee in a hidden local for repeated access
   824→                self.compile_expr_with_locals(scrutinee, code, locals)?;
   825→                let scrut_slot = *next_slot;
   826→                *next_slot += 1;
   827→                code.push(Instruction::StoreLocal(scrut_slot));
   828→                let mut arm_end_jumps: Vec<usize> = Vec::new();
   829→                for arm in arms {
   830→                    match arm {
   831→                        MatchArm::Exact(val, arm_body) => {
   832→                            code.push(Instruction::LoadLocal(scrut_slot));
   833→                            // Convert scrutinee to string via str() builtin if possible,
   834→                            // or use CmpEq directly for string comparison
   835→                            code.push(Instruction::Const(Value::String(val.clone())));
   836→                            code.push(Instruction::CmpEq);
   837→                            let arm_skip = code.len();
   838→                            code.push(Instruction::JumpIfNot(0));
   839→                            self.compile_stmts(arm_body, code, locals, next_slot, loop_ctx)?;
   840→                            let arm_end = code.len();
   841→                            code.push(Instruction::Jump(0));
   842→                            arm_end_jumps.push(arm_end);
   843→                            let next_arm = code.len();
   844→                            if let Some(Instruction::JumpIfNot(ref mut target)) = code.get_mut(arm_skip) {
   845→                                *target = next_arm;
   846→                            }
   847→                        }
   848→                        MatchArm::Contains(substr, arm_body) => {
   849→                            code.push(Instruction::LoadLocal(scrut_slot));
   850→                            code.push(Instruction::Const(Value::String(substr.clone())));
   851→                            code.push(Instruction::Contains);
   852→                            let arm_skip = code.len();
   853→                            code.push(Instruction::JumpIfNot(0));
   854→                            self.compile_stmts(arm_body, code, locals, next_slot, loop_ctx)?;
   855→                            let arm_end = code.len();
   856→                            code.push(Instruction::Jump(0));
   857→                            arm_end_jumps.push(arm_end);
   858→                            let next_arm = code.len();
   859→                            if let Some(Instruction::JumpIfNot(ref mut target)) = code.get_mut(arm_skip) {
   860→                                *target = next_arm;
   861→                            }
   862→                        }
   863→                        MatchArm::StartsWith(_prefix, _arm_body) => {
   864→                            // StartsWith not directly supported in bytecode — skip arm
   865→                            // Users should use Contains or Exact as bytecode-compatible alternatives
   866→                            let _ = (_prefix, _arm_body);
   867→                        }
   868→                        MatchArm::Compare(op, threshold, arm_body) => {
   869→                            code.push(Instruction::LoadLocal(scrut_slot));
   870→                            self.compile_expr_with_locals(threshold, code, locals)?;
   871→                            let cmp_instr = match op {
   872→                                AstCompareOp::Gt => Instruction::CmpGt,
   873→                                AstCompareOp::Lt => Instruction::CmpLt,
   874→                                AstCompareOp::Ge => Instruction::CmpGe,
   875→                                AstCompareOp::Le => Instruction::CmpLe,
   876→                                AstCompareOp::Eq => Instruction::CmpEq,
   877→                                AstCompareOp::Ne => Instruction::CmpNe,
   878→                            };
   879→                            code.push(cmp_instr);
   880→                            let arm_skip = code.len();
   881→                            code.push(Instruction::JumpIfNot(0));
   882→                            self.compile_stmts(arm_body, code, locals, next_slot, loop_ctx)?;
   883→                            let arm_end = code.len();
   884→                            code.push(Instruction::Jump(0));
   885→                            arm_end_jumps.push(arm_end);
   886→                            let next_arm = code.len();
   887→                            if let Some(Instruction::JumpIfNot(ref mut target)) = code.get_mut(arm_skip) {
   888→                                *target = next_arm;
   889→                            }
   890→                        }
   891→                    }
   892→                }
   893→                // Else body
   894→                if let Some(eb) = else_body {
   895→                    self.compile_stmts(eb, code, locals, next_slot, loop_ctx)?;
   896→                }
   897→                // Patch all arm end jumps to here
   898→                let match_end = code.len();
   899→                for &patch_addr in &arm_end_jumps {
   900→                    if let Some(Instruction::Jump(ref mut target)) = code.get_mut(patch_addr) {
   901→                        *target = match_end;
   902→                    }
   903→                }
   904→            }
   905→            // Наряд №18: Break — Jump to loop end (placeholder, patched by loop)
   906→            Statement::Break => {
   907→                if let Some(lc) = loop_ctx {
   908→                    let patch = code.len();
   909→                    code.push(Instruction::Jump(0)); // placeholder
   910→                    lc.break_patches.push(patch);
   911→                }
   912→                // Outside loop: silently ignore (matches interpreter error behavior at higher level)
   913→            }
   914→            // Наряд №18: Continue — Jump to loop condition check
   915→            Statement::Continue => {
   916→                if let Some(lc) = loop_ctx {
   917→                    code.push(Instruction::Jump(lc.continue_addr));
   918→                }
   919→            }
   920→        }
   921→        Ok(())
   922→    }

   924→    /// Compile a flow source expression into a FlowExpr.
   925→    fn compile_flow_expr(&self, expr: &Expr) -> FlowExpr {
   926→        match expr {
   927→            Expr::Ident(name) => {
   928→                if let Some(&slot) = self.global_slots.get(name) {
   929→                    FlowExpr::GlobalSlot(slot)
   930→                } else {
   931→                    FlowExpr::Ident(name.clone())
   932→                }
   933→            }
   934→            Expr::StringLit(s) => FlowExpr::Const(Value::String(s.clone())),
   935→            Expr::FloatLit(f) => FlowExpr::Const(Value::Float(*f)),
   936→            _ => FlowExpr::Ident(format!("{:?}", expr)),
   937→        }
   938→    }

   940→    /// Try to evaluate an expression to a constant Value.
   941→    fn eval_const_expr(&self, expr: &Expr) -> Value {
   942→        match expr {
   943→            Expr::StringLit(s) => Value::String(s.clone()),
   944→            Expr::FloatLit(f) => Value::Float(*f),
   945→            _ => Value::Unit,
   946→        }
   947→    }

   949→    /// Compile a rule into CompiledRule.
   950→    fn compile_rule(&self, rule: &RuleDecl) -> Result<CompiledRule, String> {
   951→        let condition = match &rule.condition {
   952→            Condition::Contains { left, right } => {
   953→                RuleCondition::Contains {
   954→                    left: self.rule_value_expr(left),
   955→                    right: self.rule_value_expr(right),
   956→                }
   957→            }
   958→            Condition::Compare { left, op, right } => {
   959→                RuleCondition::Compare {
   960→                    left: self.rule_value_expr(left),
   961→                    op: match op {
   962→                        AstCompareOp::Gt => ConditionOp::Gt,
   963→                        AstCompareOp::Lt => ConditionOp::Lt,
   964→                        AstCompareOp::Ge => ConditionOp::Ge,
   965→                        AstCompareOp::Le => ConditionOp::Le,
   966→                        AstCompareOp::Eq => ConditionOp::Eq,
   967→                        _ => ConditionOp::Eq, // Ne and others fall back to Eq
   968→                    },
   969→                    right: self.rule_value_expr(right),
   970→                }
   971→            }
   972→        };
   973→
   974→        let target_name = match &rule.target {
   975→            Expr::Ident(name) => name.clone(),
   976→            _ => return Err("rule target must be an identifier".to_string()),
   977→        };
   978→
   979→        Ok(CompiledRule {
   980→            condition,
   981→            target_name,
   982→            field: rule.field.clone(),
   983→            value_expr: self.rule_value_expr(&rule.value),
   984→            priority: rule.priority,
   985→        })
   986→    }

   988→    /// Convert an AST expression to a simplified rule value expression.
   989→    fn rule_value_expr(&self, expr: &Expr) -> RuleValueExpr {
   990→        match expr {
   991→            Expr::Ident(name) => RuleValueExpr::Ident(name.clone()),
   992→            Expr::StringLit(s) => RuleValueExpr::StringLit(s.clone()),
   993→            Expr::FloatLit(f) => RuleValueExpr::FloatLit(*f),
   994→            Expr::FieldAccess(base, field) => {
   995→                if let Expr::Ident(name) = base.as_ref() {
   996→                    RuleValueExpr::FieldAccess(name.clone(), field.clone())
   997→                } else {
   998→                    RuleValueExpr::Ident(format!("{:?}", expr))
   999→                }
  1000→            }
  1001→            _ => RuleValueExpr::Ident(format!("{:?}", expr)),
  1002→        }
  1003→    }

  1005→    /// Analyze a compiled pattern body for purity.
  1006→    /// A pattern is pure if it contains only:
  1007→    ///   - LoadLocal, Const (any type), Add, Sub, Mul, Div
  1008→    ///   - CmpGt/CmpLt/CmpGe/CmpLe/CmpEq (allowed in purity, but NOT JIT-compiled)
  1009→    ///   - Return
  1010→    /// And ALL parameter types are "Float".
  1011→    /// No globals, no builtins, no LLM calls, no struct operations, no memory.
  1012→    /// Note: is_pure is a broader check than JIT-eligibility. JIT additionally
  1013→    /// requires only arithmetic (no Cmp*). See JitCompiler::is_jit_eligible().
  1014→    fn analyze_purity(code: &[Instruction], params: &[crate::ast::Param]) -> bool {

  1050→    /// Resolve an import: find file, parse, recursively resolve sub-imports.
  1051→    fn resolve_import(&mut self, module_path: &str) -> Result<Vec<Declaration>, String> {
  1052→        if self.imported_modules.contains(module_path) {
  1053→            return Ok(Vec::new());
  1054→        }
  1055→        self.imported_modules.insert(module_path.to_string());
  1056→
  1057→        if module_path == "std/collections" {
  1058→            self.collections_loaded = true;
  1059→        }
  1060→
  1061→        let file_path = self.std_root.join(module_path).with_extension("mlog");
  1062→        let source = std::fs::read_to_string(&file_path)
  1063→            .map_err(|e| format!(
  1064→                "import '{}': cannot read {:?}: {}",
  1065→                module_path, file_path, e
  1066→            ))?;
  1067→
  1068→        let mut declarations = crate::parser::parse(&source)
  1069→            .map_err(|e| format!("import '{}': parse error: {}", module_path, e))?;
  1070→
  1071→        let mut resolved = Vec::new();
  1072→        for decl in declarations.drain(..) {
  1073→            if let Declaration::Import(sub_import) = &decl {
  1074→                // Fix 2: use `sub_import.path` instead of `sub_import.module_path`
  1075→                if !self.imported_modules.contains(&sub_import.path) {
  1076→                    let sub_decls = self.resolve_import(&sub_import.path)?;
  1077→                    resolved.extend(sub_decls);
  1078→                }
  1079→            } else {
  1080→                resolved.push(decl);
  1081→            }
  1082→        }
  1083→
  1084→        Ok(resolved)
  1085→    }
